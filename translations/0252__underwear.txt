# underwear

!! gs 'underwear', 'dispose'
!! gs 'underwear', 'remove'
!! gs 'underwear', 'wear'
!! As remove and wear but with a backup in place to ensure the correct underwear state is restored
!! gs 'underwear', 'backup'
!! gs 'underwear', 'restore'

if $ARGS[0] = 'dispose':
	if underwear['type'] = 2:
		if $bodysuitworntype ! 'none': gs 'underwear_bodysuits', 'dispose'
	else
		if $pantyworntype ! 'none': gs 'panties', 'dispose'
		if $braworntype ! 'none': gs 'bras', 'dispose'
	end
	exit
end


if $ARGS[0] = 'backup':
	lastwornunderwear['backup'] = underwear['type']
	if underwear['type'] = 2:
		$lastwornbodysuittype['backup'] = $bodysuitworntype
		lastwornbodysuitnumber['backup'] = bodysuitwornnumber
		gs 'underwear_bodysuits', 'remove'
	else
		$lastwornpantytype['backup'] = $pantyworntype
		lastwornpantynumber['backup'] = pantywornnumber
		$lastwornbratype['backup'] = $braworntype
		lastwornbranumber['backup'] = brawornnumber
		gs 'panties', 'remove'
		gs 'bras', 'remove'
	end
	exit
end


if $ARGS[0] = 'strip' or $ARGS[0] = 'remove':
	!! check for swimwear to prevent duplicate stripping stopping Sveta putting underwear on when she puts her clothes on
	if instr($clothingworntype, 'swimsuit') <= 0 and instr($clothingworntype, 'bikinis') <= 0:
		if underwear['type'] = 2:
			gs 'underwear_bodysuits', 'strip'
		else
			!! if $pantyworntype ! 'none':
			gs 'panties', 'strip'
			!! if $braworntype ! 'none':
			gs 'bras', 'strip'
		end
	end
	exit
end


if $ARGS[0] = 'restore':
	if lastwornunderwear['backup'] = 2:
		gs 'underwear_bodysuits', 'wear', $lastwornbodysuittype['backup'], lastwornbodysuitnumber['backup']
	else
		gs 'panties', 'wear', $lastwornpantytype['backup'], lastwornpantynumber['backup']
		gs 'bras', 'wear', $lastwornbratype['backup'], lastwornbranumber['backup']
	end
	exit
end


if $ARGS[0] = 'wear':
	if lastwornunderwear = 2:
		gs 'underwear_bodysuits', 'wear', $ARGS[1], ARGS[2]
	else
		gs 'panties', 'wear', $ARGS[1], ARGS[2]
		gs 'bras', 'wear', $ARGS[1], ARGS[2]
	end
	exit
end


if $ARGS[0] = 'dresser':
	'在梳妆台里，你可以翻看自己所有的内衣。其中大部分只是普通内衣，但你也能拥有一些独特的款式。'
	'这些独特款式会按胸罩和内裤分类，并按购买的商店区分。它们必须在这里挑选，而自动选项会默认使用普通内衣。'
	*nl
	'这里的“穿内裤”和“穿胸罩”选项针对的是普通内衣，那些大按钮则用于挑选独特款式。'
	*nl
	'你的梳妆台里有<<func(''panties'',''sum'',''dresser'')>>条普通内裤和<<func(''bras'',''sum'',''dresser'')>>件普通胸罩。'
	*nl
	*nl
	*nl
	*nl

	*p '<center><a href="exec:gt ''bra_view'', ''view_grid'', ''clean'', 6"><img src="images/system/icons/bra.png"></a>                               <a href="exec:gt ''panty_view'', ''view_grid'', ''clean'', 6"><img src="images/system/icons/panties.png"></a></center>'

	gs 'din_van', 'wearpan'
	gs 'din_van', 'removepan'

	act '返回': gt $loc, $loc_arg
	exit
end
