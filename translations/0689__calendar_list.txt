# calendar_list

!! Module 7: Event List
!! Contains definitions of all events that can be added to the calendar
!! After defining an event here, call gs 'calendar', 'add', 'event_function_name' to add it to the calendar
!!
!! TIMESLOT CONVERSION REFERENCE:
!! The calendar uses timeslots (0-95) where each timeslot = 15 minutes
!! To convert hour:minute to timeslot: ts = (hour * 4) + (minute / 15)
!! Examples: 08:00 → 32, 09:30 → 38, 12:00 → 48, 16:00 → 64
!! To convert minutes to timeslots: ts = minutes / 15
!! Examples: 30 min → 2 ts, 60 min → 4 ts, 90 min → 6 ts, 120 min → 8 ts

!! ============================================================================
!! Event Variables:
!! $event_vars['id']         - Unique event identifier
!! $event_vars['title']      - Event title
!! $event_vars['loc']        - Event location
!! $event_vars['desc']       - Event description
!! event_vars['daystart']    - Start day
!! event_vars['all_day']     - 0=timed event, 1=all-day event
!! event_vars['color']       - Event color number (0..20). 0 for random color
!! event_vars['blocking']    - 0=non-blocking, 1=blocking (conflicts prevent other events)
!! event_vars['priority']    - 0=low (hidden by any higher), 1=normal (hidden by 2), 2=high (never hidden)
!! $event_vars['recur']      - Repeat pattern: 'daily', 'weekly', 'biweekly', 'monthly', 'yearly',
!! 							'monthly-N-W' (Nth weekday W each month, e.g. 'monthly-3-6' = 3rd Saturday),
!! 							'yearly-mmdd-mmdd' (yearly date range, e.g. 'yearly-1231-0114' for Dec 31-Jan 14),
!! 							'1 3 5' (space-delimited weekdays), or '' (no recurrence)
!! event_vars['recur_end']   - daystart when repeating ends (0=never)
!! event_vars['holiday']     - 0=ignore holidays, 1=skip holidays, 2=only on holidays
!!
!! Flexible Time Windows:
!! event_vars['flex_type']   - 0=fixed time, 1=flexible start window, 2=flexible both (future)
!! event_vars['duration_ts'] - Event duration in timeslots (1 timeslot = 15 minutes)
!!
!! For flex_type = 0 (fixed time):
!! event_vars['start_ts']    - Start timeslot (0..95)
!!
!! For flex_type = 1 (flexible start):
!! event_vars['window_start_ts'] - Earliest possible arrival timeslot
!! event_vars['window_end_ts']   - Latest possible arrival timeslot (deadline)
!! ============================================================================

!! ============================================================================
!! init_event_vars
!! Initializes all event_vars to their default values
!! Call this before defining each new event to reset to clean state
!! Input: None
!! Output: Sets event_vars to default values matching calendar_events format
!! ============================================================================
if $ARGS[0] = 'init_event_vars':
	killvar 'event_vars'

	!! Core identification
	$event_vars['id'] = ''                    & !! Unique event ID (REQUIRED - must be set)
	$event_vars['title'] = ''                 & !! Display title (REQUIRED - must be set)
	$event_vars['loc'] = ''                   & !! Location name (optional)
	$event_vars['desc'] = ''                  & !! Detailed description (optional)
	event_vars['color'] = 0                   & !! Color index 1-20, or 0 for random (default: 0 = random)
	event_vars['priority'] = 1                & !! Priority 0-2: 0=low, 1=normal, 2=high (default: 1)

	!! Event type and timing
	event_vars['daystart'] = daystart         & !! Start day (default: today)
	event_vars['all_day'] = 0                 & !! 0=timed event, 1=all-day event (default: 0)
	event_vars['blocking'] = 0                & !! 0=non-blocking, 1=blocks other events (default: 0)
	event_vars['holiday'] = 0                 & !! 0=ignore, 1=skip holidays, 2=only on holidays (default: 0)

	!! Recurrence pattern
	$event_vars['recur'] = ''                 & !! daily, weekly, biweekly, monthly, yearly, monthly-N-W (e.g. monthly-3-6 = 3rd Saturday), yearly-mmdd-mmdd (date range), or space-delimited weekdays 1 3 5 (default: no recurrence)
gs 'time', 'to_daystart', 2024, 12, 31
	event_vars['recur_end'] = dateVars['daystart'] & !! daystart when repeating ends, 0=never (default: Dec 31, 2024)

	!! Time type: flex_type 0=fixed time, 1=flexible arrival window
	event_vars['flex_type'] = 0               & !! Default: fixed time event

	!! Event timing (applies to both fixed and flexible events)
	event_vars['duration_ts'] = 4             & !! Event duration in timeslots (default: 4 = 1 hour)

	!! Fixed time event fields (flex_type = 0)
	event_vars['start_ts'] = 0               & !! Start timeslot 0-95 (default: 0 = no start)

	!! Flexible time event fields (flex_type = 1)
	event_vars['window_start_ts'] = 0        & !! Earliest arrival timeslot (default: 0 = no window)
	event_vars['window_end_ts'] = 0          & !! Latest arrival timeslot (default: 0 = no window)

	exit
end


!! ============================================================================
!! assign_color
!! If event_vars['color'] is 0, assigns a random color (1-20)
!! Otherwise leaves the color unchanged
!! Call this before adding an event to ensure it has a valid color
!! Input: None (reads event_vars['color'])
!! Output: Sets event_vars['color'] to 1-20
!! ============================================================================
if $ARGS[0] = 'assign_color':
	if event_vars['color'] = 0:
		!! Pick random color from 1-20
		event_vars['color'] = rand(1, 20)
	end
	exit
end
