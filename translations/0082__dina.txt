if $ARGS[0] = 'hypnoDaychange':
	!! if for some reason hypnoTime is too high
	if hypnoTime > 15: hypnoTime = 15
	timeTresh = 76 - 4*hypnoTime
	!! Withdrawal effect begins if too strong influence and no visitation
	if daystart-hypnoDay > 7:
		if trait_vars['cum_addict'] = 1:
			hypnoWithdrawal = 1
		elseif trait_vars['cum_addict'] = 2:
			hypnoWithdrawal = 2
		end
	end
	!! daily weaning of therapist influence
	if hypnoStrength > 0: hypnoStrength -= 1
	if hypnoTime > 0 and hypnoStrength = 0:
		if trait_vars['cum_addict'] = 0:
			hypnoStrength = 19
			hypnoTime -= 1
		elseif trait_vars['cum_addict'] = 1:
			if rand(0, 1) = 0:
				hypnoStrength = 19
				hypnoTime -= 1
				if hypnoTime < 7: gs 'traits', 'level', 'cum_addict', 0 & hypnoWithdrawal = 0
			end
		else
			if rand(0, 4) = 0:
				hypnoStrength = 19
				hypnoTime -= 1
				if hypnoTime < 10: gs 'traits', 'level', 'cum_addict', 1 & hypnoWithdrawal = 1
			end
		end
	end
end


if $ARGS[0] = 'hypnoAddiction':
	!! Cum craving effects are now handled by traits 'cum_addict' hourly handler.
	!! This function only handles hypnoWithdrawal (therapist session dependency).
	if hypnoWithdrawal > 0:
		if hypnoWithdrawal = 1:
			if missCum >= timeTresh:
				gs 'exp_deg', 'sprt', 2 * trait_vars['cum_addict']
				gs 'mood', 'lower', 'small'
			else
				if rand(0, 1) = 0: gs 'exp_deg', 'sprt', trait_vars['cum_addict']
				gs 'mood', 'lower', 'tiny'
			end
		end
	end
end

					cumtime[i] += 1
				end

			end
		end
	end
