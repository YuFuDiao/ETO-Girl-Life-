if totminut - locat['A84_rand_time'] > 60:
	locat['A84_rand_time'] = totminut
	locat['A84_rand'] = rand(0, 9)

if hour < 9:
	$locat['A84_loc'] = 'uni_dorm'
	$locat['A84_arg'] = 'asleep'
elseif week < 4 and hour < 16:
	$locat['A84_loc'] = 'uni_lessons'
elseif week < 5 and hour >= 17 and hour < 23 and locat['A84_rand'] < 5:
	$locat['A84_loc'] = 'uni_dorm'
	$locat['A84_arg'] = 'eighth_floor'
elseif week >= 5 and week < 7 and hour < 20 and locat['A84_rand'] < 5:
	$locat['A84_loc'] = 'uni_dorm'
	$locat['A84_arg'] = 'eighth_floor'
elseif week = 7 and hour < 23 and locat['A84_rand'] < 8:
	$locat['A84_loc'] = 'uni_dorm'
	$locat['A84_arg'] = 'eighth_floor'
else
	$locat['A84_loc'] = 'unknown'
	$locat['A84_arg'] = 'unknown'
end
