		addobj '<center>-1 Hour</center>'
		addobj '<center>+1 Hour</center>'
	end

		addobj '<center>移除调试信息</center>'

if $ARGS[0] = "AddDebugVar":
	if $ARGS[1] <> '':
		i = arrsize('$ObjDebugVars')
		$ObjDebugVars[i] = '<<$ARGS[1]>>'
	end
end


if $ARGS[0] = "DeleteDebugVar":
	if $ARGS[1] <> '':
		if arrsize('$ObjDebugVars') > 0:
			i = 0
			:DeleteDebugVarLoop
			if i < arrsize('$ObjDebugVars'):
				if $ObjDebugVars[i] = $ARGS[1]:
					killvar 'ObjDebugVars', i
					i = arrsize('$ObjDebugVars')
				else
					i += 1
				end
				jump 'DeleteDebugVarLoop'
			end
		end
	end
end


if $ARGS[0] = 'wait':
	cla
	$waitStr = input ("你想等多久？<br>（最长120分钟，留空或按取消则等待15分钟）<br>或者，输入要等待到的时间（hh:mm）。现在是<<$mid(100+hour,2,2)>>:<<$mid(100+minut,2,2)>>。")

	waiting = 0
	if $waitStr = '':
		waiting = 15
	elseif isnum($waitStr) = 0:
		waiting = val(mid($waitStr, 1, 2)) * 60 + val(mid($waitStr, 4, 2)) - minut - hour * 60
	else
		waiting = val($waitStr)
	end
	if waiting > 0 and waiting <= 120:
		minut += waiting
	else
		msg '无效时间。'
	end
	gs 'stat'
	killvar 'waiting'
	killvar 'waitStr'
	gt $loc, $loc_arg
end


if $ARGS[0] = 'rest':
	cla
	waiting = val(input("你想休息多久？<br>（默认15分钟，最长120分钟）"))

	if waiting = 0:
		minut += 15
	elseif waiting > 0 and waiting <= 120:
		minut += waiting
		gs 'mood', 'raise', 'tiny'
		if pcs_sleep < 90 and waiting >= 60:
			pcs_sleep += 10
		end
	else
		msg '无效时间。'
	end
	gs 'stat'
	killvar 'waiting'
	gt $loc, $loc_arg
end
