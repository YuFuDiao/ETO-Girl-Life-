# $pcs_outfit_image

!! get the image for clothing
!! ARGS 0 - clothing group name
!! ARGS 1 - clothing index

$result = 'images/pc/items/'

if $ARGS[0] = 'misc_outfits':
	$result += 'misc/'
elseif $ARGS[0] = 'allure_swimsuit':
	$result += 'allure/one/'
elseif $ARGS[0] = 'allure_bikinis':
	$result += 'allure/two/'
elseif $ARGS[0] = 'scandalicious_swimsuit':
	$result += 'scandalicious/swim/one/'
elseif $ARGS[0] = 'scandalicious_bikinis':
	$result += 'scandalicious/swim/two/'
elseif $ARGS[0] = 'danilovich_swimsuit':
	$result += 'danilovich/swimwear/'
elseif $ARGS[0] = 'market_outfits':
	$result += 'market/clothes/'
elseif $ARGS[0] = 'materinstvo_dress':
	$result += 'mommy/'
elseif $ARGS[0] = 'nerdvana_bikinis':
	$result += 'nerdvana/swim/two/'
elseif $ARGS[0] = 'nerdvana_swimsuit':
	$result += 'nerdvana/swim/one/'
else
	temp_pcs_outfit_image_str_index = instr($ARGS[0], '_')
	$temp_pcs_outfit_image_shop_name = $mid($ARGS[0], 1, temp_pcs_outfit_image_str_index - 1)
	$temp_pcs_outfit_image_class_name = $mid($ARGS[0], temp_pcs_outfit_image_str_index + 1)

	$result += $temp_pcs_outfit_image_shop_name + '/' + $temp_pcs_outfit_image_class_name + '/'

	killvar 'temp_pcs_outfit_image_str_index'
	killvar '$temp_pcs_outfit_image_shop_name'
	killvar '$temp_pcs_outfit_image_class_name'
end

$result += '<<ARGS[1]>>.jpg'
