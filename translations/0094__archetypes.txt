# archetypes

!{
	Main archetypes, mutually exclusive, arranged in a pentagram of opposites:

		Bimbo (Opposes Prude, Punk)
		Preppy (Opposes Punk, Goth)
		Prude (Opposes Goth, Bimbo)
		Punk (Opposes Bimbo, Preppy)
		Goth (Opposes Preppy, Prude)

		Each archetype effect has an opposite effect in at least one of the opposing archetypes.
		Gaining a point in one archetype will cause you to lose a point in the opposing archetypes.
		The chosen archetype is the one with the highest score. A archetype must have a minimum of 50000 points to be active.

	Variables:
		$arch_vars['main_active'] - Name of the currently active main archetype, or '' if none
		arch_vars['X_points'] - Number of points in archetype X
		arch_vars['Y_effect'] - Current effect on Y skill/attribute, derived from active archetype points
		arch_vars['last_update'] - totminut when get_state_points last ran (for throttling)
		arch_vars['X_history_N'] - 7-day history: archetype X's snapshot points, 0=today, 6=oldest
		arch_vars['X_avg'] - 7-day running average of archetype X's points (recomputed daily)
		arch_vars['withdrawal_pct'] - Percentage deficit of active archetype vs its 7-day average (0 = no withdrawal)

	Constants:
		arch_const['point_cap'] - Maximum number of points in a archetype, default 2 mil
		arch_const['point_min'] - Minimum points to activate a archetype, default 50000
		arch_const['points_full_effect'] - Points at which effects reach full effectiveness, default 500000
		arch_const['level_third'] - One third of effect_range, used for _lvl tiers

	Helpers:
		$func('archetypes', 'get_archetype', N) - Returns archetype name for index 0-4
		$func('archetypes', 'get_opposite', 'X', 0/1)  - Returns Nth opposite of archetype X

	Point descriptors (gain/lose):
		'tiny'    =     250  (trivial: washing dishes, small remark, ~15 min passive max gain)
		'small'   =   1,000  (minor: brief interaction, small choice, ~1 hr passive max gain)
		'medium'  =   5,000  (moderate: significant scene, pill/potion, ~6 hr passive max gain)
		'large'   =  15,000  (major: transformative multi-hour event, ~1 day passive max gain)
		'huge'    =  50,000  (extreme: life-altering, crosses activation, ~3 days passive max gain)

	- Eva
!}

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Format Points													 │
!! │  Converts a raw point delta to a human-readable display-point   │
!! │  string on the same scale used by the pentagon display (÷250).  │
!! │  ARGS[1]   Raw delta (signed integer)						     │
!! │  $ARGS[2]  Mode: 'fine' (1 decimal) or 'coarse' (whole number)  │
!! │  Returns:  '+23.4' / '-<0.1' (fine) or '+23' / '-<1' (coarse)   │
!! │  Use: $func('archetypes', 'fmt_pts', delta, $mode)			     │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'fmt_pts':
	if ARGS[1] = 0:
		$result = '0'
		exit
	end
	temp_fmt['abs'] = iif(ARGS[1] < 0, -ARGS[1], ARGS[1])
	temp_fmt['whole'] = temp_fmt['abs'] / 250
	$temp_fmt['sign'] = iif(ARGS[1] > 0, '+', '-')
	if $ARGS[2] = 'fine':
		temp_fmt['tenths'] = (temp_fmt['abs'] mod 250) * 10 / 250
		if temp_fmt['whole'] = 0 and temp_fmt['tenths'] = 0:
			$result = $temp_fmt['sign'] + '&lt;0.1'
		else
			$result = $temp_fmt['sign'] + $str(temp_fmt['whole']) + '.' + $str(temp_fmt['tenths'])
		end
	else
		if temp_fmt['whole'] = 0:
			$result = $temp_fmt['sign'] + '&lt;1'
		else
			$result = $temp_fmt['sign'] + $str(temp_fmt['whole'])
		end
	end
	killvar 'temp_fmt'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Clamp Archetype												 │
!! │	Clamps a archetype''s points between 0 and the point cap.	 │
!! │	$ARGS[1]	Archetype name									 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'clamp':
	if arch_const['point_cap'] > 0:
		arch_vars['<<$ARGS[1]>>_points'] = func('math', 'int_clamp', arch_vars['<<$ARGS[1]>>_points'], 0, arch_const['point_cap'])
	else
		arch_vars['<<$ARGS[1]>>_points'] = max(0, arch_vars['<<$ARGS[1]>>_points'])
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Get Archetype Name by Index									 │
!! │	Returns the archetype name for index 0-4.					 │
!! │	ARGS[1]	 Index (0=bimbo, 1=preppy, 2=prude, 3=punk, 4=goth)	 │
!! │	Use: $func(''archetypes'', ''get_archetype'', N)			 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'get_archetype':
	if ARGS[1] = 0:
		$result = 'bimbo'
	elseif ARGS[1] = 1:
		$result = 'preppy'
	elseif ARGS[1] = 2:
		$result = 'prude'
	elseif ARGS[1] = 3:
		$result = 'punk'
	elseif ARGS[1] = 4:
		$result = 'goth'
	else
		$result = ''
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Get Opposite Archetype Name									 │
!! │	Returns the Nth opposite of the given archetype.			 │
!! │	Pentagram: bimbo↔(prude,punk), preppy↔(punk,goth),			 │
!! │	  prude↔(goth,bimbo),punk↔(bimbo,preppy),goth↔(preppy,prude) │
!! │	$ARGS[1]	Archetype name									 │
!! │	 ARGS[2]	0 or 1 (which opposite)							 │
!! │	Use: $func(''archetypes'', ''get_opposite'', ''bimbo'', 0)	 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'get_opposite':
	if $ARGS[1] = 'bimbo':
		$result = $iif(ARGS[2] = 0, 'prude', 'punk')
	elseif $ARGS[1] = 'preppy':
		$result = $iif(ARGS[2] = 0, 'punk', 'goth')
	elseif $ARGS[1] = 'prude':
		$result = $iif(ARGS[2] = 0, 'goth', 'bimbo')
	elseif $ARGS[1] = 'punk':
		$result = $iif(ARGS[2] = 0, 'bimbo', 'preppy')
	elseif $ARGS[1] = 'goth':
		$result = $iif(ARGS[2] = 0, 'preppy', 'prude')
	else
		$result = ''
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Get Archetype Percentage										 │
!! │	Returns the percentage of an archetype compared to the max   │
!! │	$ARGS[1]	Archetype name									 │
!! │	ARGS[2]		1 to use effective range						 │
!! │	Use: func(''archetypes'', ''get_percentage'', ''bimbo'')	 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'get_percentage':
	if arch_const['point_cap'] > 0:
		if ARGS[2] = 1:
			temp_gap_gp = max(0, arch_vars['<<$ARGS[1]>>_points'] - arch_const['point_min'])
			result = min(100, temp_gap_gp * 100 / (arch_const['point_cap'] - arch_const['point_min']))
			killvar 'temp_gap_gp'
		else
			result = min(100, arch_vars['<<$ARGS[1]>>_points'] * 100 / arch_const['point_cap'])
		end
	else
		result = 0
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Get Archetype Level (0-3)										 │
!! │	Returns the legacy tier for a archetype   					 │
!! │	  0 = inactive (below point_min)							 │
!! │	  1 = low (min to min + 1×level_third)						 │
!! │	  2 = mid (min + 1× to min + 2×level_third)					 │
!! │	  3 = high (min + 2×level_third and above)					 │
!! │	$ARGS[1]  Archetype name									 │
!! │	Use: func(''archetypes'', ''get_lvl'', ''bimbo'')			 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'get_lvl':
	temp_gl_pts = arch_vars['<<$ARGS[1]>>_points']
	if temp_gl_pts < arch_const['point_min']:
		result = 0
	else
		temp_gl_gap = temp_gl_pts - arch_const['point_min']
		if temp_gl_gap < arch_const['level_third']:
			result = 1
		elseif temp_gl_gap < (2 * arch_const['level_third']):
			result = 2
		else
			result = 3
		end
	end
	killvar 'temp_gl_pts'
	killvar 'temp_gl_gap'
	exit
end

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Point Gain														 │
!! │	Awards points to a single archetype and deducts from its two	 │
!! │	opposites.													 │
!! │	$ARGS[1]	Archetype name									 │
!! │	$ARGS[2]	Points: descriptor or integer					 │
!! │				Descriptors: tiny (250), small (1k),			 │
!! │				medium (5k), large (15k), huge (50k)			 │
!! │	$ARGS[3]	Story label for the event log (optional)		 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'gain':
	if cheatVars['archetypes'] = 1 or succubusflag = 1: exit
	!! Optional per-label cooldown: ARGS[4] = minimum days between gains (0 = no limit).
	!! Uses arch_vars['cooldown_<label>'] to store the last daystart this label fired.
	if ARGS[4] > 0 and $ARGS[3] ! '':
		if arch_vars['cooldown_<<$ARGS[3]>>'] > 0 and daystart - arch_vars['cooldown_<<$ARGS[3]>>'] < ARGS[4]:
			exit
		end
		arch_vars['cooldown_<<$ARGS[3]>>'] = daystart
	end
	if $ARGS[2] = 'tiny':
		temp_points = 250
	elseif $ARGS[2] = 'small':
		temp_points = 1000
	elseif $ARGS[2] = 'medium':
		temp_points = 5000
	elseif $ARGS[2] = 'large':
		temp_points = 15000
	elseif $ARGS[2] = 'huge':
		temp_points = 50000
	else
		temp_points = iif(ARGS[2] > 0, ARGS[2], 1)
	end
	arch_vars['<<$ARGS[1]>>_points'] += temp_points
	gs 'archetypes', 'clamp', $ARGS[1]
	$temp_opp0 = $func('archetypes', 'get_opposite', $ARGS[1], 0)
	$temp_opp1 = $func('archetypes', 'get_opposite', $ARGS[1], 1)
	arch_vars['<<$temp_opp0>>_points'] -= temp_points
	gs 'archetypes', 'clamp', $temp_opp0
	arch_vars['<<$temp_opp1>>_points'] -= temp_points
	gs 'archetypes', 'clamp', $temp_opp1
	!! ── Archetype gain notifications ──────────────────────────────────────
	if stat_cfg['notify_archetypes'] <> 1:
		gs 'archetypes', 'track_archetype', $ARGS[1], temp_points
		if stat_cfg['notify_archetypes_opposite'] <> 1:
			gs 'archetypes', 'track_archetype', $temp_opp0, -temp_points
			gs 'archetypes', 'track_archetype', $temp_opp1, -temp_points
		end
	end
	!! ── Event log ─────────────────────────────────────────────────────
	gs 'archetypes', 'log_event', $ARGS[1], 'story', $ARGS[3], temp_points
	gs 'archetypes', 'log_event', $temp_opp0, 'opposition', '', -temp_points
	gs 'archetypes', 'log_event', $temp_opp1, 'opposition', '', -temp_points
	killvar 'temp_points'
	killvar 'temp_opp0'
	killvar 'temp_opp1'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Point Loss														 │
!! │	Deducts points from a single archetype only; does not affect │
!! │	opposites.													 │
!! │	$ARGS[1]	Archetype name									 │
!! │	$ARGS[2]	Points: descriptor or integer					 │
!! │				Descriptors: tiny (250), small (1k),			 │
!! │				medium (5k), large (15k), huge (50k)			 │
!! │	$ARGS[3]	Story label for the event log (optional)		 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'lose':
	if cheatVars['archetypes'] = 1 or succubusflag = 1: exit
	if $ARGS[2] = 'tiny':
		temp_points = 250
	elseif $ARGS[2] = 'small':
		temp_points = 1000
	elseif $ARGS[2] = 'medium':
		temp_points = 5000
	elseif $ARGS[2] = 'large':
		temp_points = 15000
	elseif $ARGS[2] = 'huge':
		temp_points = 50000
	else
		temp_points = iif(ARGS[2] > 0, ARGS[2], 1)
	end
	arch_vars['<<$ARGS[1]>>_points'] -= temp_points
	gs 'archetypes', 'clamp', $ARGS[1]
	!! ── Archetype lose notification ───────────────────────────────────────
	if stat_cfg['notify_archetypes'] <> 1:
		gs 'archetypes', 'track_archetype', $ARGS[1], -temp_points
	end
	!! ── Event log ─────────────────────────────────────────────────────
	gs 'archetypes', 'log_event', $ARGS[1], 'story', $ARGS[3], -temp_points
	killvar 'temp_points'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Change (State Loop)											 │
!! │	Applies gains, losses, and proportional opposition for the	 │
!! │	state point loop. Unlike gain(), handles all 5 archetypes	 │
!! │	simultaneously with proportional shielding.					 │
!! │	 ARGS[1..5]	 Points: bimbo, preppy, prude, punk, goth		 │
!! │				 Positive = gain, negative = loss, 0 = skip.	 │
!! │	 ARGS[6]	 Multiplier (optional, default 1)				 │
!! │	 $ARGS[7]	 Category key for log_event (e.g. 'clothing')	 │
!! │				 Defaults to 'unknown' if omitted.				 │
!! │	 $ARGS[8]	 Optional label for log_event (e.g. for 'story'	 │
!! │				 category events with a distinct per-event name) │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'change':
	if cheatVars['archetypes'] = 1 or succubusflag = 1: exit
	killvar 'temp_change'
	temp_change['mul'] = iif(ARGS[6] > 0, ARGS[6], 1)
	temp_change['bimbo'] = ARGS[1] * temp_change['mul']
	temp_change['preppy'] = ARGS[2] * temp_change['mul']
	temp_change['prude'] = ARGS[3] * temp_change['mul']
	temp_change['punk'] = ARGS[4] * temp_change['mul']
	temp_change['goth'] = ARGS[5] * temp_change['mul']

	if temp_change['bimbo'] > 0:
		temp_change['opp_prude'] += max(0, temp_change['bimbo'] - max(0, temp_change['prude']))
		temp_change['opp_punk'] += max(0, temp_change['bimbo'] - max(0, temp_change['punk']))
	end

	if temp_change['preppy'] > 0:
		temp_change['opp_punk'] += max(0, temp_change['preppy'] - max(0, temp_change['punk']))
		temp_change['opp_goth'] += max(0, temp_change['preppy'] - max(0, temp_change['goth']))
	end

	if temp_change['prude'] > 0:
		temp_change['opp_bimbo'] += max(0, temp_change['prude'] - max(0, temp_change['bimbo']))
		temp_change['opp_goth'] += max(0, temp_change['prude'] - max(0, temp_change['goth']))
	end

	if temp_change['punk'] > 0:
		temp_change['opp_bimbo'] += max(0, temp_change['punk'] - max(0, temp_change['bimbo']))
		temp_change['opp_preppy'] += max(0, temp_change['punk'] - max(0, temp_change['preppy']))
	end

	if temp_change['goth'] > 0:
		temp_change['opp_preppy'] += max(0, temp_change['goth'] - max(0, temp_change['preppy']))
		temp_change['opp_prude'] += max(0, temp_change['goth'] - max(0, temp_change['prude']))
	end

	!! ── Log raw category deltas and opposition deductions ─────────────
	!! Log before subtracting opposition, so each is recorded separately.
	if stat_cfg['arch_log_enabled'] = 1:
		$temp_change['cat'] = $iif($ARGS[7] <> '', $ARGS[7], 'unknown')
		if temp_change['bimbo'] <> 0:			gs 'archetypes', 'log_event', 'bimbo',  $temp_change['cat'], $ARGS[8], temp_change['bimbo']
		if temp_change['preppy'] <> 0:			gs 'archetypes', 'log_event', 'preppy', $temp_change['cat'], $ARGS[8], temp_change['preppy']
		if temp_change['prude'] <> 0:			gs 'archetypes', 'log_event', 'prude',  $temp_change['cat'], $ARGS[8], temp_change['prude']
		if temp_change['punk'] <> 0:			gs 'archetypes', 'log_event', 'punk',   $temp_change['cat'], $ARGS[8], temp_change['punk']
		if temp_change['goth'] <> 0:			gs 'archetypes', 'log_event', 'goth',   $temp_change['cat'], $ARGS[8], temp_change['goth']
		if temp_change['opp_bimbo'] <> 0:		gs 'archetypes', 'log_event', 'bimbo',  'opposition', '', -temp_change['opp_bimbo']
		if temp_change['opp_preppy'] <> 0:		gs 'archetypes', 'log_event', 'preppy', 'opposition', '', -temp_change['opp_preppy']
		if temp_change['opp_prude'] <> 0:		gs 'archetypes', 'log_event', 'prude',  'opposition', '', -temp_change['opp_prude']
		if temp_change['opp_punk'] <> 0:		gs 'archetypes', 'log_event', 'punk',   'opposition', '', -temp_change['opp_punk']
		if temp_change['opp_goth'] <> 0:		gs 'archetypes', 'log_event', 'goth',   'opposition', '', -temp_change['opp_goth']
	end

	temp_change['bimbo'] -= temp_change['opp_bimbo']
	temp_change['preppy'] -= temp_change['opp_preppy']
	temp_change['prude'] -= temp_change['opp_prude']
	temp_change['punk'] -= temp_change['opp_punk']
	temp_change['goth'] -= temp_change['opp_goth']

	if temp_change['bimbo'] <> 0:
		arch_vars['bimbo_points'] += temp_change['bimbo']
		gs 'archetypes', 'clamp', 'bimbo'
	end
	if temp_change['preppy'] <> 0:
		arch_vars['preppy_points'] += temp_change['preppy']
		gs 'archetypes', 'clamp', 'preppy'
	end
	if temp_change['prude'] <> 0:
		arch_vars['prude_points'] += temp_change['prude']
		gs 'archetypes', 'clamp', 'prude'
	end
	if temp_change['punk'] <> 0:
		arch_vars['punk_points'] += temp_change['punk']
		gs 'archetypes', 'clamp', 'punk'
	end
	if temp_change['goth'] <> 0:
		arch_vars['goth_points'] += temp_change['goth']
		gs 'archetypes', 'clamp', 'goth'
	end

	!! ── State notification (opt-in) ───────────────────────────────────
	if stat_cfg['notify_archetypes'] <> 1 and stat_cfg['notify_archetypes_state'] = 1:
		if temp_change['bimbo']  <> 0: gs 'archetypes', 'track_archetype', 'bimbo',  temp_change['bimbo']
		if temp_change['preppy'] <> 0: gs 'archetypes', 'track_archetype', 'preppy', temp_change['preppy']
		if temp_change['prude']  <> 0: gs 'archetypes', 'track_archetype', 'prude',  temp_change['prude']
		if temp_change['punk']   <> 0: gs 'archetypes', 'track_archetype', 'punk',   temp_change['punk']
		if temp_change['goth']   <> 0: gs 'archetypes', 'track_archetype', 'goth',   temp_change['goth']
	end

	killvar 'temp_change'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ faith_gain - award pcs_faith restoration points from church     │
!! │ activity, once per method per day regardless of church location.│
!! │ ARGS[1] = method key ('pray'/'confession'/'vigil'/'liturgy')    │
!! │ ARGS[2] = point value                                           │
!! │ No-op for non-prude players, since pcs_faith only applies to    │
!! │ the prude archetype.                                            │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'faith_gain':
	if $arch_vars['main_active'] <> 'prude': exit
	if faithVars[$ARGS[1] + '_day'] = daystart: exit
	faithVars[$ARGS[1] + '_day'] = daystart
	pcs_faith = min(100, pcs_faith + ARGS[2])
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Sex Change														 │
!! │	Awards Bimbo/Prude for a sex act, capping cumulative daily	 │
!! │	Bimbo gain from this source (Prude loss is uncapped).		 │
!! │	 ARGS[1]	Bimbo gain (uncapped input)						 │
!! │	 ARGS[2]	Prude loss (negative)							 │
!! │	 $ARGS[3]	Label for log_event								 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'sex_change':
	if cheatVars['archetypes'] = 1 or succubusflag = 1: exit
	if arch_vars['sex_bimbo_day'] <> daystart:
		arch_vars['sex_bimbo_day'] = daystart
		arch_vars['sex_bimbo_today'] = 0
	end
	temp_sex_bimbo = max(0, min(ARGS[1], arch_const['sex_bimbo_daily_cap'] - arch_vars['sex_bimbo_today']))
	arch_vars['sex_bimbo_today'] += temp_sex_bimbo
	gs 'archetypes', 'change', temp_sex_bimbo, 0, ARGS[2], 0, 0, 1, 'story', $ARGS[3]
	killvar 'temp_sex_bimbo'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Set Active														 │
!! │	Determines and sets the active main archetype, if any.		 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'set_active':
	$temp_sa_active = ''
	temp_sa_index = 0
	:active_archetype_loop
		$temp_archetype = $func('archetypes', 'get_archetype', temp_sa_index)
		if arch_vars[$temp_archetype + '_points'] >= arch_const['point_min']:
			if arch_vars[$temp_archetype + '_points'] > arch_vars[$temp_sa_active + '_points']:
				$temp_sa_active = $temp_archetype
			end
		end
		temp_sa_index += 1
		if temp_sa_index < 5: jump 'active_archetype_loop'
	if $temp_sa_active = 'prude' and $arch_vars['main_active'] <> 'prude' and calendar_show['church'] = 0:
		calendar_show['church'] = 1
		gs 'calendar', 'pack', 'add', 'church'
	end
	$arch_vars['main_active'] = $temp_sa_active
	killvar 'temp_archetype'
	killvar 'temp_sa_active'
	killvar 'temp_sa_index'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Get state points												 │
!! │	Awards and detracts points based on the current player state.│
!! │	 ARGS[1]	Elapsed-minute multiplier, forwarded from loop.  │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'get_state_points':
	temp_gs['mul'] = ARGS[1]

	!! ── Precompute guards ────────────────────────────────────────────
	temp_gs['is_student'] = $start_type['loc'] = 'sg' and gschoolVars['school_diploma'] = 0 and kanikuli = 0
	temp_gs['school_hrs'] = temp_gs['is_student'] and week < 6 and hour >= 8 and hour < 15 and gschoolVars['last_attendance'] = daystart

	!! ── Precompute piercings/tattoos ─────────────────────────────────
	!! Piercing values > 0 = currently wearing; tattoo values > 0 = has tattoo
	temp_gs['pierce_private']	= pcs_piercings['nipples'] > 0 or pcs_piercings['pussy'] > 0

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Loop															 │
!! │	Called from stat every minute (self-throttled). Applies		 │
!! │	state-based point changes, updates active archetype,		 │
!! │	and applies archetype effects.								 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'loop':
	!! Guard against uninitialised constants
	if arch_const['effect_range'] <= 0: gs 'archetypes', 'init'

	
	!! ── Stale-clock gate ─────────────────────────────────────────────
	!! totminut is computed by gs 'time' which runs late in stat.qsrc.
	!! On the very first gs 'stat' call it may still be stale (-1 from
	!! start.qsrc).  Sync to one minute earlier so the next loop runs
	if totminut <= 0:
		arch_vars['last_update'] = (minut + hour * 60 + daystart * 1440) - 1
		exit
	end
	
	!! ── Cheat gate ───────────────────────────────────────────────────
	!! cheatVars['archetypes'] freezes the whole system: points stop
	!! moving and active effects are cleared once, then the clock is
	!! kept in sync so re-enabling doesn''t replay the frozen interval.
	!! Succubi are excluded from the archetype system entirely.
	if cheatVars['archetypes'] = 1 or succubusflag = 1:
		$arch_vars['main_active'] = ''
		killvar 'arch_effects'
		arch_vars['last_update'] = totminut
		exit
	end

	!! ── Sleep gate ───────────────────────────────────────────────────
	!! During sleep totminut advances but nothing should change.
	!! Sync the clock so the first waking tick starts fresh.
	if inSleep <> 0:
		arch_vars['last_update'] = totminut
		exit
	end

	!! ── Timing gate ──────────────────────────────────────────────────
	temp_loop_mul = totminut - arch_vars['last_update']
	if temp_loop_mul = 0: killvar 'temp_loop_mul' & exit
	arch_vars['last_update'] = totminut

	gs 'archetypes', 'get_state_points', temp_loop_mul
	gs 'archetypes', 'decay', temp_loop_mul
	gs 'archetypes', 'set_active'
	gs 'archetypes', 'compute_withdrawal'
	gs 'archetypes', 'apply_effects', temp_loop_mul
	killvar 'temp_loop_mul'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Decay															 │
!! │	Reduces all archetype points per elapsed minute, at a rate	 │
!! │	that scales with each archetype own current point total (see │
!! │	get_decay_rate below). Ensures archetypes fade without		 │
!! │	reinforcement, faster the higher they have climbed.			 │
!! │	Logs the *actual* delta (post-clamp) per archetype under	 │
!! │	cat=''decay'' so floored archetypes don''t over-report.		 │
!! │	 ARGS[1]	Elapsed minutes									 │
!! └─────────────────────────────────────────────────────────────────┘
!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Get Decay Rate													 │
!! │	Smooth (S-curve) decay/min for a given point total:			 │
!! │	 rate = decay_cap * x^2 / (x^2 + decay_half^2), x = points/1000│
!! │	Near 0 for low totals, rises through decay_half, and eases	 │
!! │	toward decay_cap for very high totals                        │
!! │	See decay_half/decay_cap in init() for calibration.	         │
!! │	 ARGS[1]	Current points									 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'get_decay_rate':
	temp_gdr_x = ARGS[1] / 1000
	temp_gdr_h = arch_const['decay_half'] / 1000
	result = arch_const['decay_cap'] * temp_gdr_x * temp_gdr_x / (temp_gdr_x * temp_gdr_x + temp_gdr_h * temp_gdr_h)
	killvar 'temp_gdr_x'
	killvar 'temp_gdr_h'
	exit
end


if $ARGS[0] = 'decay':
	!! Log only the *actual* reduction per archetype (clamped at 0) so the
	!! history page and popup don''t over-report decay on already-floored archetypes.
	!! Decay rate is piecewise-linear in the current point total (see
	!! get_decay_rate) so high totals fall faster, capping runaway passive
	!! growth, while staying at 0 below point_min so early growth isn''t punished.
	temp_decay_i = 0
	:decay_loop
		$temp_decay_archetype = $func('archetypes', 'get_archetype', temp_decay_i)
		temp_decay_before = arch_vars['<<$temp_decay_archetype>>_points']
		temp_decay_rate = func('archetypes', 'get_decay_rate', temp_decay_before)
		temp_decay_after = max(0, temp_decay_before - temp_decay_rate * ARGS[1])
		arch_vars['<<$temp_decay_archetype>>_points'] = temp_decay_after
		temp_decay_delta = temp_decay_after - temp_decay_before
		if temp_decay_delta < 0:
			gs 'archetypes', 'log_event', $temp_decay_archetype, 'decay', '', temp_decay_delta
		end
		temp_decay_i += 1
	if temp_decay_i < 5: jump 'decay_loop'
	killvar 'temp_decay_i'
	killvar 'temp_decay_before'
	killvar 'temp_decay_rate'
	killvar 'temp_decay_after'
	killvar 'temp_decay_delta'
	killvar 'temp_decay_archetype'
	exit
end

!! ┌─────────────────────────────────────────────────────────────────┐

	arch_temp['denom'] = 120 - (90 * arch_temp['gap_points']) / arch_const['effect_range']
	arch_temp['eff_mul'] = ARGS[1]
	!! numer: elapsed-minutes-scaled mood-chance numerator (preppy/goth), out of a 1500 (or 6000) denominator.
	arch_temp['numer'] = (arch_temp['gap_points'] * 100 / arch_const['effect_range']) * arch_temp['eff_mul']

		arch_effects['heels_effect']		=	 2 * arch_temp['basic_effect']	& !!  4 -> 20
		arch_effects['service_effect']		=	 1 * arch_temp['basic_effect']	& !!  2 -> 10
		arch_effects['modeling_effect']		=	 1 * arch_temp['basic_effect']	& !!  2 -> 10
		arch_effects['inhib_effect']		=	 1 * arch_temp['basic_effect']	& !!  2 -> 10
		arch_effects['chess_effect']		=	-2 * arch_temp['basic_effect']	& !! -4 -> -20
		arch_effects['intelligence_effect']	=	-2 * arch_temp['basic_effect']	& !! -4 -> -20
		arch_effects['perception_effect']	=	-1 * arch_temp['basic_effect']	& !! -2 -> -10
		arch_effects['spirit_effect']		=	-1 * arch_temp['basic_effect']	& !! -2 -> -10
		arch_effects['computer_effect']		=	-1 * arch_temp['basic_effect']	& !! -2 -> -10
		arch_effects['observation_effect']	=	-1 * arch_temp['basic_effect']	& !! -2 -> -10
		arch_effects['sewing_effect']		=	-1 * arch_temp['basic_effect']	& !! -2 -> -10

		arch_effects['persuasion_effect']		=	 1 * arch_temp['basic_effect']	& !!  2 -> 10

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Daily Snapshot													 │
!! │	Records each archetype''s current points into 7-day history	 │
!! │	array, shifting older entries down by one position.			 │
!! │	Recomputes the 7-day averages.								 │
!! │	Called from cikl (once per day).							 │
!! │															     │
!! │	arch_vars[''X_history_0''] = most recent (today)			 │
!! │	arch_vars[''X_history_6''] = oldest (6 days ago)			 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'daily_snapshot':
	temp_ds_i = 0
	:snapshot_archetype_loop
		$temp_ds_archetype = $func('archetypes', 'get_archetype', temp_ds_i)

		!! Shift history: [5]→[6], [4]→[5], ... [0]→[1]
		temp_ds_j = 6
		:snapshot_shift_loop
			if temp_ds_j > 0:
				arch_vars['<<$temp_ds_archetype>>_history_<<temp_ds_j>>'] = arch_vars[$temp_ds_archetype + '_history_' + (temp_ds_j - 1)]
				temp_ds_j -= 1
				jump 'snapshot_shift_loop'
			end

		!! Write today''s value at position 0
		arch_vars['<<$temp_ds_archetype>>_history_0'] = arch_vars[$temp_ds_archetype + '_points']

		!! Compute average
		temp_ds_sum = 0
		temp_ds_j = 0
		:snapshot_sum_loop
			temp_ds_sum += arch_vars[$temp_ds_archetype + '_history_' + temp_ds_j]
			temp_ds_j += 1
			if temp_ds_j < 7: jump 'snapshot_sum_loop'
		arch_vars['<<$temp_ds_archetype>>_avg'] = temp_ds_sum / 7

		temp_ds_i += 1
		if temp_ds_i < 5: jump 'snapshot_archetype_loop'

	killvar 'temp_ds_i'
	killvar 'temp_ds_j'
	killvar 'temp_ds_sum'
	killvar 'temp_ds_archetype'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Compute Withdrawal												 │
!! │	Computes withdrawal_pct: the percentage deficit of the		 │
!! │	active archetype''s current points vs its 7-day average.	 │
!! │	0 if at or above average, or if no archetype is active.		 │
!! │	Called every minute from loop() and from seed_snapshots().	 │
!! │																 │
!! │	Tiers (for consumers):										 │
!! │	   1-10%  mild    11-20%  moderate							 │
!! │	  21-40%  severe  41%+   critical							 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'compute_withdrawal':
	arch_vars['withdrawal_pct'] = 0
	if cheatVars['no_archetype_withdrawal'] = 0 and $arch_vars['main_active'] <> '':
		temp_cw_avg = arch_vars[$arch_vars['main_active'] + '_avg']
		if temp_cw_avg > 0:
			temp_cw_cur = arch_vars[$arch_vars['main_active'] + '_points']
			if temp_cw_cur < temp_cw_avg:
				arch_vars['withdrawal_pct'] = ((temp_cw_avg - temp_cw_cur) * 100) / temp_cw_avg
			end
		end
	end

	!! ── Mood debuffs from withdrawal ─────────────────────────────────
	if arch_vars['withdrawal_pct'] > 40:
		!! Critical: effectively cap mood at 25
		if pcs_mood > 25: gs 'mood', 'lower', 'tiny'
	elseif arch_vars['withdrawal_pct'] > 20:
		!! Severe: effectively cap mood at 40
		if pcs_mood > 40: gs 'mood', 'lower', 'small'
	end

	killvar 'temp_cw_avg'
	killvar 'temp_cw_cur'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Seed Snapshots													 │
!! │	Fills all 7 slots with each archetype''s current points.     │
!! │	Computes averages and withdrawal.							 │
!! │	Used at game start and save migration to prevent false		 │
!! │	withdrawal from an empty history.							 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'seed_snapshots':
	temp_ss_i = 0
	:seed_archetype_loop
		$temp_ss_archetype = $func('archetypes', 'get_archetype', temp_ss_i)
		temp_ss_val = arch_vars[$temp_ss_archetype + '_points']
		temp_ss_j = 0
		:seed_slot_loop
			arch_vars['<<$temp_ss_archetype>>_history_<<temp_ss_j>>'] = temp_ss_val
			temp_ss_j += 1
			if temp_ss_j < 7: jump 'seed_slot_loop'
		arch_vars['<<$temp_ss_archetype>>_avg'] = temp_ss_val
		temp_ss_i += 1
		if temp_ss_i < 5: jump 'seed_archetype_loop'

	gs 'archetypes', 'compute_withdrawal'

	killvar 'temp_ss_i'
	killvar 'temp_ss_j'
	killvar 'temp_ss_val'
	killvar 'temp_ss_archetype'
	exit
end

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Init															 │
!! │	Initializes archetype constants.							 │
!! │	Should be called in game start and saveupdater only.		 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'init':
	!! Difficulty multiplier: stored in cheatVars['arch_difficulty']
	!! 0 = Normal (1×), -1 = Easy (½×), 1 = Hard (2×), -2 = Very Easy (¼×), 2 = Very Hard (4×)
	!! Multiplier formula: 2^opt (e.g. 2^-1 = 0.5, 2^0 = 1, 2^1 = 2)
	!! We apply it by shifting the base thresholds.
	if cheatVars['arch_difficulty'] = -2:
		temp_diff_num = 1
		temp_diff_den = 4
	elseif cheatVars['arch_difficulty'] = -1:
		temp_diff_num = 1
		temp_diff_den = 2
	elseif cheatVars['arch_difficulty'] = 1:
		temp_diff_num = 2
		temp_diff_den = 1
	elseif cheatVars['arch_difficulty'] = 2:
		temp_diff_num = 4
		temp_diff_den = 1
	else
		temp_diff_num = 1
		temp_diff_den = 1
	end

	arch_const['point_cap'] = 2000000
	arch_const['point_min'] = 50000 * temp_diff_num / temp_diff_den
	arch_const['points_full_effect'] = 500000 * temp_diff_num / temp_diff_den
	arch_const['effect_range'] = arch_const['points_full_effect'] - arch_const['point_min']
	arch_const['level_third'] = arch_const['effect_range'] / 3

	!! ── Decay ──────────────────────────────────────────────────────────
	!! Smooth decay per minute, as a function of the archetype current
	!! points (see func archetypes get_decay_rate points):
	!!   rate = decay_cap * x^2 / (x^2 + decay_half^2), x = points/1000
	arch_const['decay_half'] = arch_const['points_full_effect'] * 4 / 5
	arch_const['decay_cap'] = 29

	!! ── Bimbo arousal constants ──────────────────────────────────────
	!! Proportional-gap decay: delta = gap * rate / divisor per minute.
	!! rate = ln(target) * divisor / minutes, ln(150) ≈ 5010/1000
	!!   rate_full: 2h → 120 min;  rate_min: 8h → 480 min
	arch_const['bimbo_arousal_target'] = 150
	arch_const['bimbo_rate_divisor'] = 6750
	arch_const['bimbo_rate_min'] = (5010 * arch_const['bimbo_rate_divisor']) / 480000
	arch_const['bimbo_rate_range'] = (5010 * arch_const['bimbo_rate_divisor']) / 120000 - arch_const['bimbo_rate_min']

	!! ── Prude arousal constants ──────────────────────────────────────
	!! Same approach, starting arousal 150 (gap=120): ln(120) ≈ 4780/1000
	!!   rate_full: 2h → 120 min;  rate_min: 8h → 480 min
	arch_const['prude_arousal_target'] = 30
	arch_const['prude_rate_divisor'] = 6750
	arch_const['prude_rate_min'] = (4780 * arch_const['prude_rate_divisor']) / 480000
	arch_const['prude_rate_range'] = (4780 * arch_const['prude_rate_divisor']) / 120000 - arch_const['prude_rate_min']

	killvar 'temp_diff_num'
	killvar 'temp_diff_den'

	!! ── Sex-act rebalance ─────────────────────────────────────────────
	arch_const['sex_bimbo_daily_cap'] = 2500
	arch_const['sex_chastity_days'] = 14
	arch_const['sex_chastity_bonus_prude'] = 2700

	!! ── Timing ────────────────────────────────────────────────────────
	!! Compute real totminut from daystart/hour/minut to avoid using the
	!! stale global totminut (-1 from start.qsrc) that hasn''t been
	!! recalculated by gs 'time' yet during intro initialization.
	arch_vars['last_update'] = iif(totminut <= 0, minut + hour * 60 + daystart * 1440, totminut)
	exit
end

		end

				*p '</tr>'
				thp['sj'] += 1

			*p '<tr style="border-top:1px solid <<$thp[''border'']>>;font-weight:bold;">'
			*p '<td>总计</td>'
			*p '<td width="80" align="right"><<iif(thp[''stot60'']   <> 0, $func(''archetypes'', ''fmt_pts'', thp[''stot60''],   ''coarse''), ''—'')>></td>'
			*p '<td width="80" align="right"><<iif(thp[''stot1440''] <> 0, $func(''archetypes'', ''fmt_pts'', thp[''stot1440''], ''coarse''), ''—'')>></td>'
			*p '</tr>'
			*p '</table>'
		end
		*p '</div>'

		!! Clean up per-archetype story accumulators
		killvar 'thp_story_d60'
		killvar 'thp_story_d1440'
		killvar 'thp_story_seen'
		killvar 'thp_story_labels'

		thp['ti'] += 1
	if thp['ti'] < 5: jump 'thp_archetype_loop'
	*p '</div></center>'

	!! ── Cleanup ─────────────────────────────────────────────────────
	killvar 'agg'
	killvar 'thp'
	exit
end

!! ╔═════════════════════════════════════════════════════════════════════╗
!! ║  Archetype Event Log 												 ║
!! ║  Rolling 24-hour log of individual archetype point changes.		 ║
!! ║  Disabled when stat_cfg['arch_log_enabled'] = 0. 					 ║
!! ║																	 ║
!! ║  Storage (parallel arrays, always packed at index 0):				 ║
!! ║	arch_log_minut[i]	- totminut when event was recorded  		 ║
!! ║	$arch_log_archetype[i] - 'bimbo'|'preppy'|'prude'|'punk'|'goth'	 ║
!! ║	$arch_log_cat[i]	- category key (clothing, hair, story, …)	 ║
!! ║	$arch_log_label[i]	- story label (cat='story' only), else ''	 ║
!! ║	arch_log_delta[i]	- raw points  								 ║
!! ║																	 ║
!! ║  No cursor needed: killvar arr, 0 shifts all later indices down.	 ║
!! ║  arrsize('arch_log_minut') always equals the live entry count.		 ║
!! ║																	 ║
!! ║  gs 'archetypes', 'log_event', $archetype, $cat, $label, delta		 ║
!! ║  gs 'archetypes', 'aggregate', window_minutes						 ║
!! ║	→ populates agg['arch:X, cat:Y, w:N'] = total_delta 			 ║
!! ║		agg is NOT persisted; killvar 'agg' after rendering.		 ║
!! ╚═════════════════════════════════════════════════════════════════════╝

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Log Event														 │
!! │  Appends one point-change record to the rolling 24-hour log.	 │
!! │  $ARGS[1]  Archetype name ('bimbo', 'preppy', etc.)   			 │
!! │  $ARGS[2]  Category key ('clothing', 'hair', 'story', etc.)	 │
!! │  $ARGS[3]  Story label — non-empty only when cat = 'story'		 │
!! │  ARGS[4]   Delta (signed integer; 0 is silently ignored)		 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'log_event':
	if stat_cfg['arch_log_enabled'] = 0: exit
	if ARGS[4] = 0: exit

	!! ── Trim entries older than 24 hours from the front ───────────────
	!! killvar arr, 0 removes index 0 and shifts all later indices down.
	:tl_trim_loop
	if arrsize('arch_log_minut') > 0:
		if arch_log_minut[0] < totminut - 1440:
			killvar 'arch_log_minut', 0
			killvar 'arch_log_archetype', 0
			killvar 'arch_log_cat', 0
			killvar 'arch_log_label', 0
			killvar 'arch_log_delta', 0
			jump 'tl_trim_loop'
		end
	end

	!! ── Append new entry at tail ───────────────────────────────────────
	arch_log_minut[]	= totminut
	$arch_log_archetype[]	= $ARGS[1]
	$arch_log_cat[]	= $ARGS[2]
	$arch_log_label[]	= $ARGS[3]
	arch_log_delta[]	= ARGS[4]
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Aggregate														 │
!! │	Walks the live log window and sums deltas into agg[].		 │
!! │	Call once before rendering; killvar 'agg' after.			 │
!! │	ARGS[1]  Window in minutes (e.g. 60 for last hour,			 │
!! │			1440 for last day) 									 │
!! │	Populates: agg['arch:X, cat:Y, w:N'] = total_delta			 │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'aggregate':
	temp_agg['window']	= iif(ARGS[1] > 0, ARGS[1], 1440)
	temp_agg['cutoff']	= totminut - temp_agg['window']
	temp_agg['i']		= 0
	:agg_loop
	if temp_agg['i'] < arrsize('arch_log_minut'):
		if arch_log_minut[temp_agg['i']] >= temp_agg['cutoff']:
			$temp_agg['key'] = 'arch:<<$arch_log_archetype[temp_agg[''i'']]>>, cat:<<$arch_log_cat[temp_agg[''i'']]>>, w:<<temp_agg[''window'']>>'
			agg[$temp_agg['key']] += arch_log_delta[temp_agg['i']]
		end
		temp_agg['i'] += 1
		jump 'agg_loop'
	end
	killvar 'temp_agg'
	exit
end


!! ╔═════════════════════════════════════════════════════════════════════╗
!! ║  Archetype Notification Buffer										 ║
!! ║  Accumulates point changes from 'gain' and 'lose' calls for		 ║
!! ║  display in the stat panel via stat_display_compute.				 ║
!! ║																	 ║
!! ║  Settings (stat_cfg):												 ║
!! ║    notify_archetypes          — 1 = hide (default 0 = show)		 ║
!! ║    notify_archetypes_opposite — 1 = hide opposite deductions		 ║
!! ║    notify_archetypes_state    — 1 = also track state-tick changes	 ║
!! ║                         (tracked independently of notify_archetypes;║
!! ║                          display still gated on notify_archetypes)	 ║

!! ║  Persistent window vars (not persisted across save/load):			 ║

!! ║																	 ║
!! ║  gs 'archetypes', 'track_archetype', $name, delta					 ║
!! ║  $func('archetypes', 'get_archetype_message')  → returns HTML		 ║
!! ║  gs 'archetypes', 'notify_popup', $arch → msg popup (from href)	 ║
!! ║  func('archetypes', 'is_empty_archetypes')   						 ║
!! ╚═════════════════════════════════════════════════════════════════════╝

if $ARGS[0] = 'track_archetype':

		$tnotif_archetype_names[] = $ARGS[1]
	end

end

if $ARGS[0] = 'get_archetype_message':

	if arrsize('$tnotif_archetype_names') = 0: exit

	tnotif_popup_start = tnotif_window_start
	tnotif_popup_end = totminut

	$tnotif_disp['bimbo'] = '花瓶'
	$tnotif_disp['preppy'] = '学院风'
	$tnotif_disp['prude'] = '保守派'
	$tnotif_disp['punk'] = '朋克'
	$tnotif_disp['goth'] = '哥特'

	tnotif_i = 0
	:tnotif_archetype_loop
		$tnotif_t = $tnotif_archetype_names[tnotif_i]
		if tnotif_archetype_delta[$tnotif_t] <> 0:
			!! Arrow tier: ↑(<50) / ↑↑(<250) / ↑↑↑(<1000) / ↑↑↑↑(≥1000)
			tnotif_abs = iif(tnotif_archetype_delta[$tnotif_t] < 0, -tnotif_archetype_delta[$tnotif_t], tnotif_archetype_delta[$tnotif_t])
			$tnotif_dir = iif(tnotif_archetype_delta[$tnotif_t] > 0, '↑', '↓')
			if tnotif_abs <= 50:
				$tnotif_arrow = $tnotif_dir
			elseif tnotif_abs <= 250:
				$tnotif_arrow = $tnotif_dir + $tnotif_dir
			elseif tnotif_abs <= 1000:
				$tnotif_arrow = $tnotif_dir + $tnotif_dir + $tnotif_dir
			else
				$tnotif_arrow = $tnotif_dir + $tnotif_dir + $tnotif_dir + $tnotif_dir
			end
			$result += iif($result <> '', '&ensp;', '') + '<a href="exec:gs ''archetypes'',''notify_popup'',''<<$tnotif_t>>''" style="color:<<$theme_hex[$tnotif_t]>>;text-decoration:underline"><<$tnotif_arrow>> <<$tnotif_disp[$tnotif_t]>></a>'
		end
		tnotif_i += 1
	if tnotif_i < arrsize('$tnotif_archetype_names'): jump 'tnotif_archetype_loop'

	$result = '<span style="font-weight:normal">' + $result + '</span>'

	killvar 'tnotif_i'
	killvar 'tnotif_t'
	killvar 'tnotif_abs'
	killvar 'tnotif_dir'
	killvar 'tnotif_arrow'
	killvar 'tnotif_disp'
	killvar 'tnotif_archetype_delta'
	killvar 'tnotif_archetype_names'
	tnotif_window_start = totminut
	exit
end

if $ARGS[0] = 'is_empty_archetypes':
	result = (arrsize('$tnotif_archetype_names') = 0)
	exit
end

				if $temp_np['cat'] = 'story':
					$temp_np['lbl'] = $iif($arch_log_label[temp_np['i']] <> '', $arch_log_label[temp_np['i']], '（未标注）')
					if arrpos('$temp_np_labels', $temp_np['lbl']) < 0: $temp_np_labels[] = $temp_np['lbl']
					temp_np_lbl_delta[$temp_np['lbl']] += arch_log_delta[temp_np['i']]
				else
					temp_np_cat_delta[$temp_np['cat']] += arch_log_delta[temp_np['i']]
				end
				temp_np['total'] += arch_log_delta[temp_np['i']]
			end
		end
		temp_np['i'] += 1
		jump 'np_walk_loop'
	end

	!! Header: coloured archetype name + net delta (fine dp)
	$temp_np['out'] = '<font color="<<$theme_hex[$temp_np[''archetype'']]>>"><b><<$temp_np[''disp'']>></b></font> <<$func(''archetypes'',''fmt_pts'',temp_np[''total''],''fine'')>>本次更新<br><br>'

	!! State categories (non-zero only, fine dp)
	temp_np['ci'] = 0
	:np_cats_loop
	if temp_np['ci'] < 9:
		$temp_np['ck'] = $temp_np_cats[temp_np['ci']]
		if temp_np_cat_delta[$temp_np['ck']] <> 0:
			$temp_np['out'] += '&nbsp;&nbsp;<<$temp_np_cat_disp[$temp_np[''ck'']]>>: <<$func(''archetypes'',''fmt_pts'',temp_np_cat_delta[$temp_np[''ck'']],''fine'')>><br>'
		end
		temp_np['ci'] += 1
		jump 'np_cats_loop'
	end

	!! Passive decay and opposition (fine dp)
	if temp_np_cat_delta['decay'] <> 0:
		$temp_np['out'] += '&nbsp;&nbsp；衰减：<<$func(''archetypes'',''fmt_pts'',temp_np_cat_delta[''decay''],''fine'')>><br>'
	end
	if temp_np_cat_delta['opposition'] <> 0:
		$temp_np['out'] += '&nbsp;&nbsp；对立：<<$func(''archetypes'',''fmt_pts'',temp_np_cat_delta[''opposition''],''fine'')>><br>'
	end

	!! Story events (per-label, fine dp)
	if arrsize('$temp_np_labels') > 0:
		$temp_np['out'] += '<br>'
		temp_np['li'] = 0
		:np_lbls_loop
		if temp_np['li'] < arrsize('$temp_np_labels'):
			$temp_np['lbl'] = $temp_np_labels[temp_np['li']]
			$temp_np['out'] += '&nbsp;&nbsp;<<$temp_np[''lbl'']>>: <<$func(''archetypes'',''fmt_pts'',temp_np_lbl_delta[$temp_np[''lbl'']],''fine'')>><br>'
			temp_np['li'] += 1
			jump 'np_lbls_loop'
		end
	end

	msg $temp_np['out']
	killvar 'temp_np'
	killvar 'temp_np_cats'
	killvar 'temp_np_cat_disp'
	killvar 'temp_np_cat_delta'
	killvar 'temp_np_labels'
	killvar 'temp_np_lbl_delta'
	exit
end
