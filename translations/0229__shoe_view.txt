		'你已经拥有这双鞋了。'
		exit
	end

	shop_utils_view['discount_total'] = shop_utils_view['discount'] + func('shop_utils', 'get_discount', $shop_utils_view['type'] + '_shoe', shop_utils_view['number'])
	shop_utils_view['base_price'] = ShoPrice
	shop_utils_view['price'] = shop_utils_view['base_price'] * max(0, 100 - shop_utils_view['discount_total']) / 100
	shop_utils_view['price'] = shop_utils_view['price'] / 50 * 50
	if shop_utils_view['price'] = shop_utils_view['base_price']:
		$shop_utils_view['price_string'] = $func('money', 'string_price', shop_utils_view['price'])
	else
		$shop_utils_view['price_string'] = $func('wrap', 'neg s', $func('money', 'string_price', shop_utils_view['base_price'])) + ' <b>' +  $func('money', 'string_price', shop_utils_view['price']) + '</b>'
		'现在 ' + shop_utils_view['discount_total'] + '% 折扣' + iif(shop_utils_view['discount_total'] <= 10, '', '！')
	end

	'价格：' + $shop_utils_view['price_string']

	if pcs_heels < ShoSkill:
		'你实在没法说服自己买下它们。'
		exit
	end

	if func('money', 'can_afford', shop_utils_view['price']) = 0:
		'你没有足够的钱买这双鞋。'
	else
		act '购买 (<<$shop_utils_view[''price_string'']>>)':

			gs 'shoes', 'add_item', $shop_utils_view['type'], shop_utils_view['number']
			gt $loc, $loc_arg
		end
	end

	if pcs_heels >= ShoSkill:
		act '穿上':
			gs 'shoes', 'wear', $shop_utils_view['type'], shop_utils_view['number']
			gs 'shop_utils', 'cleanup'
			gt 'wardrobe', 'main'
		end
	end
	gs 'shoe_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_storage':
	act '返回': gt 'shoe_view', 'view_list', $shop_utils_view['link']
	'这双鞋存放在存储区。'
	gs 'shoe_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_unwanted':
	act '返回': gt 'shoe_view', 'view_list', $shop_utils_view['link']
	'这双鞋被标记为不需要。'
	gs 'shoe_view', 'storage_options'
	exit
end


if $ARGS[0] = 'storage_options':
	if func('shoes', 'is_owned', $shop_utils_view['type'], shop_utils_view['number']) = 0: exit

	if func('shoes', 'in_wardrobe', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到衣柜':
			gs 'shoes', 'move_to_wardrobe', $shop_utils_view['type'], shop_utils_view['number']
			gt 'shoe_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end

	if func('shoes', 'is_immutable', $shop_utils_view['type'], shop_utils_view['number']): exit

	if func('shoes', 'in_storage', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到存储':
			gs 'shoes', 'move_to_storage', $shop_utils_view['type'], shop_utils_view['number']
			gt 'shoe_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	if func('shoes', 'in_unwanted', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到不需要':
			gs 'shoes', 'move_to_unwanted', $shop_utils_view['type'], shop_utils_view['number']
			gt 'shoe_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	act $func('wrap', 'neg', '把这些鞋子扔掉'):
		gs 'shoes', 'remove_item', $shop_utils_view['type'], shop_utils_view['number']
		gt 'shoe_view', 'view_list', $shop_utils_view['link']
	end
	exit
end
