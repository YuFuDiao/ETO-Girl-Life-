$gL_boy_z = "gt 'LakeBoyZ', 'gL_boy_z'"
if $ARGS[0] = 'gL_boy_z':
	act '<<$pcs_nickname>>':
		*clr & cla
		gs 'stat'
		if $clothingworntype = 'nude':
			'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/sunbathe_nudelake.jpg"></center>'
		else
			'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/sunbathe_lake.jpg"></center>'
		end
		'他看起来人还不错，于是你决定把你的名字告诉他。'
		if rand(0, 10) = 0 and pcs_apprnc >= rand(40, 120) and cheatVars['random_lovers'] <> 1:
			'<<$boydesc>>笑着说：“你是个很酷的女孩。也许我们可以再一起出去玩？”'
			gs 'willpower', 'misc', 'resist', 'easy'
			if pcs_willpwr < will_cost:
				act '微笑并离开' + $func('willpower', 'get_willcost_string'): $noWillpower
			else
				act '微笑并离开' + $func('willpower', 'get_willcost_string'):
					gs 'willpower', 'pay', 'resist'
					gt 'pav_lake'
				end
			end
			act '把你的手机号码给他':
				*clr & cla
				gs 'mood', 'raise', 'tiny'
				if rand(0, 2) = 0: gs 'lover', 'add_boyfriend', $boy
				minut += 5
				bmFrend[Tboynum] = 1
				gs 'stat'
				'你把你的手机号码给了<<$boydesc>>，他把它添加到了他的联系人列表里。'
				'然后他找了个借口离开，答应会给你打电话。'
				act '返回': gt 'pav_lake'
			end
		else
			'<<$boydesc>>坐得离你更近，递给你一杯冰啤酒。'
			gs 'willpower', 'drink', 'resist', 'easy'
			if pcs_willpwr < will_cost:
				act '微笑并离开' + $func('willpower', 'get_willcost_string'): $noWillpower
			else
				act '微笑并离开' + $func('willpower', 'get_willcost_string'):
					gs 'willpower', 'pay', 'resist'
					gt 'pav_lake'
				end
			end
			act '喝啤酒':
				*clr & cla
				minut += 15
				gs 'beverage', 'beer_stats'
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/drinkbeer.jpg"></center>'
				'你和那个家伙一起喝啤酒，两人聊了一会儿。'
				'突然，他把手放在你的大腿上。'
				'"再来一杯啤酒，<<$pcs_nickname>>？"'
				gs 'willpower', 'drink', 'resist', 'easy'
				if pcs_willpwr < will_cost:
					act '谢谢他的啤酒，然后离开' + $func('willpower', 'get_willcost_string'): $noWillpower
				else
					act '谢谢他的啤酒，然后离开' + $func('willpower', 'get_willcost_string'): gs 'willpower', 'pay', 'resist' & gt 'pav_lake'
				end
				act '再喝点啤酒':
					*clr & cla
					gs 'beverage', 'beer_stats'
					gs 'arousal', 'foreplay', 5
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/drinkbeer.jpg"></center>'
					'你和<<$boydesc>>又喝了不少啤酒。他开始抚摸你的腿，并倾身靠近，把手放在你的胸口上。'
					if rand(1, 100) <= 20:
						'他凑近你耳边低语：“我想尝尝你那诱人的嘴唇。”'
						gs 'willpower', 'kiss', 'resist', 'easy'
						if pcs_willpwr < will_cost:
							act '把他推开然后离开' + $func('willpower', 'get_willcost_string'): $noWillpower
						else
							act '把他推开然后离开' + $func('willpower', 'get_willcost_string'): gs 'willpower', 'pay', 'resist' & gt 'pav_lake'
						end
						act '吻他':
							*clr & cla
							gs 'arousal', 'kiss', 5
							gs 'arousal', 'end'
							if $clothingworntype = 'nude':
								'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/kissnude.jpg"></center>'
							else
								'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/kissboy.jpg"></center>'
							end
							'你热情地亲吻着<<$boydesc>>，他继续爱抚着你的身体，尤其对你的胸部情有独钟。'
							'<<$boydesc>>邀请你去他家。'
							gs 'willpower', 'sex', 'resist', 'easy'
							if pcs_willpwr < will_cost:
								act '结束今天，离开' + $func('willpower', 'get_willcost_string'): $noWillpower
							else
								act '结束今天，离开' + $func('willpower', 'get_willcost_string'): gs 'willpower', 'pay', 'resist' & gt 'pav_lake'
							end
							act '跟着他去他家':
								gs 'clothing', 'wear', $lastwornclothingtype['swim'], lastwornclothingnumber['swim']
								gs 'panties', 'wear', $lastwornpantytype['swim'], lastwornpantynumber['swim']
								gs 'bras', 'wear', $lastwornbratype['swim'], lastwornbranumber['swim']
								gt 'LBZhouse'
							end
						end
					else
						'<<$boydesc>>微笑着建议你们两个一起去散散步。'
						act '拒绝并离开': gt 'pav_lake'
						act '同意':
							*clr & cla
							minut += 15
							if $clothingworntype = 'nude':
								'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/changenude.jpg"></center>'
								gs 'clothing', 'wear', $lastwornclothingtype['swim'], lastwornclothingnumber['swim']
								gs 'panties', 'wear', $lastwornpantytype['swim'], lastwornpantynumber['swim']
								gs 'bras', 'wear', $lastwornbratype['swim'], lastwornbranumber['swim']
							else
								'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/event/changebikini.jpg"></center>'
								gs 'clothing', 'wear', $lastwornclothingtype['swim'], lastwornclothingnumber['swim']
								gs 'panties', 'wear', $lastwornpantytype['swim'], lastwornpantynumber['swim']
								gs 'bras', 'wear', $lastwornbratype['swim'], lastwornbranumber['swim']
							end
							gs 'stat'
							'你迅速换上自己的普通衣服，然后和<<$boydesc>>一起去散步。'
							'<<$boydesc>>邀请你去他家坐坐。'
							gs 'willpower', 'sex', 'resist', 'easy'
							if pcs_willpwr < will_cost:
								act '结束今天，离开' + $func('willpower', 'get_willcost_string'): $noWillpower
							else
								act '结束今天，离开' + $func('willpower', 'get_willcost_string'): gs 'willpower', 'pay', 'resist' & gt 'pav_lake'
							end
							act '跟着他去他家': gt 'LBZhouse'
						end
					end
				end
			end
		end
	end
end
