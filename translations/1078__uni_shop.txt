# uni_shop
!! 2021/10/27

if $ARGS[0] = 'start' or $ARGS[0] = '':
	*clr & cla
	$loc = 'uni_shop'
	$loc_arg = 'start'
	$menu_loc = 'uni_shop'
	$menu_arg = 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	killvar 'shoplocation'
	menu_off = 0
	gs 'themes', 'indoors'
	if sound_settings['environment_off'] = 0: PLAY 'sound/shop.mp3', 30
	gs 'stat'
	*clr & cla


	if hour < 8 or hour >= 23:
		'商店现在关门了。'
		act '离开': minut += 2 & gt 'city_island'
		exit
	end

	'商店里有一台<a href="exec: gt ''uni_shop'', ''atm''">ATM</a>，你可以从那里取钱，也可以把钱存入你的银行账户。'

	act '逛逛过道': gt 'uni_shop', 'cart'


	act '离开商店': minut += 2 & gt 'city_island'
end


if $ARGS[0] = 'atm':
	$func('wrap', 'header center b', '自动取款机')
	'<center><img <<$set_imgh>> src="images/locations/shared/store/atm.jpg"></center>'

	if bankAccount = 0:
		'你还没有银行账户！'
	else
		if karta >= bankDebtLimit:
			'你的账户里有<<$func(''money'', ''format_balance'', ''bank'')>>。'
			'你的透支额度为 ' + $func('wrap', 'accent','<<$func(''money'', ''format'', bankDebtLimit)>>!')
		else
			'你已透支 ' + $func('wrap', 'neg', '<<$func(''money'', ''format'', bankDebtLimit - karta)>> <b>₽</b>.')
			'你剩余的信用额度为 ' + $func('wrap', 'accent','<<$func(''money'', ''format'', karta)>>!')
		end

		'<br>ATM存款手续费： <<$func(''money'', ''string_price'', 100)>>'

		if money > 0:
			act '将钱存入你的银行账户':
				cla
				menu_off = 1
				minut += 5

				gs 'bank', 'deposit_cash', func('money', 'price', 100)

				gs 'stat'
				'"还有什么需要帮忙的吗？"'
				act '走开': gt 'uni_shop', 'start'
			end
		end

		if karta <= 0:
			'你的透支额已经用完了。'
		else
			act '从自动取款机取钱':
				cla
				menu_off = 1
				minut += 5

				gs 'bank', 'withdraw_cash', 0

				gs 'stat'
				'"还有什么需要帮忙的吗？"'
				act '走开': gt 'uni_shop', 'start'
			end
			if karta >= 1000:
				act '取款1000<b>₽</b>不需要买狗粮。你父母会付钱。':
					cla
					menu_off = 1
					karta -= 1000
					money += 1000
					gs 'bank', 'set_withdraw_text', 1000
					gs 'stat'
					'"还有什么需要帮忙的吗？"'
					act '走开': gt 'uni_shop', 'start'
				end
			end
		end
	end

	act '走开': gt 'uni_shop', 'start'
end


if $ARGS[0] = 'cart':
	*clr & cla
	$loc_s = 'uni_shop'
	$args_s = 'cart'
	menu_off = 0
	gs 'themes', 'indoors'

	gs 'themes', 'indoors'
	gs 'item_cart', 'shopping_aisle', 'uni'
	gs 'stat'
	$backimage = 'images/locations/city/island/unimarket.jpg'

	act '退出购物车': gt 'uni_shop', 'start'
end
