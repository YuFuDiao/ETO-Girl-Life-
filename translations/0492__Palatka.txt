if $ARGS[0] = 'start':
	menu_off = 0
	gs 'core_library', 'setloc', 'Palatka', 'start'
	$sexloc = 'Palatka'
	$location_type = 'private'
	gs 'stat'
	$func('wrap', 'header center b', '帐篷')
	'<center><img <<$set_imgh>> src="images/locations/city/shared/military/palatka.jpg"></center>'
	palrand = rand(0, 10)

	if palrand = 0: gt 'Palatka', 'pal0'
	if palrand = 1: gt 'Palatka', 'pal1'

	'令你惊讶的是，帐篷相当宽敞。里面有一张床，一个<a href="exec:gt ''mirror'', ''start''">镜子</a>，甚至还有一个简易的<a href="exec:gt ''vann'', ''start''">淋浴</a>.'

	act '离开': gt 'Military', 'start'

	act '睡眠':
		gs 'sleep_simple', 'forced', 480
		gt 'palatka', 'start'
	end
end


if $ARGS[0] = 'pal0':
	*clr & cla
	$location_type = 'event'
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/shared/military/sex/pal0.jpg"></center>'
	'你刚开始换衣服，就注意到帐篷里有两个士兵。他们看到你半裸的样子，挺着勃起的鸡巴向你走来。你渴望他们的鸡巴，心里盘算着是帮他们爽一把，还是无视他们，在辛苦一天之后好好放松一下。'

	gs 'willpower', 'sex', 'resist', 'hard'
	if pcs_willpwr < will_cost:
		act '发送' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '发送' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'pay', 'resist'
			gs 'stat'
			gt 'palatka', 'start'
		end
	end

	act '波斯特':
		gs 'npcgeneratec', 0, '士兵', rand(18, 34)
		gs 'boyStat', $npclastgenerated
		$boy[0] = $boy
		gs 'npcgeneratec', 0, '士兵', rand(18, 34)
		$boy[1] = $npclastgenerated
		guy += 2
		picrand = 22
		xgt 'sexdvoe', 'var'
	end
end


if $ARGS[0] = 'pal1':
	*clr & cla
	$location_type = 'event'
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/shared/military/pal1.jpg"></center>'
	'走进帐篷时，你绊了一下，反射性地闭上眼睛，准备脸朝下摔到地上。出乎意料的是，你并没有摔在地上，而是感到一双有力的大手接住了你。你惊讶地睁开眼睛，抬头一看，发现一个身材魁梧的黑人士兵把你抱在了怀里。'

	gs 'willpower', 'sex', 'resist', 'easy'
	if pcs_willpwr < will_cost:
		act '道谢' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '道谢 ':
			gs 'willpower', 'pay', 'resist'
			gs 'stat'
			gt 'palatka', 'start'
		end
	end

	act '好好报答他':
		sexpartkno = 1
		gs 'npcgeneratec', 0, '黑人士兵', rand(18, 35)
		gs 'boyStat', $npclastgenerated
		guy += 1
		picrand = 43
		gt 'sex', 'minet'
	end
end
