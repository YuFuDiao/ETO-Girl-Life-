!! Unified birthday handler.
!!   ARGS[1] = 'link' : inline link renderer (gs-called from start2).
!!   ARGS[1] = ''     : full picker screen (gs-called for internal date/month links,
!!                      or gt-navigated from the inline link / intro_character_custom).
!!                      Done routes back via gt $locM, $locM_arg.
if $ARGS[0] = 'core_birthday':
	if birthday = 0: gs 'intro_functions', 'set_default', 'birthday'
	gs 'time', 'init_monthnames'
	if $ARGS[1] = 'link':
		!! Inline summary link shown on start2 screens
		if $start_type['magic'] = 'tg':
			$startdob = '她的生日：<b><<mid(100 + birthday, 2, 2)>>/<<mid(100 + birthmonth, 2, 2)>>/<<birthyear>></b> (DD/MM/YYYY)'
		else
			$startdob = '你的生日：<b><<mid(100 + birthday, 2, 2)>>/<<mid(100 + birthmonth, 2, 2)>>/<<birthyear>></b> (DD/MM/YYYY)'
		end
		'<a href="exec:gt ''intro_character_creation'', ''core_birthday''"><<$startdob>></a>.'
		killvar 'startdob'
	else
		!! Full picker screen
		cls
		$func('wrap', 'header center b', '生日选择器')
		if $start_type['loc'] = 'sg' and $start_type['magic'] = 'tg':
			'<center><img <<$set_imgh>> src="images/system/1_openings/4_csb/16.jpg"></center>'
		else
			'<center><img <<$set_imgh>> src="<<FUNC(''$face_image'')>>"></center>'
		end
		'<center>选择你的出生日期：</center>'
		*nl

		if birthmonth = 2:
			if (birthyear mod 4 = 0 and birthyear mod 100 <> 0) or (birthyear mod 400 = 0):
				monthdays = 29
			else
				monthdays = 28
			end
		elseif birthmonth = 4 or birthmonth = 6 or birthmonth = 9 or birthmonth = 11:
			monthdays = 30
		else
			monthdays = 31
		end

		birthmonth = func('math', 'int_clamp', birthmonth, 1, 12)
		birthday = func('math', 'int_clamp', birthday, 1, monthdays)

		'<center><<iif($start_type[''magic''] = ''tg'', ''她'', ''你'')>>生日：<b><<birthyear>>年<<$monthName[birthmonth]>><<birthday>><<$func(''shortgs'', ''get_number_suffix'', birthday)>>日</b></center>'
		*nl
		'<center><a href="exec:birthday += 5 & gs ''intro_character_creation'', ''core_birthday''">+5 天</a></center>'
		'<center><a href="exec:birthday += 1 & gs ''intro_character_creation'', ''core_birthday''">+1 天</a></center>'
		'<center><a href="exec:birthday -= 1 & gs ''intro_character_creation'', ''core_birthday''">-1 天</a></center>'
		'<center><a href="exec:birthday -= 5 & gs ''intro_character_creation'', ''core_birthday''">-5 天</a></center>'

		$bm_td     = 'border:1px solid ' + $theme_hex['goth'] + '; background-color:' + $theme_hex['table_bg_alt'] + '; padding:6px 12px; text-align:center;'
		$bm_td_sel = 'border:1px solid ' + $theme_hex['goth'] + '; background-color:' + $theme_hex['accent']      + '; padding:6px 12px; text-align:center; color:#ffffff;'
		bm_i = 1
		:bm_build_loop
			$bm_lnk[bm_i] = '<td style="' + $bm_td + '"><a href="exec:birthmonth = ' + $str(bm_i) + ' & gs ''intro_character_creation'', ''core_birthday''">' + $monthName[bm_i] + '</a></td>'
			bm_i += 1
		if bm_i <= 12: jump 'bm_build_loop'
		killvar 'bm_i'
		$bm_lnk[birthmonth] = '<td style="' + $bm_td_sel + '"><b>' + $monthName[birthmonth] + '</b></td>'
		*p '<center><table style="border:2px solid ' + $theme_hex['goth'] + '; background-color:' + $theme_hex['table_bg'] + '; border-spacing:4px; margin:6px;" cellpadding="0" cellspacing="0">'
		*p '<tr><<$bm_lnk[1]>><<$bm_lnk[2]>><<$bm_lnk[3]>></tr>'
		*p '<tr><<$bm_lnk[4]>><<$bm_lnk[5]>><<$bm_lnk[6]>></tr>'
		*p '<tr><<$bm_lnk[7]>><<$bm_lnk[8]>><<$bm_lnk[9]>></tr>'
		*p '<tr><<$bm_lnk[10]>><<$bm_lnk[11]>><<$bm_lnk[12]>></tr>'
		*p '</table></center>'
		killvar 'bm_lnk'
		killvar 'bm_td'
		killvar 'bm_td_sel'

		act '<b>完成</b>': gt $locM, $locM_arg
		killvar 'monthdays'
	end
	exit
end

			if temp_rand = 0:
				result = 153
			elseif temp_rand = 1:
				result = 183
			elseif temp_rand = 2:
				result = 214
			else
				result = 239
			end
		elseif $start_type['loc'] = 'uni':
			result = 603
		else
			result = 521
		end
	elseif $ARGS[1] = 'location':
		if $start_type['loc'] = 'sg':
			result = rand(0, 3)
			if knowpreg = 1 or kid > 0:

					result = func('intro_character_creation', 'get_random', 'location')
					exit
				end
			end
		else
			result = 0
		end
	end

	exit

		gs 'npc_relationship', 'default_family_friends'

			killvar 'tg_init_cat'
		end
		gt 'intro_character_creation', 'start2', $ARGS[1]
	end
