!! Depracated !! Moved to 'random'
!! Randomly picks one of the items plugged in
!!
!!  Option 1: (has 6 input limit)
!!  $ARGS[1] =		'string' or 'number' to specify what you''re plugging in (can also be blank for numbers)
!!  $ARGS[2-7] =	strings or numbers to pick from, limited to 6
!!  Example 1: func('shortgs', 'rand_pick', 'string', 'alpha', 'beta', 'gamma', 'delta', 'epsilon', 'zeta')
!!  Example 2: func('shortgs', 'rand_pick', '', 2, 3, 5, 7, 11, 13)
!!
!!  Option 2: (no limitations, has to be delimited)
!!  $ARGS[1] =		'delimit' - used for larger numbers of items to pick from
!!  $ARGS[2] =		'string' or 'number' to specify what you''re plugging in (can also be blank for numbers)
!!  $ARGS[3] =		Delimited string to parse
!!  $ARGS[4] =		Delimiter used (Default = '|' pipe)
!!  Example 1: func('shortgs', 'rand_pick', 'delimit', 'string', 'alpha|beta|gamma|delta|epsilon|zeta')
!!  Example 2: func('shortgs', 'rand_pick', 'delimit', 'number', '1-12-123-1234-12345-123456', '-')
!!  Example 3: func('shortgs', 'rand_pick', 'delimit', '', '0.1.2.3.5.7.11', '.')
if $ARGS[0] = 'rand_pick':
	if $ARGS[1] = 'delimit':
		if $ARGS[2] = 'string':
			$result = $func('random', 'pick_from_delimited_string', $ARGS[3], $ARGS[4], 1)
		else
			result = func('random', 'pick_from_delimited_string', $ARGS[3], $ARGS[4])
		end
	else
		if arrsize('ARGS') <= 2:
			msg '<b>错误：''rand_pick'' 缺少 ARGS</b>在 shortgs, rand_pick'
		else
			if $ARGS[1] = 'string':
				$result = $ARGS[rand(2, arrsize('$ARGS')-1)]
			else
				result = ARGS[rand(2, arrsize('ARGS')-1)]
			end
		end
	end
	exit
end

!! converts hexadecimal string to decimal integer
!! main use - convert hex color to rgb color (to use with HTML)

!! converts ABGR integer (generated by rgb() func) to hex string
!! OMITS ALPHA CHANNEL
!!	use: func('shortgs', 'rgb_to_hex', number)
!!	example: func('shortgs', 'rgb_to_hex', bcolor)
!!	example: func('shortgs', 'rgb_to_hex', lcolor)
if $ARGS[0] = 'rgb_to_hex':
	temp_input = ARGS[1]

	if temp_input >= 0:
		temp_red = temp_input mod 256
		temp_green = (temp_input / 256) mod 256
		temp_blue = (temp_input / 65536) mod 256
	else
		!! Red (lowest 8 bits) - this always works with mod
		temp_work = (temp_input mod 256) + 256
		temp_red = temp_work mod 256

		!! Green (bits 8-15)
		temp_work = (temp_input mod 65536) + 65536
		temp_green = (temp_work / 256) mod 256

		!! Blue (bits 16-23) - same approach
		temp_work = (temp_input mod 16777216) + 16777216
		temp_blue = (temp_work / 65536) mod 256

		killvar 'temp_work'
	end

	!! Convert to hex
	$red_hex	= $func('shortgs', 'int_to_hex_str', temp_red)
	$green_hex	= $func('shortgs', 'int_to_hex_str', temp_green)
	$blue_hex	= $func('shortgs', 'int_to_hex_str', temp_blue)

	$result = $mid('00' + $red_hex,	1 + len($red_hex))
	$result += $mid('00' + $green_hex,	1 + len($green_hex))
	$result += $mid('00' + $blue_hex,	1 + len($blue_hex))
	$result = '#' + $result

	!! Clean up
	killvar 'temp_input'
	killvar 'temp_blue'
	killvar 'temp_green'
	killvar 'temp_red'
	killvar 'red_hex'
	killvar 'green_hex'
	killvar 'blue_hex'
	killvar 'temp_work'
	exit
end


!! Inverts a hexidecimal color (f.e. #31A4E2)

!! if you want to get either white or black, furthest away from the given color. Use:
!! func('shortgs', 'invert_color', $color, 'bw')
!! https://stackoverflow.com/questions/35969656/how-can-i-generate-the-opposite-color-according-to-current-color
if $ARGS[0] = 'invert_color':

		shortgs_r_int = 255 - func('shortgs', 'hex_str_to_int', mid($ARGS[1], 1, 2))
		shortgs_g_int = 255 - func('shortgs', 'hex_str_to_int', mid($ARGS[1], 3, 2))
		shortgs_b_int = 255 - func('shortgs', 'hex_str_to_int', mid($ARGS[1], 5, 2))

		$shortgs_r = func('shortgs', 'int_to_hex_str', shortgs_r_int)
		$shortgs_g = func('shortgs', 'int_to_hex_str', shortgs_g_int)
		$shortgs_b = func('shortgs', 'int_to_hex_str', shortgs_b_int)

		$shortgs_r = mid('0' + $shortgs_r, len($shortgs_r), 2)
		$shortgs_g = mid('0' + $shortgs_g, len($shortgs_g), 2)
		$shortgs_b = mid('0' + $shortgs_b, len($shortgs_b), 2)
		$result = '#' + $shortgs_r + $shortgs_g + $shortgs_b

		killvar 'shortgs_r'
		killvar 'shortgs_g'
		killvar 'shortgs_b'
		killvar 'shortgs_r_int'
		killvar 'shortgs_g_int'
		killvar 'shortgs_b_int'

end
