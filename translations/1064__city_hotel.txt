# city_hotel

if $ARGS[0] = '' or $ARGS[0] = 'start':
	if hotelRoomDays['city'] - daystart = 0 and hour > 11: HotelRoom['city'] = 0
	if hotelRoomDays['city'] - daystart < 0: HotelRoom['city'] = 0
	*clr & cla
	nom_o[1] = 100
	nom_o[2] = 1000
	nom_o[3] = 10000
	nom_o[4] = 100000
	gs 'core_library', 'setloc', 'city_hotel', ''
	$region = 'city'
	menu_off = 0
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/hotel/lobby.jpg"></center>'
	'你走进酒店大堂，只能看到一个百无聊赖的女孩站在前台。'

	act '离开': minut += 5 & gt 'city_center'
	if HotelRoom['city'] ! 0:
		act '回你的房间':
			if HotelRoom['city'] = 1: gt 'HotelRoom', 'normal'
			if HotelRoom['city'] = 2: gt 'HotelRoom', 'better'
			if HotelRoom['city'] = 3: gt 'HotelRoom', 'best'
		end
	end
	act '去前台': gt 'city_hotel', 'reception'
	if week > 5 and hour = 14 and strip_club['porn_scout_pickup'] = 1:
		act '去星探的房间': gt 'pickup_porn', 'talent_scout_meet'
	end

	gs 'lover', 'lover_events'
end


if $ARGS[0] = 'reception':
	*clr & cla
	menu_off = 1
	minut += 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/hotel/receptionist.jpg"></center>'
	'你走向前台，柜台后面一位微笑的年轻女士向你打招呼。'
	'"你好。我能帮你什么吗？"她问道。'

	act '回到门厅': minut += 1 & gt 'city_hotel'

	act '查看房间描述':
		cla
		minut += 1
		gs 'stat'
		'有 4 种类型的房间可供选择：'
		'1. 经济间：我们简单的房间，适合只想找张床睡一晚的人。浴室是与同楼层的人共用的公共浴室。'
		'2. 标准间：我们价格合理的房间。每个房间都配有私人浴室和电视。'
		'3. 套房：我们豪华的房间。除了私人浴室外，还有一间额外的起居室。'
		!! '4. ???'
		act '继续': gt 'city_hotel', 'reception'
	end

	if tobiQW = 2000:
		act '为什么这家旅馆这么拥挤？':
			cla
			tobiQW = 2001
			minut += 1
			'"为什么今天旅馆里这么拥挤？"你问道。'
			'"这里在举办一个关于土著语言的会议。他们每年聚一次，抱怨他们正在消失的语言。"她笑着说。'
			act '走开': gt 'city_hotel'
		end
	end

	act '租房间': gt 'city_hotel', 'rent_a_room'
end


if $ARGS[0] = 'pay_the_room':
	hotel_room_id = ARGS[1]
	if func('money', 'can_afford', totalCost) = 0:
		if ARGS[1] = 1: act '你付不起在旅馆住这么久的费用': gt 'city_hotel'
		if ARGS[1] = 2: act '你付不起在豪华间住这么久的费用': gt 'city_hotel'
		if ARGS[1] = 3: act '你付不起在皇家套房住这么久的费用': gt 'city_hotel'
		killvar 'totalCost'
		killvar 'hotel_room_id'
	else
		act '支付 (' + $func('money', 'string_price', totalCost) + ')':
			cla
			gs 'money', 'pay', totalCost
			HotelRoom['city'] = hotel_room_id
			hotelRoomDays['city'] = daystart + hotelRoomDays['city']
			'你付了房费，她把钥匙交给你。'
			'"谢谢 ' + iif(pavHotelMaid = 0, '小姐', $pcs_nickname) + '，祝你住得愉快！"'
			act '离开接待台': gt 'city_hotel'
			killvar 'totalCost'
			killvar 'hotel_room_id'
		end
	end

	act '改变主意':
		killvar 'totalCost'
		killvar 'hotel_room_id'
		gt 'city_hotel'
	end
end


if $ARGS[0] = 'rent_a_room':
	*clr & cla
	menu_off = 1
	gs 'stat'

	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/hotel/receptionist.jpg"></center>'
	'桌子后面的女孩对你微笑着。"你好，小姐！你想租个房间吗？"'

	if HotelRoom['city'] > 0:
		*clr & cla
		'<center><img <<$set_imgh>> src="images/locations/city/citycenter/hotel/receptionist.jpg"></center>'
		'她看着电脑。"我看到你已经在这里订了一个房间。由于房间已经预订并支付了费用，要换成不同的房间只能取消你现有的预订。如果未提前通知就取消，你的付款将被没收。我建议你不要这样做，但如果你愿意，也可以这么做。"'
		act '取消你的预订': HotelRoom['city'] = 0 & gt 'city_hotel'
		act '不': gt 'city_hotel'
	else
		act '是':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/city/citycenter/hotel/receptionist.jpg"></center>'
			'她高兴地对你微笑。"太好了！我们现在有以下房型可供选择："'
			'标准间 - ' + $func('money', 'string_price', 500) + ' 一晚；'
			'豪华间 - ' + $func('money', 'string_price', 1500) + ' 一晚；'
			'皇家套房 - ' + $func('money', 'string_price', 10000) + ' 一晚。'
			*nl
			act '标准间':
				cla
				hotelRoomDays['city'] = input ("你想住多久？")
				if hotelRoomDays['city'] > 0 :
					totalCost = hotelRoomDays['city'] * 500
					'"一间标准间住<<hotelRoomDays[''city'']>>天需要 ' + $func('money', 'string_price', totalCost) + '，"接待员说道。'
					gs 'city_hotel', 'pay_the_room', 1
				else
					gt 'city_hotel'
				end
			end

			act '豪华房':
				cla
				hotelRoomDays['city'] = input ("你想住多久？")
				if hotelRoomDays['city'] > 0:
					totalCost = hotelRoomDays['city'] * 1500
					'"一间豪华间住<<hotelRoomDays[''city'']>>天需要 ' + $func('money', 'string_price', totalCost) + '，"接待员说道。'
					gs 'city_hotel', 'pay_the_room', 2
				else
					gt 'city_hotel'
				end
			end

			act '皇家套房':
				cla
				hotelRoomDays['city'] = input ("你想住多久？")
				if hotelRoomDays['city'] > 0:
					totalCost = hotelRoomDays['city'] * 10000
					'"皇家套房住<<hotelRoomDays[''city'']>>天需要 ' + $func('money', 'string_price', totalCost) + '，"接待员说道。'
					gs 'city_hotel', 'pay_the_room', 3
				else
					gt 'city_hotel'
				end
			end

			act '离开接待台': gt 'city_hotel'
		end

		act '不，谢谢': gt 'city_hotel'
	end
end
