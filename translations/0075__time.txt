# time
!! Calculate the number of the current day since the start of the game.


if $ARGS[0] = '':
	if minut < 0 or hour < 0: gs 'time', 'time_cheat_fix'

	!! total amount of minutes since the start of the game
	prevtotmin = totminut
	!! sync totminut with daystart, hour, and minut
	totminut = minut + hour * 60 + daystart * 1440

	if totminut <> prevtotmin:
		:time_time_loop
			time_temp_jump_flag = 0
			if minut >= 60:
				!! Hourly Changes
				hour += 1
				minut -= 60
				gs 'hourly_events'
				time_temp_jump_flag = 1
			end
			if hour >= 24:
				hour -= 24
				daystart += 1
				gs 'time', 'update_date'
				gs 'cikl'
				time_temp_jump_flag = 1
			end
		if time_temp_jump_flag = 1: jump 'time_time_loop'
		killvar 'time_temp_jump_flag'

		gs 'time', 'update_daystage', totminut
	end
	killvar 'dateVars'
	exit
end


if $ARGS[0] = 'time_cheat_fix':
	if minut < 0:
		hour -= 1
		minut += 60
		jump 'time_time_loop'
	end
	if hour < 0:
		daystart -= 1
		hour += 24
		jump 'time_time_loop'
	end

	gs 'time', 'update_date'
	exit
end

if $ARGS[0] = 'to_date':
	if arrsize('ARGS') = 1: ARGS[1] = daystart
	dateVars['daystart'] = ARGS[1]

	!! daystart = 1 -> Fri 1 Jan 2016
	!! mod_daystart = 0 -> Mon 1 Jan 1901
	temp_timeVars['mod_daystart'] = dateVars['daystart'] + 151574
	temp_timeVars['week_drift'] = 6

	dateVars['odd_week'] = ((temp_timeVars['mod_daystart'] + 6 - temp_timeVars['week_drift']) / 7) mod 2
	dateVars['week'] = (temp_timeVars['mod_daystart'] + 7 - temp_timeVars['week_drift']) mod 7
	if dateVars['week'] = 0: dateVars['week'] = 7
	$dateVars['weekName'] = $weekName[dateVars['week']]

	temp_timeVars['inner_daystart'] = temp_timeVars['mod_daystart']
	dateVars['year'] = 1601

	temp_timeVars['400cycles'] = temp_timeVars['inner_daystart'] / 146097	& !! (cycle of 400 years)
	dateVars['year'] += 400 * temp_timeVars['400cycles']
	temp_timeVars['inner_daystart'] -= 146097 * temp_timeVars['400cycles']

	temp_timeVars['100cycles'] = temp_timeVars['inner_daystart'] / 36524	& !! (cycle of 100 years)
	dateVars['year'] += 100 * temp_timeVars['100cycles']
	temp_timeVars['inner_daystart'] -= 36524 * temp_timeVars['100cycles']

	temp_timeVars['4cycles'] = temp_timeVars['inner_daystart'] / 1461		& !! (cycle of 4 years)
	dateVars['year'] += 4 * temp_timeVars['4cycles']
	temp_timeVars['inner_daystart'] -= 1461 * temp_timeVars['4cycles']

	!! The remaining years in this cycle of 4 years
	if temp_timeVars['inner_daystart'] >= 1095:
		temp_timeVars['inner_years'] = 3
	elseif temp_timeVars['inner_daystart'] >= 730:
		temp_timeVars['inner_years'] = 2
	elseif temp_timeVars['inner_daystart'] >= 365:
		temp_timeVars['inner_years'] = 1
	else
		temp_timeVars['inner_years'] = 0
	end
	dateVars['year'] += temp_timeVars['inner_years']	& !! The remaining years in this cycle of 4 years
	temp_timeVars['inner_daystart'] -= 365 * temp_timeVars['inner_years']

	temp_timeVars['leapyear'] = 0
	if ((dateVars['year'] mod 4) = 0 and (dateVars['year'] mod 100) <> 0) or (dateVars['year'] mod 400 = 0):
		if temp_timeVars['inner_daystart'] >= 59:
			temp_timeVars['leapyear'] = 1
			temp_timeVars['inner_daystart'] -= 1
		end
	end

	dateVars['day_of_year'] = 1 + temp_timeVars['inner_daystart']
	!! temp_inner_days starts at 0!
	if temp_timeVars['inner_daystart'] < 31:
		dateVars['month'] = 1
	elseif temp_timeVars['inner_daystart'] < 59:
		dateVars['month'] = 2
		temp_timeVars['inner_daystart'] += temp_timeVars['leapyear']
		temp_timeVars['inner_daystart'] -= 31
	elseif temp_timeVars['inner_daystart'] < 90:
		dateVars['month'] = 3
		temp_timeVars['inner_daystart'] -= 59
	elseif temp_timeVars['inner_daystart'] < 120:
		dateVars['month'] = 4
		temp_timeVars['inner_daystart'] -= 90
	elseif temp_timeVars['inner_daystart'] < 151:
		dateVars['month'] = 5
		temp_timeVars['inner_daystart'] -= 120
	elseif temp_timeVars['inner_daystart'] < 181:
		dateVars['month'] = 6
		temp_timeVars['inner_daystart'] -= 151
	elseif temp_timeVars['inner_daystart'] < 212:
		dateVars['month'] = 7
		temp_timeVars['inner_daystart'] -= 181
	elseif temp_timeVars['inner_daystart'] < 243:
		dateVars['month'] = 8
		temp_timeVars['inner_daystart'] -= 212
	elseif temp_timeVars['inner_daystart'] < 273:
		dateVars['month'] = 9
		temp_timeVars['inner_daystart'] -= 243
	elseif temp_timeVars['inner_daystart'] < 304:
		dateVars['month'] = 10
		temp_timeVars['inner_daystart'] -= 273
	elseif temp_timeVars['inner_daystart'] < 334:
		dateVars['month'] = 11
		temp_timeVars['inner_daystart'] -= 304
	else
		dateVars['month'] = 12
		temp_timeVars['inner_daystart'] -= 334
	end
	$dateVars['monthName'] = $monthName[dateVars['month']]

	if dateVars['month'] >= 12 or dateVars['month'] < 3:
		$dateVars['season'] = 'winter'
	elseif dateVars['month'] < 6:
		$dateVars['season'] = 'spring'
	elseif dateVars['month'] < 9:
		$dateVars['season'] = 'summer'
	else
		$dateVars['season'] = 'fall'
	end

	dateVars['day'] = 1 + temp_timeVars['inner_daystart']

	$dateVars['suffix'] = $func('time', 'get_number_suffix', dateVars['day'])

	if $ARGS[2] = 'test':
		*nl
		'新的一天开始：<<ARGS[1]>> | 旧的一天开始：<<daystart>>'
		*nl
		'新的一天：<<dateVars[''day'']>> | 旧的一天：<<day>>'
		'新月份：<<dateVars[''month'']>> | 旧月份：<<month>>'
		'新一年：<<dateVars[''year'']>> | 旧一年：<<year>>'
		'新的一周：<<dateVars[''week'']>> | 旧的一周：<<week>>'
		'新的单周：<<dateVars[''odd_week'']>> | 旧的单周：<<odd_week>>'
		*nl
		'临时值：'
		'mod_daystart: <<temp_timeVars[''mod_daystart'']>>'
		'内部日开始：<<temp_timeVars[''inner_daystart'']>>'
		'闰年：<<temp_timeVars[''leapyear'']>>'
		'4周期：<<temp_timeVars[''4cycles'']>>'
		'100周期：<<temp_timeVars[''100cycles'']>>'
		'400周期：<<temp_timeVars[''400cycles'']>>'
		'内部年数：<<temp_timeVars[''inner_years'']>>'
	end

	killvar 'temp_timeVars'
	exit
end

if $ARGS[0] = 'update_daystage':
	if arrsize('ARGS') = 1: ARGS[1] = totminut

	DayLength = func('time', 'get_DayLength', totminut / 1440)

	CurTimeSun = func('time', 'get_CurTimeSun', totminut)

	!! 1 - dawn
	!! 2 - midday
	!! 3 - sunset
	!! 4 - the beginning of the night
	!! 5 - night
	!! 6 - the end of the night
	!! phases 1, 3, 4 and 6 last for 64 minutes
	if CurTimeSun < -DayLength - 64:
		DayStage = 5

		BColorBase = 0		& !! Black
	elseif CurTimeSun < -DayLength:
		DayStage = 6

		!! CurTimeSun - negative
		!! result = 128 - [64 -> 0] * 2
		BColorBase = 128 + (DayLength + CurTimeSun) * 2
	elseif CurTimeSun < -DayLength + 64:
		DayStage = 1

		!! CurTimeSun - negative
		!! result = 128 + [0 -> 64] * 2
		BColorBase = 128 + (DayLength + CurTimeSun) * 2
	elseif CurTimeSun < DayLength - 64:
		DayStage = 2

		BColorBase = 255	& !! White
	elseif CurTimeSun < DayLength:
		DayStage = 3

		!! CurTimeSun - positive
		!! result = 128 + [64 -> 0] * 2
		BColorBase = 128 + (DayLength - CurTimeSun) * 2
	elseif CurTimeSun < DayLength + 64:
		DayStage = 4

		!! CurTimeSun - positive
		!! result = 128 - [0 -> 64] * 2
		BColorBase = 128 + (DayLength - CurTimeSun) * 2
	else
		DayStage = 5

		BColorBase = 0		& !! Black
	end
	exit
end

if $ARGS[0] = 'get_CurTimeSun':
	if arrsize('ARGS') = 1: ARGS[1] = totminut

	!!  solar noon coincides with CurTimeSun of 0 which happens at 13:00
	!! Solar Noon (13:00) minus DayLength - time when day starts
	!! Solar Noon (13:00) plus DayLength - time when day ends

	!! distance from 13:00 (Solar Noon) in minutes
	!!    at midnight its -780
	!!    0 is 13:00
	!!    1 minute before midnight (23:59) is 659

if $ARGS[0] = 'get_date_string':
	!! ── Arguments ───────────────────────────────────────────────────
	!! ARGS[1] = year  (defaults to current year)
	!! ARGS[2] = month (defaults to current month)
	!! ARGS[3] = day   (defaults to current day)
	!! ARGS[4] = format (optional; defaults to 0)
	!!
	!!  All formats include the full weekday name ($weekName).
	!!
	!!   0 = Monday 14th April 2004        (default)
	!!   1 = Monday, 2004-04-14            (ISO 8601)
	!!   2 = Monday, 14/04/2004            (DD/MM/YYYY)
	!!   3 = Monday, April 14th, 2004      (Month DDth, YYYY)
	!!   4 = Monday, 04/14/2004            (MM/DD/YYYY)
	if arrsize('ARGS') <= 1: ARGS[1] = year
	if arrsize('ARGS') <= 2: ARGS[2] = month
	if arrsize('ARGS') <= 3: ARGS[3] = day

	gs 'time', 'to_daystart', ARGS[1], ARGS[2], ARGS[3]
	$temp_date_weekName = $weekName[func('time', 'get_week_from_daystart', dateVars['daystart'])]

	if ARGS[4] = 1:
		!! Monday, 2004-04-14 (ISO 8601)
		$result = str(ARGS[1]) + '-' + $mid(100 + ARGS[2], 2, 2) + '-' + $mid(100 + ARGS[3], 2, 2) + ' ' + $temp_date_weekName
	elseif ARGS[4] = 2:
		!! Monday, 14/04/2004
		$result = $mid(100 + ARGS[3], 2, 2) + '/' + $mid(100 + ARGS[2], 2, 2) + '/' + str(ARGS[1]) + ' ' + $temp_date_weekName
	elseif ARGS[4] = 3:
		!! Monday, April 14th, 2004
		$result = str(ARGS[1]) + '年' + $monthName[ARGS[2]] + str(ARGS[3]) + '日 ' + $temp_date_weekName
	elseif ARGS[4] = 4:
		!! Monday, 04/14/2004 (MM/DD/YYYY — American)
		$result = $mid(100 + ARGS[2], 2, 2) + '/' + $mid(100 + ARGS[3], 2, 2) + '/' + str(ARGS[1]) + ' ' + $temp_date_weekName
	else
		!! Format 0 (default): Monday, 14th April 2004
		$result = str(ARGS[1]) + '年' + $monthName[ARGS[2]] + str(ARGS[3]) + '日 ' + $temp_date_weekName
	end

	killvar 'temp_date_suffix'
	killvar 'temp_date_weekName'
