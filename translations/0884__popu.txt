if $ARGS[0] = 'start':
	*clr & cla
	menu_off = 1
	minut += 5
	gs 'stat'
	gs 'themes', 'indoors'

	'<center><img <<$set_imgh>> src="images/pc/items/pet_emporium/parrot_home.jpg"></center>'
	'<center><<$ParrotQW[''Name1'']>>正坐在笼子里。</center>'

	act '离开': gt 'dina', 'brodila'

	act '打开窗户，放出你的鹦鹉':
		ParrotQW['Owned1'] = 0
		minut += 10
		gt 'dina', 'brodila'
	end
end


if $ARGS[0] = 'start2':
	*clr & cla
	menu_off = 1
	minut += 5
	gs 'stat'
	gs 'themes', 'indoors'

	'<center><img <<$set_imgh>> src="images/pc/items/pet_emporium/parrot_home.jpg"></center>'
	'<center><<$ParrotQW[''Name2'']>>正坐在笼子里。</center>'

	act '离开': gt $loc, $loc_arg

	act '打开窗户，放出你的鹦鹉':
		if daystart - ParrotQW['BuyDate'] > 30:
			'你已经太舍不得<<$ParrotQW[''Name2'']>>了，你狠不下心把它放走。'
		else
			'你和<<$ParrotQW[''Name2'']>>道别，把它放飞了。'
			ParrotQW['Owned2'] = 0
			minut += 10
		end
	end
end
