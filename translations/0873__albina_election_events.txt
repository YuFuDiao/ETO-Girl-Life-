	*clr & cla
	menu_off = 1
	minut += 1

	'"你是个聪明的女孩。就在这儿签名吧，"他说着递给你一支笔和一张纸。你看了看纸，发现上面已经印好了一份陈述。'
	act '继续':
		*clr & cla
		minut += 1
		gs 'stat'
		'<center><img <<$set_imgh>> src="images/shared/misc/signature.jpg"></center>'
		'你飞快地在底部签上名字，他没有多说什么就让你走了。'
		'"你真的帮了我们大忙，<<$pcs_lastname>>小姐。谢谢你。"'
		'他示意你下车，你一关上车门，车就开走了。'
		act '打电话给阿尔比娜':
			*clr & cla
			minut += 1
			AlbinaQW['Friends'] = 1
			AlbinaQW['StarletsShutDown'] = 1
			gs 'calendar', 'pack', 'remove', 'starlets'
			starlets_missed = 0
			gs 'stat'
			'<center><img <<$set_imgh>> src="images/pc/activities/phone/calling_outdoor.jpg"></center>'
			'你迅速拨打阿尔比娜的号码，慌乱地，告诉她发生了什么。她让你冷静下来并问你是否告诉了他们什么。'
			'"我……我说了。我把一切都告诉了他们，但他们说不会牵扯到你！我别无选择，阿尔比娜！对不起！"'
			'"哦。好吧，谢谢你告诉我。"'
			*nl
			'<i>咔哒。</i>电话断了。'
			*nl
			'你低头看着手机，发现她已经挂断了。尽管切尔诺夫承诺过，你还是觉得自己出卖了阿尔比娜。然而，就在她挂断之前，听起来她似乎……<i>高兴？</i>'
			'她是不是<i>想要</i>你把一切都告诉切尔诺夫？整个遭遇让你感到困惑和迷茫。'
			act '离开': gt 'pav_residential'
		end
	end
end
