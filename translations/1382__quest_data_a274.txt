if $ARGS[0] = 'init':
	$loc_id = 'quest_data_a274'
	!! init all vars and data

	if ballet['debug'] = 1: debug['code'] = 1
	if qstage ! 0:
		gs $loc_id, 'quest_body', 'stage_<<qstage>>'
	else
		gt $loc_id, 'debug'
	end
end


if $ARGS[0] = '':
	gt $loc_id, 'debug'
end


if $ARGS[0] = 'quest_body':
	!! create quest body and navigation as required - mandatory fields:
	!! $npc_['quest_body']
	!! $npc_quest['q_back']
	!! $npc_quest['q_next']

	!! Important set q_back or q_next to null when reaching start or finish to ensure navigation works!

	!! Stage 1

	if $ARGS[1] = 'stage_1':
	!! first stage, so set to null to ensure nav works.
		$npc_quest['q_back'] = 'null'
		if mayaqw['path'] >= 1:
			$npc_quest['q_next'] = 'stage_2'
		else
			$npc_quest['q_next'] = 'null'
		end
		$npc_quest['hints'] = '参加普希金芭蕾舞学校并与学生们互动。'
	!! Quest body text:
		$npc_quest['body'] = '赢得玛雅的信任：在普希金的寄宿舞蹈学校与玛雅见面。'
	end

	!! Stage 2
	if $ARGS[1] = 'stage_2':
		$npc_quest['q_back'] = 'stage_1'
		if mayaqw['path'] >= 3:
			$npc_quest['q_next'] = 'stage_3'
		else
			$npc_quest['q_next'] = 'null'
		end
		$npc_quest['q_next'] = 'null'
		$npc_quest['hints'] = '接受玛雅的邀请。'
		if mayaqw['grave'] = 0:
			$npc_quest['body'] = '赢得玛雅的信任：继续与玛雅见面交谈。'
		elseif mayaqw['grave'] = -1:
			$npc_quest['body'] = '你整个晚上都在专心学习芭蕾，忽略了玛雅来你房间的拜访。'
		elseif mayaqw['grave'] = 1:
			$npc_quest['body'] = '玛雅来访时，你无视了她的第一次敲门。'
		elseif mayaqw['grave'] = 3:
			$npc_quest['body'] = '你邀请玛雅到你的房间，并进行了关于她过去的简短真诚谈话。'
		elseif mayaqw['grave'] = 4:
			$npc_quest['body'] = '你去了墓地，在玛雅悲伤时支持她，之后在你的房间里与她进行了真诚的交谈。'
		end
	end


	!! Stage 3


end
