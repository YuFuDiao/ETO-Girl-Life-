# treeCircle
!! The Magical Teleport Tree Circle
!!
!! ARGS[0] is your entrance point as those listed below. If the user doesn''t have the magical ability,
!!		they will only be allowed to leave back from their entry point.

!! Locations
$tpLocations[0] = 'CentralPark'
$tpLocations[1] = '巴甫洛夫斯克Park'
$tpLocations[2] = 'gad_forest'
$tpLocations[3] = '普希金Park'
$tpLocations[4] = '村庄'

!! Build Info Arrays for Locations
!!	treeCircName = The Pretty Name of the Location
!!	treeCircLoc = The Location to send the user to when leaving Circle
!!	treeCircArg = An argument to get the user to the right spot in the location.
!! Central Park Location
$treeCircName['CentralPark'] = '中央公园'
$treeCircLoc['CentralPark'] = 'city_park'
$treeCircArg['CentralPark'] = 'start'
!! Pavlovsk Park Location
$treeCircName['PavlovskPark'] = '巴甫洛夫斯克公园'
$treeCircLoc['PavlovskPark'] = 'pav_parkev'
$treeCircArg['PavlovskPark'] = 'go_for_walk2'
!! Gadukino Forest Location
$treeCircName['gad_forest'] = '加杜基诺森林'
$treeCircLoc['gad_forest'] = 'gad_forest'
$treeCircArg['gad_forest'] = 'forest_edge'
!! Pushkin Park Location
$treeCircName['PushkinPark'] = '普希金公园'
$treeCircLoc['PushkinPark'] = 'pushkin_parks'
$treeCircArg['PushkinPark'] = 'prut2'
!! Communal Village Spring Location
$treeCircName['Village'] = '村庄泉水'
$treeCircLoc['Village'] = 'ETO_village'
$treeCircArg['Village'] = 'spring_water'

$EntryPoint = $ARGS[0]

killvar 'tpRand'

*clr & cla

'<center><h2>神秘树圈</h2></center>'
'<center><img <<$set_imgh>> src="images/locations/shared/park/tree_circle.jpg"></center>'

'树圈中央的小空地大约有两个身位宽。地面覆盖着一层柔软的海绵般的苔藓。尽管头顶的枝叶几乎遮住了天空，这里的一切却出奇地清晰可见。而当你望向树圈之外时，一切都显得模糊而朦胧，让人难以分辨圆圈外面到底有什么。'
*nl

!! Determine if this user can teleport at all.

canTeleport = 0
if spellKnown['teleport'] = 1:
	if tpKnown[$EntryPoint] = 0:
		'你听从仙女的建议，熟悉了这个树圈。'
		minut += 15
		!! Learn the current entry point
		tpKnown[$EntryPoint] = 1
	end
	if pcs_mana > spellMana['teleport']:
		canTeleport = 1
	else
		*nl
		'你现在太累了，施展不了传送。'
	end
else
	'你能感觉到这个地方蕴含着某种奇异而强大的东西，却说不清到底是什么。不过这很令人着迷，于是你不慌不忙地仔细打量起这个地方，试图弄明白它。'
	minut += 15
	!! Learn the current entry point
	tpKnown[$EntryPoint] = 1
end

gs 'treeCircActs', $EntryPoint

!! Setup Actions for locations when applicable
i = 0
:LocLoop
if i < arrsize('$tpLocations'):
	if $EntryPoint ! $tpLocations[i]:
		gs 'treeCircActs', $tpLocations[i]
	end
	i += 1
	jump 'LocLoop'
end
killvar 'i'

if pcs_inhib >= 30 and pcs_horny > 60:
	act '自慰清醒头脑':
		minut += 5
		*clr & cla
		'<center><video autoplay loop <<$set_imgh>> src="images/locations/shared/park/tree_masturb.mp4"></video></center>'
		*nl
		'在树圈的寒意中自慰，感觉像是过了很久，但这里的魔法结界保护着你，即使你正在上演如此下流的表演也不会被人发现。'
		'大约五分钟后，你达到了高潮，瞬间感觉自己与树木产生了连接，瞥见了这个地方数百年的历史。'
		'就在这一瞬间，你又猛地回到了现实。'
		$orgasm_or = 'yes'
		gs 'arousal', 'clit_finger', 5, 'masturbate'
		gs 'arousal', 'end'
		cla
		act '头脑清醒了': gt 'treeCircle', $EntryPoint
	end
end
