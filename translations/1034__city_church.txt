	act '祈祷':
		*clr & cla

		minut += 10
		gs 'fame', 'church_reduction', 'city', 1
		gs 'archetypes', 'faith_gain', 'pray', 10

		if hour > 6 and hour < 20 and church_moral < 30:
			pcs_horny -= 10
			gs 'mood', 'raise', 'small'
			church_moral += 1
			'你决定祈祷一会儿。你并不是特别虔诚，但这份平静与安宁给了你思考的时间。'
		elseif hour > 6 and hour < 20 and church_moral >= 30 and church_moral < 60:
			pcs_horny -= 20
			gs 'mood', 'raise', 'medium'
			church_moral += 1
			minut += 15
			'你决定祈祷一会儿。虽然你不认为自己非常虔诚，你确实在看着周围的圣像时感受到与上帝强烈的联系。你任由自己深入审视灵魂同时思考你人生的方向。'
		elseif hour >6 and hour < 20 and church_moral >= 60:
			pcs_horny -= 30
			gs 'mood', 'raise', 'medium'
			church_moral += 1
			minut += rand(30, 60)
			'你决定祈祷。身处神的殿中的美丽与荣耀让你心醉神迷同时你沉浸在祈祷中。你不确定过去了多长时间当你终于重新意识到周围环境时，但这不重要因为你已与上帝和世界和平相处。'
		elseif (hour >= 20 or hour <= 6) and church_moral < 30:
			pcs_horny -= 40
			gs 'mood', 'raise', 'large'
			church_moral += 1
			minut += rand(30, 60)
			'中殿仅由小蜡烛照亮，周围一片昏暗。你开始感到一种奇怪的气氛，一种你无法摆脱的感觉。你突然有一种非常强烈地祈祷的冲动并且不知不觉，一个小时过去了你感到重生了。'
		elseif (hour >= 20 or hour <= 6) and church_moral >= 30 and church_moral < 60:
			pcs_horny = 0
			gs 'mood', 'raise', 'large'
			church_moral += 1
			minut += rand(30, 60)
			if drugVars['cocaine_system'] > 1: drugVars['cocaine_system'] = 3 * drugVars['cocaine_system'] / 4
			'中殿仅由小蜡烛照亮，周围一片昏暗，但你却觉得这令人安心。你有一种非常强烈地祈祷的冲动并且不知不觉，一个小时过去了你感到净化了。'
		else
			pcs_horny = 0
			gs 'mood', 'raise', 'medium'
			church_moral += 1
			minut += 60
			if drugVars['cocaine_system'] > 1: drugVars['cocaine_system'] = drugVars['cocaine_system'] / 2
			'中殿仅由蜡烛和小夜灯照亮黑暗让人感到非常舒适，就像在子宫里一样。你有一种非常强烈地祈祷的冲动并且不知不觉，一个多小时过去了你感到重生并被净化了。'
		end
		gs 'stat'
		act '回到教堂门廊': gt 'city_church', 'Narthex'
	end

			gs 'flash', 'butt', 'indoors', 2, 2
			gs 'stat'
			act '回到教堂门廊': gt 'city_church', 'Narthex'
		end
	end
