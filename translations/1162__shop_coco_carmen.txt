# shop_coco_carmen
!! 2022/06/20

if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_coco_carmen', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '可可·卡门')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/coco/shop.jpg"></center>'
	*nl
	'如果你不介意在游客中挤来挤去，可可这里有年轻人想穿的一切。这里比G&M高一个档次，除非你买得起老城区的精品店，否则你还会花不少时间在这里选购衣服。'

	act '离开可可': minut += 3 & gt 'pav_commercial'
	act '查看连衣裙': minut += 5 & gt 'shop_coco_carmen', 'dress'
	act '查看其他服装': minut += 5 & gt 'shop_coco_carmen', 'outfits'
	act '查看钱包': minut += 5 & gt 'shop_coco_carmen', 'purses'
end


if $ARGS[0] = 'dress':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_coco_carmen', 'dress'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看可可卡门的连衣裙')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'coco_dress'
			gs 'shop_utils', 'init', 'set_steps', 6, Clothingstock mod 6
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_coco_carmen', 'start'
	end
end


if $ARGS[0] = 'outfits':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_coco_carmen', 'outfits'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看可可卡门的服装')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'coco_outfits'
			gs 'shop_utils', 'init', 'set_steps', 4, (Clothingstock / 6) mod 4
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_coco_carmen', 'start'
	end
end


if $ARGS[0] = 'purses':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_coco_carmen', 'purses'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看可可卡门的手提包')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'purses', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'coco'
			gs 'shop_utils', 'init', 'set_steps', 4, (Clothingstock / 6) mod 4
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_coco_carmen', 'start'
	end
end
