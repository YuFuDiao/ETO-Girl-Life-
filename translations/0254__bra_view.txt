		!!------------------------------------------------------------!!

		!!------------------------------------------------------------!!

		gs 'shop_utils', 'filter_builder', 'init', 'bra_view', 'view_grid', $shop_display['link']
			gs 'shop_utils',	'filter_builder', 'setup_number_sort'
			gs 'shop_utils',	'filter_builder', 'setup_quality_sort'
			gs 'shop_utils',	'filter_builder', 'setup_sort_direction'

			gs 'bra_view',		'filter_builder', 'setup_sport_filter'
		gs 'shop_utils', 'filter_builder', 'finish', 'mini'
		exit
	end

	if $ARGS[1] = 'setup_shop_filter':
		gs 'shop_utils', 'filter_builder', 'init', $loc, $loc_arg
			gs 'shop_utils', 'filter_builder', 'setup_number_sort'
			gs 'shop_utils', 'filter_builder', 'setup_quality_sort'
			gs 'shop_utils', 'filter_builder', 'setup_price_sort'
			gs 'shop_utils', 'filter_builder', 'setup_discount_sort'
			gs 'shop_utils', 'filter_builder', 'setup_sort_direction'
		gs 'shop_utils', 'filter_builder', 'finish', 'mini'

	end

		!!                     INDIVIDUAL HEADERS                     !!
		!!------------------------------------------------------------!!


	if $ARGS[1] = 'setup_sport_filter':
		gs 'shop_utils', 'filter_builder', 'setup_sport_filter', 'BraSport'
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                              FILTERS                               !!
	!!                             DEPRECATED                             !!
	!!====================================================================!!


!! Filters are automatically constructed when you build them using the filter_builder
!! you can filter using if func('shop_utils', 'filters', 'apply'): {do thing if all filters pass}


!! DEPRECATED
if $ARGS[0] = 'home_filter':
	gs 'shop_utils', 'filter', 'set_block'
		if func('bra_view', 'filter', 'sport') = 0: exit
	gs 'shop_utils', 'filter', 'set_pass'
	exit
end


!! DEPRECATED
if $ARGS[0] = 'shop_filter':
	gs 'shop_utils', 'filter', 'set_pass'
	exit
end


		!!------------------------------------------------------------!!
		!!                     INDIVIDUAL FILTER                      !!
		!!------------------------------------------------------------!!


!! DEPRECATED
if $ARGS[0] = 'filter':
	if $ARGS[1] = 'sport':
		if BraSport:
			if outfitfilter['sport'] >= 0: result = 1
		else
			if outfitfilter['sport'] <= 0: result = 1
		end
		exit
	end
	exit
end



	!!====================================================================!!
	!!                                                                    !!
	!!                                INIT                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'init':
	!! $ARGS[1]: init function (add_types, set_exceptions, add_to_sorted)

	if $ARGS[1] = 'add_types':
		!! $ARGS[2]: category to add

		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 4) = 'cats':			gs 'shop_utils', 'init', 'add_type', 'cats'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 10) = 'danilovich':	gs 'shop_utils', 'init', 'add_type', 'danilovich'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 6) = 'eroto':			gs 'shop_utils', 'init', 'add_type', 'eroto'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 11) = 'fashionista':	gs 'shop_utils', 'init', 'add_type', 'fashionista'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 2) = 'gm':				gs 'shop_utils', 'init', 'add_type', 'gm'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 5) = 'lusso':			gs 'shop_utils', 'init', 'add_type', 'lusso'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 9) = 'salacious':		gs 'shop_utils', 'init', 'add_type', 'salacious'
		exit
	end


	if $ARGS[1] = 'set_exceptions':
		!! shop_display_exceptions[$type + '_bras-' + number] = 1
		!!		i.e: shop_display_exceptions['gm_bras-11'] = 1

		shop_display_exceptions['gm_bras-11'] = 1
		shop_display_exceptions['gm_bras-16'] = 1
		shop_display_exceptions['gm_bras-18'] = 1
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                               SORTED                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'sorted':
	if $ARGS[1] = 'add':
		!! $ARGS[2]: bra type
		!! ARGS[3]: bra number
		!! ARGS[4]: index

		gs 'underwear_attributes', $ARGS[2] + '_bras', ARGS[3]

		if BraQuality = 0: exit

		gs 'shop_utils', 'sorted', 'add_to_number', $ARGS[2], ARGS[3], ARGS[4]
		gs 'shop_utils', 'sorted', 'add_to_quality', $ARGS[2], ARGS[3], BraQuality
		gs 'shop_utils', 'sorted', 'add_to_price', $ARGS[2], ARGS[3], BraPrice
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                              DISPLAY                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'display':

	if $ARGS[1] = 'grid_shop':
		if $ARGS[2] = 'header':
			gs 'bra_view', 'filter_builder', 'setup_shop_filter'
			exit
		end

		if $ARGS[2] = 'main':
			!! $ARGS[3]: link
			!! $ARGS[4]: type
			!! ARGS[5]: number
			!! ARGS[6]: discount

			gs 'underwear_attributes', $ARGS[4] + '_bras', ARGS[5]
			if func('shop_utils', 'filter', 'apply'):
				*p '<a href="exec: gt ''bra_view'', ''view_item'', ''<<$ARGS[3]>>'', ''<<$ARGS[4]>>'', <<ARGS[5]>>, <<ARGS[6]>>"><img height="250" src="<<$func(''$bra_image'', $ARGS[4], ARGS[5])>>"></a>'
			end
			exit
		end

		if $ARGS[2] = 'footer':
			exit
		end
		exit
	end

	if $ARGS[1] = 'grid_wardrobe':
		if $ARGS[2] = 'header':
			gs 'shop_utils', 'display', 'wardrobe_storage_unwanted_header', 'view_grid'
			gs 'bra_view', 'filter_builder', 'setup_home_filters'
			exit
		end

		if $ARGS[2] = 'main':
			!! $ARGS[3]: link
			!! $ARGS[4]: type
			!! ARGS[5]: number
			!! ARGS[6]: discount

			gs 'underwear_attributes', $ARGS[4] + '_bras', ARGS[5]
			if func('shop_utils', 'filter', 'apply'):
				*p '<a href="exec: gt ''bra_view'', ''view_item'', ''<<$ARGS[3]>>'', ''<<$ARGS[4]>>'', <<ARGS[5]>>, <<ARGS[6]>>"><img height="150" src="<<$func(''$bra_image'', $ARGS[4], ARGS[5])>>"></a>'
			end
			exit
		end

		if $ARGS[2] = 'footer':
			exit
		end
		exit
	end
	exit
end


!! DEPRECATED
if $ARGS[0] = 'display_grid_shop':
	gs 'shop_utils', 'display', 'grid_shop'
	exit
end


!! DEPRECATED
if $ARGS[0] = 'display_grid_wardrobe':
	gs 'shop_utils', 'display', 'grid_wardrobe'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                             VIEW LIST                              !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'view_grid':
	!! $ARGS[1] - action type (wardrobe, storage, unwanted, bathroom) -> $shop_display['link']
	menu_off = 1

	!! Should not be done here
	if func('bras', 'is_owned', 'gm_bras', 1) = 0: gs 'bras', 'add_item', 'gm_bras', 1

	!! Should not be done here
	if $pursepantytype <> '' or $pursebratype <> '':
		'<center><b>你把 ' + iif($pursepantytype = '', '', '内裤') + iif($pursepantytype = '' or $pursebratype = '', '', '和') + iif($pursebratype = '', '', '胸罩') + '从你的钱包里放回了衣柜。</b></center><br>'
		killvar 'pursepantytype'
		killvar 'pursepantynumber'
		killvar 'pursebratype'
		killvar 'pursebranumber'
	end

	if shop_display['init'] = 0:
		gs 'shop_utils', 'init', 'start', 'bras', $ARGS[1]
			gs 'bra_view', 'init', 'add_types', 'all'
			if $ARGS[1] = 'wardrobe' or $ARGS[1] = 'storage' or $ARGS[1] = 'unwanted':
				gs 'shop_utils', 'init', 'set_return', 'wardrobe', 'main'
			else
				gs 'shop_utils', 'init', 'set_return', $loc, $loc_arg
			end
		gs 'shop_utils', 'init', 'end'
	end

	act '返回': gt 'shop_utils', 'return'

	if $braworntype <> 'none':
		act '脱掉胸罩':
			gs 'bras', 'strip'
			gt 'bra_view', 'view_grid', $shop_display['link']
		end
	end

	gs 'shop_utils', 'display', 'grid_wardrobe'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                             VIEW ITEM                              !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'view_item':
	$shop_utils_view['link']		= $ARGS[1]		& !! action type (wardrobe, storage, unwanted, bathroom, shop, cheat)
	$shop_utils_view['type']		= $ARGS[2]		& !! bra type
	shop_utils_view['number']		= ARGS[3]		& !! bra index
	shop_utils_view['discount']		= ARGS[4]		& !! price modifier (0-100, taken as %) {OPTIONAL}

	menu_off = 1
	gs 'stat'

	'<center><img <<$set_imgh>> src="<<$func(''$bra_image'', $shop_utils_view[''type''], shop_utils_view[''number''])>>"></center>'
	gs 'underwear_attributes', $shop_utils_view['type'] + '_bras', shop_utils_view['number']
	!! *pl $func('$short_description', $shop_utils_view['type'], shop_utils_view['number'], CloStyle2)

	gs 'underwear_descriptions', 'bras'
	if underwear['pair'] > 0: '这件胸罩与内裤 #<<underwear[''pair'']>>组成一套。'

	if $shop_utils_view['link'] = 'shop' or $shop_utils_view['link'] = 'cheat': xgt 'bra_view', 'view_item_shop'
	if func('bras', 'is_wearing', $shop_utils_view['type'], shop_utils_view['number']):
		xgt 'bra_view', 'view_item_wearing'
	end
	if $shop_utils_view['link'] = 'wardrobe': xgt 'bra_view', 'view_item_wardrobe'
	if $shop_utils_view['link'] = 'bathroom': xgt 'bra_view', 'view_item_bathroom'
	if $shop_utils_view['link'] = 'storage': xgt 'bra_view', 'view_item_storage'
	if $shop_utils_view['link'] = 'unwanted': xgt 'bra_view', 'view_item_unwanted'

	act '返回': gt 'bra_view', 'view_grid', $shop_utils_view['link']
	exit
end


if $ARGS[0] = 'view_item_shop':
	act '离开': gt $loc, $loc_arg

	if func('bras', 'is_owned', $shop_utils_view['type'], shop_utils_view['number']):
		'你已经拥有这件物品。'
		exit
	end

	shop_utils_view['bra_discount'] = min(max(0, shop_utils_view['discount'] + func('shop_utils', 'get_discount', $shop_utils_view['type'], shop_utils_view['number'])), 100)
	shop_utils_view['bra_price'] = BraPrice

	shop_utils_view['price'] = shop_utils_view['bra_price'] * (100 - shop_utils_view['bra_discount']) / 100
	shop_utils_view['price'] = shop_utils_view['price'] / 50 * 50
	if shop_utils_view['price'] = shop_utils_view['bra_price']:
		$shop_utils_view['price_string'] = $func('money', 'string_price', shop_utils_view['price'])
	else
		$shop_utils_view['price_string'] = $func('wrap', 'neg s', $func('money', 'string_price', shop_utils_view['bra_price'])) + ' <b>' +  $func('money', 'string_price', shop_utils_view['price']) + '</b>'
		'现在 ' + shop_utils_view['bra_discount'] + '% 折扣' + iif(shop_utils_view['bra_discount'] <= 10, '', '！')
	end

	'价格：' + $shop_utils_view['price_string']

	if func('money', 'can_afford', shop_utils_view['price']) = 0:
		'你买不起这件胸罩。'
	else
		act '购买 (<<$shop_utils_view[''price_string'']>>)':
			!! This should not work
			gs 'money', 'pay', shop_utils_view['price']
			gs 'bras', 'add_item', $shop_utils_view['type'], shop_utils_view['number']
			gt $loc, $loc_arg
		end
	end

	if underwear['pair'] > 0:
		if func('panties', 'is_owned', $shop_utils_view['type'], underwear['pair']):
			'你拥有配套的内裤，买下这件胸罩即可凑成整套。'
		else
			'你没有配套的内裤，你可以把这件胸罩和它们作为整套一起购买。'

			gs 'underwear_attributes', $shop_utils_view['type'] + '_panties', underwear['pair']
			shop_utils_view['pan_discount'] = min(max(0, shop_utils_view['discount'] + func('shop_utils', 'get_discount', $shop_utils_view['type'], underwear['pair'])), 100)
			shop_utils_view['pan_price'] = PanPrice
			shop_utils_view['set_price'] = 9 * (shop_utils_view['bra_price'] * (100 - shop_utils_view['bra_discount']) + shop_utils_view['pan_price'] * (100 - shop_utils_view['pan_discount'])) / 1000
			shop_utils_view['set_price'] = shop_utils_view['set_price'] / 50 * 50
			shop_utils_view['base_price'] = shop_utils_view['bra_price'] + shop_utils_view['pan_price']
			$shop_utils_view['price_string'] = $func('wrap', 'neg s', $func('money', 'string_price', shop_utils_view['base_price'])) + ' <b>' +  $func('money', 'string_price', shop_utils_view['set_price']) + '</b>'

			'套装价格：' + $shop_utils_view['price_string']

			if func('money', 'can_afford', shop_utils_view['set_price']) = 0:
				'你买不起这套。'
			else
				act '购买整套 (<<$shop_utils_view[''price_string'']>>)':
					!! Reset the underwear attributes to ensure that underwear['pair'] gets reset to the correct value
					gs 'underwear_attributes', $shop_utils_view['type'] + '_bras', shop_utils_view['number']
					gs 'money', 'pay', shop_utils_view['set_price']
					gs 'bras', 'add_item', $shop_utils_view['type'], shop_utils_view['number']
					gs 'panties', 'add_item', $shop_utils_view['type'], underwear['pair']
					gt $loc, $loc_arg
				end
			end
		end
	end
	exit
end


if $ARGS[0] = 'view_item_wearing':
	act '返回': gt 'bra_view', 'view_grid', $shop_utils_view['link']
	'你正穿着这件胸罩。'
	if $shop_utils_view['link'] = 'wardrobe':
		act '脱掉你的胸罩':
			gs 'bras', 'strip'
			gs 'shop_utils', 'cleanup'
			gt 'wardrobe', 'main'
		end
	end
	exit
end


if $ARGS[0] = 'view_item_wardrobe':
	act '返回': gt 'bra_view', 'view_grid', $shop_utils_view['link']
	act '穿上': xgt 'bra_view', 'view_item_wear_single'

	if underwear['pair'] > 0:
		if func('panties', 'is_owned', $shop_utils_view['type'], underwear['pair']) = 0:
			'你没有配套的内裤来组成这套。'
		else
			'你拥有这些内裤，可以穿这套。'
			act '穿上整套': xgt 'bra_view', 'view_item_wear_pair'
		end
	end
	gs 'bra_view', 'storage_options'
	exit
end

if $ARGS[0] = 'view_item_bathroom':
	act '返回': gt $loc, $loc_arg

	if PCloBra = 1:
		*pl '<font color=red><b>这件上衣没法配胸罩</b></font>'
	elseif hypnoBra <= 0:
		act '穿上':
			gs 'bras', 'wear', $shop_utils_view['type'], shop_utils_view['number']
			gs 'shop_utils', 'cleanup'
			gt $loc, $loc_arg
		end
	end

	if hypnoBra > 0:
		gs 'bra_view', 'view_item_wear_single_hypno', 1
		gt $loc, $loc_arg
	end

	if underwear['pair'] > 0:
		if func('panties', 'is_owned', $shop_utils_view['type'], underwear['pair']) = 0:
			'你没有配套的内裤来组成这套。'
		else
			'你拥有这些内裤，可以穿这套。'
			act '穿上整套':
				gs 'bra_view', 'view_item_wear_pair', 1
				gt $loc, $loc_arg
			end
		end
	end

	exit
end



if $ARGS[0] = 'view_item_storage':
	act '返回': gt 'bra_view', 'view_grid', $shop_utils_view['link']
	'这件胸罩存放在储物间里。'
	gs 'bra_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_unwanted':
	act '返回': gt 'bra_view', 'view_grid', $shop_utils_view['link']
	'这件胸罩被标记为不需要。'
	gs 'bra_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_wear_single':
	*nl
	if PCloBra = 1:
		*pl '<font color=red><b>这件上衣没法配胸罩</b></font>'
		act '返回': gt 'bra_view', 'view_item', $shop_utils_view['action'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		exit
	end

	if hypnoBra > 0: xgt 'bra_view', 'view_item_wear_single_hypno'

	gs 'bras', 'wear', $shop_utils_view['type'], shop_utils_view['number']
	gs 'shop_utils', 'cleanup'
	gt 'wardrobe', 'main'
end


if $ARGS[0] = 'view_item_wear_single_hypno':
	'你拿起胸罩，想着试穿一下……但你其实不喜欢穿胸罩。'
	gs 'willpower', 'misc', 'resist', 'easy'
	if cheatVars['willpower'] = 0: will_cost += hypnoTime
	if pcs_willpwr < will_cost:
		act '还是穿上它们' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '还是穿上它们' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'pay', 'resist'
			gs 'stat'

			gs 'bras', 'wear', $shop_utils_view['type'], shop_utils_view['number']
			gs 'shop_utils', 'cleanup'
			if ARGS[1] = 0: gt 'wardrobe', 'main'
		end
	end
	act '返回': gt 'bra_view', 'view_grid', $shop_utils_view['link']
	exit
end


if $ARGS[0] = 'view_item_wear_pair':
	*nl
	if PCloBra = 1:
		*pl '<font color=red><b>这件上衣没法配胸罩</b></font>'
		act '返回': gt 'bra_view', 'view_item', $shop_utils_view['action'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		exit
	end
	if PCloPanties = 1:
		*pl '<font color=red><b>你不能在这套服装里穿内裤。</b></font>'
		act '返回': gt 'bra_view', 'view_item', $shop_utils_view['action'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		exit
	end

	if hypnoBra > 0 or hypnoPanty > 0: xgt 'bra_view', 'view_item_wear_pair_hypno'

	gs 'bras', 'wear', $shop_utils_view['type'], shop_utils_view['number']
	gs 'panties', 'wear', $shop_utils_view['type'], underwear['pair']
	gs 'shop_utils', 'cleanup'
	if ARGS[1] = 0: gt 'wardrobe', 'main'
end


if $ARGS[0] = 'view_item_wear_pair_hypno':
	if hypnoBra > 0 and hypnoPanty > 0:
		'你拿起胸罩和内裤，想着试穿一下…但其实你不太喜欢穿内衣。'
		gs 'willpower', 'misc', 'resist'
	elseif hypnoBra > 0:
		'你拿起胸罩，想着试穿一下……但你其实不喜欢穿胸罩。'
		gs 'willpower', 'misc', 'resist', 'easy'
	else
		'你拿起内裤，想着试穿一下……但你其实不喜欢穿内裤。'
		gs 'willpower', 'misc', 'resist', 'easy'
	end

	if cheatVars['willpower'] = 0: will_cost += hypnoTime
	if pcs_willpwr < will_cost:
		act '还是穿上它们' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '还是穿上它们' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'pay', 'resist'
			gs 'stat'
			gs 'bras', 'wear', $shop_utils_view['type'], shop_utils_view['number']
			gs 'panties', 'wear', $shop_utils_view['type'], underwear['pair']
			gs 'shop_utils', 'cleanup'
			gt 'wardrobe', 'main'
		end
	end
	act '返回': gt 'bra_view', 'view_grid', $shop_utils_view['link']
	exit
end


if $ARGS[0] = 'storage_options':
	if func('bras', 'is_owned', $shop_utils_view['type'], shop_utils_view['number']) = 0: exit

	if func('bras', 'in_wardrobe', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到衣柜':
			gs 'bras', 'move_to_wardrobe', $shop_utils_view['type'], shop_utils_view['number']
			gt 'bra_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end

	if func('bras', 'is_immutable', $shop_utils_view['type'], shop_utils_view['number']): exit

	if func('bras', 'in_storage', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到存储':
			gs 'bras', 'move_to_storage', $shop_utils_view['type'], shop_utils_view['number']
			gt 'bra_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	if func('bras', 'in_unwanted', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到不需要':
			gs 'bras', 'move_to_unwanted', $shop_utils_view['type'], shop_utils_view['number']
			gt 'bra_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	act $func('wrap', 'neg', '扔掉这件胸罩'):
		gs 'bras', 'remove_item', $shop_utils_view['type'], shop_utils_view['number']
		gt 'bra_view', 'view_grid', $shop_utils_view['link']
	end
	exit
end
