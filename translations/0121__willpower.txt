# willpower

!! To use it call it with gs 'willpower', 'action', 'type'
!! Action: Determines what action sveta is resisting, forcing somebody to do or forcing herself to do
!! List of actions:
!!		'voyeur', 'flash', 'mast', 'hj', 'bj', 'cuni', 'sex', 'anal', 'group', 'gangbang', 'orgy', 'prostitution', 'swallow', 'cum_inside', 'cum_inside_anal', 'rape',
!!		'kiss', 'humiliation', 'footjob', 'titjob', 'trib', 'rimming', 'BDSM', 'pee', 'cum_outside', 'drink', 'drugs', 'crime', 'exhib', 'sleep', 'chore'
!!		there is also 'foreplay' for things like touching and fondling
!!		and two special cases 'misc' and 'skill'
!!		'chore' is meant to be used for menial tasks that should not greatly affect willpower max, only drain from the pool
!! Type: Determines what kind of willpower check sveta needs to do ('resist', 'force', 'self').
!!		-'resist' is used if sveta is resisting to do the 'action' somebody else wants her to do.
!!		-'force' is used if sveta forces somebody else to do the 'action'.
!!		-'self' should be used if sveta forces herself to do the 'action' (this should be used only in rare occasions)

!! Example 1: Sveta is forcing somebody to perform a blowjob: gs 'willpower', 'bj', 'force'
!! Example 2: Sveta is resisting doing drugs: gs 'willpower', 'drugs', 'resist'

!! You can modify the difficulty of the check with adding an additional argument at the end. You can either add an 'easy' or a 'hard' to modify the difficulty.
!! Example 1: Sveta is resisting doing drugs, a weak outcast is pressuring her to use them: gs 'willpower', 'drugs', 'resist', 'easy'
!! Example 2: Sveta is resisting doing drugs, a group of gopniks are pressuring her to use them: gs 'willpower', 'drugs', 'resist', 'hard'

!! There are two special cases: 'misc' and 'skill'

!! 'skill': Bridging a gap in skill for doing something, for succeeding when not quite meeting the required skill/attribute
!! Example for 'skill': gs 'willpower', 'skill', 'makupskl', 'resist', 'easy' for a willpower check regarding something with pcs_makupskl (you can also apply 'easy' or 'hard' difficulty to a skill check)
!! Example for 'skill_base': gs 'willpower', 'skill_base', 'jab', 'force' for a willpower check regarding something with jab_lvl

!! 'misc': Use this 'action' if no other 'action' fits the scene. You can set three difficulty levels for this check 'easy', 'medium', 'hard' (Difficulty is only for 'misc' checks)
!! To call 'misc' do the following: gs 'willpower', 'misc', 'type', 'difficulty'
!! Example 1 for 'misc': Sveta wants to bring herself to slap a weak looking nerd: gs 'willpower', 'misc', 'self', 'easy'
!! Example 2 for 'misc': Sveta wants to force Artem to go to Vitek and call him an asshole: gs 'willpower', 'misc', 'force', 'hard'

!! After Sveta has taken the willpower action the cost must be paid with: gs 'willpower', 'pay', 'type'(, 'chore')
!! Example: Sveta has to pay for resisting to do a blowjob: gs 'willpower', 'pay', 'resist'
!! Example: Sveta is doing a chore, which should not increase willpower: gs 'willpower', 'pay', 'self ', 'chore'

!! WARNING: When adding new options include gs 'willpower', 'difficulty', $ARGS[1], $ARGS[2] at the end of the ARGS to apply easy or hard.
!! WARNING: If you have two or more actions that need willpower at the same time, you have to recalculate the willpower cost before the costs are paid (see Code Example 2)

!{ Code Example 1 (one action needs willpower)

	gs 'willpower', 'misc', 'self', 'hard'
	if pcs_willpwr < will_cost:
		act '踢他的蛋蛋' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '踢他的蛋蛋' + $func('willpower', 'get_willcost_string'):
			*clr & cla
			gs 'willpower', 'pay', 'self'
			gs 'stat'

			'事件文本'

			act '下一步行动': gt 'farawayistan'
		end
	end
!}

!! Multiplier to make willpower cost more. Use if all total costs need adjusting.
if $ARGS[0] = 'difficulty':
	if cheatVars['willpower'] = 1:
		will_cost = 0
		exit
	end

	elseif $ARGS[2] = 'hard':
		will_cost = will_cost * 9 / 4
		if will_cost < 7: will_cost = 7
	else
		will_cost = will_cost * 3 / 2
		if will_cost < 6: will_cost = 6
	end

		will_cost = will_cost * (200 - pcs_persuas) / 100
	elseif $ARGS[1] = 'resist':
		will_cost = will_cost * (200 - pcs_sprt) / 100
		!! Doormat trait: resisting is harder
		if trait_vars['doormat'] > 0:
			will_cost = will_cost * (100 + 15 * trait_vars['doormat']) / 100
		end
	end

	!! Archetype scaling, driven by arch_effects (see archetypes.qsrc apply_effects).
	!! Effect percentages are raw (6-30%); the resulting willpower swing is clamped to 3-15 points
	!! so it stays meaningful without swinging costs too wildly on very cheap/expensive base costs.
	arch_delta = 0
	if $ARGS[3] = 'sexual':
		if $ARGS[1] = 'resist':
			arch_delta = will_cost * arch_effects['wp_sexual_effect'] / 100
		else
			arch_delta = -will_cost * arch_effects['wp_sexual_effect'] / 100
		end
	!! Resisting rape: prude makes it easier, bimbo has no effect
	elseif $ARGS[3] = 'rape':
		if $ARGS[1] = 'resist':
			arch_delta = will_cost * min(0, arch_effects['wp_sexual_effect']) / 100
		end
	elseif $ARGS[3] = 'substances':
		if $ARGS[1] <> 'resist':
			arch_delta = -will_cost * arch_effects['wp_substances_effect'] / 100
		end
	elseif $ARGS[3] = 'crime':
		if $ARGS[1] <> 'resist':
			arch_delta = -will_cost * arch_effects['wp_crime_effect'] / 100
		end
	end
	if arch_delta > 0:
		will_cost += func('math', 'int_clamp', arch_delta, 3, 15)
	elseif arch_delta < 0:
		will_cost -= func('math', 'int_clamp', -arch_delta, 3, 15)
	end
	killvar 'arch_delta'

	!! Difficulty multiplier
	if cheatVars['wp_cost_opt'] <> 0:
		will_cost = func('_difficulty', 'get_multiplied', cheatVars['wp_cost_opt'], will_cost, cheatVars['wp_cost_mult'])
	end
	exit
end

	!! Same as with BDSM concerning 'force' and 'resist'
	!! pee_check = (stat['pee'] + stat['pee_give']) / 2

	if $ARGS[1] = 'self':
		if (stat['pee'] + stat['pee_give']) / 2 <= 90:
			will_cost = (100 - ((stat['pee'] + stat['pee_give']) / 2) + will_calc - will_arousal_mod)/10
		else
			will_cost = (10 + will_calc - will_arousal_mod)/10
		end
	elseif $ARGS[1] = 'force':
		if stat['pee_give'] <= 40:
			will_cost = (160 - will_enforced - stat['pee_give'] + will_calc - will_arousal_mod)/10
		else
			will_cost = (120 - will_enforced + will_calc - will_arousal_mod)/10
		end
	else
		if stat['pee'] <= 90:
			will_cost = (10 + stat['pee'] + will_calc + will_arousal_mod)/10
		else
			will_cost = (100 + will_calc + will_arousal_mod)/10
		end
	end
	gs 'willpower', 'difficulty', $ARGS[1], $ARGS[2], 'sexual'
	exit
end


!! Calculate willpower cost based on PC skill value
!! Example Function Call: gs 'willpower', 'skill', 'skillname', 'self/force/resist', 'easy/medium/hard', 'fetish;list'(optional)
if $ARGS[0] = 'skill':
	!! Argument fixing could be deprecated starting with v0.10.0.0
	!! Remains for now to not break mods
	if $mid($ARGS[1], len($ARGS[1])-3) = '_lvl':
		$ARGS[1] = $mid($ARGS[1], 1, len($ARGS[1])-4)
		gs 'willpower', 'skill_base', $ARGS[1], $ARGS[2], $ARGS[3], $ARGS[4]
		exit
	end
	if $mid($ARGS[1], 1, 4) = 'pcs_': $ARGS[1] = $mid($ARGS[1], 5)

	gs 'willpower', 'calc', $ARGS[4]

	if dyneval('result = pcs_<<$ARGS[1]>>') >= 100:
		will_cost = will_calc / 10
	else
		dynamic 'will_cost = (100 - pcs_<<$ARGS[1]>> + will_calc) / 10'
	end
	gs 'willpower', 'difficulty', $ARGS[2], $ARGS[3]
	exit
end


!! Calculate willpower cost based on PC base skill value
!! Example Function Call: gs 'willpower', 'skill_base', 'skillname', 'self/force/resist', 'easy/medium/hard', 'fetish;list'(optional)
if $ARGS[0] = 'skill_base':
	!! Argument fixing could be deprecated starting with v0.10.0.0
	!! Remains for now to not break mods
	if $mid($ARGS[1], 1, 4) = 'pcs_':
		$ARGS[1] = $mid($ARGS[1], 5)
		gs 'willpower', 'skill', $ARGS[1], $ARGS[2], $ARGS[3], $ARGS[4]
		exit
	end
	if $mid($ARGS[1], len($ARGS[1])-3) = '_lvl': $ARGS[1] = $mid($ARGS[1], 1, len($ARGS[1])-4)

	gs 'willpower', 'calc', $ARGS[4]

	if dyneval('result = <<$ARGS[1]>>_lvl') >= 100:
		will_cost = will_calc / 10
	else
		dynamic 'will_cost = (100 - <<$ARGS[1]>>_lvl + will_calc) / 10'
	end
	gs 'willpower', 'difficulty', $ARGS[2], $ARGS[3]
	exit
end


if $ARGS[0] = 'rape':
	gs 'willpower', 'calc', $ARGS[3]

	if $ARGS[1] = 'self':
		if missCum >= timeTresh: will_calc = 0
		will_cost = (100 + will_calc - will_arousal_mod)/10
	elseif $ARGS[1] = 'force':
		will_cost = (200 - will_enforced * 2 + will_calc - will_arousal_mod)/10
	else
		if missCum >= timeTresh: will_calc += 100
		if stat['rape_count'] = 0:
			will_cost = (10 + will_calc + will_arousal_mod)/10
		else
			will_cost = (100 + stat['rape_count'] + will_calc + will_arousal_mod)/10
		end
	end
	gs 'willpower', 'difficulty', $ARGS[1], $ARGS[2], 'rape'
	exit
end


if $ARGS[0] = 'drink':
	gs 'willpower', 'calc', $ARGS[3]

	if $ARGS[1] = 'self':

	elseif $ARGS[1] = 'force':
		will_cost = (100 - will_enforced - drugVars['alcohol_exp'] * 30 + will_calc)/10
	else

		elseif alko > 6:

		elseif alko > 3:
			will_cost = (60 + drugVars['alcohol_exp'] * 30 + will_calc)/10
		else
			will_cost = (10 + drugVars['alcohol_exp'] * 20 + will_calc)/10
		end
	end
	gs 'willpower', 'difficulty', $ARGS[1], $ARGS[2], 'substances'
	exit
end


!! Drug needs already included in the will_calc so just a small base cost here

	gs 'willpower', 'calc', $ARGS[3]

	if $ARGS[1] = 'self':
		will_cost = (will_calc + 100 - trait_vars['addictive_personality'] * 30) / 10
	elseif $ARGS[1] = 'force':
		will_cost = (will_calc + 100 - will_enforced - trait_vars['addictive_personality'] * 10) / 10
	else
		will_cost = (will_calc + max(10, trait_vars['addictive_personality'] * 30)) / 10
	end
	gs 'willpower', 'difficulty', $ARGS[1], $ARGS[2], 'substances'
	exit
end

!! this is to force you to stay wake so will cost nothing if you have more than 5 sleep, it has no base cost as there will always be a sleep cost in calc

	gs 'willpower', 'calc', $ARGS[3]
	will_cost = 0
	if pcs_sleep <= 5: will_cost = will_calc/10
	!! Not sure if this one belongs. It doesn''t look like it''s used anywhere.

	exit

	gs 'willpower', 'calc', $ARGS[3]

	if $ARGS[1] = 'force':
		if $ARGS[2] = 'easy':
			will_cost = (60 - will_enforced + will_calc)/10
		elseif $ARGS[2] = 'hard':
			will_cost = (90 - will_enforced + will_calc)/10
		else
			will_cost = (70 - will_enforced + will_calc)/10
		end
	else
		if $ARGS[2] = 'easy':
			will_cost = (10 + will_calc)/10
		elseif $ARGS[2] = 'hard':
			will_cost = (40 + will_calc)/10
		else
			will_cost = (20 + will_calc)/10
		end
	end
	gs 'willpower', 'difficulty', $ARGS[1], $ARGS[2]
	exit
end


!! modifiers for drink an drugs
if $ARGS[0] = 'dnd':
	will_dnd = 0
	if alko > 10:
		will_dnd += 160
	elseif alko > 6:
		will_dnd += 80
	elseif alko > 3:
		will_dnd += 30
	end

	!! cigarettes
	if drugVars['cigarettes_need'] > 10: will_dnd += 10
	!! weed
	if drugVars['weed_high'] > 0: will_dnd -= 10
	!! amphetamines
	if drugVars['amphetamine_withdrawl'] > 2:
		will_dnd += 40
	elseif drugVars['amphetamine_high'] > 0:
		will_dnd += 80
	end
	!! heroine
	if drugVars['heroin_need'] > 0:
		will_dnd += 100
	elseif drugVars['heroin_high'] > 0:
		will_dnd += 60
	end
	!! cocaine
	if drugVars['cocaine_addict'] = 1:
		will_dnd += 60
	elseif drugVars['cocaine_system'] < 0:
		will_dnd -= 20
	end
	exit
end
