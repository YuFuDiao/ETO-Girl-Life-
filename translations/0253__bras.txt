if $ARGS[0] = 'get_total':
	if $mid($ARGS[1], 1, 4) = 'cats':
		result = 41
	elseif $mid($ARGS[1], 1, 10) = 'danilovich':
		result = 8
	elseif $mid($ARGS[1], 1, 5) = 'eroto':
		result = 27
	elseif $mid($ARGS[1], 1, 11) = 'fashionista':
		result = 42
	elseif $mid($ARGS[1], 1, 2) = 'gm':
		result = 19
	elseif $mid($ARGS[1], 1, 5) = 'lusso':
		result = 33
	elseif $mid($ARGS[1], 1, 9) = 'salacious':
		result = 37
	end
	exit
end


if $ARGS[0] = 'totals':
	total = func('bras', 'get_total', $ARGS[1])
	exit
end


if $ARGS[0] = 'sum':
	result = 0
	result += func('bras', 'sum_inner', 'lusso')
	result += func('bras', 'sum_inner', 'cats')
	result += func('bras', 'sum_inner', 'salacious')
	result += func('bras', 'sum_inner', 'fashionista')
	result += func('bras', 'sum_inner', 'gm')
	result += func('bras', 'sum_inner', 'eroto')
	result += func('bras', 'sum_inner', 'danilovich')
	if $ARGS[1] = 'dresser':
		if $braworntype <> 'none' and result > 0: result -= 1
	end
	exit
end


if $ARGS[0] = 'sum_inner':
	result = 0
	bras_i = 1
	bras_max_i = arrsize('<<$ARGS[1]>>_bras')
	:loopsumbras
		if func('bras', 'in_wardrobe', $ARGS[1], bras_i): result += 1
		bras_i += 1
	if bras_i < bras_max_i : jump 'loopsumbras'
	killvar 'bras_i'
	killvar 'bras_max_i'
	exit
end


!! $ARGS[1] set to outfit type
!! ARGS[2] for outfit number
!! ARGS[3] to skip attribute set
if $ARGS[0] = 'not_wear_reason':
	if arrsize('ARGS') = 1: $ARGS[1] = $braworntype
	if arrsize('ARGS') = 2: ARGS[2] = brawornnumber
	if arrsize('ARGS') = 3: gs 'underwear_attributes', $ARGS[1] + '_bras', ARGS[2]

	$result = ''
	if func('bras', 'is_immutable', $ARGS[1], ARGS[2]): exit
	if func('bras', 'is_owned', $ARGS[1], ARGS[2]) = 0:
		$result = 'not_owned'
		exit
	end
	if func('bras', 'in_wardrobe', $ARGS[1], ARGS[2]) = 0:
		$result = 'not_in_wardrobe'
		exit
	end
	if func('bras', 'is_strength_low', $ARGS[1], ARGS[2]):
		$result = 'low_strength'
		exit
	end
	if func('bras', 'is_hypno_approved', $ARGS[1], ARGS[2]) = 0:
		$result = 'hypno'
		exit
	end
	exit
end


if $ARGS[0] = 'can_wear':
	if arrsize('ARGS') = 1: $ARGS[1] = $braworntype
	if arrsize('ARGS') = 2: ARGS[2] = brawornnumber
	if arrsize('ARGS') = 3: gs 'underwear_attributes', $ARGS[1] + '_bras', ARGS[2]

	result = ($func('bras', 'not_wear_reason', $ARGS[1], ARGS[2], 'attributes_set') = '')
	exit
end

	exit
end

if $ARGS[0] = 'add_item':
	if $ARGS[1] = '': exit
	if ARGS[2] = 0: exit

	gs 'underwear_attributes', $ARGS[1] + '_bras', ARGS[2]
	if BraQuality = 0: exit

	dynamic "
		<<$ARGS[1]>>_bras[<<ARGS[2]>>] = 1
		<<$ARGS[1]>>_brasS[<<ARGS[2]>>] = 0
		<<$ARGS[1]>>_bras_dirt[<<ARGS[2]>>] = 0
		<<$ARGS[1]>>_bras_h[<<ARGS[2]>>] = BraStrength
	"
	exit
end


if $ARGS[0] = 'remove_item':
	if $ARGS[1] = '': $ARGS[1] = $braworntype
	if ARGS[2] = 0: ARGS[2] = brawornnumber

	if $ARGS[1] = '' or $ARGS[1] = 'none': exit

	if $ARGS[1] = 'gm' and ARGS[2] = 1:
		if $ARGS[1] = $braworntype and ARGS[2] = brawornnumber:
			gs 'bras', 'strip'
		end
		exit
	end

	dynamic "
		<<$ARGS[1]>>_bras[<<ARGS[2]>>] = 0
		<<$ARGS[1]>>_brasS[<<ARGS[2]>>] = 0
		<<$ARGS[1]>>_bras_dirt[<<ARGS[2]>>] = 0
		<<$ARGS[1]>>_bras_h[<<ARGS[2]>>] = 0
	"

	if $ARGS[1] = $braworntype and ARGS[2] = brawornnumber:
		gs 'bras', 'strip_code'
		gs 'outfit', 'set_derived_vars'
		$lastwornbratype = 'none'
		lastwornbranumber = 0
	end
	exit
end

	exit
end

if $ARGS[0] = 'reset_BraVars':
	!! BraStyle = 0
	BraMaterial = 0
	BraType = 0
	BraFun = 0
	BraQuality = 0
	BraThinness = 0
	BraCover = 0  & !! intentionally 0
	BraSport = 0
	!! BraSwim = 0
	BraPrice = 0
	BraDirt = 0
	BraStrength = 0
	BraMaxStrength = 0
	underwear['pair'] = 0
	exit
end


if $ARGS[0] = 'reset_PBraVars':
	!! PBraStyle = 0
	PBraMaterial = 0
	PBraType = 0
	PBraFun = 0
	PBraQuality = 0
	PBraThinness = 0
	PBraCover = 4  & !! intentionally 4
	PBraSport = 0
	!! PBraSwim = 0
	PBraPrice = 0
	PBraDirt = 0
	PBraStrength = 0
	PBraMaxStrength = 0
	PBraBorrowed = 0

if $ARGS[0] = 'wear':
	if $ARGS[1] = '' or $ARGS[1] = 'last_worn':
		if lastwornunderwear = 2:
			gs 'underwear_bodysuits', 'wear', 'last_worn'
			exit
		end
		$ARGS[1] = $lastwornbratype
		ARGS[2] = lastwornbranumber
	end
	if $ARGS[1] = '' or $ARGS[1] = 'none': exit

	gs 'bras', 'strip'

	gs 'underwear_attributes', $ARGS[1] + '_bras', ARGS[2]

	if BraQuality = 0:
		msg '错误：胸罩 "<<$ARGS[1]>>_bras[<<ARGS[2]>>]" 不存在'
		exit
	end

	if arrpos('$ARGS', 'check') > 0:
		$temp_not_wear_reason = $func('bras', 'not_wear_reason', $ARGS[1], ARGS[2], 'no_init')
		if $temp_not_wear_reason <> '' and $temp_not_wear_reason <> 'hypno':
			if $temp_not_wear_reason = 'not_owned':
				msg '你不再拥有这件胸罩了'
			elseif $temp_not_wear_reason = 'not_in_wardrobe':
				msg '这件胸罩不在你的衣柜里'
			elseif $temp_not_wear_reason = 'low_strength':
				msg '这件胸罩已经穿旧了'
			elseif $temp_not_wear_reason = 'hypno':
				msg '你现在不想穿这件胸罩。'
			end
			killvar '$temp_not_wear_reason'
			exit
		end
		killvar '$temp_not_wear_reason'
	end

	$braworntype = $ARGS[1]
	brawornnumber = ARGS[2]
	underwear['type'] = 0

	dynamic "
		<<$braworntype>>_brasS[<<brawornnumber>>] = 0
	"

	!! PBraStyle = BraStyle
	PBraMaterial = BraMaterial
	PBraType = BraType
	PBraFun = BraFun
	PBraQuality = BraQuality
	PBraThinness = BraThinness
	PBraCover = BraCover
	PBraSport = BraSport
	!! PBraSwim = BraSwim
	PBraPrice = BraPrice
	PBraDirt = BraDirt
	PBraStrength = BraStrength
	PBraMaxStrength = BraMaxStrength

	if arrpos('$ARGS', 'borrowed') >= 0:
		PBraBorrowed = 1
		PBraDirt = 0
		PBraStrength = PBraMaxStrength
	end

	gs 'outfit', 'set_derived_vars'
	exit
end
