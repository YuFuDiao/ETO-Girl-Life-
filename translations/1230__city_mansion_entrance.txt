# city_mansion_entrance

!! Delayed construction variant of the original mansion entrance.
!! The original location flow is preserved; only construction and furnishing timing were refactored.
!! Changes made here:
!! - construction no longer completes instantly; it now advances through staged phases
!! - staged, weekday-only mansion construction
!! - the build path is split into foundation/utilities -> shell/finishing -> room shell -> room decoration
!! - parallel room furnishing tasks
!! - completion SMS notifications for each finished phase, delivered via telefon_schedules
!! - dedicated in-game sender identity for construction updates: kirsinova_construction
!! Why:
!! - keep the location content 1:1 with the original while making the build process feel slower and more alive
!! - SMS delivery uses the game´s native SetInSMSSchedule mechanism via telefon_schedules.qsrc
!! - telefon_schedules checks on each day-change whether any construction is due and schedules
!!   a single SMS handler for a random time that day (09:00-19:00)
!! - the SMS handler performs state changes and sends the completion SMS in one pass
!! - construction_tick remains as a fallback for state changes when the player visits the location
!! Construction and furnishing only advance on working days (Monday-Friday).

if $ARGS[0] = 'calc_workday_ready_day':
	!! Returns the daystart when the requested number of working days has elapsed.
	!! ARGS[1] = start daystart
	!! ARGS[2] = number of working days to wait
	temp_mansion_ready_day = ARGS[1]
	temp_mansion_work_days = ARGS[2]
	if temp_mansion_work_days < 1: temp_mansion_work_days = 1
	:matryona_mansion_workday_loop
		temp_mansion_ready_day += 1
		temp_mansion_weekday = func('time', 'get_week_from_daystart', temp_mansion_ready_day)
		if temp_mansion_weekday <= 5:
			temp_mansion_work_days -= 1
		end
		if temp_mansion_work_days > 0: jump 'matryona_mansion_workday_loop'
	result = temp_mansion_ready_day
	killvar 'temp_mansion_ready_day'
	killvar 'temp_mansion_work_days'
	killvar 'temp_mansion_weekday'
	exit
end

if $ARGS[0] = 'calc_workdays_until_day':
	!! Returns how many working days remain until the given daystart is reached.
	!! ARGS[1] = target daystart
	temp_mansion_calc_target = ARGS[1]
	temp_mansion_calc_remaining = 0
	if temp_mansion_calc_target > daystart:
		temp_mansion_calc_day = daystart
		:matryona_mansion_remaining_loop
			temp_mansion_calc_day += 1
			if temp_mansion_calc_day <= temp_mansion_calc_target:
				temp_mansion_calc_weekday = func('time', 'get_week_from_daystart', temp_mansion_calc_day)
				if temp_mansion_calc_weekday <= 5:
					temp_mansion_calc_remaining += 1
				end
				jump 'matryona_mansion_remaining_loop'
			end
	end
	result = temp_mansion_calc_remaining
	killvar 'temp_mansion_calc_target'
	killvar 'temp_mansion_calc_remaining'
	killvar 'temp_mansion_calc_day'
	killvar 'temp_mansion_calc_weekday'
	exit
end

if $ARGS[0] = 'check_construction_due':
	!! Returns 1 if any construction completion is due today.
	!! Called by telefon_schedules on day-change to decide whether to schedule an SMS.
	result = 0
	if matryona_mansion_phase_pending = 1 and daystart >= matryona_mansion_phase_ready_day:
		result = 1
	end
	if matryona_mansion_sauna_pending = 1 and daystart >= matryona_mansion_sauna_ready_day:
		result = 1
	end
	if matryona_mansion_pool_pending = 1 and daystart >= matryona_mansion_pool_ready_day:
		result = 1
	end
	temp_mansion_check_idx = 3
	:matryona_mansion_due_room_loop
		if temp_mansion_check_idx <= 12 and result = 0:
			if matryona_mansion_room_pending[temp_mansion_check_idx] = 1 and daystart >= matryona_mansion_room_ready_day[temp_mansion_check_idx]:
				result = 1
			end
			temp_mansion_check_idx += 1
			jump 'matryona_mansion_due_room_loop'
		end
	killvar 'temp_mansion_check_idx'
	exit
end

if $ARGS[0] = 'queue_build_phase':
	!! Queue a delayed construction phase.
	!! ARGS[1] = 'foundation' | 'shell' | 'sauna' | 'pool'
	!! ARGS[2] = working-day duration
	if $ARGS[1] = 'foundation':
		if matryona_mansion_phase_pending = 1: exit
		matryona_mansion_phase_pending = 1
		matryona_mansion_phase_ready_day = func('city_mansion_entrance', 'calc_workday_ready_day', daystart, ARGS[2])
	elseif $ARGS[1] = 'shell':
		if matryona_mansion_phase_pending = 1: exit
		matryona_mansion_phase_pending = 1
		matryona_mansion_phase_ready_day = func('city_mansion_entrance', 'calc_workday_ready_day', daystart, ARGS[2])
	elseif $ARGS[1] = 'sauna':
		if matryona_mansion_sauna_pending = 1: exit
		matryona_mansion_sauna_pending = 1
		matryona_mansion_sauna_ready_day = func('city_mansion_entrance', 'calc_workday_ready_day', daystart, ARGS[2])
	elseif $ARGS[1] = 'pool':
		if matryona_mansion_pool_pending = 1: exit
		matryona_mansion_pool_pending = 1
		matryona_mansion_pool_ready_day = func('city_mansion_entrance', 'calc_workday_ready_day', daystart, ARGS[2])
	end
	exit
end

if $ARGS[0] = 'queue_room_task':
	!! Queue a room renovation or decoration step independently.
	!! This allows furnishing work to run in parallel with other construction tasks.
	!! ARGS[1] = room index (3..12)
	!! ARGS[2] = target ymanrem stage to set when the work finishes
	!! ARGS[3] = working-day duration
	if matryona_mansion_room_pending[ARGS[1]] = 1: exit
	matryona_mansion_room_pending[ARGS[1]] = 1
	matryona_mansion_room_target[ARGS[1]] = ARGS[2]
	matryona_mansion_room_ready_day[ARGS[1]] = func('city_mansion_entrance', 'calc_workday_ready_day', daystart, ARGS[3])
	exit
end
