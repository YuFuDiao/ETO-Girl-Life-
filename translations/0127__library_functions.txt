				if $lib_book_loaned = '一本冒险小说':

				elseif $lib_book_loaned = '一本奇幻小说':

				elseif $lib_book_loaned = '一本科幻小说':
					mc_inventory['scifi_books'] -= 1
				end
			elseif lib_book_read = 1:
				if $lib_book_loaned = '一本冒险小说':

				elseif $lib_book_loaned = '一本奇幻小说':

				elseif $lib_book_loaned = '一本科学书':

				end
			end
