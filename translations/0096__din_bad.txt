			$stat_texts['cycle_state'] = '距离你记得的上次月经已经过了<<temp[1]>>天。<<$stat_texts[''cycle_feel'']>>进行无保护性行为应该是安全的。'
		elseif $stat['preg_risk'] = 'prob_safe':
			$stat_texts['cycle_state'] = '距离你记得的上次月经已经过了<<temp[1]>>天。<<$stat_texts[''cycle_feel'']>>进行无保护性行为可能有点风险。'
		elseif $stat['preg_risk'] = 'danger':
			$stat_texts['cycle_state'] = '距离你记得的上次月经已经过了<<temp[1]>>天。<<$stat_texts[''cycle_feel'']>>你认为你正处于易孕期。'
		else
			$stat_texts['cycle_state'] = '距离你记得的上次月经已经过了<<temp[1]>>天。<<$stat_texts[''cycle_feel'']>>'
		end
	end
	stat['cycphase_symptoms'] = daystart
	killvar 'temp'
	gs 'stat_display_compute', 'queue_msg', 'cycle_state', '', 'status/cycle/womb', 3, "gs 'din_bad', 'd_cycreport_choice'"
end
