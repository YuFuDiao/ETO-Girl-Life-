# help
!! Help / glossary system.
!!
!! Entry points (call these from outside):
!!   gs 'help', 'open'              — open contextual page based on current location
!!   gs 'help', 'open', 'page_key'  — open a specific named page directly
!!
!! Internal navigation between help pages uses:
!!   gs 'loc', 'page_name'   — no push/pop, stays inside the glossary
!!
!! Exiting always goes through 'back', which pops the state stack.
!!
!! Context variables (set on open, cleared on back):
!!   $help_from_curloc          — $curloc at the moment the glossary was opened
!!   $help_from_loc_arg         — $loc_arg at the moment the glossary was opened
!!   $help_from_location_type   — $location_type at the moment the glossary was opened
!!
!! Page rendering — each page handler sets these then calls gs 'help', 'render':
!!   $help_page_title    — displayed in the banner
!!   $help_page_key      — set automatically by render from its ARGS[1]
!!   $help_page_loc      — location name this page lives in (omit if 'help'; set to e.g. 'help_2' if split)
!!   $help_page_content  — HTML string for the main content area


!! ── Open: entry from the stat bar icon or a direct link ─────────────
!! If ARGS[1] is provided it is used as the target page directly.
!! Otherwise the fallback-chain dispatcher (show) picks the best page.
if $ARGS[0] = 'open':
	gs 'state', 'push'
	menu_off = 1
	$help_from_curloc = $curloc
	$help_from_loc_arg = $loc_arg
	$help_from_location_type = $location_type
	gt 'help', iif($ARGS[1] <> '', $ARGS[1], 'show')
end


!! ── Show: fallback-chain dispatcher ─────────────────────────────────
!! Priority:
!!   1. Exact match on curloc + loc_arg
!!   2. Exact match on curloc alone
!!   3. Match on $help_from_location_type
!!   4. Generic fallback
if $ARGS[0] = 'show':
	!! 1. Exact loc + arg match
	!! (add: if $help_from_curloc = 'loc' and $help_from_loc_arg = 'arg': gt 'help', 'page_name')

	!! 2. Exact loc match
	!! (add: if $help_from_curloc = 'loc': gt 'help', 'page_name')

	!! 3. location_type match
	!! (add: if $help_from_location_type = 'home': gt 'help', 'topic_home')

	!! 4. Generic fallback
	gt 'help', 'generic'
end


!! ── Back: exit the glossary and restore game state ──────────────────
if $ARGS[0] = 'back':
	killvar 'help_from_curloc'
	killvar 'help_from_loc_arg'
	killvar 'help_from_location_type'
	killvar 'help_page_title'
	killvar 'help_page_key'
	killvar 'help_page_loc'
	killvar 'help_page_content'
	killvar 'help_gen_c'
	killvar 'help_gen_cs'
	gs 'state', 'pop'
end


!! ════════════════════════════════════════════════════════════════════
!! LAYOUT ENGINE
!! ════════════════════════════════════════════════════════════════════

!! ── def_section: register a top-level TOC section ──────────────────
!! ARGS[1] = key   ARGS[2] = page arg ('' = header only)   ARGS[3] = display label
!! ARGS[4] = location name (optional — omit or pass '' for 'help')
!! Children are added via def_child; the children string starts empty here.
if $ARGS[0] = 'def_section':
	$help_toc_sections[] = $ARGS[1]
	$help_toc_label[$ARGS[1]] = $ARGS[3]
	$help_toc_page[$ARGS[1]] = $ARGS[2]
	$help_toc_loc[$ARGS[1]] = iif($ARGS[4] <> '', $ARGS[4], 'help')
end


!! ── def_group: non-navigating collapsible group header ──────────────
!! ARGS[1] = parent section key ('' = top-level section list)
!! ARGS[2] = group key   ARGS[3] = display label
!! If a parent is given the group becomes a child of that section (nested).
!! Add leaf children with def_child using this group key as parent.
if $ARGS[0] = 'def_group':
	$help_toc_label[$ARGS[2]] = $ARGS[3]
	$help_toc_page[$ARGS[2]] = ''
	$help_toc_loc[$ARGS[2]] = 'help'
	help_toc_is_group[$ARGS[2]] = 1
	if $ARGS[1] = '':
		$help_toc_sections[] = $ARGS[2]
	else
		if $help_toc_children[$ARGS[1]] = '':
			$help_toc_children[$ARGS[1]] = $ARGS[2]
		else
			$help_toc_children[$ARGS[1]] += ' ' + $ARGS[2]
		end
	end
end


!! ── def_child: register a child entry under a parent section ─────────
!! ARGS[1] = parent key   ARGS[2] = child key   ARGS[3] = display label
!! ARGS[4] = location name (optional — omit or pass '' for 'help')
!! The child key is used directly as the page handler arg.
if $ARGS[0] = 'def_child':
	if $help_toc_children[$ARGS[1]] = '':
		$help_toc_children[$ARGS[1]]  = $ARGS[2]
	else
		$help_toc_children[$ARGS[1]] += ' ' + $ARGS[2]
	end
	$help_toc_label[$ARGS[2]] = $ARGS[3]
	$help_toc_page[$ARGS[2]] = $ARGS[2]
	$help_toc_loc[$ARGS[2]] = iif($ARGS[4] <> '', $ARGS[4], 'help')
end

!! ── render_toc: sidebar TOC HTML (func) ──────────────────────────────
!! 3-level tree: sections → leaf children or group children → leaf grandchildren.
!! Groups at level 2 are expandable but do not navigate.
!! Active page is bold.  ▶ = collapsed  ▼ = expanded.
if $ARGS[0] = 'render_toc':
	!! Auto-open: expand the section (and nested group) that owns the current page
	rt_ao_i = 0
	:rt_ao_loop
	if rt_ao_i < arrsize('$help_toc_sections'):
		$rt_ao_key = $help_toc_sections[rt_ao_i]
		if $help_toc_page[$rt_ao_key] = $help_page_key:
			help_toc_open[$rt_ao_key] = 1
		elseif instr(' ' + $help_toc_children[$rt_ao_key] + ' ', ' ' + $help_page_key + ' ') > 0:
			help_toc_open[$rt_ao_key] = 1
		else
			!! Scan group children of this section for the current page
			$rt_ao_grem = $help_toc_children[$rt_ao_key]
			:rt_ao_gloop
			if $rt_ao_grem <> '':
				rt_ao_gsp = instr($rt_ao_grem, ' ')
				if rt_ao_gsp > 0:
					$rt_ao_gk   = $mid($rt_ao_grem, 1, rt_ao_gsp - 1)
					$rt_ao_grem = $mid($rt_ao_grem, rt_ao_gsp + 1)
				else
					$rt_ao_gk   = $rt_ao_grem
					$rt_ao_grem = ''
				end
				if help_toc_is_group[$rt_ao_gk]:
					if instr(' ' + $help_toc_children[$rt_ao_gk] + ' ', ' ' + $help_page_key + ' ') > 0:
						help_toc_open[$rt_ao_key] = 1
						help_toc_open[$rt_ao_gk]  = 1
					end
				end
				jump 'rt_ao_gloop'
			end
		end
		rt_ao_i += 1
		jump 'rt_ao_loop'
	end
	killvar 'rt_ao_i'
	killvar 'rt_ao_key'
	killvar 'rt_ao_grem'
	killvar 'rt_ao_gsp'
	killvar 'rt_ao_gk'

	!! Build HTML
	$rt_html = '<div style="font-weight:bold;font-size:0.95em;padding:0 0 6px 0;border-bottom:2px solid;margin-bottom:8px">目录</div>'

	if arrsize('$help_toc_sections') = 0:
		$rt_html += '<small><i>暂无主题。</i></small>'
	else
		rt_i = 0
		:rt_sec_loop
		if rt_i < arrsize('$help_toc_sections'):
			$rt_key     = $help_toc_sections[rt_i]
			$rt_label   = $help_toc_label[$rt_key]
			$rt_page    = $help_toc_page[$rt_key]
			$rt_loc     = $help_toc_loc[$rt_key]
			$rt_kids    = $help_toc_children[$rt_key]
			rt_open     = help_toc_open[$rt_key]
			rt_has_kids = iif($rt_kids <> '', 1, 0)

			!! Triangle toggle
			if rt_has_kids:
				$rt_tri    = iif(rt_open, '&#9660;', '&#9654;')
				$rt_toggle = '<a style="text-decoration:none" href="exec: help_toc_open[''<<$rt_key>>''] = 1 - help_toc_open[''<<$rt_key>>''] & gs ''<<$help_page_loc>>'', ''<<$help_page_key>>''">' + $rt_tri + '</a> '
			else
				$rt_toggle = '&nbsp;&nbsp; '
			end

			!! Section label
			if $rt_page <> '':
				if $rt_page = $help_page_key:
					$rt_active = ' style="font-weight:bold;border-left:3px solid;padding-left:4px;margin-left:-7px;display:inline-block"'
				else
					$rt_active = ''
				end
				$rt_lnk    = '<a href="exec: gs ''<<$rt_loc>>'', ''<<$rt_page>>''"' + $rt_active + '>' + $rt_label + '</a>'
			elseif help_toc_is_group[$rt_key]:
				$rt_lnk    = '<a style="color:inherit;text-decoration:none;cursor:pointer" href="exec: help_toc_open[''<<$rt_key>>''] = 1 - help_toc_open[''<<$rt_key>>''] & gs ''<<$help_page_loc>>'', ''<<$help_page_key>>''">' + $rt_label + '</a>'
			else
				$rt_lnk    = $rt_label
			end

			$rt_html += '<div style="margin:4px 0;font-weight:bold;font-size:0.95em">' + $rt_toggle + $rt_lnk + '</div>'

			!! Children (level 2) — only when section is open
			if rt_open and $rt_kids <> '':
				$rt_remaining = $rt_kids
				:rt_child_loop
				if $rt_remaining <> '':
					rt_sp = instr($rt_remaining, ' ')
					if rt_sp > 0:
						$rt_ck        = $mid($rt_remaining, 1, rt_sp - 1)
						$rt_remaining = $mid($rt_remaining, rt_sp + 1)
					else
						$rt_ck        = $rt_remaining
						$rt_remaining = ''
					end

					if help_toc_is_group[$rt_ck]:
						!! Level-2 group: expandable sub-header, no navigation
						rt_cg_open    = help_toc_open[$rt_ck]
						$rt_cg_tri    = iif(rt_cg_open, '&#9660;', '&#9654;')
						$rt_cg_toggle = '<a style="text-decoration:none" href="exec: help_toc_open[''<<$rt_ck>>''] = 1 - help_toc_open[''<<$rt_ck>>''] & gs ''<<$help_page_loc>>'', ''<<$help_page_key>>''">' + $rt_cg_tri + '</a> '
						$rt_cg_lnk    = '<a style="color:inherit;text-decoration:none;cursor:pointer" href="exec: help_toc_open[''<<$rt_ck>>''] = 1 - help_toc_open[''<<$rt_ck>>''] & gs ''<<$help_page_loc>>'', ''<<$help_page_key>>''">' + $help_toc_label[$rt_ck] + '</a>'
						$rt_html += '<div style="margin:3px 0 1px 0;padding-left:14px;font-style:italic;font-size:0.88em;opacity:0.75">' + $rt_cg_toggle + $rt_cg_lnk + '</div>'

						!! Level-3 group children
						if rt_cg_open and $help_toc_children[$rt_ck] <> '':
							$rt_gc_rem = $help_toc_children[$rt_ck]
							:rt_gc_loop
							if $rt_gc_rem <> '':
								rt_gc_sp = instr($rt_gc_rem, ' ')
								if rt_gc_sp > 0:
									$rt_gc_k   = $mid($rt_gc_rem, 1, rt_gc_sp - 1)
									$rt_gc_rem = $mid($rt_gc_rem, rt_gc_sp + 1)
								else
									$rt_gc_k   = $rt_gc_rem
									$rt_gc_rem = ''
								end
								$rt_gc_l   = $help_toc_label[$rt_gc_k]
								$rt_gc_p   = $help_toc_page[$rt_gc_k]
								$rt_gc_loc = $help_toc_loc[$rt_gc_k]
								if $rt_gc_p = $help_page_key:
									$rt_gc_act = ' style="font-weight:bold;border-left:3px solid;padding-left:3px;margin-left:-6px;display:inline-block"'
								else
									$rt_gc_act = ''
								end
								$rt_html += '<div style="margin:1px 0;padding-left:28px;font-size:0.88em"><a href="exec: gs ''<<$rt_gc_loc>>'', ''<<$rt_gc_p>>''"' + $rt_gc_act + '>' + $rt_gc_l + '</a></div>'
								jump 'rt_gc_loop'
							end
						end
					else
					!! Level-2 leaf child
					$rt_cl     = $help_toc_label[$rt_ck]
					$rt_cp     = $help_toc_page[$rt_ck]
					$rt_cloc   = $help_toc_loc[$rt_ck]
					if $rt_cp = $help_page_key:
						$rt_cactive = ' style="font-weight:bold;border-left:3px solid;padding-left:4px;margin-left:-7px;display:inline-block"'
					else
						$rt_cactive = ''
					end
					$rt_html += '<div style="margin:2px 0;padding-left:14px"><a href="exec: gs ''<<$rt_cloc>>'', ''<<$rt_cp>>''"' + $rt_cactive + '>' + $rt_cl + '</a></div>'
					end
					jump 'rt_child_loop'
				end
			end
			rt_i += 1
			jump 'rt_sec_loop'
		end
	end

	$result = $rt_html
end


!! ── render: assemble and output the full help layout ─────────────────
!! ARGS[1] = page key ($ARGS[0] of the calling stub).
!! Derives $help_page_loc from the TOC registry; falls back to 'help'.
!! Auto-sets $help_page_title from the TOC label unless the stub already
!! set a custom title. Always adds a 'Back' act.
if $ARGS[0] = 'render':
	!! Page key is passed as ARGS[1]; loc is derived from the TOC registry after toc_init.
	$help_page_key = $ARGS[1]

	*clr & cla
	gs 'help', 'toc_init'

	!! Derive loc from TOC data (populated by toc_init); fall back to 'help' for
	!! unregistered pages such as the 'generic' landing page.
	$help_page_loc = iif($help_toc_loc[$help_page_key] <> '', $help_toc_loc[$help_page_key], 'help')

	!! Auto-set title from the TOC label, but only if the stub did not set a custom
	!! title already. chars_pc uses the live player name instead of the static label.
	if $help_page_title = '' and $help_toc_label[$help_page_key] <> '':
		$help_page_title = $help_toc_label[$help_page_key]
	end

	!! Sidebar extends full height; banner sits only above the content column.
	$help_layout =	'<table width="100%" cellspacing="0" cellpadding="0"><tr>'
	$help_layout +=		'<td width="20%" valign="top" style="border-right:1px solid;padding:8px 12px 8px 0">'
	$help_layout +=			$func('help', 'render_toc')
	$help_layout +=		'</td>'
	$help_layout +=		'<td valign="top" style="padding:0 0 8px 16px">'
	$help_layout +=			$func('help', 'render_banner')
	$help_layout +=			$help_page_content
	$help_layout +=		'</td>'
	$help_layout += '</tr></table>'

	*p $help_layout

	act '返回': gs 'help', 'back'
end
