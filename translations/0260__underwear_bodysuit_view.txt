# underwear_bodysuit_view


	!!====================================================================!!
	!!                                                                    !!
	!!                              FILTERS                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'filter_builder':


		!!------------------------------------------------------------!!
		!!                       GROUP HEADERS                        !!
		!!------------------------------------------------------------!!


	if $ARGS[1] = 'setup_home_filters':
		gs 'shop_utils', 'filter_builder', 'init', 'underwear_bodysuit_view', 'view_grid', $shop_display['link']
			gs 'shop_utils',				'filter_builder', 'setup_number_sort'
			gs 'shop_utils', 				'filter_builder', 'setup_quality_sort'
			gs 'shop_utils',				'filter_builder', 'setup_sort_direction'

			gs 'underwear_bodysuit_view',	'filter_builder', 'setup_sport_filter'
		gs 'shop_utils', 'filter_builder', 'finish', 'mini'
		exit
	end

	if $ARGS[1] = 'setup_shop_filter':
		gs 'shop_utils', 'filter_builder', 'init', $loc, $loc_arg
			gs 'shop_utils', 'filter_builder', 'setup_number_sort'
			gs 'shop_utils', 'filter_builder', 'setup_quality_sort'
			gs 'shop_utils', 'filter_builder', 'setup_price_sort'
			gs 'shop_utils', 'filter_builder', 'setup_discount_sort'
			gs 'shop_utils', 'filter_builder', 'setup_sort_direction'
		gs 'shop_utils', 'filter_builder', 'finish', 'mini'
		exit
	end


		!!------------------------------------------------------------!!
		!!                     INDIVIDUAL HEADERS                     !!
		!!------------------------------------------------------------!!


	if $ARGS[1] = 'setup_sport_filter':
		gs 'shop_utils', 'filter_builder', 'setup_sport_filter', 'PanSport'
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                              FILTERS                               !!
	!!                             DEPRECATED                             !!
	!!====================================================================!!


!! Filters are automatically constructed when you build them using the filter_builder
!! you can filter using if func('shop_utils', 'filters', 'apply'): {do thing if all filters pass}


!! DEPRECATED
if $ARGS[0] = 'home_filter':
	gs 'shop_utils', 'filter', 'set_block'
		if func('underwear_bodysuit_view', 'filter', 'sport') = 0: exit
	gs 'shop_utils', 'filter', 'set_pass'
	exit
end


!! DEPRECATED
if $ARGS[0] = 'shop_filter':
	gs 'shop_utils', 'filter', 'set_pass'
	exit
end


		!!------------------------------------------------------------!!
		!!                     INDIVIDUAL FILTER                      !!
		!!------------------------------------------------------------!!


!! DEPRECATED
if $ARGS[0] = 'filter':
	if $ARGS[1] = 'sport':
		if BraSport:
			if outfitfilter['sport'] >= 0: result = 1
		else
			if outfitfilter['sport'] <= 0: result = 1
		end
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                                INIT                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'init':
	!! $ARGS[1]: function name (add_types, set_exceptions, add_to_sorted)

	if $ARGS[1] = 'add_types':
		!! $ARGS[2]: category to add

		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 5) = 'lusso':			gs 'shop_utils', 'init', 'add_type', 'lusso'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 8) = 'moncheri':		gs 'shop_utils', 'init', 'add_type', 'moncheri'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 9) = 'salacious':		gs 'shop_utils', 'init', 'add_type', 'salacious'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 13) = 'scandalicious':	gs 'shop_utils', 'init', 'add_type', 'scandalicious'
		exit
	end

	if $ARGS[1] = 'set_exceptions':
		!! shop_display_exceptions[$type + '_bodysuits-' + number] = 1
		!!		i.e: shop_display_exceptions['lusso_bodysuits-3'] = 1
		exit
	end
	exit
end


if $ARGS[0] = 'sorted':
	if $ARGS[1] = 'add':
		!! $ARGS[2]: bodysuit type
		!! ARGS[3]: bodysuit number
		!! ARGS[4]: index

		gs 'underwear_attributes', $ARGS[2] + '_bodysuits', ARGS[3]

		if BraQuality = 0: exit

		gs 'shop_utils', 'sorted', 'add_to_number', $ARGS[2], ARGS[3], ARGS[4]
		gs 'shop_utils', 'sorted', 'add_to_quality', $ARGS[2], ARGS[3], BraQuality
		gs 'shop_utils', 'sorted', 'add_to_price', $ARGS[2], ARGS[3], BraPrice
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                              DISPLAY                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'display':
	if $ARGS[1] = 'grid_shop':
		if $ARGS[2] = 'header':
			gs 'underwear_bodysuit_view', 'filter_builder', 'setup_shop_filter'
			exit
		end

		if $ARGS[2] = 'main':
			!! $ARGS[3]: link
			!! $ARGS[4]: type
			!! ARGS[5]: number
			!! ARGS[6]: discount

			gs 'underwear_attributes', $ARGS[4] + '_bodysuits', ARGS[5]
			if func('shop_utils', 'filter', 'apply'):
				*p '<a href="exec: gt ''underwear_bodysuit_view'', ''view_item'', ''<<$ARGS[3]>>'', ''<<$ARGS[4]>>'', <<ARGS[5]>>, <<ARGS[6]>>"><img height="250" src="<<$func(''$bodysuit_image'', $ARGS[4], ARGS[5])>>"></a>'
			end
			exit
		end

		if $ARGS[2] = 'footer':
			exit
		end
		exit
	end

	if $ARGS[1] = 'grid_wardrobe':
		if $ARGS[2] = 'header':
			gs 'shop_utils', 'display', 'wardrobe_storage_unwanted_header', 'view_grid'
			gs 'underwear_bodysuit_view', 'filter_builder', 'setup_home_filters'
			exit
		end

		if $ARGS[2] = 'main':
			!! $ARGS[3]: link
			!! $ARGS[4]: type
			!! ARGS[5]: number
			!! ARGS[6]: discount

			gs 'underwear_attributes', $ARGS[4] + '_bodysuits', ARGS[5]
			if func('shop_utils', 'filter', 'apply'):
				*p '<a href="exec: gt ''underwear_bodysuit_view'', ''view_item'', ''<<$ARGS[3]>>'', ''<<$ARGS[4]>>'', <<ARGS[5]>>, <<ARGS[6]>>"><img height="150" src="<<$func(''$bodysuit_image'', $ARGS[4], ARGS[5])>>"></a>'
			end
			exit
		end

		if $ARGS[2] = 'footer':
			exit
		end
		exit
	end
	exit
end


!! DEPRECATED
if $ARGS[0] = 'display_grid_shop':
	gs 'shop_utils', 'display', 'grid_shop'
	exit
end


!! DEPRECATED
if $ARGS[0] = 'display_grid_wardrobe':
	gs 'shop_utils', 'display', 'grid_wardrobe'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                             VIEW LIST                              !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'view_grid':
	!! $ARGS[1] - action type (wardrobe, store, unwanted, bathroom) -> $shop_display['link']
	menu_off = 1

	if $pursepantytype <> '' or $pursebratype <> '':
		'<center><b>你把 '+iif($pursepantytype = '', '', '内裤') + iif($pursepantytype = '' or $pursebratype = '', '', '和') + iif($pursebratype = '', '', '胸罩') + '从钱包里放回衣柜。</b></center><br>'
		killvar 'pursepantytype'
		killvar 'pursepantynumber'
		killvar 'pursebratype'
		killvar 'pursebranumber'
	end

	if shop_display['init'] = 0:
		gs 'shop_utils', 'init', 'start', 'bodysuits', $ARGS[1]
			gs 'underwear_bodysuit_view', 'init', 'add_types', 'all'
			if $ARGS[1] = 'wardrobe' or $ARGS[1] = 'storage' or $ARGS[1] = 'unwanted':
				gs 'shop_utils', 'init', 'set_return', 'wardrobe', 'main'
			else
				gs 'shop_utils', 'init', 'set_return', $loc, $loc_arg
			end
		gs 'shop_utils', 'init', 'end'
	end

	act '返回': gt 'shop_utils', 'return'

	if $bodysuitworntype <> 'none':
		act '脱下连体衣':
			gs 'underwear_bodysuits', 'strip'
			gt 'underwear_bodysuit_view', 'view_grid', $shop_display['link']
		end
	end

	gs 'shop_utils', 'display', 'grid_wardrobe'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                             VIEW ITEM                              !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'view_item':
	$shop_utils_view['link']		= $ARGS[1]		& !! action type (wardrobe, store, unwanted, bathroom, shop, cheat)
	$shop_utils_view['type']		= $ARGS[2]		& !! bodysuit type
	shop_utils_view['number']		= ARGS[3]		& !! bodysuit index
	shop_utils_view['discount']		= ARGS[4]		& !! price modifier (0-100, taken as %) {OPTIONAL}

	menu_off = 1
	gs 'stat'

	'<center><img <<$set_imgh>> src="<<$func(''$pcs_outfit_image'', $shop_utils_view[''type''] + ''_bodysuits'', shop_utils_view[''number''])>>"></center>'
	gs 'underwear_attributes', $shop_utils_view['type']+'_bodysuits', shop_utils_view['number']
	gs 'underwear_descriptions', 'bodysuit'

	if $shop_utils_view['link'] = 'shop' or $shop_utils_view['link'] = 'cheat': xgt 'underwear_bodysuit_view', 'view_item_shop'
	if func('underwear_bodysuits', 'is_wearing', $shop_utils_view['type'], shop_utils_view['number']):
		xgt 'underwear_bodysuit_view', 'view_item_wearing'
	end
	if $shop_utils_view['link'] = 'wardrobe': xgt 'underwear_bodysuit_view', 'view_item_wardrobe'
	if $shop_utils_view['link'] = 'storage': xgt 'underwear_bodysuit_view', 'view_item_storage'
	if $shop_utils_view['link'] = 'unwanted': xgt 'underwear_bodysuit_view', 'view_item_unwanted'

	act '返回': gt 'underwear_bodysuit_view', 'view_grid', $shop_utils_view['link']
	exit
end


if $ARGS[0] = 'view_item_shop':
	act '离开': gt $loc, $loc_arg

	if func('underwear_bodysuits', 'is_owned', $shop_utils_view['type'], shop_utils_view['number']):
		'你已经拥有这件物品。'
		exit
	end

	shop_utils_view['discount_total'] = shop_utils_view['discount'] + func('shop_utils', 'get_discount', $shop_utils_view['type'] + '_bodysuits', shop_utils_view['number'])
	shop_utils_view['base_price'] = BraPrice
	shop_utils_view['price'] = shop_utils_view['base_price'] * max(0, 100 - shop_utils_view['discount_total']) / 100
	shop_utils_view['price'] = shop_utils_view['price'] / 50 * 50
	if shop_utils_view['price'] = shop_utils_view['base_price']:
		$shop_utils_view['price_string'] = $func('money', 'string_price', shop_utils_view['price'])
	else
		$shop_utils_view['price_string'] = $func('wrap', 'neg s', $func('money', 'string_price', shop_utils_view['base_price'])) + ' <b>' +  $func('money', 'string_price', shop_utils_view['price']) + '</b>'
		'现在 ' + shop_utils_view['discount_total'] + '% 折扣' + iif(shop_utils_view['discount_total'] <= 10, '', '！')
	end

	'价格：' + $shop_utils_view['price_string']

	if func('money', 'can_afford', shop_utils_view['price']) = 0:
		'你的钱不够买这件连体衣。'
	else
		act '购买 (<<$shop_utils_view[''price_string'']>>)':
			gs 'money', 'pay', shop_utils_view['price']
			gs 'underwear_bodysuits', 'add_item', $shop_utils_view['type'], shop_utils_view['number']
			gt $loc, $loc_arg
		end
	end
	exit
end


if $ARGS[0] = 'view_item_wearing':
	act '返回': gt 'underwear_bodysuit_view', 'view_grid', $shop_utils_view['link']
	'你现在正穿着这件连体衣。'
	if $shop_utils_view['link'] = 'wardrobe':
		act '脱掉你的连体衣':
			gs 'underwear_bodysuits', 'strip'
			gs 'shop_utils', 'cleanup'
			gt 'wardrobe', 'main'
		end
	end
	exit
end


if $ARGS[0] = 'view_item_wardrobe':
	act '返回': gt 'underwear_bodysuit_view', 'view_grid', $shop_utils_view['link']
	act '穿上': xgt 'underwear_bodysuit_view', 'view_item_wear'
	gs 'underwear_bodysuit_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_storage':
	act '返回': gt 'underwear_bodysuit_view', 'view_grid', $shop_utils_view['link']
	'这件连体衣存放在储物柜里。'
	gs 'underwear_bodysuit_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_unwanted':
	act '返回': gt 'underwear_bodysuit_view', 'view_grid', $shop_utils_view['link']
	'这件连体衣被标记为不需要。'
	gs 'underwear_bodysuit_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_wear':
	*nl
	if PCloPanties = 1 or PCloBra = 1:
		*pl '<font color=red><b>你不能在这套服装里穿连体衣</b></font>'
		act '返回': gt 'underwear_bodysuit_view', 'view_item', $shop_utils_view['action'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		exit
	end

	if hypnoPanty > 0 or hypnoBra > 0: xgt 'underwear_bodysuit_view', 'view_item_wear_hypno'

	gs 'underwear_bodysuits', 'wear', $shop_utils_view['type'], shop_utils_view['number']
	gs 'shop_utils', 'cleanup'
	gt 'wardrobe', 'main'
end


if $ARGS[0] = 'view_item_wear_hypno':
	'你拿起连体衣，想着试穿一下……但你其实不太喜欢穿连体衣。'
	gs 'willpower', 'misc', 'resist', 'easy'
	if cheatVars['willpower'] = 0: will_cost += hypnoTime
	if pcs_willpwr < will_cost:
		act '还是穿上吧' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '还是穿上吧' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'pay', 'resist'
			gs 'stat'
			gs 'underwear_bodysuits', 'wear', $shop_utils_view['type'], shop_utils_view['number']
			gs 'shop_utils', 'cleanup'
			gt 'wardrobe', 'main'
		end
	end
	act '返回': gt 'underwear_bodysuit_view', 'view_grid', $shop_utils_view['link']
	exit
end


if $ARGS[0] = 'storage_options':
	if func('underwear_bodysuits', 'is_owned', $shop_utils_view['type'], shop_utils_view['number']) = 0: exit

	if func('underwear_bodysuits', 'in_wardrobe', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到衣柜':
			gs 'underwear_bodysuits', 'move_to_wardrobe', $shop_utils_view['type'], shop_utils_view['number']
			gt 'underwear_bodysuit_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end

	if func('underwear_bodysuits', 'is_immutable', $shop_utils_view['type'], shop_utils_view['number']): exit

	if func('underwear_bodysuits', 'in_storage', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到存储':
			gs 'underwear_bodysuits', 'move_to_storage', $shop_utils_view['type'], shop_utils_view['number']
			gt 'underwear_bodysuit_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	if func('underwear_bodysuits', 'in_unwanted', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到不需要':
			gs 'underwear_bodysuits', 'move_to_unwanted', $shop_utils_view['type'], shop_utils_view['number']
			gt 'underwear_bodysuit_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	act $func('wrap', 'neg', '扔掉这件连体衣'):
		gs 'underwear_bodysuits', 'remove_item', $shop_utils_view['type'], shop_utils_view['number']
		gt 'underwear_bodysuit_view', 'view_grid', $shop_utils_view['link']
	end
	exit
end
