if $ARGS[0] = 'hair':
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'hair_color'
	if $ngpPrefResult['HasPos'] ! '':
		gs 'npc_relationship', 'modify', $npcID, rand(0, 1)
		'<<$npcdesc>>说你的头发很美。'
	elseif $ngpPrefResult['NotPos'] ! '':
		gs 'npc_relationship', 'modify', $npcID, rand(-1, 0)
		if $ngpPrefResult['NotPos'] = 'hair_color_black':
			'<<$npcdesc>>说你非常漂亮，尽管他喜欢黑发女人。'
		elseif $ngpPrefResult['NotPos'] = 'hair_color_brown':
			'<<$npcdesc>>说你非常漂亮，尽管他喜欢棕发女人。'
		elseif $ngpPrefResult['NotPos'] = 'hair_color_red':
			'<<$npcdesc>>说你非常漂亮，尽管他喜欢红发女人。'
		elseif $ngpPrefResult['NotPos'] = 'hair_color_blonde':
			'<<$npcdesc>>说你非常漂亮，尽管他喜欢金发女人。'
		end
	end
	killvar 'ngpPrefResult'
end


if $ARGS[0] = 'figure':
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'body_bmi'
	if $ngpPrefResult['HasPos'] ! '':
		gs 'npc_relationship', 'modify', $npcID, rand(0, 1)
		'<<$npcdesc>>说你身材完美。'
	elseif $ngpPrefResult['NotPos'] ! '':
		gs 'npc_relationship', 'modify', $npcID, rand(-1, 0)
		if $ngpPrefResult['NotPos'] = 'body_bmi_starving' or $ngpPrefResult['NotPos'] = 'body_bmi_underweight':
			'<<$npcdesc>>说你爱他，尽管他以前交往的都是些瘦姑娘。'
		elseif $ngpPrefResult['NotPos'] = 'body_bmi_normal':
			if func('pcs_has_attr', 'OR', 'body_bmi_starving', 'body_bmi_underweight'):
				'<<$npcdesc>>问你是否有厌食症。'
			else
				'<<$npcdesc>>问你是不是在节食。'
			end
		elseif $ngpPrefResult['NotPos'] = 'body_bmi_overweight' or $ngpPrefResult['NotPos'] = 'body_bmi_obese':
			'<<$npcdesc>>说你很漂亮，但女人应该保持好身材。'
		end
	end
	killvar 'ngpPrefResult'
end


if $ARGS[0] = 'tits':
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'body_tits'
	if $ngpPrefResult['HasPos'] ! '':
		gs 'npc_relationship', 'modify', $npcID, rand(0, 1)
		'<<$npcdesc>>说你胸部很完美。'
	elseif $ngpPrefResult['NotPos'] ! '':
		gs 'npc_relationship', 'modify', $npcID, rand(-1, 0)
		if $ngpPrefResult['NotPos'] = 'body_tits_small' :
			'<<$npcdesc>>说你爱他，尽管他其实更喜欢小胸。'
		elseif $ngpPrefResult['NotPos'] = 'body_tits_average':
			if func('pcs_has_attr', 'body_tits_small'):
				'<<$npcdesc>>说你很漂亮，虽然胸部很小。'
			else
				'<<$npcdesc>>说你胸部真的很大。'
			end
		elseif $ngpPrefResult['NotPos'] = 'body_tits_big':
			if func('pcs_has_attr', 'body_tits_huge'):
				'<<$npcdesc>>说你胸部真的很大。'
			else
				'<<$npcdesc>>说你很漂亮，虽然胸部很小。'
			end
		elseif $ngpPrefResult['NotPos'] = 'body_tits_huge':
			'<<$npcdesc>>说你很漂亮，虽然胸部很小。'
		end
	end
	killvar 'ngpPrefResult'
end
