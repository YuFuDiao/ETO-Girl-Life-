# stat_sklattrib

!{
	******************** Warning! ********************
	The code in this location is both complex and very fundamental to most systems!
	Use EXTREME caution if modifying!
	Ask if understanding is required
	****************************************************
	For Users;
		For any Attribute or Skill, outside of "stat_sklattrib", the only things that get used are;
		the variables themselves, pcs_"name", for checks
		"name"_exp to replace straight adds (but the proper way to add exp is to use the gs 'exp_gain', '"name"' since that will modify the gain for the player charcters conditions, i.e. gs 'exp_gain', 'stren',1 instead of stren_exp += 1.)
		"name"_deg to replace losses (i.e. stren_deg -= 1 instead of pcs_stren -= 1)
		"name"_muta if setting higher attribute limits due to "mutations" ("name"_muta replaces the "mutagen" variables)
		And checking "name"_lvl if and only if it''s being used to check if an NPC/Item can still train the PC. That is also the only time there should be a gate on "name"_exp += x and should be a minimum of += rand(3,5). No other gates should be used.
		The others are used in the code as controllers and modifying anything except "name"_exp, "name"_deg, and "name"_muta outside of this routine can break things

	Attributes & Skills
		$att_name[x] and $skl_name[x]; an array that stores the name of each attribute variable
		$att_desc['name'] and $skl_desc['name']; the name of each attribute and skill for use in UIs
		For each attribute and skill variable, there are the following sub-variables
		_lvl, is the base level of an attribute/skill before adjustment
		_lvlst, level yesterday, used for error checking
		_deg is the days to attribute/skill loss and is reset on XP gain
		_exp is the amount of total XP gained so far for each and will replace all the gains in the rest of the code
		_mem is the amount of XP yesterday
		_mem2 is the amount of XP at the last stat call
		_muta replaces the assorted mutagen variables for attributes, this is to replace the 30 lines of code with 1
		_flr, this is the floor through which an attribute/skill cannot fall
		_xpnxt is the XP required to reach the next level
		_xpprv was the XP required to reach this level

	Traits
		$traitattskl[x], this stores a list of all traits that can affect attribute or skill advancement or degradation, if a trait does not, it does not go in this array
		For each trait that can affect attribute or skill advancement or degradation;
		"trait"[$stat_name + '-exp_gain] = percentual increase in exp gain (handled in exp_gain)
		"trait"[$stat_name + '-deg_loss] = percentual increase in deg loss
							$stat_name is the name of the attribute or skill that this effects.
							i.e. 'stren', 'run', 'magik'. all means it effects everything
		Note, the effects of multiple traits on a single skill or attribute stack

	For Modders
		If you want to add a new skill to the list, add the following two pieces of code to your mods main location.
		1. Add the skill to the list:
			gs 'stat_sklattrib_core', 'add_skl_to_list', 'name', 'Skill Name'
		2. Set the scaling for the skill (see stat_sklattrib_lvlset)
			pcs_name = (name_lvl + (pcs_stren + pcs_agil) / 2) / 2
			$name[0] = 'stren' & $name[1] = 'agil'

		Example:
		I want to add a horseriding skill named "horseriding". It should depend on agility, reaction, and perception.
		To do that, I add the following code to my mods main location:
			gs 'stat_sklattrib_core', 'add_skl_to_list', 'horseriding', 'Horse Riding'
			pcs_horseriding = (horseriding_lvl + (pcs_react + pcs_prcptn + pcs_agil) / 3) / 2
			$horseriding[0] = 'react' & $horseriding[1] = 'agil' & $horseriding[2] = 'prcptn'


	Temporary skill explenations:
		temp_sklattrib = array with temporary variables for deg and gain loops
		temp_sklattrib_main = array with temporary variables for the main loop

			temp_sklattrib['index'] = index used for the loop over the stats (attributes and skills)
			$temp_sklattrib['name'] = name of the stat. i.e. 'stren' or 'run'
			$temp_sklattrib['stat_type'] = is the stat an attribute or skill (takes values of 'attribute' or 'skill')
			temp_sklattrib['trait_index'] = index used for the loop over the traits.
			$temp_sklattrib['trait'] = name of the trait. i.e. 'nat_dancer' or 'scholarly'
			temp_sklattrib['deg_adj'] = is the bonus to degradation. With "60" being the unmodified normal (aka 100% of intended starting degradation).
										Higher means more time until degradation sets in.
			temp_sklattrib['degradation'] = the maximum time it takes for the stat to degrade
										Used to reset the degradation time back to full
			temp_sklattrib['daycall'] = flag for if the advancement loop is called with the 'daycall' flag (from cikl).
										Forces exp/lvl variables calculations
			temp_sklattrib['update_stat'] = flag for if the exp has changed.
										If true (1) recalculates exp/lvl variables
!}

if $ARGS[0] = '' or $ARGS[0] = 'daycall':
	gs 'stat_sklattrib', 'init'

	!! This is the section that runs when the call is from cikl
	if $ARGS[0] = 'daycall':
		!!	Attribute XP with Skill XP gain
		temp_sklattrib['index'] = 0
		:sklxploop
			$temp_sklattrib['name'] = $att_name[temp_sklattrib['index']]
			dynamic "
				<<$temp_sklattrib['name']>>_exp += <<$temp_sklattrib['name']>>_exp_skill_derived / 100
				<<$temp_sklattrib['name']>>_exp_skill_derived = 0
				"
			temp_sklattrib['index'] += 1
		if temp_sklattrib['index'] < arrsize('$att_name'): jump 'sklxploop'
		killvar 'temp_sklattrib'

		!!	Degradation Loop
		gs 'stat_sklattrib', 'degradation_loop'
	end

	!! Advancement Loop
	gs 'stat_sklattrib', 'advancement_loop', $ARGS[0]

	gs 'stat_sklattrib_lvlset'
	exit
end

!! gs 'stat_sklattrib', 'degradation'
if $ARGS[0] = 'degradation_loop':
	$temp_sklattrib['stat_type'] = 'attribute'
	:degloop_outer
		temp_sklattrib['index'] = 0
		:degloop_inner
			temp_sklattrib['deg_loss'] = func('_difficulty', 'get_deg_loss')
			if $temp_sklattrib['stat_type'] = 'attribute':
				$temp_sklattrib['name'] = $att_name[temp_sklattrib['index']]
			else
				$temp_sklattrib['name'] = $skl_name[temp_sklattrib['index']]
				temp_sklattrib['deg_loss'] = 75 * temp_sklattrib['deg_loss'] / 100
			end

			temp_sklattrib['trait_index'] = 0
			:traitdegloop
				$temp_sklattrib['trait'] = $traitattskl[temp_sklattrib['trait_index']]
				if $temp_sklattrib['trait'] ! '':
					temp_sklattrib['deg_loss'] += trait_vars[$temp_sklattrib['trait'] + '-' + $temp_sklattrib['name'] + '-deg_loss']
					temp_sklattrib['deg_loss'] += trait_vars['all-' + $temp_sklattrib['name'] + '-deg_loss']
				end
				temp_sklattrib['trait_index'] += 1
			if temp_sklattrib['trait_index'] < arrsize('$traitattskl'): jump 'traitdegloop'

			dynamic "
				<<$temp_sklattrib['name']>>_lvl = <<$temp_sklattrib['name']>>_lvlst
				temp_sklattrib['max_deg'] = func('stat_funcs', 'get_max_deg', <<$temp_sklattrib['name']>>_lvl, <<$temp_sklattrib['name']>>_muta)

				if <<$temp_sklattrib['name']>>_exp > <<$temp_sklattrib['name']>>_mem:
					<<$temp_sklattrib['name']>>_deg = temp_sklattrib['max_deg']
				else
					<<$temp_sklattrib['name']>>_exp = <<$temp_sklattrib['name']>>_mem
					<<$temp_sklattrib['name']>>_deg -= temp_sklattrib['deg_loss']
				end

				if <<$temp_sklattrib['name']>>_deg <= 0:
					if <<$temp_sklattrib['name']>>_lvl <= <<$temp_sklattrib['name']>>_flr:
						<<$temp_sklattrib['name']>>_lvl = <<$temp_sklattrib['name']>>_flr
						<<$temp_sklattrib['name']>>_deg = temp_sklattrib['max_deg']
					else
						<<$temp_sklattrib['name']>>_lvl -= 1
						<<$temp_sklattrib['name']>>_deg = temp_sklattrib['max_deg']

						<<$temp_sklattrib['name']>>_xpprv = func('stat_funcs', 'get_xpprv', <<$temp_sklattrib['name']>>_lvl)
						<<$temp_sklattrib['name']>>_xpnxt = func('stat_funcs', 'get_xpnxt', <<$temp_sklattrib['name']>>_lvl)
						<<$temp_sklattrib['name']>>_exp = <<$temp_sklattrib['name']>>_xpprv + min(max(0, <<$temp_sklattrib['name']>>_exp - <<$temp_sklattrib['name']>>_xpnxt), <<$temp_sklattrib['name']>>_xpnxt - <<$temp_sklattrib['name']>>_xpprv)
					end
				end
				<<$temp_sklattrib['name']>>_lvlst = <<$temp_sklattrib['name']>>_lvl
				<<$temp_sklattrib['name']>>_mem = <<$temp_sklattrib['name']>>_exp
			"
			temp_sklattrib['index'] += 1
		if $temp_sklattrib['stat_type'] = 'attribute':
			if temp_sklattrib['index'] < arrsize('$att_name'): jump 'degloop_inner'
			$temp_sklattrib['stat_type'] = 'skill'
		else
			!! $temp_sklattrib['stat_type'] = 'skill':
			if temp_sklattrib['index'] < arrsize('$skl_name'): jump 'degloop_inner'
			$temp_sklattrib['stat_type'] = '结束'
		end
	if $temp_sklattrib['stat_type'] = 'skill': jump 'degloop_outer'

	killvar 'temp_sklattrib'
	exit
end


!! gs 'stat_sklattrib', 'advancement', 'daycall'/''
if $ARGS[0] = 'advancement_loop':
	if $ARGS[1] = 'daycall': temp_sklattrib['daycall'] = 1

	$temp_sklattrib['stat_type'] = 'attribute'
	:advloop_outer
		temp_sklattrib['index'] = 0
		:advloop_inner
			if $temp_sklattrib['stat_type'] = 'attribute':
				$temp_sklattrib['name'] = $att_name[temp_sklattrib['index']]
			else
				$temp_sklattrib['name'] = $skl_name[temp_sklattrib['index']]
			end

			if temp_sklattrib['daycall']:
				temp_sklattrib['update_stat'] = 1
			else
				dynamic "
					if <<$temp_sklattrib['name']>>_exp ! <<$temp_sklattrib['name']>>_mem2:
						temp_sklattrib['update_stat'] = 1
					elseif <<$temp_sklattrib['name']>>_exp >= <<$temp_sklattrib['name']>>_xpnxt:
						temp_sklattrib['update_stat'] = 1
					elseif <<$temp_sklattrib['name']>>_exp < <<$temp_sklattrib['name']>>_xpprv:
						temp_sklattrib['update_stat'] = 1
					else
						temp_sklattrib['update_stat'] = 0
					end
				"
			end

			if temp_sklattrib['update_stat']:
				dynamic "
					<<$temp_sklattrib['name']>>_lvl = <<$temp_sklattrib['name']>>_lvlst
					<<$temp_sklattrib['name']>>_xpprv = func('stat_funcs', 'get_xpprv', <<$temp_sklattrib['name']>>_lvl)
					<<$temp_sklattrib['name']>>_xpnxt = func('stat_funcs', 'get_xpnxt', <<$temp_sklattrib['name']>>_lvl)
					if <<$temp_sklattrib['name']>>_exp >= <<$temp_sklattrib['name']>>_xpnxt:
						<<$temp_sklattrib['name']>>_lvl += 1
					elseif <<$temp_sklattrib['name']>>_exp < <<$temp_sklattrib['name']>>_xpprv:
						<<$temp_sklattrib['name']>>_lvl -= 1
					end
					<<$temp_sklattrib['name']>>_lvlst = <<$temp_sklattrib['name']>>_lvl

					if <<$temp_sklattrib['name']>>_lvl / 5 > <<$temp_sklattrib['name']>>_flr: <<$temp_sklattrib['name']>>_flr = <<$temp_sklattrib['name']>>_lvl / 5
					if <<$temp_sklattrib['name']>>_lvl > (100 + (<<$temp_sklattrib['name']>>_muta * 50)): <<$temp_sklattrib['name']>>_lvl = (100 + (<<$temp_sklattrib['name']>>_muta * 50))

					<<$temp_sklattrib['name']>>_xpprv = func('stat_funcs', 'get_xpprv', <<$temp_sklattrib['name']>>_lvl)
					<<$temp_sklattrib['name']>>_xpnxt = func('stat_funcs', 'get_xpnxt', <<$temp_sklattrib['name']>>_lvl)
					<<$temp_sklattrib['name']>>_mem2 = <<$temp_sklattrib['name']>>_exp
				"
			end
			temp_sklattrib['index'] += 1
		if $temp_sklattrib['stat_type'] = 'attribute':
			if temp_sklattrib['index'] < arrsize('$att_name'): jump 'advloop_inner'
			$temp_sklattrib['stat_type'] = 'skill'
		else
			if temp_sklattrib['index'] < arrsize('$skl_name'): jump 'advloop_inner'
			$temp_sklattrib['stat_type'] = '结束'
		end
	if $temp_sklattrib['stat_type'] = 'skill': jump 'advloop_outer'

	killvar 'temp_sklattrib'
	exit
end

!! This is sets _lvl and _exp based on the value of pcs_"name" the first time so that _lvl and _exp do not need to set on start.
!! Theoretically this can be eventually deleted.


!! gs 'stat_sklattrib', 'init'
if $ARGS[0] = 'init_loop':
	$temp_sklattrib['stat_type'] = 'attribute'
	:initloop_outer
		temp_sklattrib['index'] = 0
		:initloop_inner
			if $temp_sklattrib['stat_type'] = 'attribute':
				$temp_sklattrib['name'] = $att_name[temp_sklattrib['index']]
			else
				$temp_sklattrib['name'] = $skl_name[temp_sklattrib['index']]
			end

			dynamic "
				if pcs_<<$temp_sklattrib['name']>> > 0 and <<$temp_sklattrib['name']>>_lvl = 0:
					<<$temp_sklattrib['name']>>_lvl = pcs_<<$temp_sklattrib['name']>>
					<<$temp_sklattrib['name']>>_lvlst = <<$temp_sklattrib['name']>>_lvl
					<<$temp_sklattrib['name']>>_xpprv = func('stat_funcs', 'get_xpprv', <<$temp_sklattrib['name']>>_lvl)
					<<$temp_sklattrib['name']>>_xpnxt = func('stat_funcs', 'get_xpnxt', <<$temp_sklattrib['name']>>_lvl)
					<<$temp_sklattrib['name']>>_exp = <<$temp_sklattrib['name']>>_xpprv + 1
					<<$temp_sklattrib['name']>>_mem = <<$temp_sklattrib['name']>>_xpprv

					temp_sklattrib['max_deg'] = func('stat_funcs', 'get_max_deg', <<$temp_sklattrib['name']>>_lvl, <<$temp_sklattrib['name']>>_muta)
					<<$temp_sklattrib['name']>>_deg = rand(max(100, temp_sklattrib['max_deg'] / 2), max(150, 3 * temp_sklattrib['max_deg'] / 4))
				end
			"
			temp_sklattrib['index'] += 1
		if $temp_sklattrib['stat_type'] = 'attribute':
			if temp_sklattrib['index'] < arrsize('$att_name'): jump 'initloop_inner'
			$temp_sklattrib['stat_type'] = 'skill'
		else
			if temp_sklattrib['index'] < arrsize('$skl_name'): jump 'initloop_inner'
			$temp_sklattrib['stat_type'] = '结束'
		end
	if $temp_sklattrib['stat_type'] = 'skill': jump 'initloop_outer'

	killvar 'temp_sklattrib'
	exit
end
