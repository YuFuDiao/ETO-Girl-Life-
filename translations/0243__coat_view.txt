# coat_view


	!!====================================================================!!
	!!                                                                    !!
	!!                              FILTERS                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'filter_builder':


		!!------------------------------------------------------------!!
		!!                       GROUP HEADERS                        !!
		!!------------------------------------------------------------!!


	if $ARGS[1] = 'setup_home_filters':
		gs 'shop_utils', 'filter_builder', 'init', 'coat_view', 'view_grid', $shop_display['link']
			gs 'shop_utils', 'filter_builder', 'setup_number_sort'
			gs 'shop_utils', 'filter_builder', 'setup_quality_sort'
			gs 'shop_utils', 'filter_builder', 'setup_sort_direction'
		gs 'shop_utils', 'filter_builder', 'finish'
		exit
	end

	if $ARGS[1] = 'setup_shop_filter':
		gs 'shop_utils', 'filter_builder', 'init', $loc, $loc_arg
			gs 'shop_utils', 'filter_builder', 'setup_number_sort'
			gs 'shop_utils', 'filter_builder', 'setup_quality_sort'
			gs 'shop_utils', 'filter_builder', 'setup_price_sort'
			gs 'shop_utils', 'filter_builder', 'setup_discount_sort'
			gs 'shop_utils', 'filter_builder', 'setup_sort_direction'
		gs 'shop_utils', 'filter_builder', 'finish'
		exit
	end
	exit
end



	!!====================================================================!!
	!!                                                                    !!
	!!                                INIT                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'init':
	!! $ARGS[1]: function name (add_types, set_exceptions, add_to_sorted)

	if $ARGS[1] = 'add_types':
		!! $ARGS[2]: category to add

		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 5) = 'bomba':			gs 'shop_utils', 'init', 'add_type', 'bomba'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 4) = 'cats':			gs 'shop_utils', 'init', 'add_type', 'cats'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 10) = 'danilovich':	gs 'shop_utils', 'init', 'add_type', 'danilovich'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 5) = 'dolls':			gs 'shop_utils', 'init', 'add_type', 'dolls'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 2) = 'gm':				gs 'shop_utils', 'init', 'add_type', 'gm'
		if $ARGS[2] = 'all' or $mid($ARGS[2], 1, 8) = 'moncheri':		gs 'shop_utils', 'init', 'add_type', 'moncheri'
		exit
	end

	if $ARGS[1] = 'set_exceptions':
		!! shop_display_exceptions[$type + '_coats-' + number] = 1
		!!		i.e: shop_display_exceptions['gm_coats-6'] = 1
		exit
	end
	exit
end


if $ARGS[0] = 'sorted':
	if $ARGS[1] = 'add':
		!! $ARGS[2]: coat type
		!! ARGS[3]: coat number
		!! ARGS[4]: index

		gs 'coat_attributes', $ARGS[2], ARGS[3]

		if CoatQuality = 0: exit

		gs 'shop_utils', 'sorted', 'add_to_number', $ARGS[2], ARGS[3], ARGS[4]
		gs 'shop_utils', 'sorted', 'add_to_quality', $ARGS[2], ARGS[3], CoatQuality
		gs 'shop_utils', 'sorted', 'add_to_price', $ARGS[2], ARGS[3], CoatPrice
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                              DISPLAY                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'display':
	if $ARGS[1] = 'grid_shop':
		if $ARGS[2] = 'header':
			gs 'coat_view', 'filter_builder', 'setup_shop_filter'
			exit
		end

		if $ARGS[2] = 'main':
			!! $ARGS[3]: link
			!! $ARGS[4]: type
			!! ARGS[5]: number
			!! ARGS[6]: discount

			gs 'coat_attributes', $ARGS[4], ARGS[5]
			*p '<a href="exec: gt ''coat_view'', ''view_item'', ''<<$ARGS[3]>>'', ''<<$ARGS[4]>>'', <<ARGS[5]>>, <<ARGS[6]>>"><img height="250" src="<<$func(''$coat_image'', $ARGS[4], ARGS[5])>>"></a>'
			exit
		end

		if $ARGS[2] = 'footer':
			exit
		end
		exit
	end

	if $ARGS[1] = 'grid_wardrobe':
		if $ARGS[2] = 'header':
			gs 'shop_utils', 'display', 'wardrobe_storage_unwanted_header', 'view_grid'
			gs 'coat_view', 'filter_builder', 'setup_home_filters'
			exit
		end

		if $ARGS[2] = 'main':
			!! $ARGS[3]: link
			!! $ARGS[4]: type
			!! ARGS[5]: number
			!! ARGS[6]: discount

			gs 'coat_attributes', $ARGS[4], ARGS[5]
			*p '<a href="exec: gt ''coat_view'', ''view_item'', ''<<$ARGS[3]>>'', ''<<$ARGS[4]>>'', <<ARGS[5]>>, <<ARGS[6]>>"><img height="250" src="<<$func(''$coat_image'', $ARGS[4], ARGS[5])>>"></a>'
			exit
		end

		if $ARGS[2] = 'footer':
			exit
		end
		exit
	end
	exit
end


!! DEPRECATED
if $ARGS[0] = 'display_grid_shop':
	gs 'shop_utils', 'display', 'grid_shop'
	exit
end


!! DEPRECATED
if $ARGS[0] = 'display_grid_wardrobe':
	gs 'shop_utils', 'display', 'grid_wardrobe'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                             VIEW LIST                              !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'view_grid':
	!! $ARGS[1] - action type (wardrobe, store, unwanted, sell) -> $shop_display['link']
	menu_off = 1

	if shop_display['init'] = 0:
		gs 'shop_utils', 'init', 'start', 'coats', $ARGS[1]
			gs 'coat_view', 'init', 'add_types', 'all'
			if $ARGS[1] = 'wardrobe' or $ARGS[1] = 'storage' or $ARGS[1] = 'unwanted':
				gs 'shop_utils', 'init', 'set_return', 'wardrobe', 'main'
			else
				gs 'shop_utils', 'init', 'set_return', $loc, $loc_arg
			end
		gs 'shop_utils', 'init', 'end'
	end

	act '返回': gt 'shop_utils', 'return'

	if $coatworntype <> 'none':
		act '脱掉外套':
			gs 'coats', 'remove'
			gt 'coat_view', 'view_grid', $shop_display['link']
		end
	end

	gs 'shop_utils', 'display', 'grid_wardrobe'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                             VIEW ITEM                              !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'view_item':
	$shop_utils_view['link']		= $ARGS[1]		& !! action type (wardrobe, store, unwanted, bathroom, shop, cheat)
	$shop_utils_view['type']		= $ARGS[2]		& !! coat type
	shop_utils_view['number']		= ARGS[3]		& !! coat index
	shop_utils_view['discount']		= ARGS[4]		& !! price modifier (0-100, taken as %) {OPTIONAL}

	menu_off = 1
	gs 'stat'

	'<center><img <<$set_imgh>> src="<<$func(''$coat_image'', $shop_utils_view[''type''], shop_utils_view[''number''])>>"></center>'
	gs 'coat_attributes', $shop_utils_view['type'], shop_utils_view['number']
	*pl $coat_description

	if CoatWarm = 1:
		'它可以让你在零下10度的天气里保持相当温暖。'
	elseif CoatWarm = 2:
		'它能应对零下20度的低温。'
	elseif CoatWarm = 3:
		'即使在深冬，你也会非常暖和。'
	end

	if $shop_utils_view['link'] = 'shop' or $shop_utils_view['link'] = 'cheat': xgt 'coat_view', 'view_item_shop'
	if func('coats', 'is_wearing', $shop_utils_view['type'], shop_utils_view['number']):
		xgt 'coat_view', 'view_item_wearing'
	end
	if $shop_utils_view['link'] = 'wardrobe': xgt 'coat_view', 'view_item_wardrobe'
	if $shop_utils_view['link'] = 'storage': xgt 'coat_view', 'view_item_storage'
	if $shop_utils_view['link'] = 'unwanted': xgt 'coat_view', 'view_item_unwanted'

	act '返回': gt 'coat_view', 'view_grid', $shop_utils_view['link']
	exit
end


if $ARGS[0] = 'view_item_shop':
	act '离开': gt $loc, $loc_arg

	if func('coats', 'is_owned', $shop_utils_view['type'], shop_utils_view['number']):
		'你已经拥有这件物品。'
		exit
	end

	shop_utils_view['discount_total'] = shop_utils_view['discount'] + func('shop_utils', 'get_discount', $shop_utils_view['type'] + '_coats', shop_utils_view['number'])
	shop_utils_view['base_price'] = CoatPrice
	shop_utils_view['price'] = shop_utils_view['base_price'] * max(0, 100 - shop_utils_view['discount_total']) / 100
	shop_utils_view['price'] = shop_utils_view['price'] / 50 * 50
	if shop_utils_view['price'] = shop_utils_view['base_price']:
		$shop_utils_view['price_string'] = $func('money', 'string_price', shop_utils_view['price'])
	else
		$shop_utils_view['price_string'] = $func('wrap', 'neg s', $func('money', 'string_price', shop_utils_view['base_price'])) + ' <b>' +  $func('money', 'string_price', shop_utils_view['price']) + '</b>'
		'现在 ' + shop_utils_view['discount_total'] + '% 折扣' + iif(shop_utils_view['discount_total'] <= 10, '', '！')
	end

	'价格：' + $shop_utils_view['price_string']

	if func('money', 'can_afford', shop_utils_view['price']) = 0:
		'你买不起这件外套。'
	else
		act '购买 (<<$shop_utils_view[''price_string'']>>)':
			gs 'money', 'pay', shop_utils_view['price']
			gs 'coats', 'add_item', $shop_utils_view['type'], shop_utils_view['number']
			gt $loc, $loc_arg
		end
	end
	exit
end


if $ARGS[0] = 'view_item_wearing':
	act '返回': gt 'coat_view', 'view_grid', $shop_utils_view['link']
	'你正穿着这件外套。'
	if $shop_utils_view['link'] = 'wardrobe':
		act '脱下外套':
			gs 'coats', 'strip'
			gs 'shop_utils', 'cleanup'
			gt 'wardrobe', 'main'
		end
	end
	exit
end


if $ARGS[0] = 'view_item_wardrobe':
	act '返回': gt 'coat_view', 'view_grid', $shop_utils_view['link']
	act '穿上':
		gs 'coats', 'wear', $shop_utils_view['type'], shop_utils_view['number']
		gs 'shop_utils', 'cleanup'
		gt 'wardrobe', 'main'
	end
	gs 'coat_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_storage':
	act '返回': gt 'coat_view', 'view_grid', $shop_utils_view['link']
	'这件外套在储藏室里。'
	gs 'coat_view', 'storage_options'
	exit
end


if $ARGS[0] = 'view_item_unwanted':
	act '返回': gt 'coat_view', 'view_grid', $shop_utils_view['link']
	'这件外套是不需要的。'
	gs 'coat_view', 'storage_options'
	exit
end


if $ARGS[0] = 'storage_options':
	if func('coats', 'is_owned', $shop_utils_view['type'], shop_utils_view['number']) = 0: exit

	if func('coats', 'in_wardrobe', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到衣柜':
			gs 'coats', 'move_to_wardrobe', $shop_utils_view['type'], shop_utils_view['number']
			gt 'coat_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end

	if func('coats', 'is_immutable', $shop_utils_view['type'], shop_utils_view['number']): exit

	if func('coats', 'in_storage', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到存储':
			gs 'coats', 'move_to_storage', $shop_utils_view['type'], shop_utils_view['number']
			gt 'coat_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	if func('coats', 'in_unwanted', $shop_utils_view['type'], shop_utils_view['number']) = 0:
		act '移到不需要':
			gs 'coats', 'move_to_unwanted', $shop_utils_view['type'], shop_utils_view['number']
			gt 'coat_view', 'view_item', $shop_utils_view['link'], $shop_utils_view['type'], shop_utils_view['number'], shop_utils_view['discount']
		end
	end
	act $func('wrap', 'neg', '扔掉这件外套'):
		gs 'coats', 'remove_item', $shop_utils_view['type'], shop_utils_view['number']
		gt 'coat_view', 'view_grid', $shop_utils_view['link']
	end
	exit
end
