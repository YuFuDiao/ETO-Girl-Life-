if $ARGS[0] = 'add':
	if ARGS[2] <= 0: exit
	if func('tattoo_management', 'get_total', $ARGS[1]) < ARGS[2]: exit

	if pcs_tattoos[$ARGS[1]] <= 0: pcs_tattoos['total'] += 1
	pcs_tattoos['any'] = 1
	pcs_tattoos[$ARGS[1]] = ARGS[2]

	!! Visible placements earn both punk and goth; private placements earn neither.
	if $ARGS[1] = 'face' or $ARGS[1] = 'lip' or $ARGS[1] = 'neck' or $ARGS[1] = 'back' or $ARGS[1] = 'shoulder' or $ARGS[1] = 'chest' or $ARGS[1] = 'side' or $ARGS[1] = 'belly' or $ARGS[1] = 'arm' or $ARGS[1] = 'wrist' or $ARGS[1] = 'hand' or $ARGS[1] = 'leg' or $ARGS[1] = 'ankle':
		gs 'archetypes', 'gain', 'punk', 'tiny', '纹了明显的纹身'
		gs 'archetypes', 'gain', 'goth', 'tiny', '纹了明显的纹身'
	end

	exit
end


if $ARGS[0] = 'remove':
	if pcs_tattoos[$ARGS[1]] <= 0: exit

	pcs_tattoos['total'] -= 1
	pcs_tattoos[$ARGS[1]] = -pcs_tattoos[$ARGS[1]]
	if pcs_tattoos['total'] <= 0:
		pcs_tattoos['total'] = 0
		pcs_tattoos['any'] = 0
	end
	exit
end
