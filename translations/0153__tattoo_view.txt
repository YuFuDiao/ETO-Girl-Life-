# tattoo_view


	!!====================================================================!!
	!!                                                                    !!
	!!                                INIT                                !!
	!!                                                                    !!
	!!====================================================================!!



if $ARGS[0] = 'init':
	!! $ARGS[1]: init function (add_types, set_exceptions, add_to_sorted)

	if $ARGS[1] = 'add_types':
		!! $ARGS[2]: category to add

		if $ARGS[2] = 'all' or instr($ARGS[2], 'ankle'):	gs 'shop_utils', 'init', 'add_type', 'ankle'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'arm'):		gs 'shop_utils', 'init', 'add_type', 'arm'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'ass'):		gs 'shop_utils', 'init', 'add_type', 'ass'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'back'):		gs 'shop_utils', 'init', 'add_type', 'back'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'belly'):	gs 'shop_utils', 'init', 'add_type', 'belly'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'breast'):	gs 'shop_utils', 'init', 'add_type', 'breast'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'chest'):	gs 'shop_utils', 'init', 'add_type', 'chest'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'face'):		gs 'shop_utils', 'init', 'add_type', 'face'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'hand'):		gs 'shop_utils', 'init', 'add_type', 'hand'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'leg'):		gs 'shop_utils', 'init', 'add_type', 'leg'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'lip'):		gs 'shop_utils', 'init', 'add_type', 'lip'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'neck'):		gs 'shop_utils', 'init', 'add_type', 'neck'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'pussy'):	gs 'shop_utils', 'init', 'add_type', 'pussy'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'shoulder'):	gs 'shop_utils', 'init', 'add_type', 'shoulder'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'side'):		gs 'shop_utils', 'init', 'add_type', 'side'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'tramp'):	gs 'shop_utils', 'init', 'add_type', 'tramp'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'under'):	gs 'shop_utils', 'init', 'add_type', 'under'
		if $ARGS[2] = 'all' or instr($ARGS[2], 'wrist'):	gs 'shop_utils', 'init', 'add_type', 'wrist'
		exit
	end


	if $ARGS[1] = 'set_exceptions':
		!! shop_display_exceptions['tattoos_' + $location + '-' + number] = 1
		!!		i.e: shop_display_exceptions['tattoos_ankle-2'] = 1

		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                               SORTED                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'sorted':
	if $ARGS[1] = 'add':
		!! $ARGS[2]: bra type
		!! ARGS[3]: bra number
		!! ARGS[4]: index

		gs 'tattoo_attributes', $ARGS[2], ARGS[3]

		if TatQuality = 0: exit

		gs 'shop_utils', 'sorted', 'add_to_number', $ARGS[2], ARGS[3], ARGS[4]
		gs 'shop_utils', 'sorted', 'add_to_quality', $ARGS[2], ARGS[3], TatQuality
		gs 'shop_utils', 'sorted', 'add_to_price', $ARGS[2], ARGS[3], TatPrice
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                              DISPLAY                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'display':
	if $ARGS[1] = 'grid_shop':
		if $ARGS[2] = 'header':
			exit
		end

		if $ARGS[2] = 'main':
			!! $ARGS[3]: link
			!! $ARGS[4]: type
			!! ARGS[5]: number
			!! ARGS[6]: discount

			gs 'tattoo_attributes', $ARGS[4], ARGS[5]
			*p '<a href="exec: gt ''tattoo_view'', ''view_item'', ''<<$ARGS[3]>>'', ''<<$ARGS[4]>>'', <<ARGS[5]>>, <<ARGS[6]>>"><img height="250" src="<<$func(''tattoo_management'', $ARGS[4] + ''_image'', ARGS[5])>>"></a>'
			exit
		end

		if $ARGS[2] = 'footer':
			exit
		end
		exit
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                             VIEW ITEM                              !!
	!!                                                                    !!
	!!====================================================================!!


!! [1] - action_type
!! [2] - slot
!! [3] - index
!! [4] - pric
if $ARGS[0] = 'view_item':
	$shop_utils_view['link']		= $ARGS[1]		& !! action type (wardrobe, store, unwanted, bathroom, shop, cheat)
	$shop_utils_view['type']		= $ARGS[2]		& !! bra type
	shop_utils_view['number']		= ARGS[3]		& !! bra index
	shop_utils_view['discount']		= ARGS[4]		& !! price modifier (0-100, taken as %) {OPTIONAL}

	gs 'tattoo_attributes', $shop_utils_view['type'], shop_utils_view['number']

	'<center><img <<$set_imgh>> src="<<$func(''tattoo_management'', $shop_utils_view[''type''] + ''_image'', shop_utils_view[''number''])>>"></center>'

	if $shop_utils_view['link'] = 'shop': xgt 'tattoo_view', 'view_item_shop'

	act '返回': gt $loc, $loc_arg
	exit
end


if $ARGS[0] = 'view_item_shop':
	shop_utils_view['discount_total'] = shop_utils_view['discount'] + func('shop_utils', 'get_discount', $shop_utils_view['type'] + '_tattoos', shop_utils_view['number'])
	shop_utils_view['base_price'] = TatPrice
	shop_utils_view['price'] = shop_utils_view['base_price'] * max(0, 100 - shop_utils_view['discount_total']) / 100
	shop_utils_view['price'] = shop_utils_view['price'] / 50 * 50
	if shop_utils_view['price'] = shop_utils_view['base_price']:
		$shop_utils_view['price_string'] = $func('money', 'string_price', shop_utils_view['price'])
	else
		$shop_utils_view['price_string'] = $func('wrap', 'neg s', $func('money', 'string_price', shop_utils_view['base_price'])) + ' <b>' +  $func('money', 'string_price', shop_utils_view['price']) + '</b>'
		'现在 ' + shop_utils_view['discount_total'] + '% 折扣' + iif(shop_utils_view['discount_total'] <= 10, '', '！')
	end

	'价格：' + $shop_utils_view['price_string']

	if func('money', 'can_afford', shop_utils_view['price']) = 0:
		'你买不起这个纹身。'
		act '返回': gt $loc, $loc_arg
		exit
	end

	act '纹这个纹身 (<<$shop_utils_view[''price_string'']>>)':
		minut += 60
		gs 'money', 'pay', shop_utils_view['price']
		gs 'tattoo_management', 'add', $shop_utils_view['type'], shop_utils_view['number']
		gs 'tattoo_management', 'count'
		gs 'stat'
		gt $loc, $loc_arg
	end
	act '返回': gt $loc, $loc_arg
end
