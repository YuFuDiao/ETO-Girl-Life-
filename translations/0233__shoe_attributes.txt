# shoe_attributes

!!{
	**Shoe Organization
	Shoe Categories: moncheri, bomba, dolls, eroto, gm, danilovich

	ShoExhibit: 0-100 Exhibitionism value required to wear the outfit

	**Shoe Variables

	ShoStyle = 1 Different styles of footwear which might give bonuses to certain npcs or events (0 if it does not fit these styles)
				1 Alternative - For all good rebels might help your willpower

	ShoStyle2:	1 Stripper, 2 Exercise

	ShoQuality = 1-7 scale, based on the general expensiveness of the shoe. Higher is more expensive.
		Notes: 7 is for shoes that might only be found on celebrities or millionaires at red carpet events.
		Ex:		6-5 are the higher and lower levels of the expensive category
				4-3 are the same for average category
				2-1 are the same for cheap category.

	ShoHeels = 1-6 scale, with higher values being taller heels.
		Ex:		6 eroto heel over 15cm (6")
				5 Ultra high heel around 15cm (6")
				4 High heel around 10cm (4")
				3 Mid heel around 7.6cm (3")
				2 low heel around 5cm (2")
				1 Flats less than 5cm (2")

	ShoCut = 1-6 scale from shoe to groin high boots
		Ex:		6 groin high
				5 Thigh high
				4 for approx. knee high
				3 for calf high
				2 for ankle high
				1 for shoes

	ShoPrice: The full price of the shoes

	ShoStrip = 1: Stripper shoes
	ShoSport = 1: Sport shoes
	ShoBimbo = 1: Bimbo shoes
	ShoGoth = 1: Goth shoes
	ShoPunk = 1: Punk shoes
!!}


gs 'shoes', 'reset_ShoVars'


if loc('$attributes_<<$ARGS[0]>>_shoes'): gs '$attributes_<<$ARGS[0]>>_shoes', ARGS[1]
!! gs 'mod_system', 'outfit', 'shoe_attributes', $ARGS[0], ARGS[1]

if ShoQuality = 0: exit


if ShoHeels < 2:
	!! Flat
	ShoSkill = 0
	ShoPain['severe'] = 0
	ShoPain['medium'] = 0
	ShoPain['mild'] = 0
elseif ShoHeels = 2:
	!! Low heel
	ShoSkill = 0
	ShoPain['severe'] = 0
	ShoPain['medium'] = 25
	ShoPain['mild'] = 40
elseif ShoHeels = 3:
	!! Mid heel
	ShoSkill = 15
	ShoPain['severe'] = 0
	ShoPain['medium'] = 35
	ShoPain['mild'] = 50
elseif ShoHeels = 4:
	!! High heel
	ShoSkill = 30
	ShoPain['severe'] = 0
	ShoPain['medium'] = 45
	ShoPain['mild'] = 60
elseif ShoHeels = 5:
	!! Ultra high heel
	ShoSkill = 45
	ShoPain['severe'] = 55
	ShoPain['medium'] = 65
	ShoPain['mild'] = 75
else
	!! Fetish heel
	ShoSkill = 60
	ShoPain['severe'] = 70
	ShoPain['medium'] = 80
	ShoPain['mild'] = 90
end

ShoMaxStrength = ShoStrength

if func('shoes', 'is_owned', $ARGS[0], ARGS[1]):
	dynamic "
		ShoStrength = <<$ARGS[0]>>_shoe_h[<<ARGS[1]>>]
	"
end
