	!! Returns: $property_name[agentned_i], $property_code[agentned_i], $property_display[agentned_i], property_rent[agentned_i], property_days[agentned_i]
	gs 'homes_properties', 'get_rented_properties', 'home'
	agentned_count = arrsize('$property_name')
	if agentned_count > 0:
		agentned_i = 0
		'<h4>租赁信息</h4>'
		:listing_rented_properties_loop
			'你的<<$property_display[agentned_i]>>租约还剩<b><<property_days[agentned_i]>>天</b>，你的月租金是<<$func(''money'', ''string_price'', property_rent[agentned_i])>>。 ' + iif(func('money', 'can_afford', property_rent[agentned_i]) = 0, '你现在付不起租金。', '你可以支付一笔租金 <a href="exec: gs ''homes_properties'', ''pay_rent'', ''<<$property_code[agentned_i]>>'' & gt ''agentned'', ''start''">付款</a>来延长租期。')
			dynamic "
				act ""取消<<$property_display[agentned_i]>>的租约"":
					cla & *nl
					!! TODO: 1 month notice period like in real life?
					'注意：一旦取消租约，你将无法再回到你的公寓。'
					act '返回': gt 'agentned', 'start'
					act '取消租约':
						gs 'homes_properties', 'cancel_rent', '<<$property_code[agentned_i]>>'
						gt 'agentned', 'start'
					end
				end
			"
			agentned_i += 1
		if agentned_i < agentned_count: jump 'listing_rented_properties_loop'
	end

	!! Get a list of the properties and their attributes that the player owns:
	!! $property_code[], $property_name[], $property_display[], $property_status_display[], $property_type[]
	!! property_construction_status, property_is_renovated[], property_sales_price[], property_renovation_value[]
	gs 'homes_properties', 'get_owned_properties', 'home'
	agentned_count = arrsize('$property_name')
	if agentned_count > 0:
		'<h4>你拥有以下房产</h4>'
		agentned_i = 0
		:listing_owned_properties_loop
			*pl $property_status_display[agentned_i]

		if agentned_i < agentned_count: jump 'listing_owned_properties_loop'
		*nl
		'如果你决定出售，我们随时为你服务<a href="exec:gt ''agentned'',''sell''">出售</a>。'
		*nl
	end

	act '离开': minut += 5 & gt 'city_center'

	gs 'homes_properties', 'clean_up_property_data'
	killvar 'agentned_count'
	killvar 'agentned_i'
end

	!! "Returns":
	!! $property_code[], $property_name[], $property_display[], $property_agency_name[],

	gs 'homes_properties', 'get_properties_for_sale'

		'我们目前没有挂牌出售的房产。'
	else
		agentned_i = 0
		'<h4>我们有以下待售房产</h4>'
		:listing_properties_for_sale_loop
			if property_is_rented[agentned_i] = 0:
				'有一处<<$property_display[agentned_i]>>可以出售，价格为<<func(''money'', ''string_price'', property_sales_price[agentned_i])>>。'
			else
				'你可以买下你目前租的<<$property_display[agentned_i]>>，价格为<<func(''money'', ''string_price'', property_sales_price[agentned_i])>>。'
			end
			if $property_code[agentned_i] = 'city_apartment' or $property_code[agentned_i] = 'village_cottage' or $property_code[agentned_i] = 'city_house' or $property_code[agentned_i] = 'matryona_mansion':
				dynamic "
						act ""买下<<$property_display[agentned_i]>>"" + $func('money', 'get_cost_string', <<property_sales_price[agentned_i]>>, 'bank'):
							if func('money', 'can_afford', <<property_sales_price[agentned_i]>>, 'bank') = 0:
								$noMoney
							else
								gt 'agentned', 'buy_property', '<<$property_code[agentned_i]>>'
						end
					end
				"
			end
			agentned_i += 1
		if agentned_i < agentned_count: jump 'listing_properties_for_sale_loop'
	end
	gs 'homes_properties', 'clean_up_property_data'
	killvar 'agentned_count'
	killvar 'agentned_i'
end

!! NOTE: When a property is sold, it can be bought back. To avoid a guaranteed income from the
!! "I buy property for X, renovate it, sell it for X+Y, then buy it for X, then repeat" loop,
!! the sales price of the property is updated with the price that it was sold for.
if $ARGS[0] = 'sell':
	*clr & cla
	menu_off = 0
	gs 'core_library', 'setloc', 'agentned', 'sell'
	$location_type = 'public_indoors'
	gs 'homes_properties', 'clean_up_property_data'

	$func('wrap', 'header center b', '房地产中介')
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/realestate/agenstvo.jpg"></center>'
	*nl
	'<h4>任何房产的购买或出售都需要银行账户。根据反洗钱条例，中介公司不接受现金交易</h4>'
	*nl

	act '返回':
		gs 'homes_properties', 'clean_up_property_data'
		gt 'agentned', 'start'
	end

	if bankAccount = 0:
		'<b>你需要开设一个银行账户才能出售你的房产</b>'
	else
		!! Get a list of the properties and their attributes that the player rented:
		!! $property_code[], $property_name[], $property_display[], $property_status_display[], $property_type[]
		!! property_construction_status, property_is_renovated[], property_sales_price[], property_renovation_value[]
		gs 'homes_properties', 'get_owned_properties', 'home'

		agentned_count = arrsize('$property_name')
		if agentned_count > 0:
			'<h4>你的房产的可用报价</h4>'
			agentned_i = 0
			:listing_offers_on_owned_properties_loop
				if $property_code[agentned_i] <> 'matryona_mansion' or ($property_code[agentned_i] = 'matryona_mansion' and property_construction_status[agentned_i] = 0):
					!! NOTE: Offer: (last sales price + rand(-10%, +10%)) + (renovation value + rand(0%,+10%)
					!!		No need to check what has been renovated, the only thing that matters is the total value of the renovation
					!!		If Sveta sells, this will be stored as the new sales price of the property

					offer = property_sales_price + property_renovation_value + (property_sales_price * rand(-10, 10)) / 100 + (property_renovation_value * rand(0, 10)) / 100
					if $property_code[agentned_i] = 'city_apartment' or $property_code[agentned_i] = 'village_cottage' or $property_code[agentned_i] = 'city_house' or $property_code[agentned_i] = 'matryona_mansion':
						'我们已经为你的<<$property_display[agentned_i]>>找到了买家。他们将出价<<$func(''money'', ''string_profit'', offer)>>，减去5%的手续费和税费，折合<<$func(''money'', ''string_profit'', offer * 5 / 100)>>，总计为<<$func(''money'', ''string_profit'', offer * 95 / 100)>>'
						dynamic "
							act ""以<<$func('money', 'string_profit', offer * 95 / 100)>>出售<<$property_display[agentned_i]>>"":
								gt 'agentned', 'finalise_sale', '<<$property_code[agentned_i]>>', <<agentned_i>>, <<offer>>
							end
						"
					end
				end
				agentned_i += 1
			if agentned_i < agentned_count: jump 'listing_offers_on_owned_properties_loop'

		end
		gs 'homes_properties', 'clean_up_property_data'
		killvar 'agentned_count'
		killvar 'agentned_i'
		killvar 'j'
		act '离开': minut += 5 & gt 'city_center'

	!! NOTE: Keeping this here for future renovation related reference.
	!!{
	if bankAccount = 1:
		if rembedr = 1 and remsitr = 1 and remkorr = 1 and remvanr = 1 and remkuhr = 1:
			!! Renovated city residential apartment
			offer1 = ((800000 + 250000) + rand(-100000, 100000))
		else
			offer1 = (800000 + rand(-100000, 100000))
		end

		if func('homes_properties', 'property_renovated', 'village_cottage') = 1:
			!! Renovated communal cottage
			offer2 = ((prop_price['village_cottage'] + 200000) + rand(-10000, 10000))
		else
			!! Communal cottage
			offer2 = (prop_price['village_cottage'] + rand(-5000, 5000))
		end
	!}
	end
end
