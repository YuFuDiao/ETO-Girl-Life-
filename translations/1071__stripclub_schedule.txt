# stripclub_schedule


if $ARGS[0] = 'do_book':
	!! Helper: book a strip club shift via v4 API
	!! gs 'stripclub_schedule', 'do_book', weekday, shift_num, next_week_flag
	!! weekday: 1=Mon..7=Sun, shift_num: 1 or 2, next_week_flag: 0=this week, 1=next week
	temp_sb_weekday = ARGS[1]
	temp_sb_shift = ARGS[2]
	temp_sb_next = ARGS[3]

	!! Calculate the target daystart from current day + weekday offset
	temp_sb_offset = (temp_sb_weekday - week + 7) mod 7
	if temp_sb_next = 1: temp_sb_offset += 7
	!! If offset is 0 (today) and it's this week, that''s valid (booking today's remaining shift)
	temp_sb_daystart = daystart + temp_sb_offset

	!! Map shift number (1-based) to slot index (0-based) for v4 API
	temp_sb_slot = temp_sb_shift - 1

	!! Book via v4 (handles count increment + calendar event + conflict check)
	!! Store shift number as booking data for downstream queries
	temp_sb_result = func('jobs', 'book_slot', 'city_strip_stripper', temp_sb_daystart, temp_sb_slot, $str(temp_sb_shift))

	killvar 'temp_sb_weekday'
	killvar 'temp_sb_shift'
	killvar 'temp_sb_next'
	killvar 'temp_sb_offset'
	killvar 'temp_sb_daystart'
	killvar 'temp_sb_slot'
	killvar 'temp_sb_result'
end


if $ARGS[0] = 'do_cancel':
	!! Helper: cancel a booked strip club shift via v4 API
	!! gs 'stripclub_schedule', 'do_cancel', weekday, shift_num, next_week_flag
	temp_sc_weekday = ARGS[1]
	temp_sc_shift = ARGS[2]
	temp_sc_next = ARGS[3]

	!! Calculate the target daystart
	temp_sc_offset = (temp_sc_weekday - week + 7) mod 7
	if temp_sc_next = 1: temp_sc_offset += 7
	temp_sc_daystart = daystart + temp_sc_offset

	!! Map shift number (1-based) to slot index (0-based) for v4 API
	temp_sc_slot = temp_sc_shift - 1

	!! Cancel via v4 (handles count decrement + calendar event removal)
	gs 'jobs', 'cancel_booking', 'city_strip_stripper', temp_sc_daystart, temp_sc_slot

	killvar 'temp_sc_weekday'
	killvar 'temp_sc_shift'
	killvar 'temp_sc_next'
	killvar 'temp_sc_offset'
	killvar 'temp_sc_daystart'
	killvar 'temp_sc_slot'
end


!! ARGS[1] = week (1: monday, …, 7: sunday)
!! ARGS[2] = shift (1: 18-00, 2: 21-03)
!! ARGS[3] = next_week (0: this week, 1: next week)
if $ARGS[0] = 'display_single_shift':
	if ARGS[3] = 0:
		$scs_return_arg = 'set_schedule'
	elseif ARGS[3] = 1:
		$scs_return_arg = 'next_week_set_schedule'
	end

	if ARGS[2] = 1:
		$scs_time_string = '18:00-00:00'
		scs_hour1 = 9
		scs_hour2 = 13
	elseif ARGS[2] = 2:
		$scs_time_string = '21:00-03:00'
		scs_hour1 = 13
		scs_hour2 = 17
	end

	!! Calculate daystart for this slot and check v4 booking data
	scs_offset = (ARGS[1] - week + 7) mod 7
	if ARGS[3] = 1: scs_offset += 7
	scs_day = daystart + scs_offset
	scs_slot = ARGS[2] - 1
	$scs_booking = $func('jobs', 'get_booking_data', 'city_strip_stripper', scs_day, scs_slot)
	scs_has_conflict = func('jobs', 'check_booking_conflict', 'city_strip_stripper', scs_day, scs_slot)

	$result +=	'<tr><td>'
	if $scs_booking = '' and no (ARGS[3] = 0 and week = ARGS[1] and hour >= scs_hour1 and hour < scs_hour2):
		!! Slot is available — show booking link (or grey if conflict)
		if scs_has_conflict = 1:
			$result +=		'<font color="grey"><<$scs_time_string>>班次（繁忙）</font>'
		else
			$result +=		'<a href="exec:gs ''stripclub_schedule'', ''do_book'', <<ARGS[1]>>, <<ARGS[2]>>, <<ARGS[3]>> & gs ''stripclub_schedule'', ''<<$scs_return_arg>>''"><<$scs_time_string>>班次</a>'
		end
	elseif ARGS[3] = 0 and week = ARGS[1] and hour >= scs_hour1 and hour < scs_hour2:
		!! Shift is currently in progress — not bookable
		$result +=		'<<$scs_time_string>>班次'
		$result +=	'</td></tr><tr><td>'
		$result +=		func('stripclub_schedule', 'random_stripper_name')
		$result +=	'</td></tr><tr><td>'
		$result +=		func('stripclub_schedule', 'random_stripper_name')
	elseif scs_day < daystart:
		!! Past day — already worked
		$result +=		'<<$scs_time_string>>班次'
		$result +=	'</td></tr><tr><td>'
		$result +=		'<i>已工作过</i>'
	else
		!! Booked by player — show cancel link
		$result +=		'<<$scs_time_string>>班次'
		$result +=	'</td></tr><tr><td>'
		$result +=		'<a href="exec:gs ''stripclub_schedule'', ''do_cancel'', <<ARGS[1]>>, <<ARGS[2]>>, <<ARGS[3]>> & gs ''stripclub_schedule'', ''<<$scs_return_arg>>''">取消班次</a>'
	end
	$result +=	'</td></tr>'

	killvar 'scs_return_arg'
	killvar 'scs_time_string'
	killvar 'scs_hour1'
	killvar 'scs_hour2'
	killvar 'scs_offset'
	killvar 'scs_day'
	killvar 'scs_slot'
	killvar 'scs_booking'
	killvar 'scs_has_conflict'
end
