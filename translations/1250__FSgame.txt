gs 'stat'

'<center><b>酒店房间</b></center>'
''

if FSmedkit > 0:
	'急救包<<FSmedkit>>个'

	if FSHP < 100:
		act '治疗':
			cls
			FShour += 1
			FSHP = 100
			FSmedkit -= 1
			minut += 1
			gs 'stat'
			'你靠急救包恢复了生命值。'

			act '进一步': gt $curloc
		end
	end
end

act '闲待着':
	cls
	FShour += 1
	minut += 1
	gs 'stat'
	'你在公寓里闲待着，瘫在沙发上琢磨着该怎么摆脱眼前的困境。'

	act '进一步': gt $curloc
end

act '离开房间':
	cls
	FShour += 1
	minut += 1
	gs 'stat'
	!! 'You left the room.'
	gt 'FSroom'
end
