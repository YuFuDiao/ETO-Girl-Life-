if $ARGS[0]='food':
	*clr & cla
	minut += 1
	menu_off = 1
	gs 'stat'
	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '杂货区')
	'<center><img <<$set_imgh>> src="images/locations/pushkin/cafe/larek.jpg" ></center>'
	'<a href="exec: gs ''larek'', ''buyfood'' ">一份食物 - 50 卢布</a>'
	'<a href="exec: gs ''larek'', ''buyfood2'' ">一份健康食品 - 250 卢布</a>'

	act '退出部门': gt 'larek', 'start'

	act '买茶点饼干' + $func('money', 'get_cost_string', 300):
		if func('money', 'can_afford', 300) = 0:
			$noMoney
		else
			cla
			gs 'money', 'pay', 300
			mc_inventory['food_biscuits'] += 10
			gs 'stat'

			'你买了一袋与茶搭配的饼干。你估计里面大概有十份。'
			act '退出部门': gt 'larek', 'food'
		end
	end
	if bottle=0 and bag>0:
		act '买一瓶水' + $func('money', 'get_cost_string', 50):
			if func('money', 'can_afford', 50) = 0:
				$noMoney
			else
				cla
				gs 'money', 'pay', 50
				mc_inventory['food_water'] = 1
				gs 'stat'

				'你买了一瓶水，放进了钱包里。'
				act '退出部门': gt 'larek', 'food'
			end
		end
	end
	if mc_inventory['food_sandwich']=0 and bag>0:
		act '买一个三明治' + $func('money', 'get_cost_string', 100):
			if func('money', 'can_afford', 100) = 0:
				$noMoney
			else
				cla
				gs 'money', 'pay', 100
				mc_inventory['food_sandwich'] = 1
				gs 'stat'

				'你买了一个三明治，放进了钱包里。'
				act '退出部门': gt 'larek', 'food'
			end
		end
	end
	if mc_inventory['food_wine'] = 0 and bag > 0:
		act '买一瓶廉价葡萄酒' + $func('money', 'get_cost_string', 500):
			if func('money', 'can_afford', 500) = 0:
				$noMoney
			else
				cla
				gs 'money', 'pay', 500
				mc_inventory['food_wine'] = 1
				gs 'stat'

				'你买了一瓶葡萄酒，放进了钱包里。'
				act '退出部门': gt 'larek', 'food'
			end
		end
	end
end


if $ARGS[0] = 'buyfood':
	if hour < 8 or hour >= 21:
		'食品杂货区已关闭'
	else
		if func('money', 'can_afford', 50) = 0:
			'你没有足够的钱'
		else
			gt 'larek', 'buyfood_1'
		end
	end

if $ARGS[0] = 'buyfood_1':
	*clr & cla
	minut += 15
	gs 'stat'

	'<center><img <<$set_imgh>> src="images/locations/pushkin/cafe/larek.jpg" ></center>'
	portion = input ("你想买多少份？")
	if portion <= 0: portion = 1
	if func('money', 'can_afford', portion * 50) = 0:
		'你的钱不够。'
	else
		mc_inventory['food_basic'] += portion
		gs 'money', 'pay', portion * 50

		'你用现金付了饭钱。'
	end
	act '离开柜台': gt 'larek', 'food'
end
