# piercing_management


if $ARGS[0] = 'get_total':
	if $ARGS[1] = 'ears':
		result = 24
	elseif $ARGS[1] = 'nose':
		result = 7
	elseif $ARGS[1] = 'brow':
		result = 2
	elseif $ARGS[1] = 'lip':
		result = 7
	elseif $ARGS[1] = 'tongue':
		result = 4
	elseif $ARGS[1] = 'navel':
		result = 4
	elseif $ARGS[1] = 'nipples':
		result = 7
	elseif $ARGS[1] = 'pussy':
		result = 10
	end
	exit
end


if $ARGS[0] = 'totals':
	total = func('piercing_management', 'get_total', $ARGS[1])
	exit
end


if $ARGS[0] = 'is_owned':
	if $ARGS[1] = '': exit
	if ARGS[2] = 0: exit

	result = (pcs_piercings['<<$ARGS[1]>>_<<ARGS[2]>>_owned'] <> 0)
	exit
end


if $ARGS[0] = 'is_pierced':
	if $ARGS[1] = '': exit

	result = pcs_piercings[$ARGS[1]] <> 0
	exit
end


if $ARGS[0] = 'is_wearing':
	if $ARGS[1] = '': exit
	if ARGS[2] = 0: exit

	result = (pcs_piercings[$ARGS[1]] = ARGS[2])
	exit
end


if $ARGS[0] = 'is_wearing_any':
	if $ARGS[1] = '':
		result = (pcs_piercings['total'] > 0)
	else
		result = (pcs_piercings[$ARGS[1]] > 0)
	end
	exit
end


if $ARGS[0] = 'set_manage_string':
	if pcs_piercings['total'] > 0:
		'<a href="exec: gt ''piercing_management'', ''main''">管理穿孔</a>'
	end
	exit
end

	end
	exit
end


if $ARGS[0] = 'pierce':
	if pcs_piercings[$ARGS[1]] <> 0: exit
	if func('piercing_management', 'get_total', $ARGS[1]) = 0: exit

	pcs_piercings['total'] += 1
	pcs_piercings[$ARGS[1]] = -999
	exit
end


if $ARGS[0] = 'add':
	if ARGS[2] <= 0: exit
	if func('piercing_management', 'get_total', $ARGS[1]) < ARGS[2]: exit

	gs 'piercing_management', 'pierce', $ARGS[1]
	pcs_piercings['<<$ARGS[1]>>_<<ARGS[2]>>_owned'] = 1
	gs 'piercing_management', 'wear', $ARGS[1], ARGS[2]

	if $ARGS[1] = 'brow' or $ARGS[1] = 'nose' or $ARGS[1] = 'lip' or $ARGS[1] = 'tongue':
		gs 'archetypes', 'gain', 'punk', 'small', '面部穿孔'
		gs 'archetypes', 'gain', 'goth', 'tiny', '面部穿孔'
	elseif $ARGS[1] = 'navel':
		gs 'archetypes', 'gain', 'punk', 'tiny', '肚脐穿孔'
	end

	exit
end


if $ARGS[0] = 'wear':
	if ARGS[2] < 0: ARGS[2] = -ARGS[2]
	if pcs_piercings[$ARGS[1]] = 0 or ARGS[2] = 0: exit
	if pcs_piercings['<<$ARGS[1]>>_<<ARGS[2]>>_owned'] = 0: exit

	if pcs_piercings[$ARGS[1]] < 0: pcs_piercings['wearing'] += 1
	pcs_piercings[$ARGS[1]] = ARGS[2]
	exit
end


if $ARGS[0] = 'wear_last':
	if pcs_piercings[$ARGS[1]] >= 0: exit

	gs 'piercing_management', 'wear', $ARGS[1], -pcs_piercings[$ARGS[1]]
	exit
end


if $ARGS[0] = 'remove':
	if pcs_piercings[$ARGS[1]] <= 0: exit

	pcs_piercings['wearing'] -= 1
	pcs_piercings[$ARGS[1]] = -pcs_piercings[$ARGS[1]]
	exit
end


if $ARGS[0] = 'image':

end

	exit
end


if $ARGS[0] = 'count':
	pcs_piercings['total'] = 0
	pcs_piercings['wearing'] = 0
	if pcs_piercings['ears'] ! 0:
		pcs_piercings['total'] += 1
		if pcs_piercings['ears'] > 0: pcs_piercings['wearing'] += 1
	end

		pcs_piercings['total'] += 1
		if pcs_piercings['nose'] > 0: pcs_piercings['wearing'] += 1
	end

		pcs_piercings['total'] += 1
		if pcs_piercings['brow'] > 0: pcs_piercings['wearing'] += 1
	end

		pcs_piercings['total'] += 1
		if pcs_piercings['lip'] > 0: pcs_piercings['wearing'] += 1
	end
	if pcs_piercings['tongue'] ! 0:
		pcs_piercings['total'] += 1
		if pcs_piercings['tongue'] > 0: pcs_piercings['wearing'] += 1
	end

		pcs_piercings['total'] += 1
		if pcs_piercings['navel'] > 0: pcs_piercings['wearing'] += 1
	end

		pcs_piercings['total'] += 1
		if pcs_piercings['nipples'] > 0: pcs_piercings['wearing'] += 1
	end

		pcs_piercings['total'] += 1
		if pcs_piercings['pussy'] > 0: pcs_piercings['wearing'] += 1
	end
	exit
end
