# billiard
$sexloc = 'billiard'
$menu_loc = 'billiard'
$menu_arg = ''
menu_off = 0
$location_type = 'public_indoors'
cla
*clr
gs 'stat'
'<center><b>台球</b></center>'
'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/pool/bil.jpg"></center>'
'这是一个相当大的台球室，主厅里有很多球桌，但也有一些球桌在单独的小隔间里。'

act '离开': gt 'city_mall'

if hour >= 8 and hour <= 20:
	act '打台球 (1:00)' + $func('money', 'get_cost_string', 125):
		if func('money', 'can_afford', 125) = 0:
			$noMoney
		else
			menu_off = 1
			gs 'money', 'pay', 125
			minut += 60
			billvar = 0
			gs 'exp_gain', 'pool', rand(0, 3)
			billrand = rand(0, 5)
			gs 'stat'

			if billrand = 0:
				cla
				'你和这里的一个男人打了一局，他问你想不想让比赛更有意思。'

				act '为比赛下注' + $func('money', 'get_cost_string', 1000, 'cash'):
					if func('money', 'can_afford', 1000, 'cash') = 0:
						$noMoney
					else
						billvar = 0 & gt 'billiard', 'randwin'
					end
				end

				if mc_inventory['strapon'] = 1: act '赌他的屁股，输了就做爱': billvar = 1 & gt 'billiard', 'randwin'

				gs 'willpower', 'anal', 'resist'
				if pcs_willpwr < will_cost:
					act '坚持练习赛' + $func('willpower', 'get_willcost_string'): $noWillpower
				else
					act '坚持练习赛' + $func('willpower', 'get_willcost_string'):
						gs 'willpower', 'anal', 'resist'
						gs 'willpower', 'pay', 'resist'
						gs 'stat'
						gt 'billiard'
					end
				end

				act '让他用桌子吧':
					gs 'money', 'earn', 125
					gs 'exp_gain', 'pool', rand(0, 1)
					gs 'stat'
					gt 'billiard'
				end
			elseif billrand = 1:
				cla
				'你和这里的几个男人打了几局，他们问你想不想让比赛更有意思。'

				act '为比赛下注' + $func('money', 'get_cost_string', 1000, 'cash'):
					if func('money', 'can_afford', 1000, 'cash') = 0:
						$noMoney
					else
						billvar = 0 & gt 'billiard', 'randwin'
					end
				end

				if mc_inventory['strapon'] = 1: act '赌他们的屁股，输了就做爱': billvar = 2 & gt 'billiard', 'randwin'

				gs 'willpower', 'anal', 'resist', 'hard'
				if pcs_willpwr < will_cost:
					act '坚持练习赛' + $func('willpower', 'get_willcost_string'): $noWillpower
				else
					act '坚持练习赛' + $func('willpower', 'get_willcost_string'):
						gs 'willpower', 'anal', 'resist', 'hard'
						gs 'willpower', 'pay', 'resist'
						gs 'stat'
						gt 'billiard'
					end
				end

				act '让他用桌子吧':
					gs 'money', 'earn', 125
					gs 'exp_gain', 'pool', rand(0, 1)
					gs 'stat'
					gt 'billiard'
				end
			elseif billrand >= 2:
				cla
				'你自己一个人打了会儿台球。'

				act '离开': gt 'billiard'
			end
		end
	end

	act '打钱局' + $func('money', 'get_cost_string', 250):
		if func('money', 'can_afford', 250) = 0:
			$noMoney
		else
			menu_off = 1
			minut += 60
			gs 'exp_gain', 'pool', rand(0, 1)
			billvar = 3
			gt 'billiard', 'randwin'
		end
	end
end


if $ARGS[0] = 'randwin':
	cla
	menu_off = 1
	if pcs_pool < 10: billwin = rand(0, 10)
	if pcs_pool >= 10 and pcs_pool < 20: billwin = rand(0, 12)
	if pcs_pool >= 20 and pcs_pool < 30: billwin = rand(0, 14)
	if pcs_pool >= 30 and pcs_pool < 40: billwin = rand(0, 16)
	if pcs_pool >= 40 and pcs_pool < 50: billwin = rand(0, 18)
	if pcs_pool >= 50 and pcs_pool < 60: billwin = rand(0, 20)
	if pcs_pool >= 60 and pcs_pool < 70: billwin = rand(0, 22)
	if pcs_pool >= 70 and pcs_pool < 80: billwin = rand(0, 24)
	if pcs_pool >= 80 and pcs_pool < 90: billwin = rand(0, 26)
	if pcs_pool >= 90: billwin = rand(0, 28)

	if billwin > 10 and billvar = 0:
		gs 'money', 'earn', 1000, 'cash'
		'你赢了并获得 ' + $func('money', 'string_profit', 1000) + '。'

		act '离开': gt 'billiard'
	elseif billwin > 10 and billvar = 1:
		*clr
		'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/pool/sex/strapon.jpg"></center>'
		'你赢了，戴上你的假阳具，看着那家伙说："好吧，弯下腰，今天你是我的婊子。"'
		'那家伙脱下裤子弯下腰，你用假阳具狠狠干了他的屁股。'

		gs 'arousal', 'anal_strap_give', 10, 'dom'
		gs 'arousal', 'end'

		act '离开': gt 'billiard'
	elseif billwin > 10 and billvar = 2:
		'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/pool/sex/strapon.jpg"></center>'
		'你赢了，戴上你的假阳具，看着这些家伙说："来吧，姑娘们，内裤脱掉，屁股翘起来，现在我要把你们整个后宫都干一遍。"'
		'你挨个把他们全都干了一遍。'

		gs 'arousal', 'anal_strap_give', 10, 'dom'
		gs 'arousal', 'end'

		act '离开': gt 'billiard'
	elseif billwin > 10 and billvar = 3:
		gs 'money', 'earn', 250, 'cash'
		'你赢了并获得 ' + $func('money', 'string_profit', 250) + '。'

		act '离开': gt 'billiard'
	elseif billwin <= 10 and billvar = 0:
		gs 'money', 'pay', 1000, 'cash'
		'你输了并支付了 ' + $func('money', 'string_price', 1000) + '。'

		act '离开': gt 'billiard'
	elseif billwin <= 10 and billvar = 1:
		'你输了，张开双腿。'
		picrand = 16

		act '报酬': xgt 'sex', 'var'
	elseif billwin <= 10 and billvar = 2:
		'你输了，张开双腿。'
		killvar 'boy'
		gs 'npcgeneratec', 0, '台球手', rand(18, 40)
		gs 'boyStat', $npclastgenerated
		gs 'npcgeneratec', 0, '台球手', rand(18, 40)
		gs 'boyStat', $npclastgenerated, '1'
		gs 'npcgeneratec', 0, '台球手', rand(18, 40)
		gs 'boyStat', $npclastgenerated, '2'
		picrand = 12

		act '报酬': gt 'paysex', 'var'
	elseif billwin <= 10 and billvar = 3:
		gs 'money', 'pay', 250, 'cash'
		'你输了并支付了 ' + $func('money', 'string_price', 250) + '。'

		act '离开': gt 'billiard'
	else
		gs 'money', 'pay', 300, 'cash'
		'你输了并支付了 ' + $func('money', 'string_price', 300) + '。'

		act '离开': gt 'billiard'
	end
end


if $ARGS[0] = 'start':
	*clr & cla
	$menu_loc = 'billiard'
	$menu_arg = 'start'
	menu_off = 0
	'<center><b>台球</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/pool/bil.jpg"></center>'

	act '离开': gt 'billiard'
end
