# NPCChanger

if $ARGS[0] = 'Start':
	n = 1
	o = 10
	gs 'NPCChanger', 'NPCLoop'
end


if $ARGS[0] = 'NPCLoop':
	*clr
	gs 'obj_din', 'cheattabs', '人物'
	cla
	act '返回': gt 'cheatmenu_din', 'npcs'
	'<center><h2>作弊菜单 - NPC编辑</h2></center>'
	'<center><table cellspacing="0" cellpadding="20" valign="top"><tr>'
	'<td width="400" cellspacing="0" cellpadding="20" valign="top">'
	'*** 这是一个尚待完善的项目 ***'
	if n = 0:
		n = 1
		o = 10
	end

	:LoopNPC
		if o > aarraynumber: o = aarraynumber
	if n <= o:
		gs 'NPCChanger', 'npcdisplay'
		n += 1
		jump 'LoopNPC'
	end

	'<hr>'
	!! *nl 'o = <<o>> & n = <<n>>'
	*p '<a href="exec:gt ''NPCChanger'', ''next10'' ">后10人</a> '
	*p '<a href="exec:gt ''NPCChanger'', ''back10'' ">前10人</a> '
	*nl
	*p '<a href="exec:gt ''NPCChanger'', ''next50'' ">跳过50人</a> '
	*p '<a href="exec:gt ''NPCChanger'', ''back50'' ">返回50人</a> '
	*nl
	*p '<a href="exec:gt ''NPCChanger'', ''selection'' ">选择要修改的NPC</a>'
	*nl '*** 这是一个尚待完善的项目 ***'
	'</td></tr></table></center>'
end


if $ARGS[0] = 'next10':
	o += 10
	n = (o - 10)
	gs 'NPCChanger', 'NPCLoop'
end


if $ARGS[0] = 'next50':
	o += 50
	n = (o - 10)
	gs 'NPCChanger', 'NPCLoop'
end


if $ARGS[0] = 'back10':
	o -= 10
	if o < 10: o = 10
	n = (o - 10)
	gs 'NPCChanger', 'NPCLoop'
end


if $ARGS[0] = 'back50':
	o -= 50
	if o < 10: o = 10
	n = (o - 10)
	gs 'NPCChanger', 'NPCLoop'
end


if $ARGS[0] = 'npcdisplay':
	*nl 'NPC[A<<n>>] = '

	if isnum($npc_firstname['A<<n>>']):
		*p ''
	else
		*p $npc_firstname['A<<n>>']
	end

	if $npc_firstname['A<<n>>'] = $npc_nickname['A<<n>>'] or isnum($npc_nickname['A<<n>>']):
		*p ''
	else
		*p ' "<<$npc_nickname[''A<<n>>'']>>"'
	end

	if isnum($npc_lastname['a<<n>>']):
		*p ''
	else
		*p ' <<$npc_lastname[''A<<n>>'']>>'
	end
end


if $ARGS[0] = 'selection':
	s = val(input("你想编辑哪个NPC？输入编号#"))
	!! *clr & cla
	if s = 0:
		n = (o - 10)
		gt 'NPCChanger', 'NPCLoop'
	else
		gt 'NPCChanger', 'selection2'
	end
end


if $ARGS[0] = 'selection2':
	'NPC名字：<<$npc_firstname[''A<<s>>'']>> <a href="exec:gt ''NPCChanger'', ''editf''">更改</a>'
	'NPC昵称：<<$npc_nickname[''A<<s>>'']>> <a href="exec:gt ''NPCChanger'', ''editn''">更改</a>'
	'NPC姓氏：<<$npc_lastname[''A<<s>>'']>> <a href="exec:gt ''NPCChanger'', ''editl''">更改</a>'
	'NPC曾用名：<<$npc_usedname[''A<<s>>'']>> <a href="exec:gt ''NPCChanger'', ''editu''">更改</a>'
	'NPC出生日期：<<npc_dob[''A<<s>>'']>> <a href="exec:gt ''NPCChanger'', ''editd''">更改</a>'
	'NPC描述：'+ $func('npc_notes', s)
	if npc_gender['A<<s>>'] = 0:
		'NPC性别：男'
		'NPC鸡巴长度：<<npc_dick[''A<<s>>'']>>厘米'
		'NPC鸡巴描述：<<$npc_thdick[''A<<s>>'']>>'
	else
		'NPC性别：女'
	end
	act '完成':
		n = (o - 10)
		gt 'NPCChanger', 'NPCLoop'
	end
end


if $ARGS[0] = 'editf':
	$npc_firstname['A<<s>>'] = input('输入新的名字')
	gt 'NPCChanger', 'selection2'
end


if $ARGS[0] = 'editn':
	$npc_nickname['A<<s>>'] = input('输入新的昵称')
	gt 'NPCChanger', 'selection2'
end


if $ARGS[0] = 'editl':
	$npc_lastname['A<<s>>'] = input('输入新的姓氏')
	gt 'NPCChanger', 'selection2'
end


if $ARGS[0] = 'editu':
	$npc_usedname['A<<s>>'] = input('输入新的曾用名')
	gt 'NPCChanger', 'selection2'
end


if $ARGS[0] = 'editd':
		npc_dob['A<<s>>'] = input('输入新的出生日期 (yyyymmdd)')
	gt 'NPCChanger', 'selection2'
end
