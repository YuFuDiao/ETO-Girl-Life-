	transportVars['train_wait_center'] = 0

	if $ARGS[1] <> 'pavlovsk':
			!! Train departs every 15 minutes (between 4:00 and 22:00), 30 minutes (between 22:00 and 01:00) and 60 minute (betweeen 01:00 and 04:00)
			!! center -> pavlovsk: 00:02
		temp_transportVars['totminut_center'] = temp_transportVars['totminut'] - func('transport_functions', 'get_train_timecost', 'center', $ARGS[1])
		if temp_transportVars['totminut_center'] >= 1320 or temp_transportVars['totminut_center'] < 60:
			transportVars['train_wait_pavlovsk']	= (1442 - temp_transportVars['totminut_center']) mod 30
		elseif temp_transportVars['totminut_center'] < 240:
			transportVars['train_wait_pavlovsk']	= (1442 - temp_transportVars['totminut_center']) mod 60
		else
			transportVars['train_wait_pavlovsk']	= (1442 - temp_transportVars['totminut_center']) mod 15
		end
	end
	if $ARGS[1] <> 'center':
			!! Train departs every 15 minutes (between 4:00 and 22:00), 30 minutes (between 22:00 and 01:00) and 60 minute (betweeen 01:00 and 04:00)
			!! pavlovsk -> center: 00:07
		temp_transportVars['totminut_pavlovsk'] = temp_transportVars['totminut'] - func('transport_functions', 'get_train_timecost', 'pavlovsk', $ARGS[1])

		end
	end
	exit
end
