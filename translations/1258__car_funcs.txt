	if car['ID'] > 0:

if $ARGS[0] = 'is_here':
	result = 0
	if car['ID'] = 0: exit
	!! if car['wreck'] = 1: exit

	if arrsize('$ARGS') = 1: $ARGS[1] = $loc
	if arrsize('$ARGS') = 2: $ARGS[2] = $loc_arg
	if arrsize('$ARGS') = 3: $ARGS[3] = $region

	if $ARGS[1] = $car['loc'] and $ARGS[2] = $car['loc_arg'] and $ARGS[3] = $car['region']:
		result = 1
	end
	exit
end

if $ARGS[0] = 'avtonorm':
	if car['ID'] >= 1 and car['ID'] <= 8:
		normhour = 20
	elseif car['ID'] = 9:
		normhour = 15
	elseif car['ID'] = 10:
		normhour = 18
	elseif car['ID'] = 11 or car['ID'] = 12:
		normhour = 15
	elseif car['ID'] = 13 or car['ID'] >= 95:
		normhour = 50
	end

	if car['wreck'] = 0:
		normneed = 0 - (car['current_condition'] - car['new_condition'])
		normrem = normneed * normhour
	else
		normneed = 0 - (car['current_condition'] - car['new_condition'])
		normrem = (normneed * normhour) * 8
	end


	killvar 'normhour'
	killvar 'normneed'
end
