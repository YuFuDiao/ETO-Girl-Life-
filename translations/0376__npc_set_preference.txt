# npc_set_preference

!! npc_code = 'A***', 'B***', 'C***', etc.

!!	Full reset of npc preferences:
!!		gs 'npc_set_preference', 'reset', npc_code
!!
!!	Add a preference to an npc:
!!		gs 'npc_set_preference', $npc_code, $trait, [$amount/amount]
!!		npc_code = 'A12', 'A51', 'B13', 'C1', etc
!!		trait: = 'body_strong', 'cum_face', 'body_tits_big', 'body_ass_bubble', etc.
!!
!!		you can either use a string or number for amount:
!!		$amount = 'hate', 'dislike', 'neutral', 'like', 'love'
!!		amount is the change in appearance that this preference causes.
!!
!!		examples:
!!			gs 'npc_set_preference', 'A12', 'body_tits_big', 'like'
!!			gs 'npc_set_preference', 'B1', 'body_bmi_overweight', -50



if $ARGS[0] = 'reset':
	$npc_preferences[$ARGS[1]] = ''
end


!! $ARGS[0] = 'A***', 'B***', 'C***', etc.
if isnum(mid($ARGS[0], 2)):
	if $ARGS[1] = '': exit

	!! Dynamically create the $npc_pref_traits and npc_pref_values arrays
	dynamic $replace($npc_preferences[$ARGS[0]], 'npc_pref', 'npc_nsp_pref')

	!! Example of these created arrays.
	!! $npc_pref_traits: npc_pref_traits[i] = 'stats_stren_strong'
	!! npc_pref_values: npc_pref_values['stats_stren_strong'] = -15

	if $ARGS[2] = '':
		temp_nsprefVars['value'] = ARGS[2]
	else
		!! Translate 'hate, dislike, etc. into a number and store it.'
		temp_nsprefVars['value'] = func('npc_set_preference', 'get_value', $ARGS[1], $ARGS[2])
	end

	temp_nsprefVars['index'] = arrpos('$npc_nsp_pref_traits', $ARGS[1])
	if temp_nsprefVars['value'] <> 0:
		if arrsize('$npc_nsp_pref_traits') = 0:
			!! replace(…, '	', '') is there to remove the tabs from the string.
			$temp_nsprefVars['pref_string'] = "killvar 'npc_pref_traits'
				killvar 'npc_pref_values'
				$npc_pref_traits[] = '<<$ARGS[1]>>'
				npc_pref_values['<<$ARGS[1]>>'] = <<temp_nsprefVars['value']>>
			"

			$npc_preferences[$ARGS[0]] = $replace($temp_nsprefVars['pref_string'], '	', '')

		else
			if temp_nsprefVars['index'] >= 0:
				!! The npc already has a preference for the trait.
				!! We overwrite the trait and then escape the preference_loop
				npc_nsp_pref_values[$ARGS[1]] = temp_nsprefVars['value']

				!! Rebuild the $npc_pref…[$ARGS[0]] strings from the grounds up
				gs 'npc_set_preference', 'rebuild_preferences', $ARGS[0]

			else
				!! We have checked that the npc has no preference for the trait.
				!! We add the preference to the npc.
				$temp_nsprefVars['pref_string'] = "$npc_pref_traits[] = '<<$ARGS[1]>>'
					npc_pref_values['<<$ARGS[1]>>'] = <<temp_nsprefVars['value']>>
				"
				$npc_preferences[$ARGS[0]] += $replace($temp_nsprefVars['pref_string'], '	', '')
			end
		end

	else
		if temp_nsprefVars['index'] >= 0:
			!! The trait exists and we are setting it to 0. Best just remove it entirely
			killvar 'npc_nsp_pref_traits', temp_nsprefVars['index']
			gs 'array', 'remove_element', 'npc_nsp_pref_values', $ARGS[1]

			!! Rebuild the $npc_pref…[$ARGS[0]] strings from the grounds up
			gs 'npc_set_preference', 'rebuild_preferences', $ARGS[0]
		end
	end


	killvar 'npc_nsp_pref_traits'
	killvar 'npc_nsp_pref_values'
	killvar 'temp_nsprefVars'
end


!! Called as gs 'npc_set_preference', 'rebuild_preferences', $npc_code
if $ARGS[0] = 'rebuild_preferences':
	!! Rebuild the $npc_pref…[$ARGS[1]] strings from the grounds up
	$temp_nsprefVars['pref_string'] = "killvar 'npc_pref_traits'
		killvar 'npc_pref_values'
	"

	!! Loop over every preference and add it to the $npc_pref…[$ARGS[1]] strings.
	temp_nsprefVars['i'] = 0
	temp_nsprefVars['max_i'] = arrsize('$npc_nsp_pref_traits')
	:rebuild_preferences_loop
		$temp_nsprefVars['trait'] = $npc_nsp_pref_traits[temp_nsprefVars['i']]
		$temp_nsprefVars['pref_string'] += "$npc_pref_traits[] = '<<$temp_nsprefVars['trait']>>'
			npc_pref_values['<<$temp_nsprefVars['trait']>>'] = <<npc_nsp_pref_values[temp_nsprefVars['i']]>>
		"
		temp_nsprefVars['i'] += 1
	if temp_nsprefVars['i'] < temp_nsprefVars['max_i']: jump 'rebuild_preferences_loop'


	!! Overwrite the $npc_pref…[$ARGS[1]] strings with the rebuild string.
	$npc_preferences[$ARGS[1]] = $replace($temp_nsprefVars['pref_string'], '	', '')

	killvar 'temp_nsprefVars'
end


!! Called as gs 'npc_set_preference', 'remove_prefence', $npc_code, $pref_trait
if $ARGS[0] = 'remove_preference':
	if $ARGS[2] = '': exit

	dynamic $replace($npc_preferences[$ARGS[1]], 'npc_pref', 'npc_nsp_pref')

	!! Check the position of the preference in the preference array.
	temp_nsprefVars['index'] = arrpos('$npc_nsp_pref_traits', $ARGS[2])

	!! If the preference exists:
	if temp_nsprefVars['index'] >= 0:
		!! Remove the preference from the array.
		killvar 'npc_nsp_pref_traits', temp_nsprefVars['index']
		gs 'array', 'remove_element', 'npc_nsp_pref_values', $ARGS[2]

		!! Then rebuild the dynamic string.
		gs 'npc_set_preference', 'rebuild_preferences', $ARGS[1]
	end

	killvar 'temp_nsprefVars'
end




	!!====================================================================!!
	!!                                                                    !!
	!!                        CALCULATE PREF_VALUE                        !!
	!!                                                                    !!
	!!====================================================================!!


!!	gs 'npc_set_preference', 'get_value', $PREFERENCE, $VALUE
!!		$PREFERENCE: body_fit, clothes_style_bimbo, etc.
!!		$VALUE: 'love', 'like', 'approve', 'neutral', 'disapprove', 'dislike', or 'hate'


if $ARGS[0] = 'get_value':
	temp_pref_values['love']		= 20
	temp_pref_values['like']		= 10
	temp_pref_values['approve']		= 5
	temp_pref_values['neutral']		= 0
	temp_pref_values['disapprove']	= -5
	temp_pref_values['dislike']		= -10
	temp_pref_values['hate']		= -20

	if $ARGS[1] = 'body_bmi_starving':
		temp_pref_values['love']		= 60
		temp_pref_values['like']		= 40
		temp_pref_values['approve']		= 20
		temp_pref_values['disapprove']	= -3
		temp_pref_values['dislike']		= -5
		temp_pref_values['hate']		= -10

	elseif $ARGS[1] = 'body_bmi_underweight':
		temp_pref_values['love']		= 10
		temp_pref_values['like']		= 5
		temp_pref_values['approve']		= 3
		temp_pref_values['disapprove']	= -10
		temp_pref_values['dislike']		= -25
		temp_pref_values['hate']		= -35

	elseif $ARGS[1] = 'body_bmi_normal':
		temp_pref_values['love']		= 5
		temp_pref_values['like']		= 2
		temp_pref_values['approve']		= 1
		temp_pref_values['disapprove']	= -25
		temp_pref_values['dislike']		= -50
		temp_pref_values['hate']		= -70

	elseif $ARGS[1] = 'body_bmi_overweight':
		temp_pref_values['love']		= 10
		temp_pref_values['like']		= 5
		temp_pref_values['approve']		= 3
		temp_pref_values['disapprove']	= -15
		temp_pref_values['dislike']		= -25
		temp_pref_values['hate']		= -35

	elseif $ARGS[1] = 'body_bmi_obese':
		temp_pref_values['love']		= 60
		temp_pref_values['like']		= 40
		temp_pref_values['approve']		= 20
		temp_pref_values['disapprove']	= -3
		temp_pref_values['dislike']		= -5
		temp_pref_values['hate']		= -10

	elseif $ARGS[1] = 'body_skin_bad':
		temp_pref_values['love']		= 3
		temp_pref_values['like']		= 2
		temp_pref_values['approve']		= 1
		temp_pref_values['disapprove']	= -1
		temp_pref_values['dislike']		= -2
		temp_pref_values['hate']		= -3

	elseif $ARGS[1] = 'body_skin_normal':
		temp_pref_values['love']		= 3
		temp_pref_values['like']		= 2
		temp_pref_values['approve']		= 1
		temp_pref_values['disapprove']	= -1
		temp_pref_values['dislike']		= -2
		temp_pref_values['hate']		= -3

	elseif $ARGS[1] = 'body_skin_good':
		temp_pref_values['love']		= 3
		temp_pref_values['like']		= 2
		temp_pref_values['approve']		= 1
		temp_pref_values['disapprove']	= -1
		temp_pref_values['dislike']		= -2
		temp_pref_values['hate']		= -3
	end

	if $ARGS[2] <> '':
		result = temp_pref_values[$ARGS[2]]
	else
		if ARGS[2] = temp_pref_values['neutral']:
			$result = 'neutral'
		elseif ARGS[2] < (temp_pref_values['hate'] + temp_pref_values['dislike']) / 2:
			$result = 'hate'
		elseif ARGS[2] > (temp_pref_values['love'] + temp_pref_values['like']) / 2:
			$result = 'love'
		elseif ARGS[2] < (temp_pref_values['dislike'] + temp_pref_values['disapprove']) / 2:
			$result = 'dislike'
		elseif ARGS[2] > (temp_pref_values['like'] + temp_pref_values['approve']) / 2:
			$result = 'like'
		elseif ARGS[2] < (temp_pref_values['disapprove'] + temp_pref_values['neutral']) / 2:
			$result = 'disapprove'
		elseif ARGS[2] > (temp_pref_values['approve'] + temp_pref_values['neutral']) / 2:
			$result = 'approve'
		else
			$result = 'neutral'
		end
	end

	killvar 'temp_pref_values'
end




	!!====================================================================!!
	!!                                                                    !!
	!!                     BACKUP CURRENT PREFERENCES                     !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'backup':
	if arrsize('$npc_pref_traits') > 0:
		killvar 'npc_backup_traits'
		killvar 'npc_backup_values'
		copyarr('$npc_backup_traits', '$npc_pref_traits')
		copyarr('npc_backup_values', 'npc_pref_values')
	end
end


if $ARGS[0] = 'restore':
	if arrsize('$npc_backup_traits') > 0:
		killvar 'npc_pref_traits'
		killvar 'npc_pref_values'
		copyarr('$npc_pref_traits', '$npc_backup_traits')
		copyarr('npc_pref_values', 'npc_backup_values')
		killvar 'npc_backup_traits'
		killvar 'npc_backup_values'
	end
end
