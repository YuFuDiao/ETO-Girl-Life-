if $ARGS[0] = 'init':

		act '召唤你的力量并进食（需要大量时间）': scpopt = 0 & gt 'succubus', 'init'

			cla
			'你试图抑制<<$sucself1>>…'
			gs 'willpower', 'misc', 'resist', 'easy'
			will_cost = will_cost * succublvl
			if pcs_willpwr < will_cost:
				'<br>你的意志力不足以控制你的冲动。'
				'<<$sucself1>>继续向你的意识表层涌上来。'
				succonfail = 1

				act '继续': scpopt = 2 & gt $sclocrt, $scargrt
			end

		gt 'succubus', 'firsttime'
	else

		'它完全充满你，用力量渗透你的全身…'

		succonfail = 0
