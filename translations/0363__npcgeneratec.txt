# npcgeneratec

!! This file requires the sex, name and age for the npc to be generated
!! You can also set their approx. home location
!! All data produced is random so if you need a specific value you will have to overwrite it

!{
	ARGS[0]:	gender (0: male, 1: female, 2: fherm, 3:mherm)
	$ARGS[1]:	name
	ARGS[2]:	age
	ARGS[3]:	location (1: pav res, 2: uni, 3: city res, 4: city center)
	ARGS[4]:	anonymous (0: no, 1: yes)
	$ARGS[2+]:	starting attraction
				('attracted' or 'like':			leaning to positive preferences)
				('unattracted' or 'dislike':	leaning to negative preferences)
!}
if $ARGS[0] = '' or isnum($ARGS[0]):
	if $ARGS[0] <> '': ARGS[0] = val($ARGS[0])

	!! npcgen_lastrun is a flag to help track whether npcgeneratec or npcpreservec has been run last
	!! npcgeneratec is 1
	!! npcpreservec is 2
	npcgen_lastrun = 1

	!! sets sex, 0 male, 1 female, 2 female herm, 3 male herm
	npcgeneratecVars['gender'] = ARGS[0]

	!! npc age
	if ARGS[2] <> 0:
		npcgeneratecVars['age'] = ARGS[2]
	else
		npcgeneratecVars['age'] = rand(18, 35)
	end

	!! Name for npc, can just be rapist or shop assistant, etc
	if $ARGS[1] <> '':
		$npcgeneratecVars['name'] = $ARGS[1]
	else
		if npcgeneratecVars['gender'] = 0:
			$npcgeneratecVars['name'] = '男性'
		elseif npcgeneratecVars['gender'] = 1:
			$npcgeneratecVars['name'] = iif(npcgeneratecVars['age'] <= 25, '女孩', '女人')
		elseif npcgeneratecVars['gender'] = 2:
			$npcgeneratecVars['name'] = iif(npcgeneratecVars['age'] <= 25, '女孩', '女人')
		else
			$npcgeneratecVars['name'] = '男性'
		end
	end

	!! npc home location area
	!! 1 - Pavlovsk Residential Area
	!! 2 - Student dorms in the city
	!! 3 - St. Petersburg Residential Area
	!! 4 - St. Petersburg City Center
	if ARGS[3] <> 0:
		npcgeneratecVars['loc'] = ARGS[3]
	else
		if $region = 'pav':
			npcgeneratecVars['loc'] = 1
		elseif $region = 'city':
			npcgeneratecVars['loc'] = rand(3, 4)
		end
	end

	npcgeneratecVars['anonymous'] = ARGS[4]

	!! For example
	!! gs 'npcgeneratec', 0, 'stranger', rand(18,45), 3, 1
	!! Which would generate a male named stranger between ages of 18 and 45 who lives in the St. Petersburg Residential Area who will be remembered as ''stranger''

	if arrpos('$ARGS', 'attracted') > 0 or arrpos('$ARGS', 'like') > 0:
		$npcgeneratecVars['attracted'] = 'like'
	elseif arrpos('$ARGS', 'unattracted') > 0 or arrpos('$ARGS', 'dislike') > 0:
		$npcgeneratecVars['attracted'] = 'dislike'
	else
		$npcgeneratecVars['attracted'] = ''
	end

	gs 'npcgeneratec', 'init'
end


if $ARGS[0] = 'init':
	gs 'npcgeneratec', 'assign_index'
	gs 'npcgeneratec', 'assign_dob', $npclastgenerated
	gs 'npcgeneratec', 'assign_stats', $npclastgenerated
	gs 'npcgeneratec', 'set_details', $npclastgenerated

	gs 'npcgeneratec', 'set_lifestyle', $npclastgenerated
	gs 'npcgeneratec', 'set_occupation', $npclastgenerated
	gs 'npcgeneratec', 'set_personality', $npclastgenerated
	gs 'npcgeneratec', 'set_preferences', $npclastgenerated
	gs 'npcgeneratec', 'set_apprnc', $npclastgenerated
	gs 'npcgeneratec', 'set_other_vars', $npclastgenerated

if $ARGS[0] = 'assign_index':
	!! This sets up a newly generated index number by overwriting the first blank number
	!! The first check is to see if array 0 in the cleanup list is available and overwrites it if it is free
	if mid($cemptyarray[0], 1, 1) = 'C':
		$npclastgenerated = $cemptyarray[0]
		killvar 'cemptyarray', 0
	else
		:npcgencsanityloop
			ngc_scheck = 0
			$npclastgenerated = 'C' + '<<carraynumber>>'
			$npcgeneratecVars['npcID'] = 'C' + '<<carraynumber-1>>'
			if $npc_perstype[$npcgeneratecVars['npcID']] = '' and carraynumber > 0:
				carraynumber -= 1
				ngc_scheck = 1
			elseif $npc_perstype[$npclastgenerated] = '':
				ngc_scheck = 0
			else
				carraynumber += 1
				ngc_scheck = 1
			end
		if ngc_scheck = 1: jump 'npcgencsanityloop'
		killvar 'ngc_scheck'
	end
	carraynumber += 1

	$npc_index[$npclastgenerated] = $npclastgenerated
end


if $ARGS[0] = 'assign_dob':
	!! Random birthday based on age given

	!! npc reached age exactly 1 year ago
	gs 'time', 'to_daystart', year - npcgeneratecVars['age'] - 1, month, day
	npcgeneratecVars['min_daystart'] = dateVars['daystart'] + 1

	npcgeneratecVars['birth_daystart'] = rand(npcgeneratecVars['min_daystart'], npcgeneratecVars['min_daystart'] + 364)
	gs 'time', 'to_date', npcgeneratecVars['birth_daystart']

	npc_dob[$ARGS[1]] = dateVars['year'] * 10000
	npc_dob[$ARGS[1]] += dateVars['month'] * 100
	npc_dob[$ARGS[1]] += dateVars['day']

	!! {we save the unique number of the npc, to be able to get the index of the array via arrpos}
