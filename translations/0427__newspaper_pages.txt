	if daystart >= 276 and daystart < 290: $np_pages[] = "gs 'newspaper_pages', 'cheerleading_tryouts'"
	if daystart >= 324 and daystart < 373:
		if daystart < 338:
			$np_front_pages[] = "gs 'newspaper_pages', 'boris_arrested'"
		elseif daystart < 345:
			$np_front_pages[] = "gs 'newspaper_pages', 'starlets_report_0'"
		elseif daystart < 352:
			$np_front_pages[] = "gs 'newspaper_pages', 'starlets_report_1'"
		elseif daystart < 359:
			$np_front_pages[] = "gs 'newspaper_pages', 'starlets_report_2'"
		elseif daystart < 366:
			$np_front_pages[] = "gs 'newspaper_pages', 'starlets_report_3'"
		else
			$np_front_pages[] = "gs 'newspaper_pages', 'starlets_report_4'"
		end
	end

	if balletqw['letter'] = 1:
		if daystart >= 196 and daystart < 203:
			$np_pages[] = "gs 'newspaper_pages', 'ballet_pass'"
		elseif daystart >= 354 and daystart < 361:
			$np_pages[] = "gs 'newspaper_pages', 'ballet_first'"
		end
	end

	if rand(0, 9) = 0 and NewspaperVars['gad_missing_day'] < daystart:
		if rand(0, 9) < 4:
			$randname = $func('npcrnamefile', 'rusMale')
		else
			$randname = $func('npcrnamefile', 'rusFemale')
		end
		$randname += ' ' + $func('npcrnamefile', 'rusSur')
		$NewspaperVars['gad_missing_name'] = $randname
		NewspaperVars['gad_missing_age'] = rand(9, 69)

		if NewspaperVars['gad_missing_counter'] = 0:
			NewspaperVars['gad_missing_counter'] = rand(12, 16)
			NewspaperVars['gad_missing_counter'] = daystart / 35
		end

		NewspaperVars['gad_missing_counter'] += 1
		NewspaperVars['gad_missing_day'] = daystart + 15 + rand(0, NewspaperVars['gad_missing_counter'] / 2)
		$np_pages[] = "gs 'newspaper_pages', 'gad_missing'"
		killvar 'randname'
	end
end


if $ARGS[0] = 'add_advert_pages':
	$np_pages[] = "gs 'newspaper_pages', 'advert_<<rand(0, 7)>>'"
end


if $ARGS[0] = 'fill_arrays':
	if arrsize('$np_front_pages') = 0:
		gs 'newspaper_pages', 'add_generic_frontpage'
	end

	:fill_newspaper_loop
	if arrsize('$np_pages') < 3:
		gs 'newspaper_pages', 'add_generic_page'
		jump 'fill_newspaper_loop'
	end
	killvar 'temp_pages'
end


if $ARGS[0] = 'add_generic_frontpage':
	$np_front_pages[] = "gs 'newspaper_pages', 'generic_front_<<rand(0, 3)>>' "
	exit
end


if $ARGS[0] = 'add_generic_page':
	temp_newspaper_i = 0
	:add_generic_page_loop
		$temp_page_chosen = "gs 'newspaper_pages', 'generic_<<rand(0, 21)>>' "
		temp_newspaper_i += 1
	if arrpos('$np_pages', $temp_page_chosen) >= 0 and temp_newspaper_i < 100:
		jump 'add_generic_page_loop'
	end
	$np_pages[] = $temp_page_chosen

	killvar 'temp_newspaper_i'
	killvar 'temp_page_chosen'
	exit
end
