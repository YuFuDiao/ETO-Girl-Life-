!! ┌─────────────────────────────────────────────────────────────────┐
!! │ Unified swap handler                                            │
!! │ All reorder callers use: gs '$menu_settings', 'swap', ...       │
!! │ Convention:                                                     │
!! │   ARGS[0] = 'swap'                                              │
!! │   ARGS[1] = array name to swap within (e.g. 'stat_order',       │
!! │             'menu_bar_order', 'rel_grp_coolkids', ...)          │
!! │   ARGS[2] = index to move                                       │
!! │   ARGS[3] = 'up' or 'down'                                      │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'swap':
	$temp_arr = $ARGS[1]
	!! --- boundary check ---
	if ARGS[2] = 0 and $ARGS[3] = 'up': jump 'swap_cleanup'
	if ARGS[2] = (arrsize('$' + $temp_arr) - 1) and $ARGS[3] = 'down': jump 'swap_cleanup'
	!! --- perform swap via dyneval/dynamic ---
	if $ARGS[3] = 'up':
		$temp_stat_feature = $dyneval('$result = $<<$temp_arr>>[<<ARGS[2] - 1>>]')
		dynamic '$<<$temp_arr>>[<<ARGS[2] - 1>>] = $<<$temp_arr>>[<<ARGS[2]>>]'
		dynamic '$<<$temp_arr>>[<<ARGS[2]>>] = $temp_stat_feature'
	elseif $ARGS[3] = 'down':
		$temp_stat_feature = $dyneval('$result = $<<$temp_arr>>[<<ARGS[2] + 1>>]')
		dynamic '$<<$temp_arr>>[<<ARGS[2] + 1>>] = $<<$temp_arr>>[<<ARGS[2]>>]'
		dynamic '$<<$temp_arr>>[<<ARGS[2]>>] = $temp_stat_feature'
	end
	:swap_cleanup
	killvar 'temp_stat_feature'
	killvar 'temp_arr'
	exit
end


!! Reorder a member within one skill/relation group.
if $ARGS[0] = 'swap_grp_member':
	$temp_sgm_base = $ARGS[1]
	temp_sgm_i = ARGS[3]
	if $ARGS[4] = 'up' and temp_sgm_i > 0:
		temp_sgm_j = temp_sgm_i - 1
	elseif $ARGS[4] = 'down':
		temp_sgm_j = temp_sgm_i + 1
	else
		jump 'swap_gm_cleanup'
	end
	$temp_sgm_a = $ARGS[2] + '_' + $str(temp_sgm_i)
	$temp_sgm_b = $ARGS[2] + '_' + $str(temp_sgm_j)
	!! Empty neighbour = past the last member; nothing to swap (would leave a hole)
	$temp_sgm_bval = $dyneval('$result = $<<$temp_sgm_base>>[''<<$temp_sgm_b>>'']')
	if $temp_sgm_bval = '': jump 'swap_gm_cleanup'
	$temp_stat_feature = $dyneval('$result = $<<$temp_sgm_base>>[''<<$temp_sgm_a>>'']')
	dynamic '$<<$temp_sgm_base>>[''<<$temp_sgm_a>>''] = $temp_sgm_bval'
	dynamic '$<<$temp_sgm_base>>[''<<$temp_sgm_b>>''] = $temp_stat_feature'
	:swap_gm_cleanup
	killvar 'temp_stat_feature'
	killvar 'temp_sgm_base'
	killvar 'temp_sgm_i'
	killvar 'temp_sgm_j'
	killvar 'temp_sgm_a'
	killvar 'temp_sgm_b'
	killvar 'temp_sgm_bval'
	exit
end

if $ARGS[0] = 'toggle_menu_sym_opt':
	!! Push one or more options, each as a "value:Label" string (up to 8 per call, ARGS[1]–ARGS[8]).
	!! value is the integer written to / compared against $var; Label is the display text.
	temp_tms_pi = 1
	:toggle_menu_sym_opt_loop
	if $ARGS[temp_tms_pi] <> '':
		temp_tms_colon = instr($ARGS[temp_tms_pi], ':')
		tms_opt_val[tms_opt_count] = val(mid($ARGS[temp_tms_pi], 1, temp_tms_colon - 1))
		$tms_opt_lbl[tms_opt_count] = mid($ARGS[temp_tms_pi], temp_tms_colon + 1)
		tms_opt_count += 1
		temp_tms_pi += 1
		jump 'toggle_menu_sym_opt_loop'
	end
	killvar 'temp_tms_pi'
	killvar 'temp_tms_colon'
	exit
end


if $ARGS[0] = 'toggle_menu_sym':
	!! $ARGS[1] = menu name
	!! $ARGS[2] = variable to set/compare (e.g. "cheatVars['deg_speed_opt']")
	!! $ARGS[3] = menu label (e.g. "Skill degradation rate")
	!! $ARGS[4] = custom option label, optional
	!! $ARGS[5] = custom option exec, optional
	!! $ARGS[6] = disable option label, optional
	*p $menu_span + $ARGS[3] + ':</span>'
	dynamic 'temp_tms_cur = ' + $ARGS[2]

	temp_tms_i = 0
	temp_tms_matched = 0

	!! Optional disable button: ARGS[6] = label string (e.g. 'Off').
	!! When provided, a button that sets the opt var to -99 (= 0×) is prepended.
	if $ARGS[6] <> '' and temp_tms_cur = -99:
		temp_tms_matched = 1
	end

	:toggle_menu_sym_loop
	if temp_tms_i < tms_opt_count:
		if temp_tms_i > 0: *p '&nbsp;|&nbsp;'
		if temp_tms_cur = tms_opt_val[temp_tms_i]:
			*p '<b><<$tms_opt_lbl[temp_tms_i]>></b>'
			temp_tms_matched = 1
		else
			*p '<a href="exec:<<$ARGS[2]>>=<<tms_opt_val[temp_tms_i]>> & gt ''$menu_settings'', ''<<$ARGS[1]>>''"><<$tms_opt_lbl[temp_tms_i]>></a>'
		end
		temp_tms_i += 1
		jump 'toggle_menu_sym_loop'
	end

	!! Custom option — only shown if both ARGS[4] and ARGS[5] are non-empty
	if $ARGS[4] <> '' and $ARGS[5] <> '':
		*p '&nbsp;|&nbsp;'
		if temp_tms_matched = 0:
			dynamic '$temp_tms_custom_val = $str(' + $ARGS[4] + ')'
			*p '<b>自定义： <<$temp_tms_custom_val>>%</b>'
			killvar 'temp_tms_custom_val'
		else
			!! normalise to single apostrophe
			$temp_tms_custom_exec = $replace($ARGS[5], "''", "'")
			*p '<a href="exec: <<$temp_tms_custom_exec>>">自定义</a>'
			killvar 'temp_tms_custom_exec'
		end
	end

	!! Off button — appended last, after the symmetric options and Custom
	if $ARGS[6] <> '':
		*p '&nbsp;|&nbsp;'
		if temp_tms_cur = -99:
			*p '<b>' + $ARGS[6] + '</b>'
		else
			*p '<a href="exec:<<$ARGS[2]>>=-99 & gt ''$menu_settings'', ''<<$ARGS[1]>>''"><<$ARGS[6]>></a>'
		end
	end

	*nl

	killvar 'temp_tms_cur'
	killvar 'temp_tms_i'
	killvar 'temp_tms_matched'
	!! Clear the option arrays for the next call
	killvar 'tms_opt_val'
	killvar 'tms_opt_lbl'
	killvar 'tms_opt_count'
	exit
end

if $ARGS[0] = 'theme_customize_input':
	if $ARGS[1] = 'rgb':
		result = ARGS[3]
		$temp_input = $ucase($trim($input('输入 <<$ARGS[2]>> 颜色分量，格式为 0 - 255 或 00 - FF')))
		if $temp_input <> '':
			if isnum($temp_input) <> 0:
				if val($temp_input) < 0 or val($temp_input) > 255:
					msg '无效输入'
				else
					temp_result = val($temp_input)
				end
			else
				if len($temp_input) > 2 or instr('0123456789ABCDEF', $mid($temp_input, 1, 1)) = 0 or instr('0123456789ABCDEF', $mid($temp_input, 2, 1)) = 0:
					msg '无效输入'
				else
					temp_result = func('shortgs', 'hex_str_to_int', $temp_input)
				end
			end
		end
		if arrsize('temp_result') = 1:
			if $ARGS[2] = 'red':
				result = ((ARGS[3] and (-256)) or temp_result)
			elseif $ARGS[2] = 'green':
				result = ((ARGS[3] and (-65281)) or (temp_result * 256))
			elseif $ARGS[2] = 'blue':
				result = ((ARGS[3] and (-16711681)) or (temp_result * 65536))
			end
		end
		killvar 'temp_result'
	elseif $ARGS[1] = 'hex':
		result = ARGS[2]
		$temp_input = $ucase($trim($input('输入颜色，格式为 ABC 或 AABBCC')))
		if $temp_input <> '':
			$temp_input = $replace($temp_input, '#')
			temp_length = len($temp_input)
			if temp_length = 3 or temp_length = 6:
				temp_index = 1
				:is_hex_loop
					if instr('0123456789ABCDEF', $mid($temp_input, temp_index, 1)) = 0:
						msg '无效输入' & jump 'invalid_input_exit'
					end
				if temp_index < temp_length: temp_index += 1 & jump 'is_hex_loop'
				killvar 'temp_index'

				if temp_length = 3:
					$temp_red = $mid($temp_input, 1, 1) & $temp_red = $temp_red + $temp_red
					$temp_green = $mid($temp_input, 2, 1) & $temp_green = $temp_green + $temp_green
					$temp_blue = $mid($temp_input, 3, 1) & $temp_blue = $temp_blue + $temp_blue
				else
					$temp_red = $mid($temp_input, 1, 2)
					$temp_green = $mid($temp_input, 3, 2)
					$temp_blue = $mid($temp_input, 5, 2)
				end

				result = rgb(func('shortgs', 'hex_str_to_int', $temp_red), func('shortgs', 'hex_str_to_int', $temp_green), func('shortgs', 'hex_str_to_int', $temp_blue))

				killvar 'temp_red'
				killvar 'temp_green'
				killvar 'temp_blue'
			else
				msg '无效输入'
			end
			:invalid_input_exit
			killvar 'temp_length'
		end
	end
	killvar 'temp_input'
end

!! ARGS[1] = current hex value (e.g. '#38C0D6'), ARGS[2] = label shown in dialog
if $ARGS[0] = 'theme_customize_input_hex_str':
	$result = $ARGS[1]
	$temp_hsi = $ucase($trim($input('输入 <<$ARGS[2]>> 颜色 (ABC 或 AABBCC 十六进制格式）：')))
	if $temp_hsi <> '':
		$temp_hsi = $replace($temp_hsi, '#')
		temp_hsi_len = len($temp_hsi)
		if temp_hsi_len = 3 or temp_hsi_len = 6:
			temp_hsi_idx = 1
			:hsi_hex_loop
				if instr('0123456789ABCDEF', $mid($temp_hsi, temp_hsi_idx, 1)) = 0:
					msg '无效输入：不是有效的十六进制颜色。' & jump 'hsi_exit'
				end
			if temp_hsi_idx < temp_hsi_len: temp_hsi_idx += 1 & jump 'hsi_hex_loop'
			killvar 'temp_hsi_idx'
			if temp_hsi_len = 3:
				$result = '#' + $mid($temp_hsi, 1, 1) + $mid($temp_hsi, 1, 1) + $mid($temp_hsi, 2, 1) + $mid($temp_hsi, 2, 1) + $mid($temp_hsi, 3, 1) + $mid($temp_hsi, 3, 1)
			else
				$result = '#' + $temp_hsi
			end
			:hsi_exit
			killvar 'temp_hsi_len'
		else
			msg '无效输入：必须是 3 或 6 个十六进制字符。'
		end
	end
	killvar 'temp_hsi'
end
