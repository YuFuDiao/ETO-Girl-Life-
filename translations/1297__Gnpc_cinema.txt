# Gnpc_cinema

menu_off = 1
cls
minut += 15
gs 'stat'
$func('wrap', 'header center h1', '电影院')
'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/cinema/cinema.jpg"></center>'
*nl
'你和<<$nameBoyfrend[numnpc]>>一起来到公园附近的电影院。<<$nameBoyfrend[numnpc]>>买了票，然后你们俩走进一间小小的影厅，找到了自己的座位。'

act '看电影':
	cls
	gs 'mood', 'raise', 'large'
	minut += 100
	otnBoyFrend[numnpc] += 5
	gs 'stat'
	'你和那个男的坐在后排的座位上。很快，灯光暗了下来，电影在银幕上开始了。'

	if otnBoyFrend[numnpc] < 80:
		'你在那家伙旁边的座位上看了好几个小时的电影。'

		act '前往出口': gt 'Gnpc2'
	elseif otnBoyFrend[numnpc] >= 80 and gnpcSex[numnpc] = 0:
		'你正看着电影，突然感觉到一只手搭在了你的膝盖上。'

		gs 'willpower', 'foreplay', 'self', 'easy'
		if pcs_willpwr < will_cost:
			act '拿开他的手' + $func('willpower', 'get_willcost_string'): $noWillpower
		else
			act '拿开他的手' + $func('willpower', 'get_willcost_string'):
				cls
				gs 'willpower', 'pay', 'self'
				otnBoyFrend[numnpc] -= 20
				gs 'stat'
				'你拿开了那家伙搭在你膝盖上的手，继续看电影。电影终于结束了，影厅里的灯亮了起来，银幕上滚动着演职员名单。'

				act '前往出口': gt 'Gnpc2'
			end
		end

		act '忽略':
			cls
			otnBoyFrend[numnpc] += 5
			gs 'stat'
			'你假装什么都没发生，而那家伙却更起劲地抚摸起你的腿。电影终于结束了，影厅里的灯亮了起来，银幕上滚动着演职员名单。'

			act '前往出口': gt 'Gnpc2'
		end
	elseif otnBoyFrend[numnpc] >= 80 and gnpcSex[numnpc] > 0:
		'你正看着电影，<<$nameBoyfrend[numnpc]>>把手放在你的脚上，开始抚摸你的膝盖。电影终于结束了，影厅里的灯亮了起来，银幕上出现了片尾字幕。'

		act '前往出口': gt 'Gnpc2'
	end
end
