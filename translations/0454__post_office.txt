!{
	Adding mail that can be picked up:
	gs 'post_office', 'add_mail', ['region'], ['code'], [time]
		['region']: The region of the post office where the mail is
			'city', 'pav', 'all'
		['code']: code to be executed in the post office
			"gs 'blackmailer', 'set_post_act'"
			NOTE: it''s highly encouraged not to have this code directly lead to a gt.
				Have it gs to an act.
		[time]: the totminut after which the mail becomes available. Defaults to 0

	gs 'post_office', 'remove_mail', ['code']
		['code'] is the same code passed to the 'add_mail' function.
			"gs 'blackmailer', 'set_post_act'"
!}

	copyarr('$temp_mail_code',		'$mail_code')

	po_i = 0
	:mail_loop
		if $temp_mail_region[po_i] = $region or $temp_mail_region[po_i] = 'all':
			if totminut >= temp_mail_time[po_i]:
				temp_mail_counter += 1
				dynamic $temp_mail_code[po_i]
			end
		end
		po_i += 1
	if po_i < arrsize('$temp_mail_region'): jump 'mail_loop'

	if temp_mail_counter > 0:
		'你有<<temp_mail_counter>>封邮件待领取。'
	else
		'你已经领取了所有邮件。'
	end

	killvar 'po_i'
	killvar 'temp_mail_counter'
	killvar 'temp_mail_region'
	killvar 'temp_mail_code'
	killvar 'temp_mail_time'
end
