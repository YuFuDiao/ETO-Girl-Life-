# stat_display_compute


!! ╔═════════════════════════════════════════════════════════════════════╗
!! ║  queue_alert — queue a transient alert into $sd_alerts[]            ║
!! ║  $ARGS[1] = message text (plain or HTML; no <br>)                   ║
!! ║  $ARGS[2] = optional theme colour token; '' or 'none' = no wrap     ║
!! ║  sec_alerts renders $sd_alerts[] with auto-bold per entry.          ║
!! ║  Alerts ignore show_positive_texts and are always shown.            ║
!! ╚═════════════════════════════════════════════════════════════════════╝
if $ARGS[0] = 'queue_alert':
	$sd_qa['text'] = iif($ARGS[2] <> '' and $ARGS[2] <> 'none', $func('wrap', $ARGS[2], $ARGS[1]), $ARGS[1])
	$sd_alerts[] = $sd_qa['text']
	killvar 'sd_qa'
	exit
end


!! ╔═════════════════════════════════════════════════════════════════════╗
!! ║  queue_msg — queue a text message + optional icon                   ║
!! ║  $ARGS[1] = label key into $stat_texts[]                            ║
!! ║  $ARGS[2] = colour token (wrap name; '' or omit = no wrap)  		 ║
!! ║  Positional shorthands (ARGS[3] disambiguated by value):            ║
!! ║   Short-A (text-only+gate): label, color, 'pos'|'const'             ║
!! ║   Short-B (icon+gate):      label, color, icon, row, 'pos'|'const'  ║
!! ║   Full:                     label, color, icon, row, action, gate   ║
!! ║  Icon: 'light|dark' path relative to images/system/icons/           ║
!! ║  Gate: 'pos' = positive-dynamic (icons+texts), 'const' = text only  ║
!! ║  Icon-only: if $stat_texts[label]='' but icon set, text is skipped. ║
!! ╚═════════════════════════════════════════════════════════════════════╝
if $ARGS[0] = 'queue_msg':
	!! Resolve positional shorthands
	if $ARGS[3] = 'pos' or $ARGS[3] = 'const':
		!! Short-A: no icon, gate at ARGS[3]
		$sd_qm['icon_arg'] = ''
		sd_qm['row_arg'] = 0
		$sd_qm['action_arg'] = ''
		$sd_qm['gate'] = $ARGS[3]
	elseif $ARGS[5] = 'pos' or $ARGS[5] = 'const':
		!! Short-B: icon present, gate at ARGS[5], action omitted
		$sd_qm['icon_arg'] = $ARGS[3]
		sd_qm['row_arg'] = ARGS[4]
		$sd_qm['action_arg'] = ''
		$sd_qm['gate'] = $ARGS[5]
	else
		!! Full form
		$sd_qm['icon_arg'] = $ARGS[3]
		sd_qm['row_arg'] = ARGS[4]
		$sd_qm['action_arg'] = $ARGS[5]
		$sd_qm['gate'] = $ARGS[6]
	end

	!! _tooltip = plain text for title="" attr; stat_texts[label] = rich text for popup/display
	$sd_qm['title'] = iif($stat_texts[$ARGS[1] + '_tooltip'] <> '', $stat_texts[$ARGS[1] + '_tooltip'], $stat_texts[$ARGS[1]])
	$sd_qm['display'] = $stat_texts[$ARGS[1]]
	!! Action: prefer action_arg, then msg via stat_texts ref (not embedded value — HTML quotes break href attrs).
	if $sd_qm['action_arg'] <> '':
		$sd_qm['action'] = $sd_qm['action_arg']
	elseif $sd_qm['display'] <> '':
		$sd_qm['action'] = 'msg $stat_texts[''' + $ARGS[1] + ''']'
	else
		$sd_qm['action'] = 'msg $stat_texts[''' + $ARGS[1] + '_tooltip'']'
	end
	!! menu_off override
	if menu_off and $sd_qm['action_arg'] <> '':
		$sd_qm['action'] = 'msg $func(''wrap'', ''v_neg'', ''你现在不能使用菜单。'')'
	end
	if $sd_qm['icon_arg'] <> '' and instr($sd_qm['icon_arg'], '|') > 0:
		if theme['is_dark'] = 0:
			$sd_qm['icon'] = $mid($sd_qm['icon_arg'], 1, instr($sd_qm['icon_arg'], '|') - 1)
		else
			$sd_qm['icon'] = $mid($sd_qm['icon_arg'], instr($sd_qm['icon_arg'], '|') + 1)
		end
	else
		$sd_qm['icon'] = $sd_qm['icon_arg']
	end

	!! Extract bg_color if present (format: "path:hex_color")
	$sd_qm['bg_color'] = ''
	if instr($sd_qm['icon'], ':') > 0:
		$sd_qm['bg_color'] = $mid($sd_qm['icon'], instr($sd_qm['icon'], ':') + 1)
		$sd_qm['icon'] = $mid($sd_qm['icon'], 1, instr($sd_qm['icon'], ':') - 1)
	end

	!! Gate: 'pos'=positive-dynamic (icons+texts togglable), 'const'=constant accessory (only text gated), ''=always shown.
	if $sd_qm['gate'] = 'pos':
		if stat_cfg['show_positive_icons'] = 0: $sd_qm['icon'] = ''
		if stat_cfg['show_positive_texts'] = 0: $sd_qm['display'] = ''
	elseif $sd_qm['gate'] = 'const':
		if stat_cfg['show_constant_texts'] = 0: $sd_qm['display'] = ''
	end

	!! Icons — title uses plain-text _tooltip, click uses rich display text or tooltip
	if $sd_qm['icon'] <> '' and (stat_cfg['msg_display_mode'] = 0 or stat_cfg['msg_display_mode'] = 1):
		if $sd_qm['bg_color'] <> '':
			$sd_qm['final_icon'] = '<a href="exec: <<$sd_qm[''action'']>>"><span style="display:inline-block;background:<<$sd_qm[''bg_color'']>>;border-radius:12px;width:<<stat_cfg[''icon_height'']>>px;height:<<stat_cfg[''icon_height'']>>px;padding:5%;box-sizing:border-box;"><img src="images/system/icons/<<$sd_qm[''icon'']>>.png" title="<<$sd_qm[''title'']>>" style="width:100%;height:100%;display:block;"></span></a>'
		else
			$sd_qm['final_icon'] = '<a href="exec: <<$sd_qm[''action'']>>"><img src="images/system/icons/<<$sd_qm[''icon'']>>.png" title="<<$sd_qm[''title'']>>" height="<<stat_cfg[''icon_height'']>>"></a>'
		end
		if sd_qm['row_arg'] = 1:
			$sd_icons_1[] = $sd_qm['final_icon']
		elseif sd_qm['row_arg'] = 2:
			$sd_icons_2[] = $sd_qm['final_icon']
		elseif sd_qm['row_arg'] = 3:
			$sd_icons_3[] = $sd_qm['final_icon']
		else
			$sd_icons_4[] = $sd_qm['final_icon']
		end
	end

	!! Text — uses rich display text (may contain HTML links etc.)
	!! Skip text queue when display is empty (icon-only mode)
	if $sd_qm['display'] <> '' and (stat_cfg['msg_display_mode'] = 0 or stat_cfg['msg_display_mode'] = 2 or $sd_qm['icon'] = ''):
		$sd_qm['final_text'] = $iif($ARGS[2] <> '', $func('wrap', $ARGS[2], $sd_qm['display']), $sd_qm['display'])
		$sd_texts[] = $sd_qm['final_text']
	end

	killvar 'sd_qm'
	exit
end

!! ╔═════════════════════════════════════════════════════════════════════╗
!! ║  COMPUTE ATTRIBUTES PROSE — render_mode=3 prose summary             ║
!! ║  Notable = absolute outlier (≥80/≤15) or relative (±15 from mean).  ║
!! ║  Result → $sd_attrs_prose                                           ║
!! ╚═════════════════════════════════════════════════════════════════════╝
if $ARGS[0] = 'compute_attributes_prose':
	!! ── Step 1: Collect visible attribute values (normalized 0-100) ──
	sd_ap['cnt'] = 0
	sd_ap['sum'] = 0

	!! Iterate $attr_bar_order[] to respect hide + ordering
	sd_ap['ai'] = 0
	sd_ap['amax'] = arrsize('$attr_bar_order')
	:sd_ap_collect
	if sd_ap['ai'] < sd_ap['amax']:
		$sd_ap['ck'] = $attr_bar_order[sd_ap['ai']]

		if stat_hide_bar[$sd_ap['ck']] = 0:
			!! Map key → value (0-100 normalized) + display name
			sd_ap['cv'] = -1
			if $sd_ap['ck'] = 'inhib':
				sd_ap['cv'] = 100 - pcs_inhib
				$sd_ap['cn'] = '抑制'
			elseif $sd_ap['ck'] = 'stren':
				sd_ap['cv'] = pcs_stren
				$sd_ap['cn'] = '力量'
			elseif $sd_ap['ck'] = 'agil':
				sd_ap['cv'] = pcs_agil
				$sd_ap['cn'] = '敏捷'
			elseif $sd_ap['ck'] = 'vital':
				sd_ap['cv'] = pcs_vital
				$sd_ap['cn'] = '耐力'
			elseif $sd_ap['ck'] = 'intel':
				sd_ap['cv'] = pcs_intel
				$sd_ap['cn'] = '智力'
			elseif $sd_ap['ck'] = 'sprt':
				sd_ap['cv'] = pcs_sprt
				$sd_ap['cn'] = '精神'
			elseif $sd_ap['ck'] = 'react':
				sd_ap['cv'] = pcs_react
				$sd_ap['cn'] = '反应'
			elseif $sd_ap['ck'] = 'chrsm':
				sd_ap['cv'] = pcs_chrsm
				$sd_ap['cn'] = '魅力'
			elseif $sd_ap['ck'] = 'prcptn':
				sd_ap['cv'] = pcs_prcptn
				$sd_ap['cn'] = '感知'
			elseif $sd_ap['ck'] = 'apprnc':
				!! Normalize 0-200 → 0-100
				sd_ap['cv'] = pcs_apprnc * 100 / 200
				$sd_ap['cn'] = '魅力'
			elseif $sd_ap['ck'] = 'magik':
				if $start_type['magic'] <> 'nomagic':
					sd_ap['cv'] = pcs_magik
					$sd_ap['cn'] = '魔法'
				end
			elseif $sd_ap['ck'] = 'sucexcess':
				if $start_type['magic'] <> 'nomagic' and succubusflag = 1:
					sd_ap['cv'] = sucexcess
					$sd_ap['cn'] = '魅魔能量'
				end
			end

			if sd_ap['cv'] >= 0:
				sd_ap['v_' + $str(sd_ap['cnt'])] = sd_ap['cv']
				$sd_ap['n_' + $str(sd_ap['cnt'])] = $sd_ap['cn']
				sd_ap['sum'] += sd_ap['cv']
				sd_ap['cnt'] += 1
			end
		end

		sd_ap['ai'] += 1
		jump 'sd_ap_collect'
	end

	!! ── Step 2: Compute mean ─────────────────────────────────────────
	if sd_ap['cnt'] > 0:
		sd_ap['mean'] = sd_ap['sum'] / sd_ap['cnt']
	else
		sd_ap['mean'] = 50
	end

	!! ── Step 3: Flag notable attributes ──────────────────────────────
	!! Buckets: high (positive notable), low (negative notable)
	sd_ap['n_high'] = 0
	sd_ap['n_low'] = 0

	sd_ap['bi'] = 0
	:sd_ap_bucket
	if sd_ap['bi'] < sd_ap['cnt']:
		sd_ap['bv'] = sd_ap['v_' + $str(sd_ap['bi'])]
		sd_ap['dev'] = sd_ap['bv'] - sd_ap['mean']

		!! High: absolute ≥80 or relative deviation ≥+15
		if sd_ap['bv'] >= 80 or sd_ap['dev'] >= 15:
			$sd_ap['high_' + $str(sd_ap['n_high'])] = $sd_ap['n_' + $str(sd_ap['bi'])]
			sd_ap['n_high'] += 1
		!! Low: absolute ≤15 or relative deviation ≤-15
		elseif sd_ap['bv'] <= 15 or sd_ap['dev'] <= -15:
			$sd_ap['low_' + $str(sd_ap['n_low'])] = $sd_ap['n_' + $str(sd_ap['bi'])]
			sd_ap['n_low'] += 1
		end

		sd_ap['bi'] += 1
		jump 'sd_ap_bucket'
	end

	!! ── Step 4: Assemble prose ───────────────────────────────────────
	$sd_attrs_prose = ''

	!! High attributes: "Your Strength and Charisma stand out."
	if sd_ap['n_high'] > 0:
		if sd_ap['n_high'] = 1:
			$sd_attrs_prose = '你的 ' + $func('wrap', 'pos', $sd_ap['high_0']) + ' 很显眼'
		elseif sd_ap['n_high'] = 2:
			$sd_attrs_prose = '你的 ' + $func('wrap', 'pos', $sd_ap['high_0']) + ' 和 ' + $func('wrap', 'pos', $sd_ap['high_1']) + ' 很显眼'
		else
			$sd_attrs_prose = '你的 ' + $func('wrap', 'pos', $sd_ap['high_0'])
			sd_ap['ji'] = 1
			:sd_ap_join_high
			if sd_ap['ji'] < (sd_ap['n_high'] - 1):
				$sd_attrs_prose += ', ' + $func('wrap', 'pos', $sd_ap['high_' + $str(sd_ap['ji'])])
				sd_ap['ji'] += 1
				jump 'sd_ap_join_high'
			end
			$sd_attrs_prose += ', 而且 ' + $func('wrap', 'pos', $sd_ap['high_' + $str(sd_ap['n_high'] - 1)]) + ' 很显眼'
		end
	end

	!! Low attributes: "but your Agility is lacking" or standalone
	if sd_ap['n_low'] > 0:
		!! Build the low list
		if sd_ap['n_low'] = 1:
			$sd_ap['low_list'] = $func('wrap', 'neg', $sd_ap['low_0'])
			$sd_ap['low_verb'] = '有所欠缺'
		elseif sd_ap['n_low'] = 2:
			$sd_ap['low_list'] = $func('wrap', 'neg', $sd_ap['low_0']) + ' 和 ' + $func('wrap', 'neg', $sd_ap['low_1'])
			$sd_ap['low_verb'] = '有所欠缺'
		else
			$sd_ap['low_list'] = $func('wrap', 'neg', $sd_ap['low_0'])
			sd_ap['ji'] = 1
			:sd_ap_join_low
			if sd_ap['ji'] < (sd_ap['n_low'] - 1):
				$sd_ap['low_list'] += ', ' + $func('wrap', 'neg', $sd_ap['low_' + $str(sd_ap['ji'])])
				sd_ap['ji'] += 1
				jump 'sd_ap_join_low'
			end
			$sd_ap['low_list'] += ', 而且 ' + $func('wrap', 'neg', $sd_ap['low_' + $str(sd_ap['n_low'] - 1)])
			$sd_ap['low_verb'] = '有所欠缺'
		end

		if $sd_attrs_prose <> '':
			!! Append as contrast: ", but your X is lacking."
			$sd_attrs_prose += '，但你的' + $sd_ap['low_list'] + ' ' + $sd_ap['low_verb']
		else
			!! Standalone: "Your X is lacking."
			$sd_attrs_prose = '你的' + $sd_ap['low_list'] + ' ' + $sd_ap['low_verb']
		end
	end

	!! Close the sentence
	if $sd_attrs_prose <> '':
		$sd_attrs_prose += '。'
	else
		$sd_attrs_prose = '你的属性很均衡。'
	end

	killvar 'sd_ap'
	exit
end

		if stat_cfg['render_mode_status'] = 3: $sd_pr_urgent[] = '痛苦不堪'
	elseif pain['total'] > 50:
		$stat_texts['pain'] = '尖锐的疼痛窜过你的四肢。'
		gs 'stat_display_compute', 'queue_msg', 'pain', 'neg', 'status/pain_2', 2, 'gt ''$menu_character'', ''pain'''
		gs 'stat_display_compute', 'queue_alert', '你感到一阵剧痛。', 'neg'
		if stat_cfg['render_mode_status'] = 3: $sd_pr_warning[] = '剧痛难忍'
	elseif pain['total'] > 25:
		$stat_texts['pain'] = '阵阵疼痛分散着你的注意力。'
		gs 'stat_display_compute', 'queue_msg', 'pain', 'neutral', 'status/pain_1', 2, 'gt ''$menu_character'', ''pain'''
		if stat_cfg['render_mode_status'] = 3: $sd_pr_mild[] = '有些酸痛'
	elseif pain['total'] > 5:
		$stat_texts['pain'] = '一阵隐痛牵扯着你的肌肉。'

			jump 'sd_pr_flatten_mild'
		end

			end
		else
			!! Build the sentence from the flat list.
			!! Pattern: "You are <f0>" / "You are <f0> and <f1>" /
			!!          "You are <f0>, <f1>, and <f2>" (Oxford comma).
			if sd_pr['total'] = 1:

			elseif sd_pr['total'] = 2:

			else
				!! 3+ items: comma-separated with "and" before last
				$sd_status_prose = '你' + $sd_pr_flat[0]
				sd_pr['i'] = 1
				:sd_pr_join_loop
				if sd_pr['i'] < (sd_pr['total'] - 1):
					$sd_status_prose += '，' + $sd_pr_flat[sd_pr['i']]
					sd_pr['i'] += 1
					jump 'sd_pr_join_loop'
				end
				$sd_status_prose += '，而且' + $sd_pr_flat[sd_pr['total'] - 1]
			end

			!! Append upbeat clause if present (e.g. ", but in great spirits")
			if arrsize('sd_pr_upbeat') > 0:
				$sd_status_prose += '—但 ' + $func('wrap', 'v_pos', $sd_pr_upbeat[0])
			end

			!! Close the sentence
			$sd_status_prose += '。'
		end
	end

	killvar 'sd_pr_urgent'
	killvar 'sd_pr_warning'
	killvar 'sd_pr_mild'
	killvar 'sd_pr_upbeat'
	killvar 'sd_pr_flat'
	killvar 'sd_pr'
	exit
end

	!! ── Active Spells ────────────────────────────────────────────────
	!! Pre-formatted HTML (bimbo-colored headers + names).
	if arrsize('spellComplete') > 0:
		$sd_cm['spell_text'] = '<br><b><<$func(''wrap'', ''bimbo'', ''Active Spells'')>></b>'

	!! Cache in $archetype_notif_pending so alert persists until player navigates away.
	if stat_cfg['notify_archetypes'] <> 1:
		if func('archetypes', 'is_empty_archetypes') = 0:

			if $loc <> $archetype_notif_pending_loc:

		$stat_texts['rent'] = '你在城市住宅区租了一套公寓，已付清<<func(''homes_properties'', ''get_rent_days'', ''city_apartment'')>>天。你也在普希金有一套公寓，已支付了<<func(''homes_properties'', ''get_rent_days'', ''old_town_apartment'')>>天。水电费应在<<$monthName>> 25日缴纳。'
	elseif accessible_property['city_house'] = 1:
		$stat_texts['rent'] = '你在城市住宅区租了一套公寓，已付清<<func(''homes_properties'', ''get_rent_days'', ''city_apartment'')>>天。水电费应在<<$monthName>> 25日缴纳。'

		$stat_texts['rent'] = '你在普希金租了一套公寓，已付清<<func(''homes_properties'', ''get_rent_days'', ''old_town_apartment'')>>天。水电费应在<<$monthName>> 25日缴纳。'

	$stat_texts['nicholas'] = $nichStatMsg

	!!   Render     — walks the day-cache each compute, skips already-past timeslots,
	if arrsize('$events_list') > 0:

				cal_upc_days[cal_r['ci']]	= upcoming_days[cal_r['ci']]

				!! Cache all_day/flex_type/window_end_ts so render-loop skip checks need no get_event call

				cal_upc_all_day[cal_r['ci']] = event_vars['all_day']
				cal_upc_end_ts[cal_r['ci']] = iif(event_vars['flex_type'] = 1, event_vars['window_end_ts'], event_vars['start_ts'])
				cal_r['ci'] += 1
				jump 'cal_cache_copy'
			end
			cal_upc_cache_cnt = upcoming_count
			cal_day_cache_key = daystart
			cal_upcoming_dirty = 0
		end

		!! All skip checks use cached data only; get_event called only for rendered entries.
		$cal_upcoming_html = ''
		cal_r['now_ts'] = hour * 4 + minut / 15

		if stat_cfg['cal_count'] = 6:

			cal_r['max'] = stat_cfg['cal_count']
			!! today_only: 0=On (default/unset), 1=Off

		end
		cal_r['shown'] = 0
		cal_r['i'] = 0
		:cal_render_loop
		if cal_r['i'] < cal_upc_cache_cnt and cal_r['shown'] < cal_r['max']:

			if cal_r['today_only'] = 1 and cal_upc_days[cal_r['i']] <> daystart:
				cal_r['i'] += 1
				jump 'cal_render_loop'
			end
			!! Skip events more than 5 days in the future
			if cal_upc_days[cal_r['i']] > daystart + 5:
				cal_r['i'] += 1
				jump 'cal_render_loop'
			end
			!! Skip all-day events if hidden (cached; cal_show_all_day: 0=Shown default, 1=Hidden)
			if stat_cfg['cal_show_all_day'] = 1 and cal_upc_all_day[cal_r['i']] = 1:
				cal_r['i'] += 1
				jump 'cal_render_loop'
			end
			!! Skip today timed events whose timeslot has already passed (cached).
			if cal_upc_days[cal_r['i']] = daystart and cal_upc_all_day[cal_r['i']] = 0 and cal_upc_end_ts[cal_r['i']] <= cal_r['now_ts']:
				cal_r['i'] += 1
				jump 'cal_render_loop'
			end
			!! Load event only for entries we are actually rendering
			gs 'calendar_events', 'get_event', $cal_upc_ids[cal_r['i']]
			!! Build entry HTML
			$cal_r['rgb'] = $func('calendar_render', 'get_event_color', event_vars['color'])
			if cal_upc_days[cal_r['i']] = daystart:
				$cal_r['day_lbl'] = '今天'
			elseif cal_upc_days[cal_r['i']] = daystart + 1:
				$cal_r['day_lbl'] = '明天'
			else
				$cal_r['day_lbl'] = $weekName[func('time', 'get_week_from_daystart', cal_upc_days[cal_r['i']])]
			end
			if event_vars['all_day'] = 1:
				$cal_r['time_lbl'] = ''
			elseif event_vars['flex_type'] = 1:
				$cal_r['time_lbl'] = $func('time', 'get_time_string', event_vars['window_start_ts'] / 4, (event_vars['window_start_ts'] mod 4) * 15, 0) + '&ndash;' + $func('time', 'get_time_string', event_vars['window_end_ts'] / 4, (event_vars['window_end_ts'] mod 4) * 15, 0)
			else
				$cal_r['time_lbl'] = $func('time', 'get_time_string', cal_upc_ts[cal_r['i']] / 4, (cal_upc_ts[cal_r['i']] mod 4) * 15, 0)
			end
			$cal_upcoming_html += iif($cal_upcoming_html <> '', '<br>', '') + '<span style="color:rgb(<<$cal_r[''rgb'']>>)">&#9679;</span>&ensp;' + $event_vars['title'] + '&ensp;<span style="opacity:0.7">' + $cal_r['day_lbl'] + iif($cal_r['time_lbl'] <> '', '&ensp;' + $cal_r['time_lbl'], '') + '</span>'
			cal_r['shown'] += 1
			cal_r['i'] += 1
			jump 'cal_render_loop'
		end
		killvar 'cal_r'

	end

	exit
end

!! │ cond_sweat — pcs_sweat: <20 '' · <30 sweating · <40 smelly      │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'cond_sweat':
	if pcs_sweat >= 40:
		$result = $func('wrap', 'v_neg', '[臭哄哄]')
	elseif pcs_sweat >= 30:
		$result = $func('wrap', 'neg', '[有异味]')
	elseif pcs_sweat >= 20:
		$result = $func('wrap', 'neutral', '[出汗]')
	else
		$result = ''
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ cond_coat_temp — PCoatWarm vs temper[0] warmth adequacy         │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'cond_coat_temp':
	if $coatworntype = 'none' or $coatworntype = '':
		$result = ''
		exit
	end
	!! Minimum temperatures each CoatWarm level can handle (from outdoors.qsrc)
	!! 1 → -10°C  |  2 → -20°C  |  3 → -30°C
	sd_cs['min_temp'] = iif(PCoatWarm = 3, -30, iif(PCoatWarm = 2, -20, -10))
	if temper[0] > 15:
		!! Coat unnecessary at this temperature, but not a problem
		$result = $func('wrap', 'v_pos', '[足够温暖]')
	elseif temper[0] >= sd_cs['min_temp']:
		$result = $func('wrap', 'v_pos', '[发热]')
	else
		$result = $func('wrap', 'v_neg', '[太冷了！]')
	end
	killvar 'sd_cs'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ cond_heel_skill — colored heel-skill adequacy for shoe image    │
!! │ Returns '' for flat shoes; mirrors shoes.qsrc pain thresholds.  │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'cond_heel_skill':
	!! Flat shoes have no skill or pain requirements — nothing to show
	if PShoSkill = 0 and PShoPain['mild'] = 0:
		$result = ''
		exit
	end
	!! Use pain thresholds (from shoe_attributes.qsrc) to reflect current pain level.
	!! Mirrors the wardrobe description in shoes.qsrc exactly.
	!! ≥mild = no pain message shown (Fine); lower tiers are genuinely painful.
	if PShoPain['severe'] > 0 and pcs_heels < PShoPain['severe']:
		$result = $func('wrap', 'v_neg', '[很疼]')
	elseif PShoPain['medium'] > 0 and pcs_heels < PShoPain['medium']:
		$result = $func('wrap', 'neg', '[疼痛]')
	elseif pcs_heels < PShoPain['mild']:
		$result = $func('wrap', 'neutral', '[不适]')
	else
		$result = $func('wrap', 'v_pos', '[还好]')
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ cond_clo_style — colored style/category tag list for clothing   │
!! │ PCloStyle2 role: 1=maid 2=server 3=stripper 4=school 5=office   │
!! │ 6=gym. Also: PCloSwimwear, PCloStyle=4, bimbo/goth/punk flags.  │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'cond_clo_style':
	$result = ''
	if $clothingworntype = 'nude': exit

	!! Role uniform (Swimwear overrides Style2 per outfit.qsrc)
	if PCloSwimwear = 1:
		$result += ', ' + $func('wrap', 'bimbo', '[泳装]')
	elseif PCloStyle2 = 6:
		$result += ', ' + $func('wrap', 'bimbo', '[健身]')
	elseif PCloStyle2 = 5:
		$result += ', ' + $func('wrap', 'bimbo', '[办公]')
	elseif PCloStyle2 = 4:
		$result += ', ' + $func('wrap', 'bimbo', '[校服]')
	elseif PCloStyle2 = 3:
		$result += ', ' + $func('wrap', 'bimbo', '[脱衣舞娘]')
	elseif PCloStyle2 = 2:
		$result += ', ' + $func('wrap', 'bimbo', '[服务员]')
	elseif PCloStyle2 = 1:
		$result += ', ' + $func('wrap', 'bimbo', '[女仆]')
	end
	if PCloStyle = 4: $result += ', ' + $func('wrap', 'bimbo', '[卖淫]')
	if PCloInhibit >= 40 or PCloThinness = 6: $result += ', ' + $func('wrap', 'bimbo', '[暴露狂]')
	if PCloBimbo = 1 and cheatVars['bimbo'] = 0: $result += ', ' + $func('wrap', 'bimbo', '[婊气]')
	if PCloGoth = 1: $result += ', ' + $func('wrap', 'goth', '[哥特]')
	if PCloPunk = 1: $result += ', ' + $func('wrap', 'punk', '[朋克]')

	!! Trim leading ', '
	if $result <> '': $result = mid($result, 3)
	exit
end


!! ╔═════════════════════════════════════════════════════════════════════╗
!! ║                      COMPUTE IMAGES                                 ║
!! ╚═════════════════════════════════════════════════════════════════════╝


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ compute_images — populate $sd_img[url/label/short/cond_KEY]     │
!! │ for each visible entry in $image_order[].                       │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'compute_images':

	killvar 'sd_img'
	sd_img['count'] = 0

	sd_ci['max'] = arrsize('$image_order')
	sd_ci['i'] = 0
	:sd_compute_images_loop
		$sd_ci['key'] = $image_order[sd_ci['i']]

		!! Skip if this image type is hidden
		if stat_hide[$sd_ci['key']] = 1: jump 'sd_compute_images_next'

		!! ── Per-key image computation ───────────────────────────────
		if $sd_ci['key'] = 'face':
			$sd_img['url_face'] = $func('$face_image')
			!! Makeup label lookup (face_style type 2 = makeup active)
			if face_style['type'] <> 2:
				$sd_img['label_face'] = '外貌'
			else
				$sd_mu_lbl[0] = '花掉的妆容'
				$sd_mu_lbl[1] = '素颜'
				$sd_mu_lbl[2] = '淡妆'
				$sd_mu_lbl[3] = '亮丽妆容'
				$sd_mu_lbl[4] = '浓妆'
				$sd_mu_lbl[5] = '专业妆容'
				$sd_mu_lbl[6] = '傻妞妆'
				$sd_mu_lbl[7] = '哥特妆'
				$sd_img['label_face'] = iif(pcs_makeup >= 1 and pcs_makeup <= 7, $sd_mu_lbl[pcs_makeup], iif(pcs_makeup <= 0, '花掉的妆容', '外貌'))
				killvar 'sd_mu_lbl'
			end
			$sd_img['short_face'] = $sd_img['label_face']
			!! Condition: cum on face, writing, or messy hair
			$sd_img['cond_face'] = $func('stat_display_compute', 'cond_face')

		elseif $sd_ci['key'] = 'body':
			$sd_img['url_body'] = $func('$body_image', 'body')
			$sd_img['label_body'] = '身体'
			$sd_img['short_body'] = '身体'
			!! Condition: sweat (only when notable ≥20) + visible cum on body (not face)
			!! check_visible from cum_manage = cumloc[6]+[7]+[11]+[13]+[16]; exclude [11] (face shown separately)
			$sd_img['cond_body'] = $func('stat_display_compute', 'cond_sweat')
			if (cumloc[6] + cumloc[7] + cumloc[13] + cumloc[16]) > 0:
				$sd_ci['cum_line'] = $func('wrap', 'v_neg', '[身上有精液]')
				$sd_img['cond_body'] = iif($sd_img['cond_body'] <> '', $sd_img['cond_body'] + ' ', '') + $sd_ci['cum_line']
			end

		elseif $sd_ci['key'] = 'coat':
			if $coatworntype = 'none' or $coatworntype = '':
				jump 'sd_compute_images_next'
			end
			$sd_img['url_coat'] = $func('$body_image', 'coat')
			$sd_img['label_coat'] = '外套'
			$sd_img['short_coat'] = $func('$item_description', 'coat', $coatworntype, coatwornnumber)
			!! Condition: warmth adequacy vs current temperature
			$sd_img['cond_coat'] = $func('stat_display_compute', 'cond_coat_temp')

		elseif $sd_ci['key'] = 'clothes':
			$sd_img['url_clothes'] = $func('$body_image', 'clothes')
			if $clothingworntype = 'nude':
				!! Towel or robe override, otherwise naked
				$sd_img['label_clothes'] = iif(towel = 1, '毛巾', iif(robe = 1, '浴袍', '裸体'))
				$sd_img['short_clothes'] = $sd_img['label_clothes']
				!! No condition for nude/covering states
			else
				$sd_img['label_clothes'] = 'Clothes'
				$sd_img['short_clothes'] = $func('$short_description', $clothingworntype, clothingwornnumber)
				!! Condition: dirt + wear, then style/category tags
				$sd_img['cond_clothes'] = $func('stat_display_compute', 'cond_string', PCloDirt, 1, dyneval('result = <<$clothingworntype>>_h[<<clothingwornnumber>>]'), 1)
				$sd_ci['style_line'] = $func('stat_display_compute', 'cond_clo_style')
				if $sd_ci['style_line'] <> '':
					$sd_img['cond_clothes'] = iif($sd_img['cond_clothes'] <> '', $sd_img['cond_clothes'] + ' ', '') + $sd_ci['style_line']
				end
			end

		elseif $sd_ci['key'] = 'bra':
			$sd_img['url_bra'] = $func('$body_image', 'bra')
			if $braworntype = 'none' or $braworntype = '':
				$sd_img['label_bra'] = '胸部'
				$sd_img['short_bra'] = '胸部'
			else
				$sd_img['label_bra'] = '胸罩'
				$sd_img['short_bra'] = $func('$item_description', 'bra', $braworntype, brawornnumber)
				!! Condition: dirt only (no durability tracking for bras)
				$sd_img['cond_bra'] = $func('stat_display_compute', 'cond_string', PBraDirt, 1, 0, 0)
			end

		elseif $sd_ci['key'] = 'panties':
			$sd_img['url_panties'] = $func('$body_image', 'panties')
			if $pantyworntype = 'none' or $pantyworntype = '':
				$sd_img['label_panties'] = '小穴'
				$sd_img['short_panties'] = '小穴'
			else
				$sd_img['label_panties'] = '内裤'
				$sd_img['short_panties'] = $func('$item_description', 'panty', $pantyworntype, pantywornnumber)
				!! Condition: dirt only (no durability tracking for panties)
				$sd_img['cond_panties'] = $func('stat_display_compute', 'cond_string', PPanDirt, 1, 0, 0)
			end

		elseif $sd_ci['key'] = 'shoes':
			$sd_img['url_shoes'] = $func('$body_image', 'shoes')
			if $shoeworntype = 'none' or $shoeworntype = '':
				$sd_img['label_shoes'] = '脚'
				$sd_img['short_shoes'] = '脚'
			else
				$sd_img['label_shoes'] = '鞋子'
				$sd_img['short_shoes'] = $func('$shoe_description', $shoeworntype, shoewornnumber)
			end
			!! Condition: heel skill adequacy (empty when no heels required)
			$sd_img['cond_shoes'] = $func('stat_display_compute', 'cond_heel_skill')

		elseif $sd_ci['key'] = 'bodysuit':
			!! Only shown when actually worn
			if bodysuitwornnumber = 0 or $bodysuitworntype = 'none' or $bodysuitworntype = '':
				jump 'sd_compute_images_next'
			end
			!! Bodysuit replaces bra+panties; skip if both are hidden
			if stat_hide['bra'] = 1 and stat_hide['panties'] = 1:
				jump 'sd_compute_images_next'
			end
			$sd_img['url_bodysuit'] = $func('$body_image', 'bodysuit')
			$sd_img['label_bodysuit'] = '连体衣'
			$sd_img['short_bodysuit'] = $func('$item_description', 'bodysuit', $bodysuitworntype, bodysuitwornnumber)
			!! Condition: dirt only (uses PPanDirt slot per outfit.qsrc)
			$sd_img['cond_bodysuit'] = $func('stat_display_compute', 'cond_string', PPanDirt, 1, 0, 0)
		end

		!! Skip if URL ended up empty (e.g. coat with no image)
		if $sd_img['url_' + $sd_ci['key']] = '': jump 'sd_compute_images_next'

		!! Record visible order for rendering
		$sd_img['order_' + $str(sd_img['count'])] = $sd_ci['key']
		sd_img['count'] += 1

		:sd_compute_images_next
		sd_ci['i'] += 1
	if sd_ci['i'] < sd_ci['max']: jump 'sd_compute_images_loop'

	killvar 'sd_ci'
	exit
end
