# kakuzu
if $ARGS[0] = 'start':
	cla & *clr
	minut += 5
	$menu_loc = 'kakuzu'
	$menu_arg = 'start'
	menu_off = 0
	gs 'stat'
	gs 'themes', 'indoors'
	'<center><b>卡库祖公馆</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/industrial/mansion/mansion.jpg"></center>'
	'一座阴森古老的大宅，连本地帮派都因恐惧而不敢靠近，他们深信那里住着某种邪恶之物。无需任何技能，你也会丧命于致命一击之下。'

	act '进入公馆': gt 'kakuzu', 'mansion'
	act '离开': gt 'city_industrial'
end


if $ARGS[0] = 'mansion':
	$location = 'event'
	*clr & cla
	minut += 5
	gs 'stat'
	'<center><b>卡库祖公馆</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/industrial/mansion/mansion1.jpg"></center>'
	'房间里坐着角都，他完全没有注意到你。'

	act '与角都交谈': gt 'kakuzu', 'mansion1'
	act '离开': gt 'kakuzu', 'start'
end


if $ARGS[0] = 'mansion1':
	$location = 'event'
	*clr & cla
	minut += 5
	gs 'stat'
	'<center><b>角都</b></center>'
	'<center><img <<$set_imgh>> src="images/characters/city/kakuzu/kakuzu.jpg"></center>'

	if tobiQW = 1: '我知道你是谁，也知道你需要什么。付给我 ' + $func('money', 'string_price', 100) + 'k，然后我就教你如何获得元素亲和。'
	if tobiQW = 2: '你有<<srodpaper>>张用来测定元素亲和的纸。如果纸用完了，你必须在角都那里再买，每张 ' + $func('money', 'string_price', 10000) + '。'

	if tobiQW = 1:
		act '钱在这里' + $func('money', 'get_cost_string', 100000):
			if func('money', 'can_afford', 100000) = 0:
				$noMoney
			else
				cla
				gs 'money', 'pay', 100000
				tobiQW = 2
				srodpaper = 10
				'你把钱放在角都面前。他连看都没看就把钱扔进了抽屉。'
				'你问：“连数都不数一下吗？”角都平静地回答：“哪怕只差一个卢布，我也会找到你并杀了你。”'
				'（角都） - 仔细听好了。元素一共有五种， '
				'火系 - 该学派拥有最强大的伤害潜力和最弱的防护。但没有比这更适合一击击败对手的学派了。'
				'雷系 - 该学派每次命中都会使对手麻痹，其伤害比火系弱，但如果你被雷电缠住，那么你很快就能达成任何目标。'
				'风系 - 伤害弱，防御弱，但迷雾或分身都无法把他们从风中拯救出来而且风系法术几乎无法躲避。'
				'土系 - 防御型学派，但攻击非常危险。如果敌方攻击落空，他们就会卡在地里无法移动。土系防护是所有防护中最强大的，大地防护能自我再生并形成对敌人攻击的绝对防御。'
				'水系 - 此学派的伤害较弱，但所有法术都会从敌人身上窃取法力，同样高级的水系防护也非常危险，它会消耗敌人的法力并自我恢复，虽然比土系防护弱。'
				'元素亲和意味着你可以学习特定的元素。在选定亲和后直到完全掌握该元素你无法探索其他元素。我会给你10张魔法纸。每张纸花费我 ' + $func('money', 'string_price', 5000) + '. 将能量集中在一张纸上，直到成功为止，然后你可以重新开始，希望得到另一种元素，或者继续使用出现的元素。'

				act '在纸上做标记': gt 'kakuzu', 'mansion1'
			end
		end
	elseif tobiQW = 2 and srodpaper <= 0:
		act '我需要再来一张纸' + $func('money', 'get_cost_string', 10000):
			if func('money', 'can_afford', 10000) = 0:
				$noMoney
			else
				cla
				gs 'money', 'pay', 10000
				srodpaper += 1
				'你交了钱，角都递给你一张纸。'

				act '在纸上做标记': gt 'kakuzu', 'mansion1'
			end
		end
	end

	if tobiQW = 2:
		act '再给我讲讲元素':
			cla
			'（角都） - 蠢货，听好了，记清楚。元素一共有五种， '
			'火系 - 该学派拥有最强大的伤害潜力和最弱的防护。但没有比这更适合一击击败对手的学派了。'
			'雷系 - 该学派每次命中都会使对手麻痹，其伤害比火系弱，但如果你被雷电缠住，那么你很快就能达成任何目标。'
			'风系 - 伤害弱，防御弱，但迷雾或分身都无法把他们从风中拯救出来而且风系法术几乎无法躲避。'
			'土系 - 防御型学派，但攻击非常危险。如果敌方攻击落空，他们就会卡在地里无法移动。土系防护是所有防护中最强大的，大地防护能自我再生并形成对敌人攻击的绝对防御。'
			'水系 - 此学派的伤害较弱，但所有法术都会从敌人身上窃取法力，同样高级的水系防护也非常危险，它会消耗敌人的法力并自我恢复，虽然比土系防护弱。'
			'元素亲和意味着你可以学习特定的元素。在选定亲和后直到完全掌握该元素你无法探索其他元素。我会给你10张魔法纸。每张纸花费我 ' + $func('money', 'string_price', 5000) + '. 将能量集中在一张纸上，直到成功为止，然后你可以重新开始，希望得到另一种元素，或者继续使用出现的元素。'

			act '走开': gt 'kakuzu', 'mansion1'
		end
	end

	if tobiQW = 2 and srodpaper > 0 and pcs_mana < 2000: '你的法力不足，无法继续集中精神。'

	if tobiQW = 2 and srodpaper > 0 and pcs_mana >= 2000:
		act '将能量集中在纸上 (1:00)':
			cla
			minut += 60
			pcs_mana -= 2000
			gs 'sweat', 'add', 40

			if srodstvo = 0: srodrand = rand(1, 5)

			'你将能量集中在纸上。你感到自己为这个举动消耗了巨量的法力。'
			srodstvo += rand(9 - 2 * cheatVars['skill_gain'], 10)

			if srodstvo < 100:
				'你累得汗流浃背，但你的努力还不够。'

				act '走开': gt 'kakuzu', 'mansion1'
			elseif srodstvo >= 100:
				srodpaper -= 1

				if srodrand = 1:
					'纸在你手中破裂。角都赞许地点了点头。火系亲和。你接受吗？'
					gs 'spellList', 'addAvailableSpells', 'fireSpells', 20
				end
				if srodrand = 2:
					'纸在你手中通了电，火花在纸上跳跃。角都赞许地点了点头。雷系亲和。你接受吗？'
					gs 'spellList', 'addAvailableSpells', 'electSpells', 20
				end
				if srodrand = 3:
					'纸在你手中变得坚硬。角都赞许地点了点头。土系亲和。你接受吗？'
					gs 'spellList', 'addAvailableSpells', 'earthSpells', 20
				end
				if srodrand = 4:
					'纸在你手中碎裂成片。角都赞许地点了点头。风系亲和。你接受吗？'
					gs 'spellList', 'addAvailableSpells', 'airSpells', 20
				end
				if srodrand = 5:
					'纸在你手中湿透了。角都赞许地点了点头。水系亲和。你接受吗？'
					gs 'spellList', 'addAvailableSpells', 'waterSpells', 20
				end

				act '确定亲和':
					cla
					tobiQW = 3
					stihia = srodrand
					srodstvo = 0
					gt 'kakuzu', 'mansion1'
				end

				act '放弃这个亲和':
					srodstvo = 0
					gt 'kakuzu', 'mansion1'
				end
			end
		end
	end

	if tobiQW = 3 and pcs_mana < 1000:
		'角都告诉你："你几乎没什么力气了，你需要休息。"'
	elseif tobiQW = 3 and pcs_mana >= 1000:
		if stihia = 1:
			gs 'spellList', 'teacherActions', '$fireSpells', 'kakuzu', 'mansion1'
		end

		if stihia = 2:
			gs 'spellList', 'teacherActions', '$electSpells', 'kakuzu', 'mansion1'
		end

		if stihia = 3:
			gs 'spellList', 'teacherActions', '$earthSpells', 'kakuzu', 'mansion1'
		end

		if stihia = 4:
			gs 'spellList', 'teacherActions', '$airSpells', 'kakuzu', 'mansion1'
		end

		if stihia = 5:
			gs 'spellList', 'teacherActions', '$waterSpells', 'kakuzu', 'mansion1'
		end
	end

	act '离开': gt 'kakuzu', 'mansion'
end
