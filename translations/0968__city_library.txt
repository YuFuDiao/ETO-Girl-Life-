# city_library

menu_off = 0

if $ARGS[0] = '':
	gs 'core_library', 'setloc', 'city_library', $ARGS[0]
	$location_type = 'public_indoors'
	frost = 0
	gs 'stat'
	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '俄罗斯国家图书馆')
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/library/downlibint.jpg"></center>'
	'一座巨大的图书馆，藏书成千上万。你就算在这里待上几年，恐怕也读不完所有的书。'

	act '离开': minut += 5 & gt 'city_center'

	if hour < 8 or hour >= 19:
		'图书馆关门了。'
		exit
	end

	'有一个很大的IT区，里面有<a href="exec: gs ''internet_mobile'', ''get_access'', ''free'', ''noporn'',''nocamshow'' & gs ''internet_mobile'', ''add_limitation'', ''general'', ''On this public computer all adult content sites are blocked'' & gt ''komp'',''start''">电脑</a>可供公众使用。'

	act '读一本书': gt 'city_library', 'read'
	act '借一本书': gt 'city_library', 'loan'

	if MagicLibrary = 1:
		act '阅读魔法书':
			*clr & cla
			menu_off = 1
			gs 'library_functions', 'read_book', 'splcstng'
			gs 'stat'
			$func('wrap', 'header center b', '魔法书')
			imageid = rand(1, 2)
			'<center><img <<$set_imgh>> src="images/locations/city/citycenter/library/mage_learning_<<imageid>>.jpg"></center>'
			killvar 'imageid'
			'你挑了一本自己真正能读懂的魔法书，坐下来翻阅。'
			'你发现这些内容不仅艰深晦涩，大部分对你来说如同天书，而且描述和解释冗长得毫无必要，仿佛作者在刻意证明什么似的。'
			'大约一个小时后，你决定先休息一下，转而测试自己的记忆。你记住的不多，但对魔法的理解比之前稍微好了一点。'

			!! Find Researchable Spells
			if func('spellList', 'numAvailableSpells', 'librarySpells') <= 0:
				'这里似乎已经没有什么新魔法可学了，不过你仍然可以在已经找到的魔法上继续提升技能。'
			else
				if rand(0, 9) = 0:
					gs 'spellList', 'addAvailableSpells', 'librarySpells'
					*nl
					*pl '你偶然发现了一条关于新魔法的线索。'
				end
			end

			!! Adding to the Actions opportunity to practice spells MC found
			gs 'spellList', 'teacherActions', '$librarySpells', 'city_library', ''

			act '把书放下': gt $loc, $loc_arg
		end
		!! Adding to the Actions opportunity to practice spells MC found
		gs 'spellList', 'teacherActions', '$librarySpells', 'city_library', ''
	end

	gs 'camera', 'check_location'
end


if $ARGS[0] = 'read':
	gs 'core_library', 'setloc', 'city_library', 'read'

	act '离开': minut += 5 & gt 'city_center'

	if hour <= 7 or hour >= 19:
		'图书馆关门了。'
		exit
	end

	act '回去': gt 'city_library'
	gs 'library_functions', 'set_library_read_acts'
end


if $ARGS[0] = 'loan':
	gs 'core_library', 'setloc', 'city_library', 'loan'

	act '离开': minut += 5 & gt 'city_center'

	if hour <= 7 or hour >= 19:
		'图书馆关门了。'
		exit
	end

	act '回去': gt 'city_library'
	gs 'library_functions', 'set_loan_acts'
end
