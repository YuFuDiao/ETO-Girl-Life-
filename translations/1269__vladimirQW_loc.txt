			cla
			'你决定无视那个男人，转身走开，但听到身后传来脚步声，你又转过身来。'

if $ARGS[0] = 'qwvladdy':

	gs 'vladimirQW_loc', 'qwnamedy'

		cla
		vladimirQW['stage'] = 1

		gs 'vladimirQW_loc', 'qwnamedy'

			cla
			vladimirQW['stage'] = 2

		end
	end

if $ARGS[0] = 'qwnamedy':
	act '<<$pcs_nickname>>':
		cla
		vladimirQW['stage'] = 10
		'"<<$pcs_nickname>>。"你自我介绍道。'
		'弗拉基米尔微笑道："一个非常适合你的美丽名字。很遗憾我得走了，但我想再见到你。如果你留下电话号码，我明天可以给你打电话，带你去餐厅共进晚餐。"'

		act '不（离开）':
			cla
			vladimirQW['stage'] = 2
			'你拒绝了，走开了，留下那个男人站在原地。'

			act '离开': gt 'city_park', 'luna'
		end

		act '把你的号码给他':
			cla
			vladimirQW['stage'] = 20
			vladimirQW['day'] = daystart
			gs 'telefon', 'AddContact', 'A108', 'icon_na', 1
			gs 'telefon', 'SetInCallSchedule', 'A108', "gs 'vladimirQW_loc', 'telephone_call'", "vladimirQW['day'] ! daystart and hour >= 15 and hour <= 17"
			gs 'stat'
			'你把电话号码告诉了弗拉基米尔，他把号码记在一个皮革封面的小笔记本上。'

			act '离开': gt 'city_park', 'luna'
		end
	end
end & !! --- qwnamedy ---
