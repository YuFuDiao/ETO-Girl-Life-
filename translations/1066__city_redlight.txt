# city_redlight


if $ARGS[0] = 'start':
	*clr & cla
	CLOSE ALL
	menu_off = 0
	if sound_settings['environment_off'] = 0:
		if month >= 11 and month <= 12 or month >= 1 and month <= 3:
			if hour >= 8 and hour <= 23:
				PLAY 'sound/street2.mp3', 30
				PLAY 'sound/street7.mp3', 30
				PLAY 'sound/zima.mp3', 30
			end
			if hour >= 0 and hour <= 7: PLAY 'sound/street_night.mp3', 30 & PLAY 'sound/zima.mp3', 30
		else
			if hour >= 8 and hour <= 23:
				PLAY 'sound/street2.mp3', 30
				PLAY 'sound/street7.mp3', 30
				PLAY 'sound/street_leto.mp3', 30
			end
			if hour >= 0 and hour <= 7: PLAY 'sound/street_night_leto.mp3' & PLAY 'sound/street_night.mp3', 30
		end
	end
	gs 'core_library', 'setloc', 'city_redlight', 'start'
	$region = 'city'
	$location_type = 'public_outdoors'

	gs 'stat'
	'<center><h2>圣彼得堡</h2></center>'
	'<center><b>红灯区</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/redlight/redlight.jpg"></center>'

	'工业区内的这个小区域是脱衣舞俱乐部、成人用品店和其他下流生意的聚集地。'
	if hour >= 18 or hour < 3:
		'这家名为<a href="exec:gt ''stripclub'', ''start''">黄金女孩</a>的脱衣舞俱乐部以明亮的灯光和轰鸣的音乐吸引着全城的男人。'
	else
		if ($job_status['city_strip_stripper'] = 'employed' and func('jobs', 'is_arrival_time', 'city_strip_stripper') = 1) or ($job_status['city_strip_bargirl'] = 'employed' and func('jobs', 'is_arrival_time', 'city_strip_bargirl') = 1):
			' <a href="exec:gt ''stripclub'', ''start''">黄金女孩</a>脱衣舞俱乐部仍然关闭，但你可以从员工入口进去。'
		else
			'脱衣舞俱乐部当天关闭，将在 ' + func('time', 'get_time_string', 18, 0) + '。'
		end
	end
	if pornstudio > 0:
		if hour >= 9 and hour <= 22:
			'一家<a href="exec:gt ''pornstudio''">色情工作室</a>设在少数几座尚未废弃的仓库之一里。'
		else
			'色情工作室已关闭，将在 ' + func('time', 'get_time_string', 9, 0) + '。'
		end
	end
	if hour < 3 or hour >= 11:
		'一家出售性玩具和其他情色商品的<a href="exec:minut += 1 & gt ''shop_erotomaniac'', ''start''">破旧小店</a>正在营业，供全城的变态者浏览其商品。'
	else
		'一家外观破旧的商店，出售性玩具和其他情色商品。目前停业，将在 ' + func('time', 'get_time_string', 11, 0) + '。'
	end
	if bordel > 0 and func('bordel', 'is_open'): '庭院深处藏着通往<a href="exec:gt ''bordel''">妓院</a>的入口。'
	if hour >= 10:
		'一家小<a href="exec:gt ''city_redlight'',''city_copy_shop''">印刷店</a>藏身于两栋较大的建筑之间。'
	else
		'一家小印刷店隐藏在两栋较大的建筑之间。目前停业，将在 '+func('time', 'get_time_string', 10, 0)+'。'
	end
	if hour >= 19 or hour < 2:
		'一家<a href="exec:gt ''hookahlounge'', ''start''">水烟休息室</a>坐落在街角，印度音乐和异国香料的香味从敞开的门里飘出。'
	else
		'水烟休息室目前关闭，将在 ' + func('time', 'get_time_string', 19, 0) + '。'
	end

	act '步行到工业区（0:05）': minut += 5 & gt 'city_industrial'
	gs 'taxi'

	if func('car_funcs', 'is_here', 'fuelstation', 'start'):
		'你的<a href="exec:minut += 5 & gs ''carF'', ''start''"><<$car[''name'']>></a>停在加油站附近。'
	elseif func('car_funcs', 'is_here', 'city_redlight', 'start'):
		'你的<a href="exec:gs ''carF'', ''start''"><<$car[''name'']>></a>停在停车场里。'
	elseif func('car_funcs', 'is_here', 'autoservisF', 'start'):
		if func('autoservisF', 'is_open'):
			'你的<a href="exec:minut += 5 & gs ''carF'', ''start''"><<$car[''name'']>></a>停在汽车服务中心。'
		else
			'看起来修车店在关门之前把你的车移到了外面。'
			$car['loc'] = 'city_industrial'
			$car['loc_arg'] = ''
		end
	elseif func('car_funcs', 'is_here', 'autotraidF', 'start'):
		if func('autotraidF', 'is_open'):
			'你的<a href="exec:minut += 5 & gs ''carF'', ''start''"><<$car[''name'']>></a>停在汽车经销商那里。'
		else
			'看起来汽车经销商在关门之前把你的车移到了外面。'
			$car['loc'] = 'city_industrial'
			$car['loc_arg'] = ''
		end
	end
	if func('car_funcs', 'is_here', 'city_industrial', ''):
		'你的<a href="exec:minut += 5 & gs ''carF'', ''start''"><<$car[''name'']>></a>停在红灯区的停车场里。'
	end

	if hour >= 20 or hour <= 4:
		*nl
		'这附近有妓女正在接客。'
		if pcs_makeup > 2 and (PCloStyle = 4 or PCloProstitute = 1) and pcs_inhib >= 40:
			act '出卖自己': prst_loc = 1 & gt 'street_walker', 'work'
		elseif pcs_inhib >= 40:
			*nl
			'你可以做妓女，但你需要穿合适的衣服，还要化浓一点的妆。'
		else
			*nl
			'你可以做妓女，但你对这个想法感到不自在。'
		end
	end
end


if $ARGS[0] = 'city_copy_shop':
	*clr & cla
	menu_off = 1
	minut += 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/redlight/bm_seller.jpg"></center>'
	'你走进这家破旧的复印店，环顾四周。房间里散落着一些小桌子，上面摆着各种布满灰尘的复印机，看起来不太可靠。柜台后面，一个头发油腻、神情猥琐的男人正百无聊赖地看着你。'
	if fakepassport = 2:
		!! only notify player 1 time after events triggered
		fakepassport = 0
		'他狠狠地盯着你。“一本假护照要花你 ' + $func('money', 'string_price', 30000) + '，”他低声说道。'
	end
	if university['fakediplom'] = 2:
		!! only notify player 1 time after events triggered
		university['fakediplom'] = 0
		'“一本学士学位文凭要花你 ' + $func('money', 'string_price', 100000) + '，”他低声说道。'
	end
	act '离开': gt 'city_redlight', 'start'
	if age < 18 and fakepassport ! 1:
		act '买假护照' + $func('money', 'get_cost_string', 30000, 'cash'):
			if func('money', 'can_afford', 30000, 'cash') = 0:
				$noMoney
			else
				*clr & cla
				minut += 60
				gs 'money', 'pay', 30000, 'cash'
				fakepassport = 1
				'<center><img <<$set_imgh>> src="images/locations/city/redlight/bm_seller.jpg"></center>'
				'他让你到后面去，快速给你拍了张脸部照片，然后让你在前面等他做完。大约一个小时后，他走出来把护照递给你，护照上是你的脸，但印着一个更大的年龄。'
				act '离开': gt 'city_redlight', 'start'
			end
		end
	end
	if university['diploma'] = 0 and university['fakediplom'] = 0:
		act '买假文凭' + $func('money', 'get_cost_string', 100000, 'cash'):
			if func('money', 'can_afford', 100000, 'cash') = 0:
				$noMoney
			else
				*clr & cla
				minut += 60
				gs 'money', 'pay', 100000, 'cash'
				university['fakediplom'] = 1
				'<center><img <<$set_imgh>> src="images/locations/city/redlight/bm_seller.jpg"></center>'
				'你买了一张假文凭。'
				act '离开': gt 'city_redlight', 'start'
			end
		end
	end
end
