# KGstart

if $ARGS[0] = '':
	minut += 1
	gs 'stat'

	if KFOnLineReaga > 0:
		'欢迎来到韦尔泰普世界！'
		'你可以游玩<<KFOnLineReaga>>更多天，但你可以随时通过网上银行延长游玩时间（' + $func('money', 'string_price', 1000) + ' 持续30天）'

		if KGOLpers > 0:
			'你可以用当前的角色进入游戏，也可以删除他们。'
			'<<$KGOLname>> - <<KGOLlvl>> <<$KGOLrace>> <<$KGOLklass>>'

			act '玩': gt 'KGOLgame'

			act '删除角色': KGOLpers = 0 & gt 'KGstart'
		else
			'你现在可以创建角色了！'

			act '创建一个角色':
				cla
				$KGOLname = input("输入你的角色名字")

				if $KGOLname = '': msg '请输入有效的名字！' & gt 'KGstart'

				'你的角色名字是：<<$KGOLname>>'
				'请选择玩家种族：'

				act '人类':	$KGOLrace = 'Human'		& gt 'KGstart', 'KGOLklassvibor'
				act '精灵':		$KGOLrace = 'Elf'		& gt 'KGstart', 'KGOLklassvibor'
				act '兽人':	$KGOLrace = 'Zveroid'	& gt 'KGstart', 'KGOLklassvibor'
			end
		end

		act '再多买几天' + $func('money', 'get_cost_string', 1000, 'bank'):
			if func('money', 'can_afford', 1000, 'bank') = 0:
				$noMoney
			else
				gs 'money', 'pay', 1000, 'bank'
				KFOnLineReaga += 30
				gt 'KGstart'
			end
		end

		act '购买游戏币' + $func('money', 'get_cost_string', 1, 'bank') + ' = 1 枚游戏币':
			if func('money', 'can_afford', 1, 'bank') = 0:
				$noMoney
			else
				cla
				KGOLdonat = input("你想购买多少硬币？")

				if KGOLdonat <= 0 or func('money', 'can_afford', KGOLdonat, 'bank') = 0:
					msg '请输入有效的数量！'
				else
					gs 'money', 'pay', KGOLdonat, 'bank'
					KGOLmoney += KGOLdonat
					msg '谢谢！<<KGOLdonat>>硬币已添加到你的账户中。'
				end

				gt 'KGstart'
			end
		end
	else
		'欢迎来到韦尔泰普世界！'
		'要进入这个游戏，你必须拥有通过网上银行支付的有效订阅（' + $func('money', 'string_price', 1000) + ' 持续30天）'

		act '付费解锁权限' + $func('money', 'get_cost_string', 1000, 'bank'):
			if func('money', 'can_afford', 1000, 'bank') = 0:
				$noMoney
			else
				gs 'money', 'pay', 1000, 'bank'
				KFOnLineReaga += 30
				gt 'KGstart'
			end
		end
	end

	act '离开': gt 'komp', 'start'
end


if $ARGS[0] = 'KGOLklassvibor':
	cla
	'请选择角色职业：'

	if $KGOLrace = 'Human':
		'人类可选职业：战士、盗贼、法师'

		act '战士':	$KGOLklass = 'Warrior'		& gt 'KGstart', 'KGOLklassvibor2'
		act '盗贼':	$KGOLklass = 'Rogue'		& gt 'KGstart', 'KGOLklassvibor2'
		act '法师':	$KGOLklass = 'Mage'				& gt 'KGstart', 'KGOLklassvibor2'
	elseif $KGOLrace = 'Elf':
		'精灵可选职业：牧师、弓箭手'

		act '牧师':	$KGOLklass = 'Priest'		& gt 'KGstart', 'KGOLklassvibor2'
		act '弓手':	$KGOLklass = 'Archer'		& gt 'KGstart', 'KGOLklassvibor2'
	elseif $KGOLrace = 'Zveroid':
		'兽人可选职业：野蛮人'

		act '野蛮人': $KGOLklass = 'Barbarian'	& gt 'KGstart', 'KGOLklassvibor2'
	end
end


if $ARGS[0] = 'KGOLklassvibor2':
	cla
	KGOLexp = 0
	KGOLlvl = 1
	KGOLbrona = 0
	KGOLwpower = 0
	'<<$KGOLname>> <<$KGOLrace>> <<$KGOLklass>> <<KGOLlvl>>级别'
	if $KGOLklass = 'Warrior':
		KGOLstren = 30
		KGOLspeed = 20
		KGOLvital = 30
		KGOLintel = 20
	elseif $KGOLklass = 'Rogue':
		KGOLstren = 20
		KGOLspeed = 30
		KGOLvital = 20
		KGOLintel = 30
	elseif $KGOLklass = 'Mage':
		KGOLstren = 10
		KGOLspeed = 20
		KGOLvital = 20
		KGOLintel = 50
	elseif $KGOLklass = 'Priest':
		KGOLstren = 10
		KGOLspeed = 10
		KGOLvital = 10
		KGOLintel = 70
	elseif $KGOLklass = 'Archer':
		KGOLstren = 20
		KGOLspeed = 40
		KGOLvital = 10
		KGOLintel = 30
	elseif $KGOLklass = 'Barbarian':
		KGOLstren = 40
		KGOLspeed = 10
		KGOLvital = 40
		KGOLintel = 10
	end

	KGHP = KGOLvital * 10
	KGMana = KGOLintel * 10

	act '玩': KGOLpers = 1 & gt 'KGOLgame'
	act '删除角色': KGOLpers = 0 & gt 'KGstart'
end
