!! Ballet Summer School
!!
!! This is the initiator for the ballet story arc. Triggers in June, 2016 on start, and is open to the Dancer at the start of July.
!!
!! Variables
!!
!! balletqw[blocker]- quest line initiator to start/stop the quests.
!! balletqw[letter]- temp blocker to prevent the letter being read multiple times.
!! ballet[hypno] = 1 - hypno content trapdoor
!! school_ballet_grade_score[''total''] combines ballet_grade_score[''class''] and ballet_grade_score[''homework''] to assess attendance and homework completion.
!! balletqw[school] - Holding variable for final score 2 - passed, 3 - passed with Rudolph''s help, 4 - passed with Maya''s help, 5 = Merit, 1 - failed
!! balletqw[rank] - What stage of her career, 0 failed, 1 apprenticeship, 2 sponsored apprenticeship, 3 part of Corp de ballet, 4 full member of the Theatre.
!! Ranks Corps De Ballet, coryphée, demi-soloist, soloist, principle dancer.
!! balletqw[performances] - How many shows performed
!! mayaqw[grave] - Did you attend Anitchka''s grave with Maya, 3 means you talked to Maya, 4 means you were empathic. Has impact later on Maya''s actions towards the player.
!! mayaqw[trust] - How much Maya trusts you and works in conjunction with npc_rel
!! mayaqw[path] - progress tracker for maya quests
!! Rudolph quest is broken into two variables the main tracker being rudolphqw[stage] and rudolphqw[corruption], the latter reflects his slow insidious undermining of Sveta''s
!! confidence and the former is how much she has submitted to his authority and demands. Corruption opens dialogue options.
!!
!! AlbinaQW[StarletsJoined] - 0 not yet invited, 1 joined, -1 kicked out, -2 quit,
!!   -3 declined the invitation. Any non-zero value prevents the invite from firing again.
!!

	if $start_type['cat'] = 'dancer':
		!! Pushkin residential block

			act '去学校（'+func('time', 'get_time_string', 7, 30)+' 开始）': gt 'pushkin_ballet_class', 'start'

			gs 'array', 'remove_element', 'balletqw', 'blocker'

		end
	end
end

if $ARGS[0] = 'daily_assessment':
	!! Grade appearance, behaviour and other factors.
	!! Daily Appearance
	if school_daily_check ! daystart and (nclass > 0 or nclass <= 5):
		gs 'shortgs', 'clothing_status'
		if ballet_debug = 1: if nclass > 0: $ballet_class_debug += nclass + ' - ' + $loc_arg + iif(nclass = 5, ' -|- <br>', ', ')
		!! Positive additions:
		ballet_grade_attendance[week] += 1
		if pcs_makeup > 1:
			ballet_grade_mua[week] += 1
		else
			ballet_grade_mua[week] -= 1
		end
		if hbraids > 0 or hpingripw > 0 or pcs_hairlng <= 80:
			ballet_grade_braids[week] += 1
		elseif hscrunchw > 0:
			ballet_grade_braids[week] -= 1
		end
		if pcs_pubes < 4 and pcs_leghair < 4:
			ballet_grade_shave[week] += 1
		else
			ballet_grade_shave[week] -= 1
		end
		if $apparel['status'] = 'sport' or $apparel['status'] = 'dance':
			ballet_grade_uniform[week] += 1
		else
			ballet_grade_uniform[week] -= 1
		end
		!! Negative deductions:
		!! large busts can affect balance so will penalise performances - penalise if over a c cup
		if pcs_cupsize > 20: ballet_grade_appearance[week] -= 1
		!! pregnancy check - unlikely but in place due to cheats - automatic fail on course
		if thinkpreg > 1 or knowpreg > 1: ballet_grade_discipline[week] -= 200
		!! Teeth checks, appearance is everything and should only affects if custom or cheat stats
		if pcs_missing_teeth > 0 or pcs_teeth ! -1: ballet_grade_appearance[week] -= 5
		if pcs_brace > 1: ballet_grade_discipline[week] -= 1
		!! Don''t do drugs kids - school is zero tolerance to any form of drugs, this only affects the residency week.
		if drugVars['cigarettes_used'] > 1: ballet_grade_discipline[week] -= 5
		if alko > 1: ballet_grade_discipline[week] -= 1
		if drugVars['heroin_high'] > 0 or drugVars['weed_high'] > 0 or drugVars['heroin_high'] > 0 or drugVars['cocaine_system'] > 0 or drugVars['amphetamine_high'] > 0: ballet_grade_discipline[week] -= 50
	end
	if nclass = 5 and school_daily_check ! daystart:
		!! Daily class score
		ballet_daily_score[week] = ballet_grade_attendance[week] + ballet_grade_mua[week] + ballet_grade_braids[week] + ballet_grade_shave[week] + ballet_grade_uniform[week] + ballet_homework[week] - ballet_grade_discipline[week]
		!! Cumulative daily score
		ballet_grade_score['class'] += ballet_daily_score[week]
	end

	if $ARGS[1]= 'grade':
		!! theoretical max score is 175 allowing for 30pts a day, lessons + 5 points (less exam day) in homework lessons.
		!! Bonus points will be added on the assessment day such as taking care of feet.

		ballet_grade_score['total'] = 100 * (ballet_grade_score['class'] + ballet_grade_score['homework'] + ballet_grade_health + (danc_lvl / 10)) / 210
		!! Three score levels, fail, pass and merit. Fail is one route for corruption path, pass is either low or no pay, merit is low to low +10%.
		if ballet_grade_score['total'] <= 40:
			balletqw['school'] = 0
		elseif ballet_grade_score['total'] >= 41 and ballet_grade_score['total'] < 70:
			!! Interim pass until quest lines are integrated for Maya/Rudolph - see header for stage no.
			!! balletqw['school'] = 4
			balletqw['school'] = 2
			balletqw['rank'] = 1
		elseif ballet_grade_score['total'] >= 70 and ballet_grade_score['total'] <= 90:
			balletqw['school'] = 2
			balletqw['rank'] = 1
		elseif ballet_grade_score['total'] >= 91:
			gs 'pushkin_ballet_init', 'famemodifier', ballet_grade_score['total']
			balletqw['school'] = 5
			balletqw['rank'] = 1
		end
		$balletqw['final_report'] += '<centre><h2>普希金芭蕾舞学校 - 成绩报告</h2></centre>'
		$balletqw['final_report'] += '<br>'
		if ballet_grade_score['class'] < 125:
			$balletqw['final_report'] += '这真的值得你花时间去上学吗？你没有参加全部课程或存在纪律问题。很明显你对芭蕾或艺术毫无兴趣。'
		elseif ballet_grade_score['class'] >= 125:
			$balletqw['final_report'] += '你参加了所有指定的课程，并且没有表现出纪律问题。你对艺术持续的奉献和专注值得称赞。'
		else
			$balletqw['final_report'] += '<font color="red"><b>课堂出勤调试：</b></font> <<ballet_grade_score[''class''] >>'
		end
		$balletqw['final_report'] += '<br>'
		if ballet_grade_score['homework'] >=6 and ballet_grade_score['homework'] <= 14:
			$balletqw['final_report'] += '你在完成所有指定作业时表现出的奉献精神和自律在学生中是值得称赞的，但如果你希望成为首席舞者我们期望你未来付出更多努力。'
		elseif ballet_grade_score['homework'] >= 15:
			$balletqw['final_report'] += '你表现出的非凡才能和自律超出了我们对你家庭作业的预期。这正是我们对那些立志并致力于成为首席舞者的舞者的期望。'
		elseif ballet_grade_score['homework'] <= 5:
			$balletqw['final_report'] += '你没有完成布置给你的全部作业，你缺乏成为我校柯里菲的纪律和才能。'
		else
			$balletqw['final_report'] += '<font color="red"><b>作业调试：</b></font> <<ballet_grade_score[''homework'']>> '
		end
		$balletqw['final_report'] += '<br><br>'
		if balletqw['school'] = 5:
			$balletqw['final_report'] += '你的技艺和奉献让我们非常高兴地授予你优秀奖并且你已被接纳为获得我们音乐学院赞助的学徒。'
		elseif balletqw['school'] = 3 or balletqw['school'] = 4:
			$balletqw['final_report'] += '经过深思熟虑我们很高兴为你提供我们学校的名额。不幸的是你在学校的表现不足以让你获得赞助。你需要在指定日期参加与鲁道夫的入学面试以讨论额外训练然后才能被录取进入柯里菲。'
		elseif balletqw['school'] = 2:
			$balletqw['final_report'] += '我们很高兴为你提供我们学校的名额，不幸的是你在学校的表现不足以让你获得赞助。你将按照老师的指示就读音乐学院。'
		elseif balletqw['school'] = 1:
			$balletqw['final_report'] += '你的表现和纪律严重不足，令人失望。你对艺术的态度是一种耻辱，我们从未有过如此漠视你所获特权的学生。因此，你将不会被授予学校柯里菲的席位，也不会被邀请参加进一步的机会。'
		else
			$balletqw['final_report'] += '<font color="red"><b>奖项调试：</b></font> <<balletqw[''school'']>>'
		end
		gs $loc, 'cleanup_var'
		balletqw['letter'] = 1
	end
end
