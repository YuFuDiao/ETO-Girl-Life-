if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_lusso', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '奢内衣')
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/lusso/lusso.jpg"></center>'
	*nl
	'店内装饰明亮而现代，鲜花都是真的，梳妆台用的也是名贵木材。'
	'这显然是一家高端内衣店。光是知道自己身上穿着这些内衣，你的自信就会暴涨，从此再也不想穿回那些廉价又不合身的内衣了。当然，前提是你买得起这里的价格。'

	act '离开': gt 'city_mall'
	act '查看内裤': minut += 5 & gt 'shop_lusso', 'panties'
	act '查看胸罩': minut += 5 & gt 'shop_lusso', 'bras'
	act '查看连体衣': minut += 5 & gt 'shop_lusso', 'bodysuit'
end


if $ARGS[0] = 'panties':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_lusso', 'panties'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看露索内裤')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'panties', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'lusso'
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_lusso', 'start'
	end
end
