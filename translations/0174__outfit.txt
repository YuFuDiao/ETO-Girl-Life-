# outfit


!{
Every outfit type should have the following functions:

!! $ARGS[1] = $...worntype
if $ARGS[0] = 'totals': -> result
	Function that returns the number of elements per type
	i.e.: if $ARGS[1] = 'gm': result = 100


!! $ARGS[1] = $...worntype
!!  ARGS[2] =  ...wornnumber
!! [ARGS[3] = optional flag whether to skip the `gs '..._attributes'` call]
if $ARGS[0] = 'not_wear_reason': -> $result
	Function that returns an empty string if Sveta can wear the item
	Returns a value when she cannot
	i.e. $result = 'low_strength', 'not_owned', etc.


!! $ARGS[1] = $...worntype
!!  ARGS[2] =  ...wornnumber
if $ARGS[0] = 'is_owned': -> result
	Function that returns whether Sveta owns the item
	Should return the internal representation of "truth" or "false"
		QSP 5.7.0- (truth = -1); QSP 5.8.0+ (truth = +1)


!! $ARGS[1] = $...worntype
!!  ARGS[2] =  ...wornnumber
!!  [ARGS[3] = item strength]
if $ARGS[0] = 'add_item':
	Adds the item to your wardrobe
	If ARGS[3] is passed, overrides the items strength (if applicable)


!! [$ARGS[1] = $...worntype]
!! [ ARGS[2] =  ...wornnumber]
if $ARGS[0] = 'remove_item':
	Removes the given item.
	Missing values are filled with currentworn values


if $ARGS[0] = 'dispose':
	Removes the currently worn item.


!! $ARGS[1] = $...worntype
!!  ARGS[2] =  ...wornnumber
OR
!! $ARGS[1] = 'last_worn'
if $ARGS[0] = 'wear':
	Wears the given item


if $ARGS[0] = 'strip':
	Removes the worn item

!}

	gs 'outfit', 'reset_derived_vars'

		if $braworntype <> 'none':
			gs 'bras', 'strip'
			!! "strip" will call "set_derived_vars" again
			exit
		end

	elseif PCloBra = 2:
		PXCloTopCut = 500
	end

	if PCloBimbo = 1 and PShoBimbo = 1: PBimbo = 1
	if PCloGoth = 1 and PShoGoth = 1: PGoth = 1
	if PCloPunk = 1 and PShoPunk = 1: PPunk = 1
	if PCloPrep = 1: PPrep = 1
	if PCloPrude = 1: PPrude = 1

	!! Sets PProstitute
	gs 'prostitution_functions', 'work_clothes'

	if PCloStyle2 = 1 or PCloMaid = 1: PMaid = 1
	if PCloStyle2 = 2 or PCloServer = 1: PServer = 1
	if (PCloStyle2 = 3 or PCloStrip = 1) and (PShoStyle2 = 1 or PShoStrip = 1): PStrip = 1
	if PCloStyle2 = 4 or PCloSchool = 1: PSchool = 1
	if PCloStyle2 = 5 or PCloOffice = 1: POffice = 1
	if (PCloStyle2 = 6 or PCloSport = 1) and (PShoStyle2 = 2 or PShoSport = 1): PSport = 1

	PSwim = PCloSwim
	!!{
	if $clothingworntype <> 'nude':
	elseif PPanSwim = 0 and $pantyWorntype <> 'none':
	elseif PBraSwim = 0 and $braworntype <> 'none':
	else
		PSwim = 1
	end
	!!}

	PCoverTop = max(0, PCloCoverTop + PBraCover - 4)
	PCoverFront = max(0, PCloCoverFront + PPanCoverFront - 4)
	PCoverBack = max(0, PCloCoverBack + PPanCoverBack - 4)


			!! --------------------DEPRECATED-------------------- !!

	!! Non-P variables are depracated and will be removed from 0.10 onwards
	CoverTop = PCoverTop
	CoverBack = PCoverBack
	CoverFront = PCoverFront
	exit
end

!{
	This compares the player heels skill to the skill required to wear any heels the player
	currently wears. If the skill needed is greater than the player currently possesses,
	the pain in the feet increases and the player earns heels skill. If the skill required
	is lower, the player might still get experience, depending on how much greater the
	player skill is.

	Variables
	pcs_heels is the main heels skill
	PShoHeels tracks the height of the heels currently worn
	PShoPain tracks the various brackets where Sveta feels pain from the heels
	heelstime tracks the elapsed time since last check
	heelsminut tracks when the heels skill was last changed

	Only run if at least X minutes have passed since last time


!}
if $ARGS[0] = 'heel_stat_skill_pain':
	!! fix for time cheat
	if totminut < heelsminut or heelsminut = 0: heelsminut = totminut

	if totminut < heelsminut + 30: exit

	temp_outfitVars['heels_time'] = totminut - heelsminut
	heelsminut = totminut
	if inSleep <> 0: exit

	!! Degrade heels skill when wearing flats/barefoot
	if PShoHeels <= 1:
		gs 'exp_deg', 'heels', rand(temp_outfitVars['heels_time'] / 30, temp_outfitVars['heels_time'] / 15)
	end

	!! ── Dependency pain (heel_preference > 0) ────────────────────────
	!! Dependent players feel pain wearing shoes below their dependency.
	!! Pain scales with how far below: flats hurt most, low heels less, mid least.
	!! Tier 3: pain from mid heels (mild), low heels (moderate), flats / barefoot (severe)
	!! Tier 2: pain from low heels (mild), flats / barefoot (moderate)
	!! Tier 1: pain from flats / barefoot (mild)
	temp_dep_pain = 0
	if trait_vars['heel_preference'] >= 3:
		if PShoHeels <= 1:
			if pain['feet'] < 80: gs 'pain', rand(heelstime / 7, heelstime / 2), 'feet', 'ache'
			temp_dep_pain = 1
		elseif PShoHeels = 2:
			if pain['feet'] < 60: gs 'pain', rand(heelstime / 15, heelstime / 5), 'feet', 'ache'
			temp_dep_pain = 1
		elseif PShoHeels = 3:
			if pain['feet'] < 40: gs 'pain', rand(heelstime / 30, heelstime / 10), 'feet', 'ache'
			temp_dep_pain = 1
		end

		if PShoHeels <= 1:
			if pain['feet'] < 60: gs 'pain', rand(heelstime / 15, heelstime / 5), 'feet', 'ache'
			temp_dep_pain = 1
		elseif PShoHeels = 2:
			if pain['feet'] < 40: gs 'pain', rand(heelstime / 30, heelstime / 10), 'feet', 'ache'
			temp_dep_pain = 1
		end

		if PShoHeels <= 1:
			if pain['feet'] < 40: gs 'pain', rand(heelstime / 30, heelstime / 10), 'feet', 'ache'
			temp_dep_pain = 1
		end
	end

	!! Dependency pain already handled this tick — skip normal pain
	if PShoHeels <= 1 or temp_dep_pain = 1:
		killvar 'temp_dep_pain'
		killvar 'temp_outfitVars'
		exit
	end
	killvar 'temp_dep_pain'

	if trait_vars['heel_preference'] < 0 and trait_vars['heel_preference_daily_limit'] > 0:
		temp_time = min(temp_outfitVars['heels_time'], trait_vars['heel_preference_daily_limit'])
		!! Gains are scaled x10 and banked in heel_preference_exp_rem
		if pcs_heels < PShoPain['severe']:
			trait_vars['heel_preference_exp_rem'] += 10 * temp_time

	!! Average dirt across worn non-immutable items; immutables have constant dirt (1440) so
	!! they are excluded. If every worn item is immutable, the icon shows fully clean (wash11).
	!! wash11 = spotless, wash0 = filthy. Each dirt point / 240 moves one step down the scale.
	if $stat_texts['outfit_clean'] <> '':
		if temp_outfitVars['wash_max'] = 0:
			$temp_outfitVars['clean_icon'] = 'wash11'
		else
			temp_outfitVars['wash_index'] = 11 - (11 * temp_outfitVars['wash_sum'] + temp_outfitVars['wash_max'] / 2) / temp_outfitVars['wash_max']
			temp_outfitVars['wash_index'] = min(max(0, temp_outfitVars['wash_index']), 11)
			$temp_outfitVars['clean_icon'] = 'wash' + $str(temp_outfitVars['wash_index'])
		end
	end

	if $stat_texts['clothes'] <> '' or $temp_outfitVars['clothes_style'] <> '':
		gs 'stat_display_compute', 'queue_msg', 'clothes', $stat_texts['clothes_color'], 'status/clothing/' + $temp_outfitVars['clothes_style'], 4
	end

	if $stat_texts['panties'] <> '':
		if $pantyworntype = 'none':
			gs 'stat_display_compute', 'queue_msg', 'panties', $stat_texts['panties_color'], 'status/clothing/commando_black|status/clothing/commando_white', 4
		elseif isprok = 0 and isprokp = 0 and mesec > 0 and preg = 0:
			gs 'stat_display_compute', 'queue_msg', 'panties', $stat_texts['panties_color'], 'status/clothing/panties_red', 4
		else
			gs 'stat_display_compute', 'queue_msg', 'panties', $stat_texts['panties_color'], 'status/clothing/panties', 4
		end
	end

	!! Pair clean+repair icons side-by-side in one cell (modes 0=both, 1=icon-only)
	!! Icons are narrow (~31px), so wrap in a min-width div to keep the table column aligned.
	if stat_cfg['msg_display_mode'] <> 2:
		$temp_outfitVars['clo_paired'] = ''
		if $temp_outfitVars['clean_icon'] <> '':
			$temp_outfitVars['clo_paired'] += '<a href="exec: msg $stat_texts[''outfit_clean'']"><img src="images/system/icons/status/clothing/' + $temp_outfitVars['clean_icon'] + '.png" title="<<$stat_texts[''outfit_clean'']>>" height="<<stat_cfg[''menu_icon_height'']>>"></a>'
		end
		if $temp_outfitVars['repair_icon'] <> '':
			$temp_outfitVars['clo_paired'] += '<a href="exec: msg $stat_texts[''outfit_repair'']"><img src="images/system/icons/status/clothing/' + $temp_outfitVars['repair_icon'] + '.png" title="<<$stat_texts[''outfit_repair'']>>" height="<<stat_cfg[''menu_icon_height'']>>"></a>'
		end
		if $temp_outfitVars['clo_paired'] <> '': $sd_icons_4[] = '<div style="min-width:<<stat_cfg[''menu_icon_height'']>>px;text-align:center;">' + $temp_outfitVars['clo_paired'] + '</div>'
	end

	killvar 'temp_outfitVars'
	exit
end
