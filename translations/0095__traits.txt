# traits

!! ╔════════════════════════════════════════════════════════════════════════╗
!! ║  Each trait is an $ARGS[0] function with $ARGS[1] sub-args:            ║
!! ║	gs 'traits', 'panty_preference', 'minute'                           ║
!! ║	gs 'traits', 'academic',     'daily'                                ║
!! ║	gs 'traits', 'exhibitionist', 'cheat', 2                            ║
!! ║                                                                        ║
!! ║  The $ARGS[0] function name = trait_vars key = registry ID.            ║
!! ║                                                                        ║
!! ║  Dispatchers aggregate per-trait calls:                                ║
!! ║	gs 'traits', 'init' - builds trait list                             ║
!! ║	gs 'traits', 'daily' - calls each trait''s ''daily'', _aliases      ║
!! ║	gs 'traits', 'minute' - calls each trait''s ''minute'',_aliases     ║
!! ║	gs 'traits', 'cheat', 'X', N - set trait X to level N               ║
!! ║	  (N = -99 for full reset)                                          ║
!! ║                                                                        ║
!! ║  Per-trait sub-functions:                                              ║
!! ║	''details'' — populate $trait_temp / trait_temp with:               ║
!! ║	  $trait_temp[''name'']        — display name                       ║
!! ║	  $trait_temp[''icon'']        — icon path (in images/)             ║
!! ║	  $trait_temp[''desc'']        — description (level-aware)          ║
!! ║	  trait_temp[''hidden'']       — 1 = hide when inactive             ║
!! ║	  $trait_temp[''up_msg, N'']   — level-up notification              ║
!! ║	  $trait_temp[''down_msg, N''] — level-down notification            ║
!! ║	''daily'' / ''minute'' / ''hourly'' — periodic logic                ║
!! ║	''cheat'', N — set level N with stable exp (N=-99 = reset)          ║
!! ║                                                                        ║
!! ║  Shared utilities:                                                     ║
!! ║	gs 'traits', 'level', 'X', val — set level, notify on change        ║
!! ║	gs 'traits', 'register_attskl', 'X' — indicate that this trait      ║
!! ║                                          affects exp gain and deg loss ║
!! ║	gs 'traits', 'deregister_attskl', 'X' — it no longer affects        ║
!! ║                                            exp gain and deg loss       ║
!! ║	$func('traits', 'card', 'X', lvl) — HTML card for one trait         ║
!! ║	$func('traits', 'show_all')  — HTML cards for all traits            ║
!! ║                                                                        ║
!! ║  Registry (built by init):                                             ║
!! ║	$trait_vars['list, <i>'] — trait ID                                 ║
!! ║                                                                        ║
!! ║  Storage:                                                              ║
!! ║	trait_vars['X']             — trait level (integer)                 ║
!! ║	trait_vars['X_exp']         — trait experience                      ║
!! ║                                                                        ║
!! ║  EXP / DEG impacts:                                                    ║
!! ║    traits_vars['X-attrskl-exp_gain'] = N: — Privide an additional      ║
!! ║              N% exp boost to an attribute or skill with the given name ║
!! ║              N can be negative for an exp penalty                      ║
!! ║    traits_vars['X-attrskl-deg_loss'] = N: — Privide an additional      ║
!! ║              N% deg loss to an attribute or skill with the given name  ║
!! ║              N can be negative for a slower decay                      ║
!! ║    Register the skill using the register function                      ║
!! ║    Deregister the skill when it becomes inactve                        ║
!! ╚════════════════════════════════════════════════════════════════════════╝

!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ minute — called from stat, limited to 1 per minute                     │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'minute':
	!! ── Stale-clock gate ──────────────────────────────────────────────
	!! totminut is computed by gs 'time' which runs late in stat.qsrc.
	!! On the very first gs 'stat' call it may still be stale (-1 from
	!! start.qsrc).  Sync to one minute earlier so the next loop runs
	if totminut <= 0:
		trait_vars['last_update'] = (minut + hour * 60 + daystart * 1440) - 1
		exit
	end
	!! ── Sleep gate ────────────────────────────────────────────────────
	!! During sleep totminut advances but nothing should change.
	!! Sync the clock so the first waking tick starts fresh.
	if inSleep <> 0:
		trait_vars['last_update'] = totminut
		exit
	end

	!! ── Timing gate ───────────────────────────────────────────────────
	temp_loop_mul = totminut - trait_vars['last_update']
	if temp_loop_mul = 0: killvar 'temp_loop_mul' & exit
	trait_vars['last_update'] = totminut

	traits_i = 0
	:traits_minute_loop
	if $trait_vars['list, <<traits_i>>'] ! '':
		gs 'traits', $trait_vars['list, <<traits_i>>'], 'minute'
		traits_i += 1
		jump 'traits_minute_loop'
	end
	killvar 'traits_i'
	exit
end

!! ┌────────────────────────────────────────────────────────────────────────┐

if $ARGS[0] = 'level':
	if ARGS[2] <> trait_vars[$ARGS[1]]:
		$temp_level['dir'] = iif(ARGS[2] > trait_vars[$ARGS[1]], 'up', 'down')

		gs 'traits', '_notify', $ARGS[1], $temp_level['dir'], ARGS[2]
		gs 'traits', $ARGS[1], 'level', ARGS[2]
		killvar 'temp_level'
	end
	exit

!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ register_attskl($trait) — Register the trait to have an exp effect     │
!! └────────────────────────────────────────────────────────────────────────┘

	if arrpos('$traitattskl', $ARGS[1]) < 0: $traitattskl[] = $ARGS[1]
	exit
end


!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ deregister_attskl($trait) — Deregister the trait                       │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'deregister_attskl':
	if arrpos('$traitattskl', $ARGS[1]) >= 0:
		gs 'array', 'remove_element', '$traitattskl', $ARGS[1]
	end
	exit
end


!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ _notify($trait, $dir, level) — show level-change notification          │
!! │ Calls 'details' to get $trait_temp['up_msg, N'] etc.                   │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = '_notify':
	if opPRE > 0: exit
	gs 'traits', 'get_details', $ARGS[1]
	!! Prefer directional message (msg-up / msg-down), fall back to generic msg
	if $trait_temp['msg-<<$ARGS[2]>>'] <> '':
		$temp_notify = $trait_temp['msg-<<$ARGS[2]>>']
	else
		$temp_notify = $trait_temp['msg']
	end
	if $temp_notify <> '': msg $temp_notify
	killvar 'temp_notify'
	exit
end

!! │ Calls 'details' for metadata.                                          │

if $ARGS[0] = 'card':

	$temp_card['desc'] = $trait_temp['desc']
	$temp_card['name'] = $trait_temp['name']
	$temp_card['prog'] = ''
	if $trait_temp['tooltip'] <> '':
		if trait_temp['exp_down'] = -1:

		elseif trait_temp['exp_down'] >= 1000 or trait_temp['exp_down'] <= -1000:
			$temp_pw = '↓&nbsp' + $str(trait_temp['exp_down'] / 1000) + 'k' + '&nbsp|&nbsp'

		elseif trait_temp['exp_up'] >= 1000 or trait_temp['exp_up'] <= -1000:

	end

	if trait_temp['hidden'] = 1 and ARGS[2] = 0 and trait_vars[$ARGS[1] + '_discovered'] = 0 and cheatVars['show_hidden_traits'] = 0:

	killvar 'temp_bcolor'

!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ show_all — return HTML for all trait cards, sorted:                    │
!! │   1. Active traits (level > 0)         — registry order                │
!! │   2. Inactive visible (not hidden)    — registry order                 │
!! │   3. Hidden traits (hidden + inactive) — registry order                │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'show_all':
	!! ── Bucket traits by state ────────────────────────────────────────
	'<center><font size="2" color="#888888">将鼠标悬停在卡片右上角的EXP控件上，即可了解如何获得或失去经验值。</font></center>'
	temp_show_all['i'] = 0
	:show_all_loop
	if temp_show_all['i'] < trait_vars['count']:
		killvar 'trait_temp'
		$temp_show_all['trait'] = $trait_vars['list, ' + str(temp_show_all['i'])]
		gs 'traits', $temp_show_all['trait'], 'details'
		if trait_vars[$temp_show_all['trait']] <> 0:
			$temp_active[] = $temp_show_all['trait']
		elseif trait_temp['hidden'] = 0 or trait_vars[$temp_show_all['trait'] + '_discovered'] = 1:
			$temp_visible[] = $temp_show_all['trait']
		else
			$temp_hidden[] = $temp_show_all['trait']
		end
		killvar 'trait_temp'
		temp_show_all['i'] += 1
		jump 'show_all_loop'
	end

	!! ── Render cards ──────────────────────────────────────────────────
	$result = ''
	temp_show_all['i'] = 0
	:sa_render_act
	if temp_show_all['i'] < arrsize('temp_active'):
		$result += $func('traits', 'card', $temp_active[temp_show_all['i']], trait_vars[$temp_active[temp_show_all['i']]])
		temp_show_all['i'] += 1
		jump 'sa_render_act'
	end
	temp_show_all['i'] = 0
	:sa_render_vis
	if temp_show_all['i'] < arrsize('temp_visible'):
		$result += $func('traits', 'card', $temp_visible[temp_show_all['i']], 0)
		temp_show_all['i'] += 1
		jump 'sa_render_vis'
	end
	temp_show_all['i'] = 0
	:sa_render_hid
	if temp_show_all['i'] < arrsize('temp_hidden'):
		$result += $func('traits', 'card', $temp_hidden[temp_show_all['i']], 0)
		temp_show_all['i'] += 1
		jump 'sa_render_hid'
	end

	killvar 'temp_active' & killvar 'temp_visible'
	killvar 'temp_hidden' & killvar 'temp_show_all'
	exit
end

!! ║       $ARGS[0] = trait name                                            ║
!! ║       $ARGS[1] = initlevel||daily|hourly|minute|reset                  ║

!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ cumeater — binary, EXP from known swallowing                           │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'cumeater':
	if $ARGS[1] = 'daily':
		if cheatVars['cumeater_disabled']: exit
		if trait_vars['cumeater'] = 0:
			trait_vars['cumeater_exp'] += (stat['swallow'] - stat['swallow_unaware']) - trait_vars['cumeater_prev']
			trait_vars['cumeater_prev'] = stat['swallow'] - stat['swallow_unaware']
			if trait_vars['cumeater_exp'] >= 50: gs 'traits', 'level', 'cumeater', 1
		end
	end

	if $ARGS[1] = 'details':
		$trait_temp['name'] = '吞精者'
		$trait_temp['icon'] = 'cumeater.png'
		trait_temp['hidden'] = 0
		if trait_vars['cumeater'] = 0:
			$trait_temp['desc'] = '你对精液并没有特别的喜好。'
			$trait_temp['msg'] = '你失去了特质<b>吞精者</b>。'
			trait_temp['exp'] = trait_vars['cumeater_exp']
			trait_temp['exp_down'] = -1
			trait_temp['exp_up'] = 50
			$trait_temp['tooltip'] = '你有<<trait_vars[''cumeater_exp'']>>经验值。通过有意吞下精液来获得经验值。'
		elseif trait_vars['cumeater'] = 1:
			$trait_temp['desc'] = '经验让你逐渐喜欢上精液的味道。吞精变得自然而然，处理精液的新可能性也随之打开。'
			$trait_temp['msg'] = '你获得了特质<b>吞精者</b>。'
			trait_temp['exp'] = trait_vars['cumeater_exp']
			trait_temp['exp_down'] = -1
			trait_temp['exp_up'] = -1
			$trait_temp['tooltip'] = '一生吞精总量： <<trait_vars[''cumeater_exp'']>>。'
		end
	end

	if $ARGS[1] = 'cheat':
		if ARGS[2] = -99 or ARGS[2] = 0:
			trait_vars['cumeater_exp'] = 0
			gs 'traits', 'level', 'cumeater', 0
		elseif ARGS[2] = 1:
			trait_vars['cumeater_exp'] = 50
			gs 'traits', 'level', 'cumeater', 1
		end
	end

	exit
end

!! │ cum_addict — leveled 0-2, hypno-only                                   │
!! │ Absorbs hypnoAddict. Player craves cum inside her (any source).        │
!! │ 0 = Normal: no dependency                                              │
!! │ 1 = Cum Craving: restless without cum; minimum arousal floor 30        │
!! │ 2 = Cum Addicted: strong penalties; minimum arousal floor 50           │
!! │ missCum counts hours without cum. timeTresh in same unit.              │
!! │ Minute sensor detects cum; hourly handler processes craving.           │
!! │ hypnoWithdrawal is separate (therapist session dependency).            │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'cum_addict':

	!! ── Minute sensor: detect cum presence ────────────────────────────
	if $ARGS[1] = 'minute':
		if trait_vars['cum_addict'] = 0: exit
		if cumloc[0] > 0 or cumloc[3] > 0:
			trait_vars['cum_addict_sensed'] = 1
		end
	end

	!! ── Hourly: cum craving / satisfaction ────────────────────────────
	if $ARGS[1] = 'hourly':
		if trait_vars['cum_addict'] = 0: exit

		!! Cum was present at some point this hour → satisfaction
		if trait_vars['cum_addict_sensed'] = 1:
			killvar 'missCum'
			trait_vars['cum_addict_sensed'] = 0
			gs 'mood', 'raise', 'tiny'
		else
			missCum += 1
		end

		!! Craving penalty when too long without cum
		if missCum > timeTresh:
			if trait_vars['cum_addict'] = 1:
				gs 'mood', 'lower', 'tiny'
				pcs_horny += 10
			else
				gs 'mood', 'lower', 'small'
				pcs_horny += 20
			end
		end
	end

	!! ── Daily: arousal floor + condom sabotage ────────────────────────
	if $ARGS[1] = 'daily':
		if trait_vars['cum_addict'] = 1:
			min_arousal = max(min_arousal, 30)
		elseif trait_vars['cum_addict'] = 2:
			min_arousal = max(min_arousal, 50)
		end

		!! Desperation: force high arousal + sabotage condoms
		if trait_vars['cum_addict'] > 0 and missCum > timeTresh:
			pcs_horny = max(pcs_horny, 90)
			if preziktype = 0 or preziktype = 2:
				if mc_inventory['normal_condoms'] > 0 and mc_inventory['sabotaged_condoms'] < rand(2, 3): gs 'din_bad', 'sabotage_a_condom', 'no_text'
				if preziktype = 0 and mc_inventory['sabotaged_condoms'] > 0: gs 'din_bad', 'set_condom_used', 'sabotaged'
			end
		end
	end

	if $ARGS[1] = 'details':
		$trait_temp['icon'] = 'cum_addict.png'
		trait_temp['hidden'] = 1
		if trait_vars['cum_addict'] = 1:
			$trait_temp['name'] = '渴望精液'
			$trait_temp['desc'] = '体内有精液时你会感觉更好。太久没有它会让你焦躁不安、心情低落。你总是至少保持一定程度的欲望（最低30）。'
			$trait_temp['msg'] = '你已经形成了<b>渴望精液</b>的状况。没有它你会觉得不完整。'
		elseif trait_vars['cum_addict'] = 2:
			$trait_temp['name'] = '精液成瘾'
			$trait_temp['desc'] = '你的身体需要精液。没有它，你几乎无法正常思考——你的心情跌入谷底，身体渴求着它。你总是处于欲望之中（最低50）。'
			$trait_temp['msg'] = '你已经变得<b>精液成瘾</b>。你极度需要它。'
		elseif trait_vars['cum_addict'] = 0:
			$trait_temp['name'] = '精液依赖'
			$trait_temp['desc'] = '你对精液没有特别的渴望。'
			$trait_temp['msg'] = '你的精液依赖已经消退。'
		end
	end

	if $ARGS[1] = 'hypno_grant':
		!! Called from therapist.qsrc when hypnoTime crosses thresholds
		!! ARGS[2] = target level (1 or 2)
		gs 'traits', 'level', 'cum_addict', ARGS[2]
	end

	if $ARGS[1] = 'cheat':
		if ARGS[2] = -99 or ARGS[2] = 0:
			gs 'traits', 'level', 'cum_addict', 0
			killvar 'missCum'
		elseif ARGS[2] = 1 or ARGS[2] = 2:
			gs 'traits', 'level', 'cum_addict', ARGS[2]
		end
	end

	exit
end
