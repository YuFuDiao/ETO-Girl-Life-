!! gs 'stat_funcs', 'get_xpprv', stat_lvl
if $ARGS[0] = 'get_xpprv':
	if ARGS[1] = 0:
		result = 0
	else
		result = 1 + (146 * (ARGS[1] - 1) * (ARGS[1] - 1) / 91)
	end
	exit
end


!! gs 'stat_funcs', 'get_xpnxt', stat_lvl
if $ARGS[0] = 'get_xpnxt':
	result = 1 + (146 * ARGS[1] * ARGS[1] / 91)
	exit
end


!! gs 'stat_funcs', 'get_max_deg', stat_lvl, stat_muta
if $ARGS[0] = 'get_max_deg':
	result = max(200, (102 - min(ARGS[1], 100) + ARGS[2]) * 100)
	exit
end


!! gs 'stat_funcs', 'add_attr_to_list', att_name, att_desc
if $ARGS[0] = 'add_att_to_list':
	if arrpos('$att_name', $ARGS[1]) < 0:
		$att_name[] = $ARGS[1]
	end
	if $ARGS[2] <> '': $att_desc[$ARGS[1]] = $ARGS[2]
	exit
end


!! gs 'stat_funcs', 'add_skl_to_list', skl_name, skl_desc

		$skl_name[] = $ARGS[1]
	end
	if $ARGS[2] <> '': $skl_desc[$ARGS[1]] = $ARGS[2]
	exit
end


!! gs 'stat_funcs', 'add_trait_to_list', skl_name
