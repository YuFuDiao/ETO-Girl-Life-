# KGOLgame

if $ARGS[0] = '':
	minut += 1
	gs 'stat'

	if $KGOLset = '': $KGOLset = '你头上什么也没戴'
	if $KGOLset2 = '': $KGOLset2 = '你穿着简单的衣服'
	if $KGOLset3 = '': $KGOLset3 = '你穿着简单的鞋子'
	if $KGOLset4 = '': $KGOLset4 = '你手上没有戴手套'
	if $KGOLset5 = '': $KGOLset5 = '你没有装备武器'
	KGHPMAX = KGOLvital * 10
	KGManaMax = KGOLintel * 10

	if KGHP > KGHPMAX: KGHP = KGHPMAX
	if KGMana > KGManaMax: KGMana = KGManaMax

	KGOLbrona -= KGOLbronaMod
	KGOLbronaMod = 0
	KGOLstren -= KGOLstrenMod
	KGOLstrenMod = 0
	KGOLspeed -= KGOLspeedMod
	KGOLspeedMod = 0
	KGOLvital -= KGOLvitalMod
	KGOLvitalMod = 0
	KGOLintel -= KGOLintelMod
	KGOLintelMod = 0
	KGOLwpower -= KGOLwpowerMod
	KGOLwpowerMod = 0
	KGOLcooldown = 0

	if KGHP < KGHPMAX and KGOLhirka > 0:
		kghneed = KGHPMAX - KGHP

		if KGOLhirka >= kghneed:
			KGOLhirka -= kghneed
			KGHP = KGHPMAX
		else
			KGHP = KGOLhirka
			KGOLhirka = 0
		end
	end

	if KGHP < 0: KGHP = 0

	gs 'KGOLexpa'
	'<a href="exec: gt ''KGstart''">退出游戏</a>'
	'<a href="exec: gt ''KGOLgame'', ''KGOLstat''">查看统计数据</a>'

	'<<$KGOLname>>, 等级<<KGOLlvl>> <<$KGOLrace>> <<$KGOLklass>>'
	'生命：<<KGHP>>'
	'法力：<<KGMana>>'

	if KGHP < KGHPMAX: '你的生命值还剩<<KGHP*100/KGHPMAX>>%。'
	if KGMana < KGManaMax: '你的法力值还剩<<KGMana*100/KGManaMax>>%。'
	if KGOLhirka > 0: '希尔卡会自动恢复<<KGOLhirka>>点生命值。'
	if KGOLPrizeM > 0: '你有<<KGOLmoney>>枚金币。'
	if KGOLneedExp <= 0: '<a href="exec:gs ''KGOLexpa'', ''KGOLrise''">' + $func('wrap', 'neg', '通过获得经验值来提升你的属性。') + '</a>'
	if KGOLpoint > 0: '<a href="exec:gs ''KGOLexpa'', ''KGOLrise''">' + $func('wrap', 'neg', '你有尚未分配的升级点数。') + '</a>'
	if KGOLspellpoint >= 4: '<a href="exec:gs ''KGOLexpa'', ''KGOLrise2''">' + $func('wrap', 'neg', '你有可以使用的技能点。') + '</a>'

	'<<KGOLspellpoint>>'
	''
	'<center><b>乡间</b></center>'
	'你和其他玩家角色一起待在一个风景如画的村庄里。一个NPC正站在几根柱子旁。'

	act '走近NPC':
		cla
		minut += 1

		if KGOLmqw = 0:
			KGOLmqw = 1
			'(NPC) - "<<$KGOLname>>，我需要你去猎杀5朵捕食花。"'
			'奖励：武器（职业限定）'
		elseif KGOLmqw = 1 and KGOLqwvip >= 5:
			KGOLqwvip = 0
			KGOLmqw = 2
			KGOLwpower = 10

			if $KGOLklass = 'Warrior' or $KGOLklass = 'Barbarian':
				$KGOLset5 = '你装备了一把简单的剑'
			elseif $KGOLklass = 'Rogue':
				$KGOLset5 = '你装备了一套匕首'
			elseif $KGOLklass = 'Magician' or $KGOLklass = 'Priest':
				$KGOLset5 = '你装备了一根简单的法杖'
			elseif $KGOLklass = 'Archer':
				$KGOLset5 = '你装备了一把简单的弓'
			end
			'(NPC) - "谢谢你，<<$KGOLname>>！这是你的武器。"'
		elseif KGOLmqw = 2:
			KGOLmqw = 3
			'(NPC) - "<<$KGOLname>>，我需要你去杀死10只彩虹甲虫。"'
			'奖励：皮甲'
		elseif KGOLmqw = 3 and KGOLqwvip >= 10:
			KGOLqwvip = 0
			KGOLmqw = 4
			KGOLbrona += 10
			$KGOLset2 = '你穿着简单的皮甲'
			'(NPC) - "谢谢你，<<$KGOLname>>！这是你的护甲。"'
		elseif KGOLmqw = 4:
			KGOLmqw = 5
			'(NPC) - "<<$KGOLname>>，我需要你去杀死20株茅膏菜。"'
			'奖励：皮靴'
		elseif KGOLmqw = 5 and KGOLqwvip >= 20:
			KGOLqwvip = 0
			KGOLmqw = 6
			KGOLbrona += 2
			$KGOLset3 = '你穿着一双结实的皮靴'
			'(NPC) - "谢谢你，<<$KGOLname>>！这是你的靴子。"'
		elseif KGOLmqw = 6:
			KGOLmqw = 7
			'(NPC) - "<<$KGOLname>>，我需要你去杀死25只无牙狗。"'
			'奖励：皮革护手'
		elseif KGOLmqw = 7 and KGOLqwvip >= 25:
			KGOLqwvip = 0
			KGOLmqw = 8
			KGOLbrona += 2
			$KGOLset4 = '你穿着一双皮革护手'
			'(NPC) - "谢谢你，<<$KGOLname>>！这是你的护手。"'
		elseif KGOLmqw = 8:
			KGOLmqw = 9
			'(NPC) - "<<$KGOLname>>，我需要你去杀死30株小仙人掌。"'
			'奖励：皮头盔'
		elseif KGOLmqw = 9 and KGOLqwvip >= 30:
			KGOLqwvip = 0
			KGOLmqw = 10
			KGOLbrona += 2
			$KGOLset = '你戴着一顶皮头盔'
			'(NPC) - "谢谢你，<<$KGOLname>>！这是你的头盔。"'
		elseif KGOLmqw = 10 and KGOLlvl >= 5:
			KGOLmqw = 11
			'(NPC) - "<<$KGOLname>>，我需要你去杀死10只蟾蜍。"'
			'奖励：100点经验值'
		elseif KGOLmqw = 11 and KGOLqwvip >= 10:
			KGOLqwvip = 0
			KGOLmqw = 12
			KGOLexp += 100
			'(NPC) - "谢谢你，<<$KGOLname>>！这是你的奖励！"'
		elseif KGOLmqw = 12 and KGOLlvl >= 5:
			KGOLmqw = 13
			'(NPC) - "<<$KGOLname>>，我需要你去杀死10只皇家甲虫。"'
			'奖励：200点经验值'
		elseif KGOLmqw = 13 and KGOLqwvip >= 10:
			KGOLqwvip = 0
			KGOLmqw = 14
			KGOLexp += 200
			'(NPC) - "谢谢你，<<$KGOLname>>！这是你的奖励！"'
		end

		act '离开': gt 'KGOLgame'
	end

	act '去商店':
		cla
		minut += 1

		if KGOLmoney >= 300:
			act '购买生命药水 (300金币）':
				KGOLmoney -= 300
				KGOLpotionH += 1
				gt 'KGOLgame'
			end
		end

		if KGOLmoney >= 300:
			act '购买法力药水 (300金币）':
				KGOLmoney -= 300
				KGOLpotionM += 1
				gt 'KGOLgame'
			end
		end

		if KGOLmoney >= 1500 and KGOLhirka <= 0:
			act '购买再生药水 (1500金币）':
				KGOLmoney -= 1500
				KGOLhirka += 10000
				gt 'KGOLgame'
			end
		end

		act '离开': gt 'KGOLgame'
	end

	if KGHP < KGHPMAX or KGMana < KGManaMax:
		act '恢复 (1:00)':
			cla
			minut += 60
			KGHP = KGHPMAX
			KGMana = KGManaMax
			'你坐了一个小时，等待生命值和法力值恢复。虽然很无聊，但至少是免费的。'

			act '离开': gt 'KGOLgame'
		end
	end

	if KGHP > 0:
		act '在村庄附近打怪':
			cla
			minut += 1
			KGOLdist = rand(1, 10)
			'村庄附近的田野里游荡着各种各样的怪物。'

			act '捕食花':
				gs 'KGOLenemy', '0'
				gt 'KGOLfight'
			end

			act '彩虹甲虫':
				gs 'KGOLenemy', '1'
				gt 'KGOLfight'
			end

			act '茅膏菜':
				gs 'KGOLenemy', '2'
				gt 'KGOLfight'
			end

			act '无牙狗':
				gs 'KGOLenemy', '3'
				gt 'KGOLfight'
			end

			act '小仙人掌':
				gs 'KGOLenemy', '4'
				gt 'KGOLfight'
			end
		end
	end

	if KGOLlvl >= 5 and KGHP > 0:
		act '在磨坊附近打怪':
			cla
			minut += 1
			KGOLdist = rand(1, 10)
			'磨坊附近的田野里游荡着各种各样的怪物。'

			act '蟾蜍':
				gs 'KGOLenemy', '5'
				gt 'KGOLfight'
			end

			act '皇家甲虫':
				gs 'KGOLenemy', '6'
				gt 'KGOLfight'
			end

			act '斑点螳螂':
				gs 'KGOLenemy', '7'
				gt 'KGOLfight'
			end

			act '瘟疫鼠':
				gs 'KGOLenemy', '8'
				gt 'KGOLfight'
			end

			act '多刺仙人掌':
				gs 'KGOLenemy', '9'
				gt 'KGOLfight'
			end
		end

		act '5级首领':
			gs 'KGOLboss', '1'
			gt 'KGOLfight'
		end
	end
end


if $ARGS[0] = 'KGOLstat':
	cla
	'<<$KGOLname>>, 等级<<KGOLlvl>> <<$KGOLrace>> <<$KGOLklass>>'
	'<a href="exec: gt ''KGOLgame'', ''KGOLstat2''">关闭统计</a>'

	if KGOLneedExp > 0:
		pl '总经验值：<<KGOLexp>>升级所需经验：<<KGOLneedExp>>'
	else
		pl '<a href="exec:gs ''KGOLexpa'', ''KGOLrise''">' + $func('wrap', 'neg', '通过获得经验值来提升你的属性。') + '</a>'
	end

	'力量：<<KGOLstren>>'
	'速度：<<KGOLspeed>>'
	'耐力：<<KGOLvital>>'
	'智力：<<KGOLintel>>'
	'护甲防护：<<KGOLbrona>>'
	'武器伤害：<<KGOLwpower>>'
	''
	'<center>装备</center>'
	'<<$KGOLset>>'
	'<<$KGOLset2>>'
	'<<$KGOLset3>>'
	'<<$KGOLset4>>'
	'<<$KGOLset5>>'
	''
	'<center>补给品</center>'

	if KGOLpotionH > 0: pl '你有<<KGOLpotionH>>瓶生命药水。 <a href="exec:KGOLpotionH -= 1 & KGHP += 100 & gs ''stat'' & gt $curloc">' + $func('wrap', 'neg', '喝一瓶') + '</a>'
	if KGOLpotionM > 0: pl '你有<<KGOLpotionM>>瓶法力药水。 <a href="exec:KGOLpotionM -= 1 & KGMana += 100 & gs ''stat'' & gt $curloc">' + $func('wrap', 'neg', '喝一瓶') + '</a>'

	'<center>PVE</center>'
	'你已经赢了<<KGOLWin>>次，输了<<KGOLLoss>>次。'
	'<center>PVP</center>'
	'你已经赢了<<KGOLpvpWin>>次，输了<<KGOLpvpLoss>>次。'
end


if $ARGS[0] = 'KGOLstat2':
	gs 'stat'
	gt 'KGOLgame'
end
