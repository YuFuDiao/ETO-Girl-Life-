!!	Birth Control Management variables
!!
!!	tabletkimm		: The modifier for the current mood of the MC.
!!	tabletkiap		: Autopill factor affecting the liklihood you took your pill as you should have
!!	tabletkipt		: Pill Dicipline Take increase
!!	tabletkipi		: Pill Dicipline Dont take increase
!!	tabletkipd		: Current MC Pill Dicipline
!!	tabletkichday	: Current day that the Birth Control is being evaluated for
!!	tabletkicheck	: If the birth control from shot (2) or pill (1)
!!	mc_inventory['contraceptive_pill']	: How many packages of Birth Control Pills there are
!!	tabletkiold		: Total number packages of expired pills
!!	tabletkifake	: Total number packages of fake/counterfeit pills
!!	tabletkirej		: Total number packages of factory rejected pills
!!	tabletkifert	: Total number packages of subsituted fertility pills
!!	trait_vars['fertility_bc_conditioning']	: 0–50 counter; rises +1 per fertility pill taken, decays -2/day missed, -4/day on real BC
!!	tabletkisug		: Total number packages of sugar pills
!!	pillsleft		: Total number packages of pills in current package
!!	tabletkishot	: The flag for being on the Birth Control shot.
!!	pillcon			: The number for the current birth control chemical level.
!!	pilldays		: The number of consecutive days (you believe) you have taken your pill

if $ARGS[0] = 'auto_pill':
	if daystart <= tabletkichday: exit

	tabletkichday = daystart

	tabletkipd += tabletkipi
	tabletkipi = 0
	tabletkipt = 0

	if tabletkicheck = 1:
		$temp_img = '<img height = <<stat_cfg[''icon_height'']>> src="images/system/icons/status/bc_pill.png"><br>'
		if mc_inventory['contraceptive_pill'] = 0 and pillsleft[ptype] = 0:
			gs 'notification', 'add', '<<$temp_img>>你的避孕药吃完了。'
			tabletkipi = -5
			jump 'auto_pill_break_goto'
		elseif mc_inventory['contraceptive_pill'] = 0 and pillsleft[ptype] = 28:
			gs 'notification', 'add', '<<$temp_img>>你刚打开了最后一盒避孕药。你需要再买一些。'
		elseif mc_inventory['contraceptive_pill'] = 0 and pillsleft[ptype] <= 5:
			gs 'notification', 'add', '<<$temp_img>>你的避孕药快吃完了，只剩几片了。'
		end

		birth_control['using_bc'] = 3

		tabletkimm = (100 * (pcs_willpwr / willpowermax)) + pcs_mood
		if pcs_sleep < 10:
			tabletkimm -= (100 - (pcs_sleep * 10))
		end
		if pcs_willpwr < willpowermax / 4:
			tabletkimm = tabletkimm / 4
		elseif pcs_mood < 10:
			tabletkimm = tabletkimm / 2
		end
		tabletkiap = tabletkimm
		tabletkiap += (pcs_intel * 4) + (pcs_sprt * 2) + (age * 3)
		tabletkiap -= ((kid * 30) + (abort * 2) - (tabletkipd * ((60 - age) / 10)))
		temprand = rand(0, (200 + (tabletkimm - ((100 - pcs_sprt) * 3))))
		tabletkiap += temprand
		if tabletkiap < 300:
			if tabletkipd > -10:
				if pcs_willpwr < willpowermax / 4:
					!! msg 'You stare at your package of birth control pills, you know you should take one but you can''t quite work up enough effort to care.'
					gs 'notification', 'add', '<<$temp_img>>你盯着那盒避孕药，你知道自己该吃一片，但实在提不起劲去在意。'
				elseif pcs_mood < 10:
					!! msg 'You stare at your package of birth control pills, but you don''t open it.'
					gs 'notification', 'add', '<<$temp_img>>你盯着那盒避孕药，却没有打开它。'
				elseif pcs_sleep < 10:
					!! msg 'You fumble with your birth control pills, trying to get a pill out. You finally give up too tired to manage it.'
					gs 'notification', 'add', '<<$temp_img>>你笨手笨脚地摆弄避孕药，想把药片取出来。你最终放弃了，太累了，实在做不到。'
				else
					!! msg 'You just realized you forgot to take your birth control pill today.'
					gs 'notification', 'add', '<<$temp_img>>你刚刚意识到今天忘了吃避孕药。'
				end
			end
			tabletkipt = 2
			tabletkipi = -4
		else
			if tabletkiap < 350:
				temprand = rand(0, 5)
			else
				temprand = rand(0, 1000)
			end
			if temprand = 0:
				temprand = rand(0, 1)
				if temprand = 0:
					gs 'din_bad', 'takepill'
					tabletkipt = -2
					tabletkipi = 1
				else
					tabletkipt = 3
					tabletkipi = -3
				end

				if tabletkipd > -10:
					gs 'notification', 'add', '<<$temp_img>>你发现自己拿着那瓶避孕药，却想不起来今天到底吃没吃。'
				end
				if tabletkipd > 0:
					if rand(0, 9) = 0:
						if tabletkipt < 0:
							gs 'notification', 'add', '<<$temp_img>>你觉得你今天还没吃药。'
						else
							pilldaychk = daystart
							gs 'notification', 'add', '<<$temp_img>>你觉得你今天已经吃过药了。'
						end
					else
						if tabletkipt > 0:
							gs 'notification', 'add', '<<$temp_img>>你觉得你今天还没吃药。'
						else

				tabletkipt = 0
				tabletkipi = 0
			end

	end

		if pilldaychk = daystart:

			birth_control['using_bc'] = 3
		else
			pilldays -= 1
			if placebopart <= 0: birth_control['using_bc'] -= 1
		end
	end

	if birth_control['using_bc'] < 0: pilldays = 0

	:auto_pill_break_goto

if $ARGS[0] = 'daily_update':

	if tabletkicheck = 2:
		if tabletkishot ! 0:
			!! cycle = 1

		elseif shotdays > 10 and shotdays < 14:
			msg '你很快就要打避孕针了，如果不去打，你可能会怀孕。'
		elseif shotdays = 10:
			msg '你的避孕针明天就需要更新了。'
		end
	end
	!! dropped the decrement out of the above if-statement so that players on pill would still run out of BC shot
	!! as in for example the case of the therapist forcing you on the pill while you already had the shot.
	if tabletkishot > 0: tabletkishot -= 1
	if shotdays > 0: shotdays -= 1

	!! ----- processing BC spell -----
	!! birth_control['implant_status']
	!! 0: No implant
	!! 3: Implant (more than 30 days left)
	!! 2: Implant (less than 30 days left)
	!! 1: Empty implant
	if $spellTarget['birth_control'] = 'self':
		pillcon = 40000
		pillcon2 = 40000
		spellTime['birth_control'] -= 1
		if spellTime['birth_control'] > 0 and spellTime['birth_control'] < 30:
			birth_control['implant_status'] = 2
			msg '你的避孕法术只能持续<<spellTime[''birth_control'']>>天' + iif(spellTime['birth_control'] = 1, '。', '。')
		elseif spellTime['birth_control'] < 1:
			$spellTarget['birth_control'] = ''
			pillcon = 0
			pillcon2 = 0
			msg '你的避孕法术已经结束。如果你不想怀孕，就需要重新施放。'
		end
	!! ----- processing BC implant -----
	elseif birth_control['implant_status'] > 1:
		pcs_skin += rand(0, 1)

		pillcon += ((pillcon + 15000) / 25) + 4000
		if pillcon > 40000: pillcon = 40000

		pillcon2 += ((pillcon2 + 15000) / 25) + 4000
		if pillcon2 > 40000: pillcon2 = 40000

		birth_control['implant_timer'] -= 1
		if (birth_control['implant_timer'] - 1090) > 0:
			gs 'pain', '<<(birth_control[''implant_timer''] - 1090)*2>>', 'armL', 'ache'
		elseif birth_control['implant_timer'] > 0 and birth_control['implant_timer'] < 30:
			birth_control['implant_status'] = 2
			'你的避孕植入物只能再维持<<birth_control[''implant_timer'']>> ' + iif(birth_control['implant_timer'] = 1, '天。', '天。')
		elseif birth_control['implant_timer'] < 0:
			birth_control['implant_status'] = 1
			msg '你应该更换或取出你的避孕埋植剂。'
		end
	elseif birth_control['implant_status'] = 1:
		msg '你应该更换或取出你的避孕埋植剂。'
	end

	!! ----- cycle tracking, tracks last 60 days -----
	$MenCal[] = $str(cycle)
	if arrsize('$MenCal') = 60: killvar 'MenCal', 0

	gs 'fertility', 'birth_control_status_update'
end


if $ARGS[0] = 'birth_control_status_update':
	birth_control['old_pillcon'] = pillcon
	birth_control['old_pillcon2'] = pillcon2

	if pillcon > 26000:
		birth_control['safe'] = 1
	elseif placebopart <= 0:
		birth_control['safe'] = 0
	end

	if pillcon2 > 26000:
		if tabletkishot > 0 and $birth_control['type'] ! 'shot':
			$birth_control['type'] = 'shot'
		elseif $birth_control['type'] ! 'pill':
			$birth_control['type'] = 'pill'
		end
		birth_control['think_safe'] = 1
	elseif placebopart <= 0:
		birth_control['think_safe'] = 0
		$birth_control['type'] = 'none'
	end
end

!! ------------------------------------------------------------------------------------------------------------
!!  FORCE PREGNANCY HELPER (parameterized)
!! ------------------------------------------------------------------------------------------------------------
!! Usage examples:
!!   gs 'fertility', 'force_preg_by', 'A12'                 <> known NPC father (must be male)
!!   gs 'fertility', 'force_preg_by', 'unknown', 'John'     <> unknown father name
!!   gs 'fertility', 'force_preg_by', '', 'Random guy'      <> '' is treated as unknown
!!
!! ARGS:
!!   $ARGS[1] = father id (e.g. 'A12') OR 'unknown' OR ''
!!   $ARGS[2] = father display name for unknown fathers (optional, defaults to 'unknown')
!!
!! If a known NPC is supplied and npc_gender is not male (0), it will show an error message and exit.
!!
!! Force pregnancy and set father (known NPC A1-999 or unknown name)
!! Usage:
!!   gs 'fertility', 'force_preg_by', 'A12'                <> known NPC father (must be male)
!!   gs 'fertility', 'force_preg_by', 'unknown', 'John'    <> unknown father name
!!
!!	Dont run the function if PC already pregnant


if $ARGS[0] = 'force_preg_by':
	if preg = 1: exit
	if func('npc', 'is_npcID', $ARGS[1]) = 0: exit
	if $npc_usedname[$ARGS[1]] = '': exit
	if npc_gender[$ARGS[1]] <> 0:
		if npc_spermpot[$ARGS[1]] <= 0:
			msg '错误：无法让主角怀孕。父亲 "' + $ARGS[1] + '" 既不是男性也不具备生育能力。'
			exit
		end
	end

	$fp_fathid = $ARGS[1]
	$fp_fathname = $ARGS[2]
	if $fp_fathname = '': $fp_fathname = $npc_usedname[$fp_fathid]

	!! Determine father strings + validate gender for known NPC fathers
	$fp_bio = $fp_fathid
	$fp_bel = $fp_fathid
	$fp_show = $fp_fathname
	$wombthfath = $fp_show

	!! Reset non-preg cycle hormones (mirrors cheat menu style)
	mesec = 0
	focH = 0
	EggRH = 0
	Ovulate = 0
	UnfertEgg = 0
	lutH = 0
	RecovH = 0

	!! Seed pregnancy state (conceived "now")
	preg = 1
	cycle = 5
	lastovulation = daystart
	if pregChem <= 150: pregChem = 150

	!! Add unborn child entry + father linkage (same arrays used by normal fertilization)
	babyembryo += 1
	if fertegg > 0:
		fertegg -= 1
	else
		nextBaby = arrsize('$ChildFath')
		Temppolkid = rand(0, 1)
		polkid[nextBaby] = Temppolkid
		$kidname[nextBaby] = 'unborn'
		kidage[nextBaby] = 0
		daykid[nextBaby] = 0
		monthkid[nextBaby] = 0
		yearkid[nextBaby] = 0
		babyptype[nextBaby] = 0

		$ChildFath[nextBaby] = $fp_bio
		$ChildThFath[nextBaby] = $fp_bel

		hairkid[nextBaby] = rand(0, 3)
		eyeskid[nextBaby] = rand(0, 3)
		ChildConType[nextBaby] = 0
		babyptype[nextBaby] = 1
	end

	gs 'din_bad', 'd_cycreport_upduedate'

	killvar 'fp_fathid'
	killvar 'fp_fathname'
	killvar 'fp_bio'
	killvar 'fp_bel'
	killvar 'fp_show'
	killvar 'dateVars'
end
