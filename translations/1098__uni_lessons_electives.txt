!! This is a modified copy of the regular study code from uni_library.
!! It will go down the list of classes to study, and determine if there is benefit from studying it
!! If there isn''t, it will go down the list. If there are none left, it will give intel exp
if $ARGS[0] = 'study_with_friends':
	temp_hours_studied_with_friend = min(max(1, (18 - hour)), 5)
	minut += min(max(0, 60 * (hour - 17) + 45 - minut), 285)


	!! Occurs 'uni_<<$university[''enrolled_in'']>>_semester_<<university[''enrolled_in_semester'']>>' only once in the array???
	if university['semester_week'] > 0:
		temp_ule_i = 0
		temp_ule_i_max = arrsize('class_list_institution')
		:study_with_friends_loop
			if $class_list_institution[temp_ule_i] = 'uni_<<$university[''enrolled_in'']>>_semester_<<university[''enrolled_in_semester'']>>':
				if class['<<$class_list_institution[temp_ule_i]>>_<<$class_list_name[temp_ule_i]>>_optional_weekly_grade_gain'] < class['<<$class_list_institution[temp_ule_i]>>_<<$class_list_name[temp_ule_i]>>_optional_weekly_max']:
					gs 'grades', 'optional_activity_attribute', '<<$class_list_institution[temp_ule_i]>>', '<<$class_list_name[temp_ule_i]>>', 'yes', '<<temp_hours_studied_with_friend>>'
					temp_ule_i = temp_ule_i_max + 10
				end
			end
			temp_ule_i += 1
		if temp_ule_i < temp_ule_i_max: jump 'study_with_friends_loop'
		if temp_ule_i < temp_ule_i_max + 10:
			gs 'exp_gain', 'intel', temp_hours_studied_with_friend + rand(-1, 1)
		end
	end

	killvar 'temp_ule_i'
	killvar 'temp_ule_i_max'
	killvar 'temp_hours_studied_with_friend'
	gs 'stat'
end
