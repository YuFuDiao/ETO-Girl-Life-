# shop_flamingos
!! 2022/03/12

if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_flamingos', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '火烈鸟')
	'<center><img <<$set_imgh>> src="images/locations/city/island/flamingos/shop.jpg"></center>'
	*nl
	'火烈鸟商店时尚又现代，价格也压得足够低，适合岛上众多的学生顾客。'
	'它模仿面向年轻人的最新潮流，但在材料和制作上削减成本。这正适合那些往往还没等衣服穿坏就已经穿腻了的顾客。'

	act '离开': minut += 3 & gt 'pushkin_sq'
	act '查看连衣裙': minut += 5 & gt 'shop_flamingos', 'dresses'
	act '查看其他服装': minut += 5 & gt 'shop_flamingos', 'clothes'
	act '查看钱包': minut += 5 & gt 'shop_flamingos', 'purses'
end


if $ARGS[0] = 'dresses':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_flamingos', 'dresses'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看火烈鸟的连衣裙')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'flamingos_dress'
			gs 'shop_utils', 'init', 'set_steps', 3, Clothingstock mod 3
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_flamingos', 'start'
	end
end


if $ARGS[0] = 'clothes':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_flamingos', 'clothes'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看火烈鸟的其他服装')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'flamingos_outfits'
			gs 'shop_utils', 'init', 'set_steps', 2, (Clothingstock / 3) mod 2
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_flamingos', 'start'
	end
end


if $ARGS[0] = 'purses':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_flamingos', 'purses'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看火烈鸟的钱包')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'purses', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'flamingos'
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_flamingos', 'start'
	end
end
