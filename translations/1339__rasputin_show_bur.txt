# rasputin_show_bur

if $ARGS[0] = '':
!! First half of the show
	if hour = 17 or (hour = 18 and minut < 30):
		gt 'rasputin_show_bur', 'first_half'
	else
		gt 'rasputin_show_bur', 'second_half'
	end
end


if $ARGS[0] = 'first_half':
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/rasputin_show_bur_1.jpg"></center>'
	'工作人员带你到你的座位。还没有其他观众到场。'
	act	'离开':
		minut += 1
		rasputin['burlesque_ticket'] = 1
		gt 'rasputin_walkway'
	end

	act	'等待演出开始': gt 'rasputin_show_bur', 'first_half_start'

	!! if (hour = 17 and minut > 30) or minut < 20:
		act	'要一杯免费香槟':
			minut += 10
			gs 'mood', 'raise', 'tiny'
			gs 'beverage', 'champagne_stats'
			*clr & cla
			gs 'stat'
			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/drinks_free_4.jpg"></center>'
			'服务员微笑着递给你一杯香槟享用。'
			act '继续': gt 'rasputin_show_bur', 'first_half_start'
		end
	!! end
end


if $ARGS[0] = 'first_half_start':
	if hour = 17:
		minut = 90 - minut
	else
		minut = 30 - minut
	end
	rasputin['burlesque_ticket'] = 2
	*clr & cla
	gs 'stat'
	if sound_settings['music_off'] = 0:
		$track_loop = 'sound/rasputin/music_bur_1.mp3'
		volume = 100
		music_loop = 1
	end

	!! if month <= 6:
		'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/singer_1.jpg"></center>'
		'幕布缓缓升起，一位女歌手开始演唱，由店内乐队伴奏。'
		act '好的':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/1.jpg"></center>'
			'歌手退到一旁，让你的目光聚焦到舞台上的三位表演者身上。'
			act '好的':
				*clr & cla
				minut += 5
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/2.jpg"></center>'
				'舞者们转过身，穿着华丽夺目的滑稽歌舞戏服，自信地展示着她们健美有型的身材。'
				act '好的':
					minut += 10
					*clr & cla
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/3.jpg"></center>'
					'舞者们在椅子上表演时伸展身体，保持平衡。'
					act '好的':
						minut += 25
						*clr & cla
						gs 'stat'
						'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/4.jpg"></center>'
						'她们随着音乐起舞，奉献了一场同样轻松、有趣又挑逗的表演。'
						'演出的第一部分结束了，你走出来休息一下，伸展伸展双腿。'
						act '好的':
							minut += 20
							gs 'mood', 'raise', 'small'
							gt 'rasputin_walkway'
						end
					end
				end
			end
		end
	!! end
end


if $ARGS[0] = 'second_half':
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/rasputin_show_bur_2.jpg"></center>'
	'工作人员带你到你的座位。许多观众已经就座等候。'
	act	'离开':
		minut += 1
		rasputin['burlesque_ticket'] = 2
		gt 'rasputin_walkway'
	end

	act	'等待演出开始': gt 'rasputin_show_bur', 'second_half_start'

	!! if minut < 50:
		act	'要一杯免费香槟':
			minut += 10
			gs 'mood', 'raise', 'tiny'
			gs 'beverage', 'champagne_stats'
			gs 'stat'
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/drinks_free_4.jpg"></center>'
			'服务员微笑着递给你一杯香槟享用。'
			act '继续': gt 'rasputin_show_bur', 'second_half_start'
		end
	!! end
end


if $ARGS[0] = 'second_half_start':
	minut += 60 - minut
	rasputin['burlesque_ticket'] = 0
	if sound_settings['music_off'] = 0:
		$track_loop = 'sound/rasputin/music_bur_2.mp3'
		volume = 100
		music_loop = 1
	end

	!! if month <= 6:
		*clr & cla
		'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/singer_1.jpg"></center>'
		'舞台重新亮起，舞者们伴着音乐登场。'
		'滑稽歌舞秀的下半场开始了。'
		act '好的':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/5.jpg"></center>'
			'舞者们有节奏而充满活力地舞动，她们头上的蓝色羽毛头饰自由飘扬。'
			act '好的':
				minut += 10
				*clr & cla
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/6.jpg"></center>'
				'随着另一段音乐响起，其他表演者舞向一旁，让你把注意力集中在舞台上的领舞身上。'
				act '好的':
					minut += 10
					*clr & cla
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/7.jpg"></center>'
					'领舞随着节拍性感地舞动，自由而自信地展示着她的舞技与魅力。'
					'她的表演结束时，观众们为之疯狂，报以雷鸣般的掌声和口哨欢呼。'
					act '好的':
						minut += 10
						*clr & cla
						gs 'stat'
						'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/bur_1/8.jpg"></center>'
						'演出的最后一部分是香槟杯之舞。你被那华丽而性感的舞姿深深吸引。'
						'滑稽歌舞秀在一阵掌声和兴奋的议论声中落下帷幕。'
						act '好的':
							gs 'mood', 'raise', 'small'
							if pcs_inhib >= 80:
								inhib_exp += rand(2, 6)
							elseif	pcs_inhib >= 50 and pcs_inhib < 80:
								inhib_exp += rand(1, 3)
							end
							gs 'arousal', 'erotic', 15
							gs 'arousal', 'end'
							gt 'rasputin_walkway'
						end
					end
				end
			end
		end
	!! end
end
