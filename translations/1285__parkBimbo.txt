# parkBimbo

if $ARGS[0] = 'pos1':
	*clr & cla
	minut += 10
	pcs_horny += 5
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/park/parkbimbo1.jpg"></center>'
	'你注意到一个六十多岁的老头正盯着你看。接着你发现他掏出了那根相当粗大的鸡巴，而且正硬得发烫。'
	act '帮帮这位老人': gt 'parkBimbo', 'pos2'
	gs 'willpower', 'bj', 'resist'
	if pcs_willpwr < will_cost:
		act '朝他竖中指' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '朝他竖中指' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'bj', 'resist'
			gs 'willpower', 'pay', 'resist'
			gs 'stat'
			gt 'pav_park', 'start'
		end
	end
end


if $ARGS[0] = 'pos2':
	*clr & cla
	gs 'cum_call', 'mouth_swallow'
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/park/sex/parkbimbo2.jpg"></center>'
	'你抓住老头的手，把他拉到一棵树旁。然后你跪下来，立刻开始吮吸他的鸡巴。他一开始吓了一跳，想要抗议，但当你温暖湿润的嘴巴开始吸他的鸡巴时，他就不再反抗了。'
	'你开始卖力地吸吮，同时用手揉搓他的蛋蛋。'
	'没过多久，你就感到他的精液一股股地射进你的嘴里，热乎乎的。'

	gs 'arousal', 'bj', 10, 'sub'
	gs 'stat'

	horand = rand (1, 100)
	if horand > pcs_horny:
		pcs_horny += 10
		'你感觉更饥渴了一点。'
	else
		'你太享受这个过程了，身体在高潮中不住地抽搐。你费了好大劲才忍住，始终没让他的精液从嘴里漏出来。'
		$orgasm_or = 'yes'
		gs 'stat'

	end
	'你站起身来，咽下他的精液，临走时还朝他抛了个媚眼。'

	act '结束':
		gs 'arousal', 'end'
		gt 'pav_park', 'start'
	end
end
