# drugs

!! How to use:
!!		Optional flags:
!!			'gift': pass this along as $ARGS[1] if Sveta is gifted the drug.
!!					This can technically be any string or number except '' and 0.
!! SOFT DRUGS:
!!		Alcohol:		gs 'drugs', 'alcohol', 'type', [count (defaults to 1)]
!!			Types: beer, champagne, moonshine, rum, scotch, vodka, wine, whiskey
!!			Count: Number of drinks of the chosen type sveta drinks (if the number is 1, you don''t have add the count, but it isn''t a problem if you do)
!!			Example 1: sveta drinks vodka gs 'drugs', 'alcohol', 'vodka' (or gs 'drugs', 'alcohol', 'vodka', 1)
!!			Example 2: sveta drinks 2 beer gs 'drugs', 'alcohol', 'beer', 2
!!		Coffee:			gs 'drugs', 'caffeine', [strength (defaults to 4)], [count (defaults to 1)]
!!		Smoking:		gs 'drugs', 'smoke', ['gift']
!!		Joint/Weed:		gs 'drugs', 'joint', ['gift']
!! HARD DRUGS:
!!		Amphetamine:	gs 'drugs', 'amphetamine', ['gift']
!!		Heroin:			gs 'drugs', 'heroin'
!!		Cocaine:		gs 'drugs', 'cocaine', ['gift']
!! ENHANCERS:
!!		Painkiller:		gs 'drugs', 'painkiller'
!!		Aphrodisiac:	gs 'drugs', 'aphrodisiac', ['gift']
!!		Neurobooster:	gs 'drugs', 'mentats', ['gift']
!!		Steroids:		gs 'drugs', 'steroids', ['gift']
!!		Breast Cream:	gs 'drugs', 'breastcream', ['gift']
!!		Hair Cream:		gs 'drugs', 'hair_extensioncream', ['gift']
!!		Butt Injection:	gs 'drugs', 'butt_injection', ['gift']




	!!====================================================================!!
	!!                                                                    !!
	!!                         GENERAL MANAGEMENT                         !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'cikl':

	gs 'drugs', 'aphrodisiac_cikl'
	gs 'drugs', 'hair_extensioncream_cikl'
	gs 'drugs', 'butt_injection_cikl'
	gs 'drugs', 'breastcream_cikl'
	gs 'drugs', 'caffeine_cikl'
	gs 'drugs', 'painkiller_cikl'
	exit
end


if $ARGS[0] = 'hourly_events':
	gs 'drugs', 'cigarette_hourly_events'
	gs 'drugs', 'weed_hourly_events'
	gs 'drugs', 'heroin_hourly_events'
	gs 'drugs', 'cocaine_hourly_events'
	gs 'drugs', 'amphetamine_hourly_events'
	gs 'drugs', 'steroids_hourly_events'
	gs 'drugs', 'aphrodisiac_hourly_events'
	gs 'drugs', 'caffeine_hourly_events'
	gs 'drugs', 'painkiller_hourly_events'
	exit
end


if $ARGS[0] = 'stat':
	gs 'drugs', 'heroin_stat'
	gs 'drugs', 'cocaine_stat'
	gs 'drugs', 'amphetamine_stat'
	gs 'drugs', 'caffeine_stat'
	gs 'drugs', 'painkiller_stat'
	exit
end


!! compute messages and icons for stat_display, called from stat_display_compute
if $ARGS[0] = 'compute_stat_display':

	!! ━━━━ ALCOHOL ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
	!! alko scale: 1=buzz, 2-3=tipsy, 4-5=drunk, 6-11=very drunk, 12+=wasted
	if alko >= 12:
		$stat_texts['alcohol'] = '你已经烂醉如泥了。'
		gs 'stat_display_compute', 'queue_msg', 'alcohol', 'v_neg', 'drugs/alcohol_5', 3
		gs 'stat_display_compute', 'queue_alert', '你已经烂醉如泥。', 'v_neg'
	elseif alko >= 6:
		$stat_texts['alcohol'] = '你醉得很厉害。'
		gs 'stat_display_compute', 'queue_msg', 'alcohol', 'neg', 'drugs/alcohol_4', 3
		gs 'stat_display_compute', 'queue_alert', '你醉得很厉害。', 'neg'
	elseif alko >= 4:
		$stat_texts['alcohol'] = '你喝醉了。'
		gs 'stat_display_compute', 'queue_msg', 'alcohol', 'neutral', 'drugs/alcohol_3', 3
	elseif alko >= 2:
		$stat_texts['alcohol'] = '你感觉有点微醺。'
		gs 'stat_display_compute', 'queue_msg', 'alcohol', 'neutral', 'drugs/alcohol_2', 3
	elseif alko = 1:
		$stat_texts['alcohol'] = '你喝酒后感到有点微醺。'
		gs 'stat_display_compute', 'queue_msg', 'alcohol', 'neutral', 'drugs/alcohol_1', 3
	end

	!! ━━━━ CIGARETTES ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
	!! Addiction threshold: cigarettes_used >= 20 (or relapse at >=5 with 150+ lifetime)
	!! Craving: cigarettes_need > 0, ramps hourly; severe at >10, >25, >50
	!! Partaking: totminut <= cigarettes_minute (3-minute window)

	!! ── Partaking: actively smoking ──
	if totminut <= drugVars['cigarettes_minute']:
		$stat_texts['drug_cig_partake'] = '你正在抽一支烟。'
		if mc_inventory['cigarettes'] > 0:
			gs 'stat_display_compute', 'queue_msg', 'drug_cig_partake', 'v_pos', 'drugs/cig_use', 3, 'gs ''drugs'', ''smoke'''
		else
			gs 'stat_display_compute', 'queue_msg', 'drug_cig_partake', 'v_pos', 'drugs/cig_use', 3
		end
	else
	!! ── Inventory icon: have cigarettes, not currently smoking ──
		if mc_inventory['cigarettes'] > 0:
			$stat_texts['drug_cig_inv_tooltip'] = '你有<<mc_inventory[''cigarettes'']>>支烟。点击抽烟。'
			gs 'stat_display_compute', 'queue_msg', 'drug_cig_inv', '', 'drugs/cig_have', 3, 'gs ''drugs'', ''smoke'''
		end
	end

	!! ── Withdrawal / Craving ──
	if drugVars['cigarettes_need'] > 0 and cheatVars['drugs_immune'] = 0:
		if drugVars['cigarettes_need'] > 50:
			$stat_texts['drug_cig_withdraw'] = '你迫切地想抽烟。你的双手在颤抖，什么都无法专注。'
			gs 'stat_display_compute', 'queue_alert', '你极度渴望抽根烟。', 'v_neg'
		elseif drugVars['cigarettes_need'] > 25:
			$stat_texts['drug_cig_withdraw'] = '你真的需要一支烟。你的神经紧张，什么都让你烦躁。'
		elseif drugVars['cigarettes_need'] > 10:
			$stat_texts['drug_cig_withdraw'] = '你想抽烟。烟瘾正在折磨你。'
		else
			$stat_texts['drug_cig_withdraw'] = '你现在正需要来一根烟。'
		end
		if mc_inventory['cigarettes'] > 0:
			$stat_texts['drug_cig_withdraw'] += ' 你有<<mc_inventory[''cigarettes'']>>支香烟。'
			$stat_texts['drug_cig_withdraw_tooltip'] = $stat_texts['drug_cig_withdraw'] + ' 点击抽烟。'
			gs 'stat_display_compute', 'queue_msg', 'drug_cig_withdraw', 'v_neg', 'drugs/cig_crave', 3, 'gs ''drugs'', ''smoke'''
		else
			$stat_texts['drug_cig_withdraw'] += ' 你已经没有香烟了。'
			$stat_texts['drug_cig_withdraw_tooltip'] = $stat_texts['drug_cig_withdraw']
			gs 'stat_display_compute', 'queue_msg', 'drug_cig_withdraw', 'v_neg', 'drugs/cig_crave', 3
		end
	end

	!! ── Addicted (persistent flag — shown when not actively craving) ──
	if drugVars['cigarettes_used'] >= 20 and cheatVars['drugs_immune'] = 0 and drugVars['cigarettes_need'] = 0:
		!! Withdrawal threshold: 80 + used*3/5 hours of craving to quit
		$stat_texts['drug_cig_addict_tooltip'] = '你对尼古丁上瘾了。如果你停止抽烟，几小时内烟瘾就会发作。'
		gs 'stat_display_compute', 'queue_msg', 'drug_cig_addict', 'neg', 'drugs/cig_addict', 3
	end


	!! ━━━━ WEED ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
	!! weed_high: set to 1–2 on use, decays 1/hour. No addiction system.
	!! weed_minute: totminut + 2 on use (2-min smoking window)

	!! ── Partaking: actively smoking a joint ──
	if totminut <= drugVars['weed_minute']:
		$stat_texts['drug_weed_partake'] = '你正在抽大麻烟。'
		gs 'stat_display_compute', 'queue_msg', 'drug_weed_partake', 'v_pos', 'drugs/weed_use', 3
	end

	!! ── High: ongoing weed effect ──
	if drugVars['weed_high'] > 0:
		if drugVars['weed_high'] >= 2:
			$stat_texts['drug_weed_high'] = '你彻底上头了。一切都显得滑稽又遥远。你饿得厉害。'
			gs 'stat_display_compute', 'queue_alert', '你彻底嗨翻了。', 'accent'
		else
			$stat_texts['drug_weed_high'] = '你上头了。世界变得柔软而平静。你感到有点饿了。'
		end
		gs 'stat_display_compute', 'queue_msg', 'drug_weed_high', 'accent', 'drugs/weed_high', 3
	end


	!! ━━━━ HEROIN ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
	!! heroin_high: +25 on use, decays 1/hour. At >20: also clears alcohol.
	!! heroin_need: acute withdrawal, ramps hourly (mood penalty >10/24/48), capped at 5 days, then sets heroin_taper.
	!! heroin_taper: post-acute craving tail (small mood penalty) — an hour countdown set
	!!   once, scaled by how long the habit lasted (heroin_hours_addicted), capped at 50 days.
	!!   Recovery (used/hours_addicted reset) when it counts down to 0.

	!! ── High: under heroin effects ──
	if drugVars['heroin_high'] > 0:
		if drugVars['heroin_high'] >= 20:
			$stat_texts['drug_heroin_high'] = '你正处于强烈的海洛因快感中。其他一切都不重要了。'
			gs 'stat_display_compute', 'queue_alert', '你正处于强烈的海洛因快感中。', 'accent'
		elseif drugVars['heroin_high'] >= 10:
			$stat_texts['drug_heroin_high'] = '你感到温暖而麻木。海洛因仍在你的体内流淌。'
		else
			$stat_texts['drug_heroin_high'] = '海洛因的快感正在消退。世界正慢慢重新袭来。'
		end
		gs 'stat_display_compute', 'queue_msg', 'drug_heroin_high', 'accent', 'drugs/heroin_high', 3
	end

	!! ── Withdrawal: heroin_need > 0 and not currently high ──
	if drugVars['heroin_need'] > 0 and drugVars['heroin_high'] = 0:
		if drugVars['heroin_taper'] > 0:
			$stat_texts['drug_heroin_withdraw'] = '身体的剧痛已经消退，但对海洛因的渴望仍在啃噬着你。要彻底放下还需要时间。'
		elseif drugVars['heroin_need'] > 48:
			$stat_texts['drug_heroin_withdraw'] = '你极度痛苦。全身每一根骨头都在疼。你拼命需要海洛因。你的情绪和性情都受到严重影响。'
			gs 'stat_display_compute', 'queue_alert', '你对海洛因的渴望难以抗拒。', 'v_neg'
		elseif drugVars['heroin_need'] > 24:
			$stat_texts['drug_heroin_withdraw'] = '你非常虚弱，痛彻骨髓。戒断反应让你痛苦不堪。'
			gs 'stat_display_compute', 'queue_alert', '海洛因戒断反应来势汹汹。', 'v_neg'
		elseif drugVars['heroin_need'] > 10:
			$stat_texts['drug_heroin_withdraw'] = '你的身体在疼痛，你感觉糟透了。你需要来一针。'
			gs 'stat_display_compute', 'queue_alert', '你需要来一剂海洛因。', 'neg'
		else
			$stat_texts['drug_heroin_withdraw'] = '你开始感到不适。你的身体需要更多海洛因。'
		end
		$stat_texts['drug_heroin_withdraw_tooltip'] = $stat_texts['drug_heroin_withdraw']
		!! No inventory for heroin — player must seek it out. Show withdrawal info on click.
		gs 'stat_display_compute', 'queue_msg', 'drug_heroin_withdraw', iif(drugVars['heroin_taper'] > 0, 'neg', 'v_neg'), 'drugs/heroin_crave', 3

	!! ── Addicted: used >= 3 but not yet in withdrawal (currently high) ──
	elseif drugVars['heroin_used'] >= 3 and drugVars['heroin_high'] > 0:
		!! heroin_high decays exactly 1/hr — this is a precise countdown, not an estimate
		if drugVars['heroin_high'] = 1:
			$stat_texts['drug_heroin_addict_tooltip'] = '你对海洛因上瘾了。快感几乎消失 —一小时内戒断反应就会开始。'
		else
			$stat_texts['drug_heroin_addict_tooltip'] = '你对海洛因上瘾了。药效消退后，戒断反应将在<<drugVars[''heroin_high'']>>小时后开始。'
		end
		gs 'stat_display_compute', 'queue_msg', 'drug_heroin_addict', 'neg', 'drugs/heroin_addict', 3
	end


	!! ━━━━ COCAINE ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
	!! cocaine_high:   acute high (0–200), hourly decay –20. Status + effects.
	!! cocaine_system: cumulative dependency (0–60), daily decay. Feeds cocaine_addict.
	!! cocaine_crash:  post-high crash (0–60), hourly decay –10. Set when high ends.
	!! cocaine_day:    set to daystart on use.
	!! cocaine_need:   hourly counter for addicts who haven''t used today.

	!! ── High ──
	if drugVars['cocaine_high'] > 0:
		if drugVars['cocaine_high'] >= 100:
			$stat_texts['drug_cocaine_high'] = '你感到无所不能。可卡因正奔涌在你的血管里。'
			gs 'stat_display_compute', 'queue_alert', '你可卡因上头了。', 'accent'
		elseif drugVars['cocaine_high'] >= 40:
			$stat_texts['drug_cocaine_high'] = '你感到强大而警觉。你的心脏因可卡因而狂跳。'
		else
			$stat_texts['drug_cocaine_high'] = '可卡因的兴奋感正在消退但你仍感到一丝余威。'
		end
		gs 'stat_display_compute', 'queue_msg', 'drug_cocaine_high', 'accent', 'drugs/cocaine_high', 3
	end

	!! ── Crash (a few hours after high ends) ──
	if drugVars['cocaine_crash'] > 0 and drugVars['cocaine_high'] = 0:
		if drugVars['cocaine_crash'] > 30:
			$stat_texts['drug_cocaine_crash'] = '你正在经历强烈的药效消退。你精疲力竭、烦躁易怒，渴望更多的可卡因。'
		else
			$stat_texts['drug_cocaine_crash'] = '可卡因的药效正在消退。你感到空虚而疲惫。'
		end
		gs 'stat_display_compute', 'queue_msg', 'drug_cocaine_crash', 'neg', 'drugs/cocaine_crave', 3
	end

	!! ── Withdrawal: addict + haven''t used today ──
	if drugVars['cocaine_addict'] = 1 and drugVars['cocaine_day'] <> daystart:
		if drugVars['cocaine_need'] > 144:
			$stat_texts['drug_cocaine_withdraw'] = '你在毁灭自己。没有可卡因的每一小时都在损耗你的健康和理智。你的情绪和性情在急剧恶化。'
			gs 'stat_display_compute', 'queue_alert', '你<i>需要</i>可卡因。', 'v_neg'
		elseif drugVars['cocaine_need'] > 72:
			$stat_texts['drug_cocaine_withdraw'] = '你急需可卡因。你感到非常虚弱，你的健康在迅速恶化。'
			gs 'stat_display_compute', 'queue_alert', '可卡因戒断反应来势汹汹。', 'v_neg'
		elseif drugVars['cocaine_need'] > 24:
			$stat_texts['drug_cocaine_withdraw'] = '你需要可卡因。渴望非常强烈，你的身体正在变弱。'
			gs 'stat_display_compute', 'queue_alert', '你需要可卡因。', 'neg'
		else
			$stat_texts['drug_cocaine_withdraw'] = '你今天还没碰过可卡因。渴望正在累积。'
		end
		if mc_inventory['cocaine'] > 0:
			$stat_texts['drug_cocaine_withdraw'] += ' 你有<<mc_inventory[''cocaine'']>>剂可卡因。'
			$stat_texts['drug_cocaine_withdraw_tooltip'] = $stat_texts['drug_cocaine_withdraw'] + ' 点击使用。'
			gs 'stat_display_compute', 'queue_msg', 'drug_cocaine_withdraw', 'v_neg', 'drugs/cocaine_crave', 3, 'gs ''drugs'', ''cocaine'''
		else
			$stat_texts['drug_cocaine_withdraw_tooltip'] = $stat_texts['drug_cocaine_withdraw']
			gs 'stat_display_compute', 'queue_msg', 'drug_cocaine_withdraw', 'v_neg', 'drugs/cocaine_crave', 3
		end

	!! ── Addicted: used today, no active withdrawal ──
	elseif drugVars['cocaine_addict'] = 1 and drugVars['cocaine_day'] = daystart:
		$stat_texts['drug_cocaine_addict_tooltip'] = '你对可卡因上瘾了。如果明天不再使用，戒断反应就会开始。'
		gs 'stat_display_compute', 'queue_msg', 'drug_cocaine_addict', 'neg', 'drugs/cocaine_addict', 3
	end


	!! ━━━━ AMPHETAMINE ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
	!! amphetamine_high: set to 12 on use, decays 4/hour (~3h high).
	!! amphetamine_withdrawl (sic): complex state machine.
	!!   0 = clean, 2 = coming down, >=5 = active withdrawal (up to 169h).
	!!   Withdrawal: mood loss = withdrawl/5 per hour. Resets at >169.

	!! ── High: wired on amphetamines ──
	if drugVars['amphetamine_high'] > 0:
		if drugVars['amphetamine_high'] >= 8:
			$stat_texts['drug_amphet_high'] = '你兴奋极了！你的心脏怦怦直跳，你觉得自己势不可挡。'
			gs 'stat_display_compute', 'queue_alert', '你被安非他命顶得亢奋不已。', 'accent'
		elseif drugVars['amphetamine_high'] >= 4:
			$stat_texts['drug_amphet_high'] = '你感到亢奋且紧张不安。安非他命的冲击感依然强烈。'
		else
			$stat_texts['drug_amphet_high'] = '安非他命的兴奋感正在消退。你感到烦躁不安。'
		end
		gs 'stat_display_compute', 'queue_msg', 'drug_amphet_high', 'accent', 'drugs/amphet_high', 3
	end

	!! ── Withdrawal: amphetamine_withdrawl >= 5 and not high ──
	if drugVars['amphetamine_withdrawl'] >= 5 and drugVars['amphetamine_high'] <= 0:
		if drugVars['amphetamine_withdrawl'] > 100:
			$stat_texts['drug_amphet_withdraw'] = '安非他命的药效消退令人难以忍受。你几乎无法正常行动。你的情绪急速低落。'
			gs 'stat_display_compute', 'queue_alert', '安非他命的药效回落让人难以忍受。', 'v_neg'
		elseif drugVars['amphetamine_withdrawl'] > 50:
			$stat_texts['drug_amphet_withdraw'] = '安非他命戒断反应十分剧烈。你感到精疲力竭，情绪低沉绝望。'
			gs 'stat_display_compute', 'queue_alert', '安非他命戒断反应来势汹汹。', 'v_neg'
		else
			$stat_texts['drug_amphet_withdraw'] = '你正在经历安非他命药效褪去。你感到疲倦、烦躁、精疲力尽。'
			gs 'stat_display_compute', 'queue_alert', '你的安非他命药效正在退去。', 'neg'
		end
		if mc_inventory['amphetamine'] > 0:
			$stat_texts['drug_amphet_withdraw'] += ' 你有<<mc_inventory[''amphetamine'']>>片药丸。'
			$stat_texts['drug_amphet_withdraw_tooltip'] = $stat_texts['drug_amphet_withdraw'] + ' 点击领取一个。'
			gs 'stat_display_compute', 'queue_msg', 'drug_amphet_withdraw', 'v_neg', 'drugs/amphet_crave', 3, 'gs ''drugs'', ''amphetamine'''
		else
			$stat_texts['drug_amphet_withdraw_tooltip'] = $stat_texts['drug_amphet_withdraw']
			gs 'stat_display_compute', 'queue_msg', 'drug_amphet_withdraw', 'v_neg', 'drugs/amphet_crave', 3
		end

	!! ── Coming down: withdrawl = 2 and not high (transition state) ──
	elseif drugVars['amphetamine_withdrawl'] = 2 and drugVars['amphetamine_high'] <= 0:
		$stat_texts['drug_amphet_comedown'] = '安非他命的兴奋感已经消退。你感到疲惫无力，反应迟钝。'
		gs 'stat_display_compute', 'queue_msg', 'drug_amphet_comedown', 'neg', 'drugs/amphet_crave', 3
	end

	!! ── Addicted: significant use history (>=10), fully clean (withdrawl=0) ──
	!! used resets only after full 169h detox; withdrawl=0 = calm window between comedown and next use.
	if drugVars['amphetamine_used'] >= 10 and drugVars['amphetamine_high'] <= 0 and drugVars['amphetamine_withdrawl'] = 0:
		$stat_texts['drug_amphet_addict'] = '你已经对安非他命产生了依赖。'
		gs 'stat_display_compute', 'queue_msg', 'drug_amphet_addict', 'neg', 'drugs/amphet_addict', 3

		!!------------------------------------------------------------!!
		!!                        AMPHETAMINE                         !!
		!!------------------------------------------------------------!!


!! $ARGS[1] = 'gift' to indicate that Sveta is borrowing a pill
!!					(defaults to using her own)
if $ARGS[0] = 'amphetamine':
	if drugVars['amphetamine_intake'] >= 2:
		'你的心脏已经在胸腔里狂跳不止，你应该避免再服用任何药丸。'
		exit
	end

	*pl $func('wrap', 'accent', '你吞下了白色药丸。')

	if ARGS[1] = 0 and $ARGS[1] = '':
		mc_inventory['amphetamine'] -= 1
		if mc_inventory['amphetamine'] <= 0: mc_inventory['amphetamine'] = 0 & msg '那是你最后一粒安非他命药丸了。'
	end

	if cheatVars['drugs_immune'] = 0:
		drugVars['amphetamine_used'] += 1
		if drugVars['amphetamine_used'] > 5 and drugVars['amphetamine_exp'] = 0: drugVars['amphetamine_exp'] = 1
	end

	drugVars['amphetamine_intake'] += 1
	drugVars['amphetamine_high'] = 12
	gs 'stat'
	exit
end


if $ARGS[0] = 'amphetamine_hourly_events':
	if drugVars['amphetamine_withdrawl'] >= 5:
		drugVars['amphetamine_withdrawl'] += 1
		gs 'mood', 'lower', drugVars['amphetamine_withdrawl'] / 5
	end

	if drugVars['amphetamine_high'] > 0:
		if drugVars['amphetamine_withdrawl'] < 2: drugVars['amphetamine_withdrawl'] = 2
		drugVars['amphetamine_bonus'] = 2
		drugVars['amphetamine_high'] -= 4
		gs 'mood', 'raise', 2 * drugVars['amphetamine_high']

	elseif drugVars['amphetamine_withdrawl'] = 2:
		if drugVars['amphetamine_intake'] > 0:
			drugVars['amphetamine_intake'] -= 1
		else
			drugVars['amphetamine_withdrawl'] = 0
		end

		drugVars['amphetamine_high'] = 0
		drugVars['amphetamine_bonus'] = 1
		gs 'mood', 'lower', 'small'

	elseif drugVars['amphetamine_intake'] > 0:
		drugVars['amphetamine_withdrawl'] += 1

	elseif drugVars['amphetamine_withdrawl'] = 0:
		drugVars['amphetamine_bonus'] = 0
	end

	if drugVars['amphetamine_withdrawl'] > 169:
		drugVars['amphetamine_withdrawl'] = 0
		drugVars['amphetamine_bonus'] = 0
		drugVars['amphetamine_used'] = 0
	end

	if drugVars['amphetamine_high'] <= 0:
		if inSleep = 1:
			pcs_energy -= 4 * drugVars['amphetamine_bonus']
			pcs_hydra -= 8 * drugVars['amphetamine_bonus']
		else
			pcs_energy -= 8 * drugVars['amphetamine_bonus']
			pcs_hydra -= 16 * drugVars['amphetamine_bonus']
		end
	else
		if inSleep = 1:
			pcs_energy += 4 * drugVars['amphetamine_bonus'] / (1 + drugVars['amphetamine_bonus'])
			pcs_hydra += 8 * drugVars['amphetamine_bonus'] / (1 + drugVars['amphetamine_bonus'])
		else
			pcs_energy += 8 * drugVars['amphetamine_bonus'] / (1 + drugVars['amphetamine_bonus'])
			pcs_hydra += 16 * drugVars['amphetamine_bonus'] / (1 + drugVars['amphetamine_bonus'])
		end
	end
	exit
end


if $ARGS[0] = 'amphetamine_stat':
	if drugVars['amphetamine_used'] <= 3 or drugVars['amphetamine_high'] > 0: exit

	if rand(1, 5) > 4: drugVars['amphetamine_withdrawl'] = 2
	exit
end


		!!------------------------------------------------------------!!
		!!                          COCAINE                           !!
		!!------------------------------------------------------------!!


!! $ARGS[1] = 'gift' to indicate that Sveta is receiving the drugs
!!					(defaults to using her own)
if $ARGS[0] = 'cocaine':
	if ARGS[1] = 0 and $ARGS[1] = '':
		mc_inventory['cocaine'] -= 1
		if mc_inventory['cocaine'] <= 0: mc_inventory['cocaine'] = 0 & msg '那是你最后一条可卡因了。'
	end

	if cheatVars['drugs_immune'] = 0:
		drugVars['cocaine_used'] += 1
		if drugVars['cocaine_used'] > 3 and drugVars['cocaine_exp'] = 0: drugVars['cocaine_exp'] = 1
	end

	drugVars['cocaine_high'] = min(200, drugVars['cocaine_high'] + 100)
	drugVars['cocaine_system'] = min(60, drugVars['cocaine_system'] + 8)
	drugVars['cocaine_day'] = daystart
	drugVars['cocaine_crash'] = 0
	drugVars['cocaine_need'] = 0
	if drugVars['cocaine_willday'] <> daystart:
		drugVars['cocaine_willday'] = daystart
		pcs_willpwr += 10
	end

	pcs_health = pcs_vital * 10 + pcs_stren * 5
	gs 'mood', 'raise', 'huge'
	pcs_horny = max(100, pcs_horny)
	gs 'stat'
	exit
end


if $ARGS[0] = 'cocaine_cikl':
	if drugVars['cocaine_system'] < 20:
		drugVars['cocaine_system'] = max(0, drugVars['cocaine_system'] - 1)
	elseif drugVars['cocaine_system'] < 40:
		drugVars['cocaine_system'] -= 2
	elseif drugVars['cocaine_system'] < 60:
		drugVars['cocaine_system'] -= 3
	else
		drugVars['cocaine_system'] = min(60, drugVars['cocaine_system'] - 4)
	end
	exit
end


if $ARGS[0] = 'cocaine_hourly_events':
	!! Decay the high; trigger crash when it hits zero
	if drugVars['cocaine_high'] > 0:
		drugVars['cocaine_high'] = max(0, drugVars['cocaine_high'] - 20)
		if drugVars['cocaine_high'] = 0:
			drugVars['cocaine_crash'] = 60
		end
	end

	!! Crash effects (everyone who used recently, ~6 hrs after high ends)
	if drugVars['cocaine_crash'] > 0 and drugVars['cocaine_high'] = 0:
		drugVars['cocaine_crash'] = max(0, drugVars['cocaine_crash'] - 10)
		gs 'mood', 'lower', 'tiny'
		pcs_stam -= 10
	end

	!! Withdrawal (addicts only, starting the day after last use)
	if drugVars['cocaine_addict'] = 0: exit
	if drugVars['cocaine_high'] > 0: exit
	if drugVars['cocaine_day'] = daystart: exit

	drugVars['cocaine_need'] += 1
	pcs_health -= 2 + drugVars['cocaine_need'] / 10

	if drugVars['cocaine_need'] > 144:
		gs 'mood', 'lower', 'large'
		if drugVars['disp_day_cocaine'] < daystart:
			drugVars['disp_day_cocaine'] = daystart
			gs 'mood', 'lower_disposition', 'tiny'
		end
	elseif drugVars['cocaine_need'] > 72:
		gs 'mood', 'lower', 'medium'
	elseif drugVars['cocaine_need'] > 24:
		gs 'mood', 'lower', 'small'
	elseif drugVars['cocaine_need'] > 6:
		gs 'mood', 'lower', 'tiny'
	end
	exit
end


if $ARGS[0] = 'cocaine_stat':
	if cheatVars['drugs_immune'] = 1:
		drugVars['cocaine_addict'] = 0
		drugVars['cocaine_need'] = 0
	elseif drugVars['cocaine_system'] >= 25:
		drugVars['cocaine_addict'] = 1
	elseif drugVars['cocaine_system'] <= 0:
		drugVars['cocaine_addict'] = 0
		drugVars['cocaine_need'] = 0
	end

if $ARGS[0] = 'heroin':

		drugVars['heroin_used'] += 1
		if drugVars['heroin_used'] > 1 and drugVars['heroin_exp'] = 0: drugVars['heroin_exp'] = 1
	end

	gs 'mood', 'raise', 'max'
	gs 'mood', 'hold', 180

	drugVars['heroin_day'] = daystart

	gs 'stat'
	exit
end


if $ARGS[0] = 'heroin_hourly_events':
	!! Entrenchment time — scales the taper length below.
	if drugVars['heroin_used'] >= 3: drugVars['heroin_hours_addicted'] += 1

	if drugVars['heroin_high'] > 0:
		gs 'mood', 'lower', 40 / drugVars['heroin_high']
		drugVars['heroin_high'] -= 1
	elseif drugVars['heroin_taper'] > 0:
		!! heroin_taper doubles as the countdown of hours left.
		gs 'mood', 'lower', 'small'
		drugVars['heroin_taper'] -= 1
		if drugVars['heroin_taper'] <= 0:
			drugVars['heroin_need'] = 0
			drugVars['heroin_used'] = 0
			drugVars['heroin_hours_addicted'] = 0
		end
	elseif drugVars['heroin_need'] > 0:
		drugVars['heroin_need'] += 1
		if drugVars['heroin_need'] > 48:
			gs 'mood', 'lower', 'large'
			if drugVars['disp_day_heroin'] < daystart:
				drugVars['disp_day_heroin'] = daystart
				gs 'mood', 'lower_disposition', 'small'
			end
		elseif drugVars['heroin_need'] > 24:
			gs 'mood', 'lower', 'medium'
		elseif drugVars['heroin_need'] > 10:
			gs 'mood', 'lower', 'small'
		end
		!! Acute phase caps at 5 days, then starts the taper countdown.
		if drugVars['heroin_need'] >= min(120, 24 + drugVars['heroin_used'] * 2):
			drugVars['heroin_taper'] = min(1200, 24 + drugVars['heroin_hours_addicted'] * 2)
		end
	end
	exit
end


if $ARGS[0] = 'heroin_stat':
	if drugVars['heroin_high'] > 0:
		if drugVars['heroin_high'] >= 20 and alko > 0: alko = 0
		drugVars['heroin_need'] = 0
		drugVars['heroin_taper'] = 0
		gs 'mood', 'raise', 'large'
	elseif drugVars['heroin_used'] >= 3:
		drugVars['heroin_need'] = max(1, drugVars['heroin_need'])
	end
	if drugVars['heroin_used'] >= drugVars['heroin_penalty'] * 25:
		if drugVars['heroin_penalty'] > 0: vidage += 1
		drugVars['heroin_penalty'] += 1
	end
	exit
end

if $ARGS[0] = 'count_painkiller_timer':
	gs 'drugs', 'refresh_painkiller_timer'
	result = arrsize('painkiller_timer')
	exit
end


if $ARGS[0] = 'refresh_painkiller_timer':
	temp_pk_gap = totminut - drugVars['painkiller_timer_refresh']
	drugVars['painkiller_timer_refresh'] = totminut
	if arrsize('painkiller_timer') > 0:
		temp_pk_index = 0
		:painkiller_refresh_loop
			painkiller_timer[temp_pk_index] -= temp_pk_gap
			if painkiller_timer[temp_pk_index] <= 0:
				killvar 'painkiller_timer', temp_pk_index
			else
				temp_pk_index += 1
			end
		if temp_pk_index < arrsize('painkiller_timer'): jump 'painkiller_refresh_loop'
		killvar 'temp_pk_index'
	end
	killvar 'temp_pk_gap'
	exit
end


if $ARGS[0] = 'painkiller_act_str':
	temp_pk_num = func('drugs', 'count_painkiller_timer')
	$temp_pk_str = $str(temp_pk_num + 1) + $func('shortgs', 'get_number_suffix', temp_pk_num + 1)
	if temp_pk_num = 0:
		$result = '服用一片止痛药'
	elseif temp_pk_num <= 2:
		$result = '再吃一片止痛药 ' + $func('wrap', 'neutral', '（最近三小时内的第<<$temp_pk_str>>片）')
	else
		$result = '再吃一片止痛药 ' + $func('wrap', 'v_neg', '（最近三小时内的第<<$temp_pk_str>>片！）')
	end
	killvar 'temp_pk_num'
	killvar 'temp_pk_str'
	exit
end

		gs 'mood', 'lower', 'tiny'
		if drugVars['painkiller_need'] >= 12: gs 'mood', 'lower', 'tiny'

		pain['tummy'] += drugVars['painkiller_need'] / 6

		end
	end
