if $ARGS[0] = 'get_est_bmi':
	!! Approximate weight via the same formula as body.qsrc CalcWeight (preg=0 at start).
	!! At overview time bust = bust_gen and butt = butt_gen.
	!! BMI×10 for one decimal place: BMI = weight×10000 / hgt²
	!! Work from the weight numerator directly to avoid losing precision from the /100 integer division.
	!! weight_numerator = 2820 + 33*(body+bust+butt) + 70*(hgt-165)
	!! BMI×10 = (weight_numerator/100) × 10000 / hgt² = weight_numerator × 1000 / hgt²
	temp_est_weight = (2820 + 33 * (pcs_mass['body'] + pcs_mass['bust_gen'] + pcs_mass['butt_gen']) + 70 * (pcs_hgt - 165))
	temp_est_bmi2 = temp_est_weight * 1000 / (pcs_hgt * pcs_hgt)
	$temp_est_bmi_str = (temp_est_bmi2 / 10) + '.' + (temp_est_bmi2 mod 10)
	!! Categories match body_structure.qsrc get_class thresholds exactly.
	if temp_est_bmi2 < 160:
		$temp_est_bmi_cat = '极度消瘦'
	elseif temp_est_bmi2 < 190:
		$temp_est_bmi_cat = '偏瘦'
	elseif temp_est_bmi2 < 220:
		$temp_est_bmi_cat = '苗条'
	elseif temp_est_bmi2 < 250:
		$temp_est_bmi_cat = '健康'
	elseif temp_est_bmi2 < 300:
		$temp_est_bmi_cat = '超重'
	elseif temp_est_bmi2 < 350:
		$temp_est_bmi_cat = '严重超重'
	elseif temp_est_bmi2 < 400:
		$temp_est_bmi_cat = '中度肥胖'
	elseif temp_est_bmi2 < 450:
		$temp_est_bmi_cat = '重度肥胖'
	else
		$temp_est_bmi_cat = '病态肥胖'
	end
	$result = $func('intro_overview', 'render_cell', '估算BMI', $temp_est_bmi_str + ' (' + $temp_est_bmi_cat + ')', 2)
	killvar 'temp_est_weight'

if $ARGS[0] = 'cycle_group':
	!! ARGS[1] = direction: 1 (right/forward) or -1 (left/backward)
	!! Groups cycle: nerd(0) → jock(1) → cool(2) → gopnik(3) → outcast(4)
	!! Custom starts are locked — cycling would silently overwrite the player''s custom stats.
	if $start_type['cat'] = 'custom': exit
	if $start_type['group'] = 'nerd':
		temp_group_idx = 0
	elseif $start_type['group'] = 'jock':
		temp_group_idx = 1
	elseif $start_type['group'] = 'cool':
		temp_group_idx = 2
	elseif $start_type['group'] = 'gopnik':
		temp_group_idx = 3
	elseif $start_type['group'] = 'outcast':
		temp_group_idx = 4
	end
	temp_group_idx = (temp_group_idx + ARGS[1] + 5) mod 5
	if temp_group_idx = 0:
		$start_type['group'] = 'nerd'
		$start_type['cat'] = 'nerdqueen'
	elseif temp_group_idx = 1:
		$start_type['group'] = 'jock'
		$start_type['cat'] = 'volleyball'
	elseif temp_group_idx = 2:
		$start_type['group'] = 'cool'
		$start_type['cat'] = 'socialite'
	elseif temp_group_idx = 3:
		$start_type['group'] = 'gopnik'
		$start_type['cat'] = 'gopnikstart'
	elseif temp_group_idx = 4:
		$start_type['group'] = 'outcast'
		$start_type['cat'] = 'friendless'
	end
	gs 'intro_character_templates', 'set_template', $start_type['cat']
	killvar 'temp_group_idx'
	exit
end


if $ARGS[0] = 'cycle_subgroup':
	!! ARGS[1] = direction: 1 (right/forward) or -1 (left/backward)
	!! Custom starts are locked — cycling would silently overwrite the player''s custom stats.
	if $start_type['cat'] = 'custom': exit
	if $start_type['group'] = 'nerd':
		!! nerdqueen → goodstudent → computergeek → chessplayer → (wrap)
		if $start_type['cat'] = 'nerdqueen':
			$start_type['cat'] = iif(ARGS[1] = 1, 'goodstudent', 'chessplayer')
		elseif $start_type['cat'] = 'goodstudent':
			$start_type['cat'] = iif(ARGS[1] = 1, 'computergeek', 'nerdqueen')
		elseif $start_type['cat'] = 'computergeek':
			$start_type['cat'] = iif(ARGS[1] = 1, 'chessplayer', 'goodstudent')
		elseif $start_type['cat'] = 'chessplayer':
			$start_type['cat'] = iif(ARGS[1] = 1, 'nerdqueen', 'computergeek')
		end
	elseif $start_type['group'] = 'jock':
		!! volleyball → dancer → runner → football → (wrap)
		if $start_type['cat'] = 'volleyball':
			$start_type['cat'] = iif(ARGS[1] = 1, 'dancer', 'football')
		elseif $start_type['cat'] = 'dancer':
			$start_type['cat'] = iif(ARGS[1] = 1, 'runner', 'volleyball')
		elseif $start_type['cat'] = 'runner':
			$start_type['cat'] = iif(ARGS[1] = 1, 'football', 'dancer')
		elseif $start_type['cat'] = 'football':
			$start_type['cat'] = iif(ARGS[1] = 1, 'volleyball', 'runner')
		end
	elseif $start_type['group'] = 'cool':
		!! socialite → beautiful → anorexic → bimbo → (wrap)
		if $start_type['cat'] = 'socialite':
			$start_type['cat'] = iif(ARGS[1] = 1, 'beautiful', 'bimbo')
		elseif $start_type['cat'] = 'beautiful':
			$start_type['cat'] = iif(ARGS[1] = 1, 'anorexic', 'socialite')
		elseif $start_type['cat'] = 'anorexic':
			$start_type['cat'] = iif(ARGS[1] = 1, 'bimbo', 'beautiful')
		elseif $start_type['cat'] = 'bimbo':
			$start_type['cat'] = iif(ARGS[1] = 1, 'socialite', 'anorexic')
		end
	elseif $start_type['group'] = 'gopnik':
		!! gopnikstart → troublemaker → [vitekgf if tsg=1] → alternative → (wrap)
		if $start_type['cat'] = 'gopnikstart':
			$start_type['cat'] = iif(ARGS[1] = 1, 'troublemaker', 'alternative')
		elseif $start_type['cat'] = 'troublemaker':
			if ARGS[1] = 1:
				$start_type['cat'] = iif(tsg = 1, 'vitekgf', 'alternative')
			else
				$start_type['cat'] = 'gopnikstart'
			end
		elseif $start_type['cat'] = 'vitekgf':
			$start_type['cat'] = iif(ARGS[1] = 1, 'alternative', 'troublemaker')
		elseif $start_type['cat'] = 'alternative':
			if ARGS[1] = 1:
				$start_type['cat'] = 'gopnikstart'
			else
				$start_type['cat'] = iif(tsg = 1, 'vitekgf', 'troublemaker')
			end
		end
	elseif $start_type['group'] = 'outcast':
		!! friendless → uglyduckling → goodgirl → slut → goth → poor → (wrap)
		if $start_type['cat'] = 'friendless':
			$start_type['cat'] = iif(ARGS[1] = 1, 'uglyduckling', 'poor')
		elseif $start_type['cat'] = 'uglyduckling':
			$start_type['cat'] = iif(ARGS[1] = 1, 'goodgirl', 'friendless')

		elseif $start_type['cat'] = 'slut':
			$start_type['cat'] = iif(ARGS[1] = 1, 'goth', 'goodgirl')
		elseif $start_type['cat'] = 'goth':
			$start_type['cat'] = iif(ARGS[1] = 1, 'poor', 'slut')
		elseif $start_type['cat'] = 'poor':
			$start_type['cat'] = iif(ARGS[1] = 1, 'friendless', 'goth')
		end
	end
	gs 'intro_character_templates', 'set_template', $start_type['cat']
	exit
end

	elseif ARGS[1] <= 100000:
		$result = '中等'
	elseif ARGS[1] <= 125000:
		$result = '强'
	elseif ARGS[1] <= 150000:
		$result = '很强'
	elseif ARGS[1] <= 200000:
		$result = '主导'
	elseif ARGS[1] <= 300000:
		$result = '强烈'
	elseif ARGS[1] <= 400000:
		$result = '极端'
	else
		$result = '完全'
	end
	exit
end
