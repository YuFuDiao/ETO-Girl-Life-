act '开始新游戏':
	*clr & cla
	'<center><b>Disopl</b></center>'
	killvar 'KGD'
	KGD['save'] = 1
	KGD['money'] = 1000
	KGD['HP'] = 100
	KGD['lvl'] = 1
	KGD['damage'] = 10
	KGD['day'] = 1
	gt 'KGDgame'
end


if $ARGS[0] = 'loadsave':
	*clr & cla
	'<center><b>Disopl</b></center>'
	$func('wrap', 'neg b', '你确定要载入此存档吗？')
	act '确认':
		KGDsavegame = ARGS[1]
		copyarr 'KGD', 'KGD<<ARGS[1]>>'
		gt 'KGDgame'
	end
	act '取消': gt 'KGDstart'
end


if $ARGS[0] = 'deletesave':
	*clr & cla
	'<center><b>Disopl</b></center>'
	$func('wrap', 'neg b', '你确定要删除此存档吗？')
	act '确认':
		killvar 'KGD<<ARGS[1]>>'
		gt 'KGDstart'
	end
	act '取消': gt 'KGDstart'
end

act '离开': gt 'KGDgame', 'exit'
