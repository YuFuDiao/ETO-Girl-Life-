!! ------------- Pay off debt -------------
if $ARGS[0] = 'minor_debt_payoff':
	cls
	menu_off = 1
	workDisk = 0
	gs 'stat'

	if $loc = $home['location']:
		debt_payoff_at_home = 1
	else
		debt_payoff_at_home = 0
	end

	'<center><img <<$set_imgh>> src="images/locations/city/residential/street/bug.jpg"></center>'
	if debt_payoff_at_home:
		'你走近公寓入口时，看到两个壮汉朝你走来。他们看起来带着枪。'
		'"该还债了，"其中一人走近时说道。'
	else
		'两个壮汉不知从哪里冒出来，把你堵在角落里。他们看起来带着枪。'
		'"该还债了，"其中一人走近时说道。'
	end
	if func('money', 'can_afford_debt', workDolg, 'cash'):
		gs 'money', 'debt_pay', 'workDolg', 'cash'
		'你赶紧翻出钱包，把钱拿出来递过去。其中一人一把抓过钱开始清点。确认数目没错后，他只是哼了一声，两人转身离开了。'
	elseif debt_payoff_at_home and func('money', 'can_afford_debt', workDolg, 'desk'):
		gs 'money', 'debt_pay', 'workDolg', 'desk'
		'"钱在我楼上的公寓里。我去拿，"你说着朝门口走去。'
		'他们点了点头，你赶紧跑进屋，从抽屉里拿出钱回来付给他们。他们接过钱数了数，互相看了一眼，点点头便离开了。'
	else
		gs 'money', 'debt_pay', 'workDolg', iif(debt_payoff_at_home, 'desk', 'cash')
		gs 'money', 'debt_add', 'workDolg', 10000
		if debt_payoff_at_home:
			'你告诉他们你的钱不够。他们打了你一巴掌，把你推进公寓，跟着进了屋。他们翻遍你的东西，找到了你的积蓄并拿走了。'
		else
			'你告诉他们你的钱不够。他们打了你一巴掌，翻遍你的包和口袋，把你身上的东西全都拿走了。'
		end
		'这两个恶棍拿走了你所有的东西，然后向你说明了情况。简单来说，你欠了 ' + $func('money', 'string_debt', workDolg) + ' 的债。你的债务又增加了 ' + $func('money', 'string_debt_addition', 10000) + '，由于你无力偿还，他们决定"帮"你一把。'
		'他们把你拖到街上，扔进汽车后座。他们拒绝回答你的任何问题，你只能安静地坐着，看他们把你带去哪里。'
		act '继续':
			cls
			minut += 30
			gs 'stat'
			'你被带到了桑拿房，那两个人把你拖进一个房间。"你要住在这里，干活还债。要是敢联系警察或者逃跑，我们就在你脑袋里塞颗子弹，把你扔到林子里，听明白了吗？"'
			'你吓坏了，只能点头。'
			act '环顾四周': gt 'city_sauna', 'saunaroom'
		end
	end
	killvar 'debt_payoff_at_home'
	exit
end


!! ---------- Didn''t deliver disk -------------
if $ARGS[0] = 'minor_didnt_deliver_disk':
	menu_off = 1
	workDisk = 1
	workDiskAtkPlayer = 0
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/residential/street/car.jpg"></center>'
	'你突然被人抓住，扔进一辆汽车的后座。你左右各坐着一个怒气冲冲的男人，而色情片工作室的经理坐在前排。'
	'"信封在哪儿？"他质问道。'
	if konvert = 0:
		gs 'money', 'debt_add', 'workDolg', 200000
		workDolgDay = 7
		$job_termination_reason['city_pornstudio_delivery'] = 'blacklisted'
		gs 'jobs', 'set_fired', 'city_pornstudio_delivery'
		'"被偷了，"你回答。'
		'"<i>什么</i>？这怎么会发生？！你知不知道自己干了什么？！你欠我 ' + $func('money', 'string_debt_addition', 200000) + '，下周五之前把它送到我办公室。要是办不到，就等着受罪吧。"'
	else
		konvert = 0
		gs 'money', 'earn', 5000, 'cash'
		'你把信封交给他，他警告你说，以后再这样拖延可能会影响你们的合作关系。'
	end
	act '离开': gt $loc
	exit
end
