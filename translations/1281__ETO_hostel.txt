# ETO_hostel
CLOSE ALL

if $ARGS[0] = 'town_hostel':
	gs 'core_library', 'setloc', $curloc, $ARGS[0]
	$location_type = 'private'
	*clr & cla
	minut += 15
	gs 'stat'

	$func('wrap', 'header center b', '城镇旅馆')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/hostel/dom_gor.jpg"></center>'
	'旅馆里你的房间。条件很差，但总比没有强。院子里有一根晾衣绳和一个存放杂物的小棚子。奇怪的是，尽管周围似乎没有什么住户，旅馆却挤得满满的。'

	if hostel['status'] > 0: act '去你的房间': gt 'dom_gor'
	if hostel['status'] = 0: act '见管理员': gt 'comendant', 'pos1'
	if hostel['status'] = 2: act '寻找管理员': gt 'comendant', 'pos3'
	if hostel['status'] >= 3: act '检查棚屋': gt 'ETO_hostel', 'shed'

	act '离开': gt 'pav_commercial'
end


if $ARGS[0] = 'shared_bathroom':
	gs 'core_library', 'setloc', $curloc, $ARGS[0]
	$location_type = 'bathroom'
	*clr & cla
	minut += 10
	gs 'stat'

	$func('wrap', 'header center b', '共用浴室')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/hostel/comm_vann.jpg"></center>'
	'浴室状况糟糕透顶：墙壁肮脏，地上乱七八糟，管道老旧，几乎没什么水流。管道里飘出一股奇怪的气味——要是你更聪明一点，或许能明白那是什么。'

	gs 'din_van', 'private'

	if pcs_intel > 50 and hostel['status'] = 1: act '检查管道': gt 'ETO_hostel', 'rusty_pipes'

	act '回到你的房间': gt 'dom_gor'
end


if $ARGS[0] = 'rusty_pipes':
	gs 'core_library', 'setloc', $curloc, $ARGS[0]
	*clr & cla
	minut += 10
	hostel['status'] = 2
	gs 'stat'
	$func('wrap', 'header center b', '生锈的管道')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/hostel/comm_tru.jpg"></center>'
	'所有管道都锈穿了，按照建筑规范这明显是违规的。而且，管道还散发着一股化学品的味道。'

	act '回到你的房间': gt 'dom_gor'
end


if $ARGS[0] = 'shed':
	gs 'core_library', 'setloc', $curloc, $ARGS[0]
	*clr & cla
	minut += 10

	if hostel['status'] = 3: hostel['status'] = 4

	gs 'stat'
	$func('wrap', 'header center b', '旅馆棚屋')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/hostel/gor_saray.jpg"></center>'
	'除了堆积如山的垃圾，地板上还有化学品。其中一个罐子上，你看到化学危险标志和"河豚毒素"的字样。最好别碰它。'
	'嗯……危险化学品不该出现在这种地方。你应该想办法多了解这种物质。'

	act '回到你的房间': gt 'dom_gor'
end


if $ARGS[0] = 'bathroom':
	gs 'core_library', 'setloc', $curloc, $ARGS[0]
	$location_type = 'bathroom'
	*clr & cla
	gs 'stat'
	$func('wrap', 'header center b', '共用浴室')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/hostel/vann_gor.jpg"></center>'
	'宿舍里的浴室。'

	gs 'din_van', 'private'

	act '回你的房间': gt 'dom_gor'
end
