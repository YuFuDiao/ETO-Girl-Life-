!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_bar — $func('stat_display', 'helper_bar',                │
!! │   $gradient, $name, value [, max, phantom, overlay,             │
!! │   $overlay_gradient, $thresholds])                              │
!! │ Reads sd['render_mode'] and sd['name_side'].                    │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_bar':
	!! render_mode offset: sd uses 0/1/2, progressbar ARGS[8] expects 1/2/3
	sd_hb['rm'] = sd['render_mode'] + 1
	!! ── text-only mode (render_mode = 2) ────────────────────────────
	if sd['render_mode'] = 2:
		$result = $ARGS[2] + ': ' + $func('progressbar', $ARGS[1], ARGS[3], ARGS[4], ARGS[5], ARGS[6], $ARGS[7], '', $ARGS[8], sd_hb['rm'])
	!! ── bar modes ───────────────────────────────────────────────────
	else
		if sd['name_side'] = 1:
			$result = '<tr><td nowrap style="font-size: <<sd_font_pct>>%;">&nbsp;' + $ARGS[2] + '&nbsp;</td><td>' + $func('progressbar', $ARGS[1], ARGS[3], ARGS[4], ARGS[5], ARGS[6], $ARGS[7], '', $ARGS[8], sd_hb['rm']) + '</td></tr>'
		else
			$result = '<tr><td>' + $func('progressbar', $ARGS[1], ARGS[3], ARGS[4], ARGS[5], ARGS[6], $ARGS[7], '', $ARGS[8], sd_hb['rm']) + '</td><td nowrap style="font-size: <<sd_font_pct>>%;">&nbsp;' + $ARGS[2] + '</td></tr>'
		end
	end
	killvar 'sd_hb'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_bar_cell — like helper_bar but returns two <td> cells    │
!! │ Used by skills/relations multi-column layouts.                  │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_bar_cell':
	$sd_bc['bar'] = $func('progressbar', $ARGS[1], ARGS[3], ARGS[4], ARGS[5], ARGS[6], $ARGS[7], '', $ARGS[8], sd['render_mode'] + 1)
	if sd['name_side'] = 1:
		$result = '<td nowrap style="font-size: <<sd_font_pct>>%;">&nbsp;' + $ARGS[2] + '&nbsp;</td><td>' + $sd_bc['bar'] + '</td>'
	else
		$result = '<td>' + $sd_bc['bar'] + '</td><td nowrap style="font-size: <<sd_font_pct>>%;">&nbsp;' + $ARGS[2] + '</td>'
	end
	killvar 'sd_bc'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_skill_entry($name, value, cell_index)                    │
!! │ Mode 0=text, 1+=bars. Multi-column via sd['skills_columns'].    │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_skill_entry':
	sd_se['cols'] = sd['skills_columns']
	$sd_se['name'] = $ARGS[1]
	if stat_cfg['skills_mode'] > 0:
		!! Bar mode
		if ARGS[3] mod sd_se['cols'] = 0: $result = '<tr>'
		$result += $func('stat_display', 'helper_bar_cell', 'smooth_positive', $sd_se['name'], ARGS[2])
		if ARGS[3] mod sd_se['cols'] = (sd_se['cols'] - 1): $result += '</tr>'
	else
		!! Text mode
		if ARGS[3] mod sd_se['cols'] = 0: $result = '<tr>'
		$result += '<td style="font-size: <<sd_font_pct>>%;">' + $str(ARGS[2]) + ' ' + $sd_se['name'] + '</td>'
		if ARGS[3] mod sd_se['cols'] = (sd_se['cols'] - 1): $result += '</tr>'
	end
	killvar 'sd_se'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_rel_entry($name, value, cell_index)                      │
!! │ Mode 0=text, 1+=bars. Multi-column, optional color.             │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_rel_entry':
	sd_re['cols'] = sd['relations_columns']
	!! Resolve display name
	$sd_re['dname'] = $ARGS[1]
	if stat_cfg['relations_mode'] > 0:
		!! Bar mode
		if stat_cfg['rel_color_mode'] = 1:
			$sd_re['color'] = $func('progressbar', 'color', 'smooth_positive', ARGS[2], '10,35,60,80')
			!! Inlined themes.wrap_hex (one-line function, not worth a call here)
			$sd_re['dname'] = '<font color="' + $sd_re['color'] + '">' + $sd_re['dname'] + '</font>'
		end
		if ARGS[3] mod sd_re['cols'] = 0: $result = '<tr>'
		$result += $func('stat_display', 'helper_bar_cell', 'smooth_positive', $sd_re['dname'], ARGS[2])
		if ARGS[3] mod sd_re['cols'] = (sd_re['cols'] - 1): $result += '</tr>'
	else
		!! Text mode
		if ARGS[3] mod sd_re['cols'] = 0: $result = '<tr>'
		if stat_cfg['rel_color_mode'] = 1:
			$sd_re['color'] = $func('progressbar', 'color', 'smooth_positive', ARGS[2], '10,35,60,80')
			!! Inlined themes.wrap_hex (one-line function, not worth a call here)
			$result += '<td style="font-size: <<sd_font_pct>>%;"><font color="' + $sd_re['color'] + '">' + $sd_re['dname'] + ': ' + $str(ARGS[2]) + '</font></td>'
		else
			$result += '<td style="font-size: <<sd_font_pct>>%;">' + $sd_re['dname'] + ': ' + $str(ARGS[2]) + '</td>'
		end
		if ARGS[3] mod sd_re['cols'] = (sd_re['cols'] - 1): $result += '</tr>'
	end
	killvar 'sd_re'
	exit
end

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_rel_name($npc_key [, $npcid]) — NPC display name         │
!! │ Priority: per-NPC override > global mode > fallback chain.      │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_rel_name':
	$sd_hn['key'] = $ARGS[1]
	!! Use explicit NPC ID if provided, otherwise the key IS the NPC ID
	if $ARGS[2] <> '':
		$sd_hn['npcid'] = $ARGS[2]
	else
		$sd_hn['npcid'] = $sd_hn['key']
	end
	!! Determine effective mode: per-NPC override > global
	sd_hn['mode'] = stat_cfg['rel_name_mode']
	if $sd_hn['key'] <> '' and sd_rel_name_override[$sd_hn['key']] > 0:
		sd_hn['mode'] = sd_rel_name_override[$sd_hn['key']]
	end
	if sd_hn['mode'] = 1:
		$result = $npc_firstname[$sd_hn['npcid']]
	elseif sd_hn['mode'] = 2:
		$result = $npc_lastname[$sd_hn['npcid']]
		if $result = '': $result = $npc_firstname[$sd_hn['npcid']]
	elseif sd_hn['mode'] = 3:
		$result = $npc_nickname[$sd_hn['npcid']]
		if $result = '': $result = $npc_firstname[$sd_hn['npcid']]
	else
		$result = $npc_usedname[$sd_hn['npcid']]
	end
	!! Fallback chain: usedname > firstname > nickname > lastname
	if $result = '': $result = $npc_usedname[$sd_hn['npcid']]
	if $result = '': $result = $npc_firstname[$sd_hn['npcid']]
	if $result = '': $result = $npc_nickname[$sd_hn['npcid']]
	if $result = '': $result = $npc_lastname[$sd_hn['npcid']]
	if $result = '': $result = $sd_hn['key']
	killvar 'sd_hn'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_resolve_align — resolve section alignment                │
!! ├─────────────────────────────────────────────────────────────────┤
!! │ func('stat_display', 'helper_resolve_align', $section_key)      │
!! │ Returns: 0=left, 1=center, 2=right  (numeric result)            │
!! │ Per-section values: 0=Global, 1=Left, 2=Center, 3=Right.        │
!! │ Default (0) inherits from stat_cfg['global_align'].             │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_resolve_align':
	sd_ha['sec'] = stat_cfg['align_' + $ARGS[1]]
	if sd_ha['sec'] = 0:
		!! 0 = Global: inherit from global_align (0=left, 1=center, 2=right)
		result = stat_cfg['global_align']
	elseif sd_ha['sec'] = 2:
		!! 2 = Center
		result = 1
	elseif sd_ha['sec'] = 3:
		!! 3 = Right
		result = 2
	else
		!! 1 = Left
		result = 0
	end
	killvar 'sd_ha'
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_align_wrap — wrap content with alignment markup          │
!! ├─────────────────────────────────────────────────────────────────┤
!! │ $func('stat_display','helper_align_wrap',$section_key/align     │
!! │ , $content)                                                     │
!! │ ARGS[1] = 0=left (no-op), 1=center, 2=right                     │
!! │ Returns content wrapped, or unchanged if align=0 or empty.      │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_align_wrap':
	if $ARGS[1] <> '':
		temp_align = func('stat_display', 'helper_resolve_align', $ARGS[1])
	else
		temp_align = ARGS[1]
	end
	if temp_align = 1 and $ARGS[2] <> '':
		$result = '<center>' + $ARGS[2] + '</center>'
	elseif temp_align = 2 and $ARGS[2] <> '':
		!! Use a full-width wrapper table so block-level tables (progress bars etc.) are also pushed right
		$result = '<table width="100%" cellpadding="0" cellspacing="0"><tr><td align="right">' + $ARGS[2] + '</td></tr></table>'
	else
		$result = $ARGS[2]
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_font_wrap($html) — wrap non-table section content in a   │
!! │ block div with an explicit font-size. No-op when font_size=100  │
!! │ or $html is empty. Use for non-table sections; table cells use  │
!! │ $sd_font_wrap_o/c (or an explicit td style) directly.           │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_font_wrap':
	if sd_font_pct <> 100 and $ARGS[1] <> '':
		$result = '<div style="font-size: <<sd_font_pct>>%;">' + $ARGS[1] + '</div>'
	else
		$result = $ARGS[1]
	end
	exit
end


!! ┌─────────────────────────────────────────────────────────────────┐
!! │ helper_section_header — prepend section title to $result        │
!! ├─────────────────────────────────────────────────────────────────┤
!! │ $func('stat_display', 'helper_section_header',                  │
!! │       $section_key, $result_html)                               │
!! │ Returns: header HTML + $result_html, or $result_html unchanged  │
!! │          if sec_headers = 0 or $result_html = ''                │
!! │ Alignment: inherits from helper_resolve_align($section_key).    │
!! │ Header text sourced from $sd_dn[key].                           │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'helper_section_header':
	!! Alignment creates break, so no extra br needed
	$sd_sh['align'] = func('stat_display', 'helper_resolve_align', $ARGS[1])
	$temp_br = iif(sd_font_pct <> 100, '', '<br>')
	if stat_cfg['sec_headers'] = 0:
		$result = $ARGS[2]
	elseif stat_cfg['sec_headers'] = 2:
		!! Collapsible mode: toggle link always shown; body appended only when non-empty
		$sd_sh['lbl'] = $func('stat_display', 'helper_toggle', $ARGS[1], '&nbsp;' + $sd_dn[$ARGS[1]])
		$result = $func('stat_display', 'helper_font_wrap', $sd_sh['lbl'])
		if $ARGS[2] <> '': $result += $temp_br + $ARGS[2]
		killvar 'sd_sh'
	else
		!! Labels mode (sec_headers = 1): only shown when content is non-empty
		if $ARGS[2] = '':
			$result = ''
		else
			$result = $func('stat_display', 'helper_font_wrap', '<b>' + $sd_dn[$ARGS[1]] + '</b>') + $temp_br + $ARGS[2]
		end
	end
	$result = $func('stat_display', 'helper_align_wrap', $ARGS[1], $result)
	exit
end

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ get_separator($prev_type, $cur_type)                            │
!! │ separator_style: 0=matrix  1=<br>  2=<hr>                       │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'get_separator':
	if stat_cfg['separator_style'] = 1:
		$result = '<br style="font-size: <<sd_font_pct>>%;">'
	elseif stat_cfg['separator_style'] = 2:
		$result = '<hr style="margin:12px 10px;">'
	else
		!! ── Mode 0 (default): intelligent content-type matrix ───────
		!! structured ↔ structured → plain line break
		if ($ARGS[1] = 'bars' or $ARGS[1] = 'table') and ($ARGS[2] = 'bars' or $ARGS[2] = 'table'):
			$result = '<br style="font-size: <<sd_font_pct>>%;">'
		!! structured ↔ anything else → visual divider
		elseif ($ARGS[1] = 'bars' or $ARGS[1] = 'table') or ($ARGS[2] = 'bars' or $ARGS[2] = 'table'):
			$result = '<hr style="margin:<<sep_margin_px>>px 10px;">'
		!! everything else → small margin gap (safe with block elements)
		else
			$result = '<div style="margin:<<sep_margin_px>>px 0;"></div>'
		end
	end
	exit
end

!! ┌─────────────────────────────────────────────────────────────────┐
!! │ sec_weather — weather visual + temperature                      │
!! │ weather_visual: 0=image  1=text.  temp_pos: 0=R 1=↓ 2=↑ 3=L     │
!! └─────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'sec_weather':
	!! ── Build the temperature string (scale-aware) ──────────────────
	$sd_w['temp'] = $func('outdoors', 'get_temp_string', temper[0], temper[1], stat_cfg['temp_scale'])

	!! ── Build the weather visual (image or text description) ────────
	if stat_cfg['weather_visual'] = 1:
		!! Text mode: same theme color as the temperature, no underline, no trailing period
		if temper[0] < 0 or (temper[0] = 0 and temper[1] < 0):
			!! temper[1] < 0 signals a -0.x°C reading truncation would otherwise
			!! render sign-less (temper[0] = 0 has no way to show "-0" alone)
			$sd_w['theme'] = 'accent'
		elseif temper[0] < 10:
			$sd_w['theme'] = 'v_pos'
		elseif temper[0] < 20:
			$sd_w['theme'] = 'pos'
		elseif temper[0] < 30:
			$sd_w['theme'] = 'neg'
		else
			$sd_w['theme'] = 'v_neg'
		end
		!! Escape double-quotes inside $weather (e.g. from font color attrs)
		!! so they don''t break the href attribute delimiter.
		$sd_w['safe_weather'] = $replace($weather, '"', '&quot;')
		$sd_w['visual'] = '<a href="exec: msg ''<<$sd_w[''safe_weather'']>>''" style="text-decoration:none;">' + $func('wrap', $sd_w['theme'], $mid($osadki, 1, len($osadki) - 1)) + '</a>'
	else
		!! Image mode (default)
		$sd_w['safe_weather'] = $replace($weather, '"', '&quot;')
		$sd_w['visual'] = '<a href="exec: msg ''<<$sd_w[''safe_weather'']>>''">' + $weatherImage + '</a>'
	end

	!! ── Cache alignment once (used for table attr and below/above wrap) ──
	sd_w['center'] = func('stat_display', 'helper_resolve_align', 'weather')

	!! ── Layout visual and temperature by temp_pos ──────────────────
	!! Resolve align attr once for table modes (text-align on a div doesn''t move block tables)
	$sd_w['align_attr'] = iif(sd_w['center'] = 1, ' align="center"', iif(sd_w['center'] = 2, ' align="right"', ''))
	if stat_cfg['temp_pos'] = 1:
		!! below: visual on its own line, temp underneath
		$result = $sd_w['visual'] + '<br>' + $sd_w['temp']
	elseif stat_cfg['temp_pos'] = 2:
		!! above: temp first, then visual
		$result = $sd_w['temp'] + '<br>' + $sd_w['visual']
	elseif stat_cfg['temp_pos'] = 3:
		!! left: temp in left column, visual in right column
		$result = '<table' + $sd_w['align_attr'] + '><tr><td style="font-size: <<sd_font_pct>>%;">' + $sd_w['temp'] + '</td><td style="font-size: <<sd_font_pct>>%;">' + $sd_w['visual'] + '</td></tr></table>'
	else
		!! right (default, 0): visual in left column, temp in right column
		$result = '<table' + $sd_w['align_attr'] + '><tr><td style="font-size: <<sd_font_pct>>%;">' + $sd_w['visual'] + '</td><td style="font-size: <<sd_font_pct>>%;">' + $sd_w['temp'] + '</td></tr></table>'
	end

	$result = $func('stat_display', 'helper_font_wrap', $result)

	!! ── Optional alignment (below/above layouts only) ───────────────
	!! These are inline/text content, so helper_align_wrap (text-align) works correctly
	if stat_cfg['temp_pos'] = 1 or stat_cfg['temp_pos'] = 2:
		$result = $func('stat_display', 'helper_align_wrap', 'weather', $result)
	end

	killvar 'sd_w'
	exit
end

			if $sd_si['cond'] <> '':
				if $sd_si['ann'] <> '': $sd_si['ann'] += '<br>'
				$sd_si['ann'] += '<small>' + $sd_si['cond'] + '</small>'
			end

			!! ── Image tag ────────────────────────────────────────────
			$sd_si['img'] = '<a href="exec: view ''<<$sd_si[''url'']>>''"><img <<$sd_si[''size'']>> src="<<$sd_si[''url'']>>"></a>'

			!! ── One self-contained flow cell per image ───────────────
			$sd_si['cell'] = '<div style="display:inline-block; vertical-align:top; text-align:center; margin:<<sd_si[''cellspacing'']>>px;">'
			if sd_si['ann_pos'] = 0 and $sd_si['ann'] <> '':
				$sd_si['cell'] += $sd_font_wrap_o + $sd_si['ann'] + $sd_font_wrap_c + '<br>'
			end
			$sd_si['cell'] += $sd_si['img']
			if sd_si['ann_pos'] = 1 and $sd_si['ann'] <> '':
				$sd_si['cell'] += '<br>' + $sd_font_wrap_o + $sd_si['ann'] + $sd_font_wrap_c
			end
			$sd_si['cell'] += '</div>'
			$result += $sd_si['cell']

			sd_si['ci'] += 1
		if sd_si['ci'] < sd_img['count']: jump 'sd_si_wrap_loop'
		$result += '</div>'

		!! ── Apply section alignment (left/center/right) ─────────────────
		$result = $func('stat_display', 'helper_align_wrap', 'images', $result)

		killvar 'sd_si'
		exit
	end

	!! ── Build table output ───────────────────────────────────────────
	if sd_si['fit_width'] = 1:
		$result = '<table width="100%" cellspacing="<<sd_si[''cellspacing'']>>" cellpadding="<<sd_si[''cellpadding'']>>">'
		$sd_si['td_width'] = ' width="' + $str(100 / sd_si['cols']) + '%"'
	else
		$result = '<table cellspacing="<<sd_si[''cellspacing'']>>" cellpadding="<<sd_si[''cellpadding'']>>">'
		$sd_si['td_width'] = ''
	end

	!! ── Whether annotation (header + condition) goes above or below ──
	sd_si['ann_pos'] = stat_cfg['image_header_pos']

	sd_si['ci'] = 0
	$sd_si['img_cells'] = ''
	$sd_si['ann_cells'] = ''
	sd_si['has_ann'] = 0
	:sd_si_cell_loop

		$sd_si['key'] = $sd_img['order_' + $str(sd_si['ci'])]
		$sd_si['url'] = $sd_img['url_' + $sd_si['key']]

		!! ── Header text ─────────────────────────────────────────────
		$sd_si['hdr'] = ''
		if stat_cfg['image_headers'] = 1:
			$sd_si['hdr'] = $sd_img['label_' + $sd_si['key']]
		elseif stat_cfg['image_headers'] >= 2:
			!! Mode 2 (Item name) and 3 (Name & info) both use the short name
			$sd_si['hdr'] = $sd_img['short_' + $sd_si['key']]
			$sd_si['hdr'] = $replace($sd_si['hdr'], 'number ', '#')
			if $mid($sd_si['hdr'], len($sd_si['hdr']), 1) = '.':
				$sd_si['hdr'] = $mid($sd_si['hdr'], 1, len($sd_si['hdr']) - 1)
			end
		end
		if $sd_si['hdr'] <> '':
			if stat_cfg['image_header_font'] = 0:
				$sd_si['hdr'] = '<small><<$sd_si[''hdr'']>></small>'
			elseif stat_cfg['image_header_font'] = 2:
				$sd_si['hdr'] = '<b><<$sd_si[''hdr'']>></b>'
			end
		end

		!! ── Condition text (only in mode 3: Name & info) ─────────────
		$sd_si['cond'] = ''
		if stat_cfg['image_headers'] = 3:
			$sd_si['cond'] = $sd_img['cond_' + $sd_si['key']]
		end

		!! ── Build annotation cell content (header + cond combined) ───
		$sd_si['ann'] = ''
		if $sd_si['hdr'] <> '':
			$sd_si['ann'] = $sd_si['hdr']
		end
		if $sd_si['cond'] <> '':
			if $sd_si['ann'] <> '': $sd_si['ann'] += '<br>'
			$sd_si['ann'] += '<small>' + $sd_si['cond'] + '</small>'
		end

		!! ── Image tag ───────────────────────────────────────────────
		$sd_si['img'] = '<a href="exec: view ''<<$sd_si[''url'']>>''"><img <<$sd_si[''size'']>> src="<<$sd_si[''url'']>>"></a>'

		!! ── Accumulate cells for this image-row ─────────────────────
		$sd_si['img_cells'] += '<td valign="middle" align="center"<<$sd_si[''td_width'']>>>' + $sd_si['img'] + '</td>'
		$sd_si['ann_cells'] += '<td valign="top"    align="center"<<$sd_si[''td_width'']>>>' + $sd_font_wrap_o + $sd_si['ann'] + $sd_font_wrap_c + '</td>'
		if $sd_si['ann'] <> '': sd_si['has_ann'] = 1

		!! ── Flush completed image-row ────────────────────────────────
		if (sd_si['ci'] + 1) mod sd_si['cols'] = 0 or sd_si['ci'] + 1 >= sd_img['count']:
			!! Annotation above
			if sd_si['ann_pos'] = 0 and sd_si['has_ann'] = 1:
				$result += '<tr>' + $sd_si['ann_cells'] + '</tr>'
			end
			!! Images
			$result += '<tr>' + $sd_si['img_cells'] + '</tr>'
			!! Annotation below
			if sd_si['ann_pos'] = 1 and sd_si['has_ann'] = 1:
				$result += '<tr>' + $sd_si['ann_cells'] + '</tr>'
			end
			$sd_si['img_cells'] = ''
			$sd_si['ann_cells'] = ''
			sd_si['has_ann'] = 0
		end

		sd_si['ci'] += 1
	if sd_si['ci'] < sd_img['count']: jump 'sd_si_cell_loop'
	$result += '</table>'

	!! ── Wrap in <center> if alignment = center ──────────────────────
	$result = $func('stat_display', 'helper_align_wrap', 'images', $result)

	killvar 'sd_si'
	exit
end
