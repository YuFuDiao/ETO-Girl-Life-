# Serpent

menu_off = 0
gs 'core_library', 'setloc', 'Serpent', ''
$location_type = 'secluded'

if findneck = 1:
	*clr & cla
	gs 'stat'
	'<center><video autoplay loop <<$set_imgh>> src="images/locations/gadukino/serpent/serpentintro.mp4"></video></center>'
	'你听到谷仓某个角落传来一句低语。“啊——我看看，小小的精灵凡人竟然找到了我的魂器。你手中握着的，是我最伟大的珍宝。”你环顾四周想找到说话的人，但谷仓里除了你空无一人。'
	'"我们做个交易吧，你和我。你把我的宝贝还给我，我就好好教你许多关于你所拥有的魔法。"你听到头顶阴暗的房梁上传来一阵刮擦声。'
	'"你是谁？你在哪里？"你大声喊道，寻找着这个奇怪的嘶嘶声的主人。'
	'"我是Zzathrezzz，众蛇之王。"声音继续从上方传来，你看到一条大蛇从阴影中现身。谷仓外，你听到Bandit开始大声吠叫。Zathrez恼怒地嘶嘶作响。'
	'"那条狗真是个麻烦！"'
	'<center><img <<$set_imgh>> src="images/locations/gadukino/serpent/serpent1_0.jpg"></center>'
	'"那么，我们成交吗？"嘶嘶的声音继续说道。'

	gs 'willpower', 'misc', 'self'
	if pcs_willpwr < will_cost:
		act '不，离我远点你这变态蛇' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '不，离我远点你这变态蛇' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'misc', 'self', 'hard'
			gs 'willpower', 'pay', 'self'
			serpent_path = 0
			gt 'gad_gpyard', 'start'
		end
	end

	gs 'willpower', 'skill', 'magik', 'self', 'medium'
	if pcs_willpwr < will_cost:
		act '也许，但先告诉我你能教我什么' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '也许，但先告诉我你能教我什么' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'skill', 'magik', 'self', 'medium'
			gs 'willpower', 'pay', 'self'
			serpent_path = 1
			gt 'serpent'
		end
	end

	act '你僵在原地，看着蛇头向你滑来': gt $loc
end

if serpent_path = 1:
	*clr & cla
	gs 'stat'
end
