		'你还没有银行账户！'

			'你的账户里有 ' + $func('money', 'format', karta - bankDebtLimit) + '。'
			'你的透支额度为 ' + $func('wrap', 'accent',$func('money', 'format', bankDebtLimit) + '！')
		else
			'你已透支 ' + $func('wrap', 'neg',$func('money', 'format', bankDebtLimit - karta) + '。')
			'你剩余的信用额度为 ' + $func('wrap', 'accent',$func('money', 'format', karta) + '！')
		end

		'<br>ATM存款手续费： ' + $func('money', 'string_price', 100)

		if money > 0:
			act '将钱存入你的银行账户':
				cla
				menu_off = 1
				minut += 5

				gs 'bank', 'deposit_cash', func('money', 'price', 100)

				gs 'stat'
				'交易完成后，ATM 轻轻响了一声。屏幕回到欢迎界面。'
				act '离开 ATM': minut += 1 & gt 'shop', 'main'
			end
		end

		if karta <= 0:
			'你的透支额已经用完了。'
		else
			act '从自动取款机取钱':
				cla
				menu_off = 1
				minut += 5

				gs 'bank', 'withdraw_cash', 0

				gs 'stat'
				'交易完成后，ATM 轻轻响了一声。屏幕回到欢迎界面。'
				act '离开 ATM': minut += 1 & gt 'shop', 'main'
			end
			if karta >= 1000:
				act '取款 ' + $func('money', 'format', 1000) + '不需要买狗粮。你父母会付钱。':
					cla
					menu_off = 1
					karta -= 1000
					money += 1000
					gs 'bank', 'set_withdraw_text', 1000
					gs 'stat'
					'交易完成后，ATM 轻轻响了一声。屏幕回到欢迎界面。'
					act '离开 ATM': minut += 1 & gt 'shop', 'main'
				end
			end
		end
	end

	act '走开': minut += 1 & gt 'shop', 'main'
end
