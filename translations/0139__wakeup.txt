# wakeup

!! Deals with all the triggers that happen when waking up.
!!
!! Moved here from sleep or sleep_events
!! When done: Move to wakeup_events

if $ARGS[0] = 'start' or $ARGS[0] = 'wake' or $ARGS[0] = '':
	menu_off = 1
	inSleep = 1
	sleepVars['stat_display'] = 1
	gs 'stat'

	sleepVars['no_sleep_loss'] = 1

	xgt 'wakeup', 'mod_sleeptriggers'
end


if $ARGS[0] = 'mod_sleeptriggers':
	!! This location is here to allow mods to hook into the system.
	!! Check for: if $ARGS[0] = 'pre_sleep' and $ARGS[1] = 'mod_sleeptriggers'
	!!
	!! This is NOT for events!!

	gs 'mod_system', 'sleep', 'wakeup', 'mod_sleeptriggers'
	xgt 'wakeup_events', 'start'
end


if $ARGS[0] = 'get_out':
	menu_off = 1
	droutine['woke_at_min'] = totminut
	sleepVars['time_now'] = daystart * 1440 + hour * 60 + minut
	if sleepVars['slept_in'] = 1:
		'你醒来时是<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>，比计划多睡了一会儿，不过至少你睡得很充足。'
	elseif pcs_sleep >= 100:
		if alarmVars['alarmOn'] = 1:
			if sleepVars['time_now'] < sleepVars['alarm_time'] - 15:
				'你醒来时是<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>，远在闹钟响起之前。你已经睡得很充足了。'
			else
				'你醒来时是<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>，就在闹钟响起之前。你已经睡得很充足了。'
			end
		else
			'你醒来时是<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>，已不再疲惫，睡得很充足，准备好开始新的一天了。'
		end
	elseif alarmVars['alarmOn'] = 1 and sleepVars['time_now'] = sleepVars['alarm_time']:
		gs 'mood', 'lower', 'tiny'
		'你的闹钟在<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>，'+iif(pcs_sleep < 90, '但你还能再睡一会儿。', '而且你已经睡得很充足了。')+''
		gs 'wakeup', 'snooze_alarm'
	else
		!! I do not think that it could go here
		''
	end

	if kanikuli = 0 and $start_type['loc'] = 'sg' and gschoolVars['school_diploma'] = 0 and gschoolVars['block'] = 0 and week < 6 and hour >= 4 and hour < 9:
		act '起床穿好校服 (0:10)':
			minut += 10
			gs 'wardrobe', 'school_outfit'
			xgt 'bed_get_out', 'start'
		end
	end

	if strip_here = 1:
		act '起床穿好衣服 (0:10)':
			minut += 10
			gs 'wakeup', 'wear_bed_clothes'
			xgt 'bed_get_out', 'start'
		end
	end

	act '起床 (0:05)':
		minut += 5
		sleepVars['slept_in'] = 0
		xgt 'bed_get_out', 'start'
	end

	if alarmVars['alarmOn'] = 1 and sleepVars['time_now'] < sleepVars['alarm_time']:
		temp_hour = (sleepVars['alarm_time'] - sleepVars['time_now']) / 60
		temp_minut = (sleepVars['alarm_time'] - sleepVars['time_now']) mod 60
		act '在闹钟响前再睡一会儿 (<<temp_hour>>:<<$mid(100+temp_minut, 2)>>)':
			cla
			'你在床上翻了个身，闭上了眼睛。'
			*nl
			gs 'sleep_simple', 'forced', sleepVars['alarm_time'] - sleepVars['time_now']
			xgt 'wakeup', 'start'
		end
		killvar 'temp_hour'
		killvar 'temp_minut'
	end
end


if $ARGS[0] = 'snooze_alarm':
	act '"再睡几分钟"…':
		menu_off = 1
		sleepVars['slept_in'] = 1
		gs 'stat'
		gt 'sleep', 'start'
	end
end


if $ARGS[0] = 'wear_bed_clothes':
	gs 'outfit', 'restore', 'bed'
end


--- wakeup ---------------------------------
