if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_tsar_bomba', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '沙皇炸弹（Tsar Bomba）')
	'<center><img <<$set_imgh>> src="images/locations/pushkin/bomba/shop.jpg"></center>'
	*nl
	'沙皇炸弹（Tsar Bomba），奥赫洛普科夫广场上一家小小的服装店。它相当独特，过去曾以朋克风格而酷劲十足，但现在普通了许多，很容易被误认为是一家另类时尚连锁店。'
	'简单来说，这个地方非常酷，而且大部分衣服虽然奇怪，却妙不可言。'

	act '离开沙皇炸弹（Tsar Bomba）': minut += 3 & gt 'pushkin_sq'

	act '查看连衣裙':
		minut += 5
		gt 'shop_tsar_bomba', 'dresses'
	end
	act '查看服装':
		minut += 5
		gt 'shop_tsar_bomba', 'clothes'
	end
	act '查看鞋子':
		minut += 5
		gt 'shop_tsar_bomba', 'shoes'
	end
	act '查看钱包':
		minut += 5
		gt 'shop_tsar_bomba', 'purses'
	end
	act '查看外套':
		minut += 5
		gt 'shop_tsar_bomba', 'coats'
	end
end


if $ARGS[0] = 'dresses':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_tsar_bomba', 'dresses'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看沙皇炸弹（Tsar Bomba）连衣裙')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'bomba_dress'
			gs 'shop_utils', 'init', 'set_steps', 2, Clothingstock mod 2
			gs 'clothing_view', 'init_sorted'
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_tsar_bomba', 'start'
	end
end
