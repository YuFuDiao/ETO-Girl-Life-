# rasputin_entrance

music_loop = 0
CLOSE ALL
$menu_loc = 'rasputin_entrance'
$menu_arg = ''
menu_off = 0
$loc_arg = ''
$loc = 'rasputin_entrance'
$location_type = 'public_outdoors'
gs 'stat'

if hour >= 2 and hour < 15:
	'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/building_day.jpg"></center>'
	'正门已经关闭，如果你在这里工作，可以走员工通道。'

	act	'返回广场':
		if rasputin['burlesque_ticket'] = 2 or rasputin['variety_ticket'] = 2:
			'你将无法再回来观看演出的下半场。你确定要离开吗？'
			act '是':
				if rasputin['burlesque_ticket'] = 2: rasputin['burlesque_ticket'] = 0
				if rasputin['variety_ticket'] = 2: rasputin['variety_ticket'] = 0
				minut += 1
				gt 'pushkin_sq'
			end
			act '否': gt 'rasputin_entrance'
		else
			minut += 1
			gt 'pushkin_sq'
		end
	end

	act	'前往员工通道':
		if rasputin['work'] = 1:
		!! Not currently possible to trigger as you cannot work here. Location does not exist
			minut += 1
			gt 'rasputin_room_staff'
		else
			msg '你不在这里工作，没有理由进去。'
			minut += 1
			gt 'rasputin_entrance'
		end
	end
else
	'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/building_night.jpg"></center>'
	'俱乐部正在营业。'

	act	'返回广场':
		minut += 1
		gt 'pushkin_sq'
	end

	act	'去见女迎宾':
		minut += 1
		gt 'rasputin_host', 'start'
	end

	act	'进入俱乐部':
		minut += 1
		gt 'rasputin_walkway'
	end

	if rasputin['work'] = 1:
		act	'前往员工休息室':

			if rasputin['work'] = 1:
				minut += 1
				gt 'rasputin_room_staff'
			end
		end
	end
end
