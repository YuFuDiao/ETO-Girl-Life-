!! Used to get price of an outfit uses the variable "price"
!! $ARGS[1] set to outfit type
!! ARGS[2] for outfit number
!! ARGS[3] to skip attribute set
if $ARGS[0] = 'get_price':
	if ARGS[3] = 0 and $ARGS[3] = '': gs 'clothing_attributes', $ARGS[1], ARGS[2]
	result = (CloPrice * ((5 * CloQuality) + 100) / 100) * 1000 / (1250 - Clothingstock[ARGS[2]]) * 3 / 2
	result = result / 50 * 50
	exit
end

!! $ARGS[1] = location
!!  ARGS[2] = what to do with the clothes:
!!			0 : return to wardrobe
!!			1 : wear the clothes
!!			2 : Return to shop (lost FOREVAH!)
if $ARGS[0] = 'recover_lost_clothes':
	$cloLoc = $ARGS[1]
	cloAction = ARGS[2]

	$clothType = $CloLosTyp[$cloLoc]
	clothNumber = CloLosNum[$cloLoc]
	dayLost = CloLosDay[$cloLoc]

	killvar 'CloLos<<$clothingworntype>>', clothingwornnumber
	findLocInd = arrpos('$CloLosTyp', $clothType)
	killvar 'CloLosTyp', findLocInd
	findLocInd = arrpos('CloLosNum', clothNumber)
	killvar 'CloLosNum', findLocInd
	findLocInd = arrpos('CloLosDay', CloLosDay[$cloLoc])
	killvar 'CloLosDay', findLocInd
	findLocInd = arrpos('$CloLosLoc', $cloLoc)
	killvar 'CloLosLoc', findLocInd

	if cloAction = 1:
		gs 'clothing', 'wear', $clothType, clothNumber
	elseif cloAction = 2 :
		dynamic $clothType + '_s[<<clothNumber>>] = 1'
	end

	killvar 'findLocInd'
	killvar 'cloLoc'
	killvar 'cloAction'
	killvar 'clothType'
	killvar 'clothNumber'
	exit
end
