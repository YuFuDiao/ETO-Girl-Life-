if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_gm', 'start'
	$location_type = 'event'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '欢迎来到 G & M ——满足你一切衣着需求')
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/gandm/shop.jpg"></center>'
	'G&M服饰并不花哨，当然算不上时髦，但它很便宜，而且有专门的工作服和校服可选。'

	act '离开商店':
		$shoplocation = ''
		if torg = 1:
			minut += 3
			gt 'city_mall'
		else
			gt 'pav_industrial'
		end
	end
	act '查看服装部': gt 'shop_gm', 'clothing_department'

	act '查看鞋子': minut += 5 & gt 'shop_gm', 'shoes'
	act '查看外套': minut += 5 & gt 'shop_gm', 'coats'
	act '查看钱包': minut += 5 & gt 'shop_gm', 'purses'
	act '查看内裤': minut += 5 & gt 'shop_gm', 'panties'
	act '查看胸罩': minut += 5 & gt 'shop_gm', 'bras'
end

if $ARGS[0] = 'clothes':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_gm', 'clothes'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '正在浏览 G&M 套装')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'gm_outfits'
			gs 'shop_utils', 'init', 'set_steps', 2, (Clothingstock / 6) mod 2
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_gm', 'clothing_department'
	end
end
