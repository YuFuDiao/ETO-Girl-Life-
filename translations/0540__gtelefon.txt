# gtelefon


menu_off = 1
boyinday[numnpcboy] = daystart
Gcall = 0
minut += 3
gs 'stat'
'你拿起电话，"喂？"'

if boyvstre[numnpcboy] > 0:
	'电话那头传来一个男声："嗨，<<$pcs_nickname>>！是我，<<$nameBoyfrend[numnpcboy]>>。我们出去走走吧。"'
else
	'电话那头传来一个男声："你好，我找<<$pcs_nickname>>？"'
	'你告诉来电者你是<<$pcs_nickname>>。对方听起来很高兴。"嗨！我是<<$nameBoyfrend[numnpcboy]>>，我们前几天见过面。还记得吗？我们出去走走吧。"'
end

act '告诉他你今天没时间':
	cls
	minut += 3

	if boyvstre[numnpcboy] = 0: otnBoyFrend[numnpcboy] -= 10
	if boyvstre[numnpcboy] > 0: otnBoyFrend[numnpcboy] -= 5

	gs 'stat'
	''

	if otnBoyFrend[numnpcboy] > 20:
		!! otnBoyFrend[numnpcboy] = 0
		'<<$nameBoyfrend[numnpcboy]>>听起来很失落。"我知道了。那好吧，也许下次吧。"'
	else
		otnBoyFrend[numnpcboy] = 0
		'<<$nameBoyfrend[numnpcboy]>>听起来很生气。"随便你，去你的吧！"他说完立刻挂断了电话。你觉得他不会再联系你了。'
	end

	act '挂断': gt 'korrPar'
end

if GvstreBoy = 0:
	act '在公园见面':
		cls
		minut += 3
		GvstreBoy = 1
		GvstreBoyAfter[numnpcboy] = 1
		SkverBoy = numnpcboy
		gs 'stat'
		'你同意在公园和他见面。'

		act '挂断': gt 'korrPar'
	end
end
