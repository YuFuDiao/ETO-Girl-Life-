# KGZfight
minut += 1
gs 'stat'
gs 'KGZdyn'

if KGZVars['HP'] <= 0:
	*clr & cla
	gs 'mood', 'lower', 'tiny'
	'<center><b>游戏结束</b></center>'

	act '退出': gt 'KGZstart'

	exit
end

if KGZVars['dead'] <= 0:
	cla
	KGZVars['batl'] = 1
	gs 'mood', 'raise', 'tiny'
	'你消灭了所有的僵尸。'

	act '离开': gt 'KGZgame'

	exit
end

if KGZVars['dist'] < 0: KGZVars['dist'] = 0

!! '<a href="exec:gt ''KGZstart''">Exit the game</a>'
'生命<<KGZVars[''HP'']>>。第<<KGZVars[''day'']>>天，小时<<KGZVars[''hour'']>>。'
'利刃武器：<<$KGZVars[''weapon_text'']>>'

if $KGZVars['gun_text'] ! '':	*pl $KGZVars['gun_text']
if $KGZVars['gun2_text'] ! '':	*pl $KGZVars['gun2_text']
if $KGZVars['gun3_text'] ! '':	*pl $KGZVars['gun3_text']

'<<KGZVars[''dead'']>>僵尸们正朝你走来。它们离你<<KGZVars[''dist'']>>米远。'

if KGZVars['dost'] = 1:
	KGZVars['dead'] -= 1
	'Vincent 用步枪开火，一个僵尸头上多了个洞，倒在了地上。'

	if KGZVars['dead'] <= 0: gt $curloc
end

if KGZVars['gun'] > 0 and KGZVars['bullet'] > 0:
	act '用手枪射击':
		cla
		KGZVars['bullet'] -= 1
		gs 'KGZdyn', 'KGZgunatk'
		gs 'KGZdyn', 'KGZdeadmanatk'

		act '继续': gt $curloc
	end
end


if KGZVars['gun2'] > 0 and KGZVars['bullet2'] > 0:
	act '用霰弹枪射击':
		cla
		KGZVars['bullet2'] -= 1
		gs 'KGZdyn', 'KGZgunatk2'
		gs 'KGZdyn', 'KGZdeadmanatk'

		act '继续': gt $curloc
	end
end

if KGZVars['gun3'] > 0 and KGZVars['bullet3'] > 0:
	act '用机枪射击（单发）':
		cla
		KGZVars['bullet3'] -= 1
		gs 'KGZdyn', 'KGZgunatk3'
		gs 'KGZdyn', 'KGZdeadmanatk'

		act '继续': gt $curloc
	end

	if KGZVars['bullet3'] >= 3:
		act '用机枪射击（短点射）':
			cla
			KGZVars['bullet3'] -= 3
			KGZVars['bulgun'] = 3
			gs 'KGZdyn', 'KGZgunatk4'
			gs 'KGZdyn', 'KGZdeadmanatk'

			act '继续': gt $curloc
		end
	end

	if KGZVars['bullet3'] >= 10:
		act '用机枪射击（连发）':
			cla
			KGZVars['bullet3'] -= 10
			KGZVars['bulgun'] = 10
			gs 'KGZdyn', 'KGZgunatk5'
			gs 'KGZdyn', 'KGZdeadmanatk'

			act '继续': gt $curloc
		end
	end
end

if KGZVars['granade'] > 0:
	act '扔手榴弹':
		cla
		gs 'KGZdyn', 'KGZgranatk'

		act '继续': gt $curloc
	end
end

if KGZVars['dist'] <= 2:
	act '用<<$KGZVars[''weapon_texta'']>>':
		cla
		gs 'KGZdyn', 'KGZfizatk'

		act '继续': gt $curloc
	end
else
	act '等待':
		cla
		KGZVars['dist'] -= 1
		gt $curloc
	end

	act '靠近':
		cla
		KGZVars['dist'] -= 2
		gt $curloc
	end

	act '跑开':
		cla

		if KGZVars['dead'] >= 10:
			KGZVars['dist'] -= 5
			'僵尸们包围了你。你无法逃脱。'

			act '操！': gt $curloc
		else
			KGZVars['dedrandrun'] = rand(0, 1)

			if KGZVars['dedrandrun'] = 0:
				KGZVars['batl'] = 1
				'你设法从僵尸群中逃了出来。'

				act '离开': gt 'KGZgame'
			elseif KGZVars['dedrandrun'] = 1:
				KGZVars['dist'] -= 5
				KGZVars['dead'] += rand(KGZVars['dead'], KGZVars['dead'] * 5)
				'你小心翼翼地靠近，却惊动了更多的僵尸。'

				act '操！': gt $curloc
			end
		end
	end
end
