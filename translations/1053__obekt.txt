# obekt

if $ARGS[0] = '':
	*clr & cla
	menu_off = 0
	gs 'core_library', 'setloc', 'obekt', ''
	$location_type = 'secluded'
	gs 'stat'

	'<center><h4>建筑工地</h4></center>'
	'<center><img <<$set_imgh>> src="images/locations/construction/obekt.jpg"></center>'
	'施工已经停止，四周一片寂静，只有警卫在那里来回走动。'

	if func('car_funcs', 'is_here'):
		' <a href="exec:gs ''carF'', ''start''"> <<$car[''name'']>></a>停在停车场里。'
	end

	act '去路上': minut += 15 & gt 'road', '16'

	act '潜入施工现场':
		cla
		menu_off = 1
		minut += 5
		gs 'stat'

		if obekt['open'] = 0:
			if obekt['bil'] = 0:
				obekt['bil'] = 1
				*nl
				'你还没来得及进去，就听到一个声音问你在做什么。是警卫抓住了你。他指了指你应该改去的方向。'

				act '离开': gt $loc, $loc_arg
			else
				gt 'obekt', 'obekt2'
			end
		end
	end
end


if $ARGS[0] = 'obekt2':

	$location_type = 'secluded'

	!}

if $ARGS[0] = 'dungeon2':
	gs 'stat'

	'<center><img <<$set_imgh>> src="images/pre/shared/dunpre2.jpg"></center>'
	'破败的走廊，在凹进处可以看到通往其他房间的门。'

	act '离开走廊': minut += 5 & gt 'obekt', 'dungeon1'

	if obekt['svitokRTS'] = 1: exit

	act '进入1号房间': minut += 5 & gt 'obekt', 'dungeon3'

	act '进入2号房间':
		*clr & cla

		if obekt['openstateDM1'] = 1:
			minut += 5 & gt 'obekt', 'dungeon4'
		elseif obekt['openstateDM1'] = 0:
			:markanotherrand
			$obekt['DMN'] = '<<$mid($str(10000 + rand(0, 9999)), 2, 4)>>'
			if $obekt['DMN'] = '<<$mid($obekt[''DMN''], 2)>><<$mid($obekt[''DMN''], 1, 1)>>': jump 'markanotherrand'
			if $obekt['DMN'] = '8809': jump 'markanotherrand'
			if $obekt['DMN'] = '7111': jump 'markanotherrand'
			if $obekt['DMN'] = '2172': jump 'markanotherrand'
			if $obekt['DMN'] = '3213': jump 'markanotherrand'
			if $obekt['DMN'] = '7662': jump 'markanotherrand'
			if $obekt['DMN'] = '9312': jump 'markanotherrand'
			if $obekt['DMN'] = '8193': jump 'markanotherrand'
			if $obekt['DMN'] = '8096': jump 'markanotherrand'
			if $obekt['DMN'] = '7756': jump 'markanotherrand'
			if $obekt['DMN'] = '6855': jump 'markanotherrand'
			if $obekt['DMN'] = '9881': jump 'markanotherrand'
			if $obekt['DMN'] = '5531': jump 'markanotherrand'

			obekt['i'] = 1
			obekt['numdmr'] = 0
			:markdnm
				if strpos('069', $mid($obekt['DMN'], obekt['i'], 1)) > 0:
					obekt['numdmr'] += 1
				elseif $mid($obekt['DMN'], obekt['i'], 1) = '8':
					obekt['numdmr'] += 2
				end
				obekt['i'] += 1
			if obekt['i'] < 5: jump 'markdnm'

			obekt['numberOSDM'] = input("<<$obekt['DMN']>> >= ?")

			if obekt['numberOSDM'] < 0 or obekt['numberOSDM'] >= 9 or obekt['numberOSDM'] <> obekt['numdmr']:
				msg ("一片寂静，什么都没有改变")
			else
				obekt['openstateDM1'] = 1
				msg ("门里传来咔哒一声。")
			end
			gt 'obekt', 'dungeon2'
		end
	end

	if obekt['openstateDM2'] = 2: act '在第三个房间签名': minut += 5 & gt 'obekt', 'dungeon5'
end

if $ARGS[0] = 'dungeon4':
	'<center><img <<$set_imgh>> src="images/pre/shared/dunpre4.jpg"></center>'
	'小房间里有一些<a href="exec: gt ''obekt'', ''dungeon4_drums''">鼓</a>.'

	act '离开房间': minut += 5 & gt 'obekt', 'dungeon2'

	if obekt['openstateDM2'] = 1:
		'在另一个桶里，你看到一个可以输入数字的装置。'

		act '输入数字':
			*clr & cla
			obekt['numberOSDM2'] = input("12, 1, 1, 1, 2, 1, 3, ?, ?, ?, ?")

			if obekt['numberOSDM2'] = 1415:
				obekt['openstateDM2'] = 2
				msg ("外面传来一阵嘎嘎声。")
			else
				msg ("一片寂静，什么都没有改变")
			end
			gs 'array', 'remove_element', 'dungeon', 'numberOSDM2'

			gt 'obekt', 'dungeon4'
		end
	end
