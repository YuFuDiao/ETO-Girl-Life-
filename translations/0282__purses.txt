!! ARGS[2] for outfit number
!! ARGS[3] to skip attribute set
if $ARGS[0] = 'not_wear_reason':
	if arrsize('ARGS') = 1: $ARGS[1] = $currentpursetype
	if arrsize('ARGS') = 2: ARGS[2] = currentpursenumber
	if arrsize('ARGS') = 3: gs 'purse_attributes', $ARGS[1] + '_purses', ARGS[2]

	$result = ''
	if func('purses', 'is_immutable', $ARGS[1], ARGS[2]): exit
	if func('purses', 'is_owned', $ARGS[1], ARGS[2]) = 0:
		$result = 'not_owned'
		exit
	end
	if func('purses', 'in_wardrobe', $ARGS[1], ARGS[2]) = 0:
		$result = 'not_in_wardrobe'
		exit
	end
	exit
end


if $ARGS[0] = 'can_wear':

	result = ($func('purses', 'not_wear_reason', $ARGS[1], ARGS[2], 'attributes_set') = '')
	exit

if $ARGS[0] = 'add_item':
	if $ARGS[1] = '': exit
	if ARGS[2] = 0: exit

	gs 'purse_attributes', $ARGS[1], ARGS[2]
	if PursePrice = 0: exit

		<<$ARGS[1]>>_purses[<<ARGS[2]>>] = 1
	"
	exit
end


if $ARGS[0] = 'remove_item':
	if $ARGS[1] = '': $ARGS[1] = $currentpursetype
	if ARGS[2] = 0: ARGS[2] = currentpursenumber

	if $ARGS[1] = '' or $ARGS[1] = 'none': exit

	dynamic "
		<<$ARGS[1]>>_purses[<<ARGS[2]>>] = 0
	"

	if $ARGS[1] = $currentpursetype and ARGS[2] = currentpursenumber:
		gs 'purses', 'strip'
		gs 'outfit', 'set_derived_vars'
		$lastwornpursetype = 'none'
		lastwornpursenumber = 0
	end
	exit
end

if $ARGS[0] = 'strip':

		$currentpursetype = 'none'
		currentpursenumber = 0
	end

	purselastwornnumber = currentpursenumber

	exit

if $ARGS[0] = 'strip_code':
	$currentpursetype = 'none'
	currentpursenumber = 0

	gs 'purses', 'reset_PPurseVars'
	gs 'outfit', 'set_derived_vars'
	exit
end


if $ARGS[0] = 'reset_PurseVars':
	PurseQuality = 0
	PursePrice = 0
	PurseStrength = 0
	exit
end


if $ARGS[0] = 'reset_PPurseVars':
	bag = 0
	PPurseQuality = 0
	PPursePrice = 0
	PPurseStrength = 0
	exit
end


!! $ARGS[1] - shoe type name / last worn
!! $ARGS[2] - shoe index
if $ARGS[0] = 'wear':
	if $ARGS[1] = 'last_worn':
		if $purselastworntype = '':
			$purselastworntype = 'none'
			purselastwornnumber = 0
		end
		$ARGS[1] = $purselastworntype
		ARGS[2] = purselastwornnumber
	end
	if $ARGS[1] = '' or $ARGS[1] = 'none': exit

	gs 'purses', 'strip'

	gs 'purse_attributes', $ARGS[1], ARGS[2]

	if PursePrice = 0:
		msg '错误：手提包 "<<$ARGS[1]>>_purses[<<ARGS[2]>>]" 不存在'
		exit
	end

	if arrpos('$ARGS', 'check') > 0:
		$temp_not_wear_reason = $func('purses', 'not_wear_reason', $ARGS[1], ARGS[2], 'attributes_set')
		if $temp_not_wear_reason <> '':
			if $temp_not_wear_reason = 'not_owned':
				msg '你没有这个手提包'
			elseif $temp_not_wear_reason = 'not_in_wardrobe':
				msg '这个手提包不在你的衣柜里'
			end
			killvar '$temp_not_wear_reason'
			exit
		end
		killvar '$temp_not_wear_reason'
	end

	$currentpursetype = $ARGS[1]
	currentpursenumber = ARGS[2]

	bag = 1
	PPurseQuality = PurseQuality
	PPursePrice = PursePrice

	gs 'outfit', 'set_derived_vars'
	exit
end
