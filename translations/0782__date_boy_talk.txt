				cla & *nl
				'"经常，"你咧嘴一笑。"这是我慢跑时最喜欢的环节。"'
				if $npc_humor[$npcID] = 'childish':
					'"真性感，"他咧嘴笑道。'
				else
					'"嗯，如果那是你的话，我不记得了，"他咧嘴笑道。"老实说，那种时候我根本没在看她的脸。"'
				end
				act '继续': gt 'date_talk', 'continue_rate', 'improve'
			end
		end
	end
