	!! Evaluate NPC preferences across multiple categories
	!! Body preferences
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'body_tits', 'body_ass', 'body_bmi', 'body_skin', 'body_lips'

	!! Positive body preferences increase limit
	if $ngpPrefResult['HasPos'] ! '':
		sex_ev['haggle_limit'] += rand(2,5)
	end
	!! Negative body preferences decrease limit
	if $ngpPrefResult['HasNeg'] ! '':
		sex_ev['haggle_limit'] -= rand(3,6)
	end

	!! Clothing and appearance preferences
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'clothes_quality', 'clothes_cleavage', 'shoes_heels'

	if $ngpPrefResult['HasPos'] ! '':
		sex_ev['haggle_limit'] += rand(1, 3)
	end
	if $ngpPrefResult['HasNeg'] ! '':
		sex_ev['haggle_limit'] -= rand(2,4)
	end

	!! Cosmetics preferences
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'cosmetics_piercings', 'cosmetics_tattoos'

	if $ngpPrefResult['HasPos'] ! '':
		sex_ev['haggle_limit'] += rand(1, 3)
	end
	if $ngpPrefResult['HasNeg'] ! '':
		sex_ev['haggle_limit'] -= rand(1, 3)
	end

	!! Hair preferences
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'hair_color', 'hair_length', 'hair_pube_length'

	if $ngpPrefResult['HasPos'] ! '':
		sex_ev['haggle_limit'] += rand(1, 2)
	end
	if $ngpPrefResult['HasNeg'] ! '':
		sex_ev['haggle_limit'] -= rand(1, 2)
	end

	!! NPC personality modifiers
	!! Generous NPCs are more willing to pay
	if npc_generous[$npcID] = 1:
		sex_ev['haggle_limit'] += rand(3,6)
	end

	!! Selfish NPCs are less willing
	if npc_selfish[$npcID] = 1:
		sex_ev['haggle_limit'] -= rand(3,6)
	end


	!! Ensure minimum and maximum bounds
	if sex_ev['haggle_limit'] < 10: sex_ev['haggle_limit'] = 10
	if sex_ev['haggle_limit'] > 50: sex_ev['haggle_limit'] = 50

	!! Clean up temporary variables
	killvar 'ngpPrefResult'

	result = sex_ev['haggle_limit']
end
