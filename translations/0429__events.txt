				minut += 5
				drugUser = 1
				gs 'drugs', 'heroin'
				gs 'stat'
				'<center><b>巴甫洛夫斯克火车站</b></center>'
				'<center><img <<$set_imgh>> src="images/locations/pavlovsk/trainstation/event/usingheroin.jpg"></center>'
