if $ARGS[0] = 'weekly grade update':
	if kanikuli = 0 and week = 1 and gschoolVars['school_diploma'] = 0:
		if lernHome > 0 and petkaQW['homework'] = 1: gs 'grades', 'homework', 'school', 'yes', rand(0, lernHome), 2, 'A6'
		trait_vars['academic_exp'] -= min(lernHome, 5)
		if lernHome <= 1:
			trait_vars['academic_exp'] += trait_vars['nerd_learn_home']
			if trait_vars['nerd_learn_home'] < 5:
				trait_vars['nerd_learn_home'] += 1
			else
				trait_vars['nerd_learn_home'] = 0
			end
		end
		lernHome = 0
		if class['first_grade_check'] > 0:
			gs 'grades', 'calculate_grade', 'school'
			gs 'grades', 'assign_grade_description', 'school'
		else
			i = 0
			:grade_loop
			dynamic 'class[''school_<<$class_list_name[i]>>_weekly_grade_gain''] = 0'
			i += 1
			if arrsize('class_list_institution') > i: jump 'grade_loop'
		end
	end
	exit
end
