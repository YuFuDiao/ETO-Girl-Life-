		nichTempEval += 3
		nichChoreState[8] = 0
	elseif nichChoreState[8] = 0:
		if rand(1, 10) <= 3:
			'<font color = green>你今天把客厅打扫得很好。</font>'
			nichTempEval += 1
		end
	elseif nichChoreState[8] <= 5:
		'<font color = orange>客厅有点脏。看起来你今天没有打扫。我希望以后它能被打扫得干干净净。</font>'
		nichTempEval -= 5
	elseif nichChoreState[8] <= 15:
		'<font color = red>客厅的状况令人无法接受。</font>'
		nichTempEval -= 10
	else
		'<font color = red>客厅的状况糟透了。</font>'
		nichTempEval -= 20
	end

	!! Kitchen
	if nichChoreState[9] = -1:
		'<font color = green>我真的很喜欢你今天对银餐具的处理。</font>'
		nichTempEval += 3
		nichChoreState[9] = 0
	elseif nichChoreState[9] = 0:
		if rand(1, 10) <= 1:
			'<font color = green>你今天把厨房打扫得很好。</font>'
			nichTempEval += 1
		end
	elseif nichChoreState[9] <= 5:
		'<font color = orange>厨房有点脏。看起来你今天没有打扫。我希望以后它能被打扫得干干净净。</font>'
		nichTempEval -= 5
	elseif nichChoreState[9] <= 15:
		'<font color = red>厨房的状况令人无法接受。</font>'
		nichTempEval -= 10
	else
		'<font color = red>厨房的状况糟透了。</font>'
		nichTempEval -= 20
	end

	!! Study
	if nichChoreState[10] = 0:
		if rand(1, 10) <= 2:
			'<font color = green>你今天把书房打扫得很好。</font>'
			nichTempEval += 1
		end
	elseif nichChoreState[10] <= 5:
		'<font color = orange>书房有点脏。看起来你今天没有打扫。我希望以后它能被打扫得干干净净。</font>'
		nichTempEval -= 5
	elseif nichChoreState[10] <= 15:
		'<font color = red>书房的状况令人无法接受。</font>'
		nichTempEval -= 10
	else
		'<font color = red>书房的状况糟透了。</font>'
		nichTempEval -= 20
	end

	!! Influence of other characters
	if nichEvalGala = 1:
		'<font color = green>我妻子告诉我，你今天是个<i>好女孩</i>。</font>'
		nichTempEval += 1
		killvar 'nichEvalGala'
	elseif nichEvalGala = 2:
		'<font color = green>我妻子告诉我，你今天是个<i>非常好的女孩</i>。</font>'
		nichTempEval += 2
		killvar 'nichEvalGala'
	end

	if nichGalaOpponent >= 10 and nichGalaOpponent < 20:
		'<font color = red>我妻子告诉我你今天表现不佳。她没有告诉我你具体做了什么，但无论如何我都会考虑进去。</font>'
		'<i>看来加拉想通过贬低你的工作表现来赶走你。</i>'
		nichTempEval -= 5
	end

	!! Special events
	if nichGentleclubE1 > 0 and nichGentleclubE1 < 10 and nichGentleclubDayE1 < daystart:
		'<font color = red>我昨晚让你来陪我，但你没有出现。</font>'
		nichTempEval -= 10
		killvar 'nichGentleclubE1'
	end

	*nl
	if nichTempEval > 5:
		if nichPerformance >= 80:
			'总的来说，你今天表现出色，而且一直让我很满意。做得好。'
		elseif nichPerformance < 20:
			'总的来说，你今天表现出色。如果你继续保持，我愿意忘记过去的不足。'
		else
			'总的来说，你今天表现出色。做得好。'
		end
	elseif nichTempEval >= 0:
		if nichPerformance >= 80:
			'总的来说，你今天工作做得很好，而且一直让我很满意。做得好。'
		elseif nichPerformance < 20:
			'总的来说，你今天工作做得很好。如果你继续保持，我也许愿意忘记过去的不足。'
		else
			'总的来说，你今天工作做得很好。'
		end
	elseif nichTempEval >= -5:
		if nichPerformance >= 80:
			'总的来说，你今天没有完全达到我的期望。因为你通常表现良好，所以这不是什么大问题。'
		elseif nichPerformance < 20:
			'总的来说，你今天没有完全达到我的期望。我已经厌倦了对你的失望。'
		else
			'总的来说，你今天没有完全达到我的期望。你以后的表现必须提高。'
		end
	else
		'总的来说，你今天完全没有达到我的期望。'
	end

	if nichDebug = 1:
		'DEBUG: nichTempEval = <<nichTempEval>>'
	end

	nichPerformance += nichTempEval

	killvar 'nichTempEval'

	nichPerformance = min(100, max(0, nichPerformance))

	if $ARGS[1] = 'silent':
		*clr & cla
		!! note: the calling script has to make sure nichEvaluationLast is set to the right value
	else

		*nl
		if nichPerformance = 0:
			'我受够了你的懒惰。收拾好你的东西，你被解雇了！'
			*nl
			act '离开':
				gt 'nichUtil', 'fired'
			end
		else
			if nichPerformance <= 20:
				'总的来说，你表现不佳。如果你继续这样干下去，你在这里剩下的时间会很短。'
			elseif nichPerformance <= 40:
				'总的来说，我对你的表现不满意。如果你想保住这份工作，就必须改进。'
			elseif nichPerformance <= 60:
				'总的来说，你做得平平无奇。既谈不上批评，也谈不上表扬。'
			elseif nichPerformance <= 80:
				'总的来说，你表现不错，但仍有改进的空间。'
			else
				'总的来说，你表现得非常出色。我很高兴雇了你。'
			end

			if nichWorkState = 0 and daystart >= nickHired + 7:
				'过去这些天你工作得很好，足以通过试用期。你不用担心法律手续，我已经替你处理好了。'
				'从现在起，你不再需要每天来我这里进行工作表现评估了。当然，如果你想对你的工作进行详细分析，仍然可以来问我。'
				nichWorkState = 1
			end

			'今天就到这里。你现在可以走了。'
			nichEvaluationLast = daystart

			act '离开':
				gt $loc
			end
		end
	end
end
