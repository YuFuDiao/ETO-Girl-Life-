if $ARGS[0] = 'on_booking_miss':
	!! Miss handler invoked by daily_check for a missed booking-mode shift.
	!! gs 'pornschedule', 'on_booking_miss', $job_id, $booking_data
	!! $ARGS[2] = booking_data string holding the film type number set by do_contract.
	!! Sets pornmiss/missedshoot for the manager dialog and accumulates a debt
	!! sized by the film''s production cost.

	$temp_obm_job = $ARGS[1]
	$temp_obm_data = $ARGS[2]
	temp_obm_film_type = val($temp_obm_data)

	pornmiss = 1
	missedshoot = 0

	!! dynamic $pornfilmdesc sets pcash, pfactor, etc. for the given film type.
	if temp_obm_film_type > 0:
		dynamic $pornfilmdesc, temp_obm_film_type
		!! pcash * 10 stands in for studio overhead (crew, actors, equipment).
		temp_obm_cost = pcash * 10
	else
		!! Unknown film type — flat penalty.
		temp_obm_cost = 500
	end

	job_booking_debt[$temp_obm_job] += temp_obm_cost
	!! pornstudio.qsrc reads job_last_miss_cost when offering forgiveness.
	job_last_miss_cost[$temp_obm_job] = temp_obm_cost

	if job_booking_debt[$temp_obm_job] > job_penalty_max_debt[$temp_obm_job]:
		$job_termination_reason[$temp_obm_job] = 'blacklisted'
		gs 'jobs', 'set_fired', $temp_obm_job
		pfilmNO = 1
	end

	killvar 'temp_obm_job'
	killvar 'temp_obm_data'
	killvar 'temp_obm_film_type'
	killvar 'temp_obm_cost'
end

if $ARGS[0] = 'schedule':
	missedshoot = 0

	gs 'jobs', 'get_job_definition', 'city_pornstudio_actress'

	!! porndays = days until the player''s next booked shoot.
	temp_sched_next = func('jobs', 'get_next_booking_day', 'city_pornstudio_actress', daystart, job_booking_window_days['city_pornstudio_actress'])
	if temp_sched_next > 0:
		porndays = temp_sched_next - daystart
	else
		porndays = -1
	end
	killvar 'temp_sched_next'

	days = 0
	:fillschedule
		porns = 0
		:moveschedule
			if porntaken['<<days+1>>,<<porns>>'] = 0 and days >= 14:
				if rand(1, 100) = 1: pornplan['<<days+1>>,<<porns>>'] = 0
				!! Change of plans if there is no contract on the movie, shooting some other kind of porn instead
			end
			if days = job_booking_window_days['city_pornstudio_actress'] or pornplan['<<days+1>>,<<porns>>'] = 0:
			!! If there is a missing slot in the schedule, it fills in. It also regenerates the last selection on the end, so no endless repetition
				pornvariable = rand(0, 90)
				if pornvariable <= 1:
					!! 50 person
					pornplan['<<days+1>>,<<porns>>'] = 14
				elseif pornvariable <= 10:
					!! 4 person
					!! some way to randomize the scene
					pornplan['<<days+1>>,<<porns>>'] = rand(10, 13)
				elseif pornvariable <= 30:
					!! 2 person
					!! some way to randomize the scene
					pornplan['<<days+1>>,<<porns>>'] = rand(8, 9)
				elseif pornvariable <= 60:
					!! single person
					!! some way to randomize the scene
					pornplan['<<days+1>>,<<porns>>'] = rand(1, 7)
				elseif pornvariable <= 80:
					!! single person
					!! some way to randomize the scene
					pornplan['<<days+1>>,<<porns>>'] = rand(15, 20)
				else
					!! masturbation scene
					!! some way to randomize the scene - if there will be more than one(???)
					pornplan['<<days+1>>,<<porns>>'] = 15
				end
			end
			pornplan['<<days>>,<<porns>>'] = pornplan['<<days+1>>,<<porns>>']
			porntaken['<<days>>,<<porns>>'] = porntaken['<<days+1>>,<<porns>>']
			!! moving the shooting schedule and the contracts ahead by one day. "days = 0" is the current day
			if porntaken['<<days>>,<<porns>>'] = 0:
				if rand(1, days*3/2) = 1: porntaken['<<days>>,<<porns>>'] = 1
				!! some other porn actress contracted the movie. The more closer the shooting day is, the more likely, that all of them is taken
			end
			if porntaken['<<days>>,<<porns>>'] = 1:
				if rand(1, 100) = 1: porntaken['<<days>>,<<porns>>'] = 0
				!! some other porn actress contracted to the movie has canceled her contract. Not likely, but it happens.
			end
		if porns < 3: porns += 1 & jump 'moveschedule'
	if days < job_booking_window_days['city_pornstudio_actress']: days += 1 & jump 'fillschedule'
	!! Finally deleting the unnecessary (and the next day overwritten) schedule day to reduce array size
	temp_sched_tail = (job_booking_window_days['city_pornstudio_actress'] + 1) * 4
	killvar 'pornplan', temp_sched_tail + 3
	killvar 'pornplan', temp_sched_tail + 2
	killvar 'pornplan', temp_sched_tail + 1
	killvar 'pornplan', temp_sched_tail
	killvar 'temp_sched_tail'
end

	*p '<a href="exec:killvar ''datefilter'' & killvar ''selected'' & weekdayfilter = 0 & weekendfilter = 0 & gs ''pornschedule'', ''selection''">清除日期筛选</a>'
end
