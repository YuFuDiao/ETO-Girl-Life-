# bouling

if $ARGS[0] = 'start' or $ARGS[0] = '':
	$sexloc = 'bouling'
	*clr & cla
	$loc = 'bouling'
	$loc_arg = 'start'
	$menu_loc = 'bouling'
	$menu_arg = 'start'
	menu_off = 0
	$location_type = 'public_indoors'
	gs 'stat'
	'<center><b>保龄球场</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/bowling/boul.jpg"></center>'

	act '离开': gt 'city_mall'

	if hour >= 8 and hour <= 20:
		act '玩一局 (1:00)' + $func('money', 'get_cost_string', 300):
			if func('money', 'can_afford', 300) = 0:
				$noMoney
			else
				menu_off = 1
				minut += 60
				gs 'money', 'pay', 300
				boulvar = 0
				boulingnav += rand(0, 3)
				boulrand = rand(0, 9)
				gs 'npcgeneratec', 0, '保龄球手', rand(18, 40)
				$boy[0] = $npclastgenerated
				$boydesc[0] = $npc_usedname[$npclastgenerated]
				gs 'npcgeneratec', 0, '保龄球手', rand(18, 40)
				$boy[1] = $npclastgenerated
				$boydesc[1] = $npc_usedname[$npclastgenerated]
				gs 'stat'

				act '离开': gt 'bouling'

				if boulrand = 0:
					killvar 'boulrand'
					'你看到两个小伙子，他们向你发起一场比赛挑战。'
					act '赌一局 (' + $func('money', 'string_price', 2000) + ')': boulvar = 1 & gt 'bouling', 'randwin'
					gs 'willpower', 'sex', 'resist', 'hard'
					if pcs_willpwr < will_cost:
						act '拒绝，打普通的一局' + $func('willpower', 'get_willcost_string'): $noWillpower
					else
						act '拒绝，打普通的一局' + $func('willpower', 'get_willcost_string'):
							gs 'willpower', 'sex', 'resist', 'hard'
							gs 'willpower', 'pay', 'resist'
							gs 'stat'
							gt 'bouling', 'randwin'
						end
					end
				elseif boulrand = 1:
					killvar 'boulrand'
					'你遇到两个女孩，她们提议去挑战两个小伙子打一局：奖金 ' + $func('money', 'string_price', 5000) + '，输了他们就和你做爱。'
					act '开始': boulvar = 2 & gt 'bouling', 'randwin'
					gs 'willpower', 'sex', 'resist', 'hard'
					if pcs_willpwr < will_cost:
						act '拒绝，打普通的一局' + $func('willpower', 'get_willcost_string'): $noWillpower
					else
						act '拒绝，打普通的一局' + $func('willpower', 'get_willcost_string'):
							gs 'willpower', 'sex', 'resist', 'hard'
							gs 'willpower', 'pay', 'resist'
							gs 'stat'
							gt 'bouling', 'randwin'
						end
					end
				elseif boulrand > 1:
					killvar 'boulrand'
					gt 'bouling', 'randwin'
				end
			end
		end
	end
end


if $ARGS[0] = 'randwin':
	cla

	if boulingnav < 10:
		boulwin = rand(0, 10)
	elseif boulingnav < 20:
		boulwin = rand(0, 12)
	elseif boulingnav < 30:
		boulwin = rand(0, 14)
	elseif boulingnav < 40:
		boulwin = rand(0, 16)
	elseif boulingnav < 50:
		boulwin = rand(0, 18)
	elseif boulingnav < 60:
		boulwin = rand(0, 20)
	elseif boulingnav < 70:
		boulwin = rand(0, 22)
	elseif boulingnav < 80:
		boulwin = rand(0, 24)
	elseif boulingnav < 90:
		boulwin = rand(0, 26)
	else 
		boulwin = rand(0, 28)
	end

	if boulwin > 10:
		if boulvar = 0:
			gs 'money', 'earn', 300
			'你赢了，获得 ' + $func('money', 'string_profit', 300) + '。'
	
			act '离开': gt 'bouling'
		elseif boulvar = 1:
			gs 'money', 'earn', 2000
			'你赢了，获得 ' + $func('money', 'string_profit', 2000) + '。'
	
			act '离开': gt 'bouling'
		elseif boulvar = 2:
			gs 'money', 'earn', 5000
			'你赢了，获得 ' + $func('money', 'string_profit', 5000) + '。'
	
			act '离开': gt 'bouling'
		end
	else
		if boulvar = 0:
			gs 'money', 'pay', 300
			'你输了，支付 ' + $func('money', 'string_price', 300) + '。'
			act '离开': gt 'bouling'
		elseif boulvar = 1:
			picrand = 14
			if func('money', 'can_afford', 2000, 'cash') = 1:
				'你输了，必须支付 ' + $func('money', 'string_price', 2000) + '。'
				gs 'willpower', 'sex', 'self', 'hard'
				if pcs_willpwr < will_cost:
					act '改用身体抵债' + $func('willpower', 'get_willcost_string'): $noWillpower
				else
					act '改用身体抵债' + $func('willpower', 'get_willcost_string'):
						gs 'willpower', 'sex', 'self', 'hard'
						gs 'willpower', 'pay', 'self'
						gs 'stat'
						xgt 'sexdvoe', 'var'
					end
				end
				act '付款 (' + $func('money', 'string_price', 2000) + ')': gs 'money', 'pay', 2000, 'cash' & gt 'bouling'
			else
				'你输了，但你的钱不够付。只有一个办法了。'
				act '改用身体抵债': xgt 'sexdvoe', 'var'
			end
		elseif boulvar = 2:
			picrand = 11
			'你输了，把腿张开吧。'
			act '付款': gt 'sexorg', 'var'
		end
	end
	killvar 'boulwin'
end
