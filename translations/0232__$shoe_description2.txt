# $shoe_description2


if ShoQuality <= 1:
	$result = '这是一双劣质廉价的'
elseif ShoQuality = 2:
	$result = '这是一双品质低劣的'
elseif ShoQuality = 3:
	$result = '这是一双品质普通的'
elseif ShoQuality = 4:
	$result = '这是一双品质尚可的'
elseif ShoQuality = 5:
	$result = '这是一双品质上乘的'
elseif ShoQuality = 6:
	$result = '这是一双做工精良的'
else
	$result = '这是一双做工精美的'
end


if $ARGS[0] = 'danilovich':
	$result += '训练鞋。'
else
	if ShoHeels <= 1:
		$result += '平底'
	elseif ShoHeels = 2:
		$result += '低跟'
	elseif ShoHeels = 3:
		$result += '中跟'
	elseif ShoHeels = 4:
		$result += '高跟'
	elseif ShoHeels = 5:
		$result += '超高跟'
	else
		$result += '恋物高跟'
	end

	if ShoCut <= 1:
		$result += '鞋。'
	elseif ShoCut = 2:
		$result += '短靴。'
	elseif ShoCut = 3:
		$result += '中筒靴。'
	elseif ShoCut = 4:
		$result += '及膝靴。'
	elseif ShoCut = 5:
		$result += '过膝靴。'
	else
		$result += '及裆长靴。'
	end
end

if ShoStyle = 1:
	$result += ' 这些鞋子被视为另类鞋。'
end
if ShoStyle2 = 1 or ShoStrip = 1:
	$result += ' 这些鞋子被视为脱衣舞鞋。'
elseif ShoStyle2 = 2 or ShoSport = 1:
	$result += ' 这些鞋子被视为运动鞋。'
end

killvar 'temp_shoe_desc'
if ShoBimbo = 1:	$temp_shoe_desc[] = 'bimbo'
if ShoGoth = 1:		$temp_shoe_desc[] = 'goth'
if ShoPunk = 1:		$temp_shoe_desc[] = 'punk'

if arrsize('$temp_shoe_desc') > 0:
	$result += ' 这些鞋子被视为 '
	$result += func('string', 'enumerate_list', '$temp_shoe_desc')
	$result += ' 鞋。'
end
killvar 'temp_shoe_desc'
