# gdktoilet

if $ARGS[0] = 'start':
	menu_off = 0
	gs 'core_library', 'setloc', 'gdktoilet', 'start'
	$location_type = 'public_indoors'
	minut += 1
	ghnow = 0
	gs 'stat'
	gs 'themes', 'indoors'

	*clr & cla
	$func('wrap', 'header center b', '社区中心 - 洗手间')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/community/bathroom/bathroom_entrance.jpg"></center>'

	'洗手间的门旁有一个<a href="exec: gt ''beverage'', ''watercooler''">饮水机</a>。'

	act '出去':
		if $clothingworntype ! 'nude':
			gs 'arousal', 'end'
			gt 'pav_commcenter'
		else
			msg '<b><font color = red>你需要穿好衣服。</font></b>'
		end
	end

	if hour >= 8:
		gs 'willpower', 'exhib', 'self'
		if pcs_willpwr < will_cost:
			act '进入男厕所' + $func('willpower', 'get_willcost_string'): $noWillpower
		else
			act '进入男厕所' + $func('willpower', 'get_willcost_string'):
				gs 'willpower', 'exhib', 'self'
				gs 'willpower', 'pay', 'self'
				gs 'stat'
				gt 'gdktoilet_mens', 'mens_entrance_events'
			end
		end
		act '进入女洗手间': gt 'gdktoilet', 'womens'
	end

	if hour >= 20 and week >= 5 and week < 7:
		if sound_settings['music_off'] = 0:
			$track_loop = 'sound/komb.mp3'
			volume = 50
			music_loop = 1
		end
		if gdkincum = daystart and (cumloc[6] = 0 and cumloc[7] = 0) and cumloc[11] = 0:
			gdkincum = 0
		end
		if gdkincum ! daystart:
			if pav_disco_in = daystart:
				act '回到舞会': gt 'pav_disco'
			else
				act '去参加舞会' + $func('money', 'get_cost_string', 25):
					if func('money', 'can_afford', 25) = 0:
						$noMoney
					else
						gs 'money', 'pay', 25
						minut += 5
						gt 'pav_disco'
					end
				end
			end
		else
			*nl
			if (cumloc[6] = 1 or cumloc[7] = 1) and cumloc[11] = 1:
				'你应该在回舞会前把脸上和衣服上的精液清理干净。'
			elseif cumloc[11] = 1:
				'你应该在回舞会前把脸上的精液清理干净。'
			elseif cumloc[6] = 1 or cumloc[7] = 1:
				'你应该在回舞会前把衣服上的精液清理干净。'
			end
		end
	else
		music_loop = 0
		CLOSE ALL
	end

	if hour >= 8 and hour < 18: act '去图书馆': gt 'pav_library'
	if hour >= 12 and hour < 20: act '上楼去兴趣小组': gt 'pav_commclubs'
	if (week < 6 and hour >= 14 and hour < 20) or (week >= 6 and hour >= 9 and hour < 20): act '去健身房': gt 'gdksport', 'start'
	act '在更衣室换衣服': gt 'wardrobe', 'start'
end


if $ARGS[0] = 'womens':
	menu_off = 0
	gs 'core_library', 'setloc', 'gdktoilet', 'womens'
	$location_type = 'bathroom'
	$locclass = 'restroom'
	gs 'stat'
	*clr & cla
	$func('wrap', 'header center b', '女洗手间')
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/community/bathroom/womens/bathroom_women.jpg"></center>'
	'你走进女洗手间。你看到一些洗手池和<a href="exec:gt ''mirror'', ''start''">镜子</a>. 你注意到最后那个隔间的门大开着。'

	act '离开': gt 'gdktoilet', 'start'
	act '去最后一个隔间': gt 'gloryhole', 'start'

	gs 'din_van', 'tampon'
	gs 'din_van', 'quickwash'
	gs 'din_van', 'basin'
	gs 'din_van', 'publicpan'
end
