# gad_backwater

menu_off = 0
gs 'core_library', 'setloc', 'gad_backwater', ''
$location_type = 'secluded'
gs 'gadukino_event', 'sound'
clothesAtLocation = func('clothing', 'lost_clothes_here', 'gad_backwater')

if $ARGS[0] = '':
	if hunterVars['Rape'] = 2:
		clr & cla
		'<center><img <<$set_imgh>> src="images/locations/gadukino/sex/hunter/backwaterrelax.jpg"></center>'
		*nl '你坐在岸边，开始休息。'
		menu_off = 1
		minut += 5
		hunterVars['Rape'] = 3
		gs 'stat'

		act '进一步': gt 'hunters', 'start'
	else
		if $clothingworntype = 'nude' and hunterVars['outside'] = 1 and hunterVars['available'] = 1 and rand(1, 10) = 1: gt 'gad_backwater', 'hunter' else gt 'gad_backwater', 'alone'
	end
end


if $ARGS[0] = 'alone':
	'<center><b>死水</b></center>'
	if month >= 4 and month <= 10:
		'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/'+iif(DayStage < 4, 'backwater.jpg', 'backwater_night.jpg')+'"></center>'
	else
		'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/'+iif(DayStage < 4, 'backwater_winter.jpg', 'backwater_winter_night.jpg')+'"></center>'
	end
	*nl
	'一条水还算干净的小溪。如果天气暖和，你可以在这里清洗身体。另外，如果你的衣服脏了，也可以在这里洗。'
	*nl

	if clothesAtLocation = 1: '你的衣服躺在岸上。'


	act '回到小屋':
		minut += 5
		hunterVars['refuse'] = 0
		gt 'gad_swamp_yard', 'start'
	end


	if $clothingworntype = 'nude':
		act '洗澡':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/'+iif($pantyworntype = 'none', 'backwaterswim2.jpg', 'backwaterswim1.jpg')+'"></center>'
			*nl '你在小溪里洗了个澡。'
			menu_off = 1
			noshampoo = 1
			gs 'din_van', 'showerdin'
			face_write = 0
			body_write = 0
			minut += 30
			gs 'mood', 'raise', 'tiny'
			if willday_bath ! daystart: willday_bath = daystart & pcs_willpwr += 1
			inhib_exp += rand(1, 3)
			if hunterVars['Rape'] = 1: hunterVars['Rape'] = 2
			if deodorant_on = 1: gs 'sweat', 'remove_deo' & '<br>你的除臭剂被水冲走了。'
			act '完成': gt 'gad_backwater'
		end
	end
	if cumloc[11]+cumloc[12]+cumloc[16] > 0:
		act '清理脸上的精液 (0:10)':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/backwaterswim1.jpg"></center>'
			*nl '你在小溪里小心地洗掉精液。'
			menu_off = 1
			pcs_makeup = 1
			cumspclnt = 9
			gs 'cum_cleanup'
			cumspclnt = 13
			gs 'cum_cleanup'
			if pcs_sweat > 19 : pcs_sweat -= 5
			minut += 10
			if hunterVars['Rape'] = 1: hunterVars['Rape'] = 2
			gs 'stat'

			act '完成': gt 'gad_backwater'
		end
	end
	if body_write > 0 or face_write > 0:
		act '洗掉身上的字迹 (0:10)':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/shared/home/bathroom/wash1.jpg"></center>'
			*nl '你洗掉了身上那些不雅的字迹。'
			menu_off = 1
			body_write = 0
			face_write = 0
			minut += 10
			gs 'stat'

			act '完成': gt 'gad_backwater'
		end
	end
	if cumloc[2]+cumloc[5]+cumloc[6]+cumloc[7] > 0:
		act '擦掉衣服上的精液 (0:10)':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/cumwash.jpg"></center>'
			*nl '你把衣服上的精液污渍清洗干净。'
			menu_off = 1
			cumspclnt = 6
			gs 'cum_cleanup'
			minut += 10
			if hunterVars['Rape'] = 1: hunterVars['Rape'] = 2
			gs 'stat'

			act '完成': gt 'gad_backwater'
		end
	end
	if cumsumbod-cumsumvag-cumsumass > 0 or pcs_sweat > 20:
		act '快速清洗 (0:15)':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/pc/activities/facewash.jpg"></center>'
			*nl '你仔细地清洗身体，去除任何污垢。'
			menu_off = 1
			cumspclnt = 4
			gs 'cum_cleanup'
			if pcs_sweat < 30: pcs_sweat = 15 + rand(0, 4) else pcs_sweat -= 10 + rand(0, 4)
			minut += 15
			if hunterVars['Rape'] = 1: hunterVars['Rape'] = 2
			gs 'stat'
			act '完成': gt 'gad_backwater'
		end
	end

	if clothesAtLocation = 0:
		if $clothingworntype <> 'nude' and temper >= 15:
			act '脱衣服':
				*clr & cla
				menu_off = 1
				if hunterVars['KnowSlut'] > 0 or (hunterVars['outside'] = 0 and hunterVars['available'] = 0) or (hour < 7 or hour > 22) or (pcs_inhib >= 25 or trait_vars['exhibitionist'] > 1):
					'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/'+iif(PCloSkirt > 0, 'backwaterundress_ski.jpg', 'backwaterundress.jpg')+'"></center>'
					'你毫不犹豫地脱下衣服，毫不在意是否有人在看。'
					minut += 5
					if hunterVars['Rape'] = 1: hunterVars['Rape'] = 2
					gs 'outfit', 'strip_all', 'gad_backwater'

					act '返回': gt 'gad_backwater'
				else
					'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/noundress.jpg"></center>'
					'你犹豫着要不要脱衣服。如果有人突然看到你怎么办？'

					act '返回': gt 'gad_backwater'
				end
			end
		else
		'天太冷了，不能脱衣服。也许你应该在小屋里洗澡'
		end
	else
		if $clothingworntype <> 'nude':
			act '捡起你的衣服':
				gs 'outfit', 'recover_lost_outfit', 'gad_backwater'
				!! 'You pick up your clothes lying on the ground.'
				gt 'gad_backwater'
			end
		elseif forced_naked_day < daystart:
			act '穿好衣服':
				gs 'outfit', 'recover_lost_outfit', 'gad_backwater', 1
				gs 'underwear', 'wear'
				!! 'You pick up your clothes lying on the ground and get dressed.'
				gt 'gad_backwater'
			end
		end
	end

end


if $ARGS[0] = 'hunter':
	'<center><b>死水</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/hunter_looks.jpg"></center>'
	'当你在死水中清洗自己的时候，你突然感到这里并不只有你一个人。转过身，你看到一个猎人。他只是站在那里看着你，他的迷彩服和灌木丛、树木几乎难以分辨。'
	menu_off = 1
	minut += 5
	gs 'stat'

	if clothesAtLocation = 1:
		act '穿好衣服':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/swampbackwaternude1.jpg"></center>'
			*nl '注意到猎人，你冲向自己的衣服，惊慌失措地开始穿衣服。穿好衣服后，你回头张望，以为猎人还站在那里，但那里一个人也没有。'
			gs 'underwear', 'wear'
			gs 'outfit', 'recover_lost_outfit', 'gad_backwater', 1
			minut += 5

			act '进一步': gt 'gad_backwater'
		end
	else
		act '隐藏':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/swampbackwaternude2.jpg"></center>'
			'你捂住胸部和下体，跳进最近的灌木丛。在恐惧和慌乱中，你环顾四周，却没有看到任何人。于是你飞快地跑回小屋。'
			minut += 1

			act '回到小屋': gt 'gad_swamp_yard', 'start'
		end
	end
	act '保持裸体':
		*clr & cla
		'<center><img <<$set_imgh>> src="images/locations/gadukino/hunters/swampbackwaternude3.jpg"></center>'
		'你决定没什么好怕的，继续全裸着待着。猎人看了你大约一分钟，突然，你听到一声"哼"，他转身离开了。'
		hunterVars['slut'] += 1
		hunterVars['knowday'] = day
		gs 'stat'

		act '进一步': gt 'gad_backwater'
	end
end
