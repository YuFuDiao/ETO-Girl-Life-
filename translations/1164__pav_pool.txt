if $ARGS[0] = 'building' or $ARGS[0] = '':
	menu_off = 0
	gs 'core_library', 'setloc', 'pav_pool', 'building'
	$location_type = 'public_outdoors'
	gs 'stat'

	act '离开': gt 'pav_residential'

		'游泳池目前关闭。开放时间从 '+func('time', 'get_time_string', 8, 0)+' 到 '+func('time', 'get_time_string', 21, 0)+'。'
		exit
	end

	'门票价格是<<$func(''money'', ''string_price'', 100)>>全天，但学生有折扣，他们只需付<<$func(''money'', ''string_price'', 60)>>.'

		pav_swimpool['entrancefee'] = iif(gschoolVars['school_diploma'] <= 0, 60, 100)
	end

	if $job_status['pav_pool_lifeguard'] <> 'employed' and (job_hiring_step['pav_pool_lifeguard'] = 1 and week = 6 or job_hiring_step['pav_pool_lifeguard'] = 2 and week = 7):
		if hour = 8:
			'你走进大楼，首席救生员迎接了你，并立刻把你带到员工更衣室。'
			act '继续': gt 'pav_pool_lifeguard', 'training'
			exit
		else
			'你今天本应参加救生员训练于 ' + $func('time', 'get_time_string', 8, 0) + '！'
		end
	elseif $job_status['pav_pool_lifeguard'] = 'employed' and func('jobs', 'is_work_day', 'pav_pool_lifeguard', daystart) = 1 and job_last_work_day['pav_pool_lifeguard'] < daystart:
		if hour = 11:
			'人群开始排队入场，你走进去，前往员工更衣室为你的班次做准备。'
			act '等待上班 (<<iif(minut=0, ''1:00'', ''0:<<60-minut>>'')>>)':
				minut += (60 - minut)
				gt 'pav_pool_lifeguard', 'start_shift'
			end
			exit
		elseif hour = 12:
			'人群已经到齐了，你匆匆走进去，前往员工更衣室为你的班次做准备。你希望没人介意你迟到了。'
			act '去上班': gt 'pav_pool_lifeguard', 'start_shift'
			exit
		elseif hour > 12:
			'你错过了救生员值班！你应该在 ' + $func('time', 'get_time_string', 12, 0) + '！'
		end
	end

	if $job_status['pav_pool_lifeguard'] = 'employed':
		'因为你在这里工作，休息日可以免费入场。'
		act '进入': minut += 2 & gt 'pav_pool', 'entrance'
	elseif pav_swimpool['entrancepaid'] = daystart:
		'<b>你已经付过门票费，可以进去了。</b>'
		act '进入': minut += 2 & gt 'pav_pool', 'entrance'
	else
		act '付款并进入' + $func('money', 'get_cost_string', pav_swimpool['entrancefee']):

			else

				gs 'money', 'pay', pav_swimpool['entrancefee']
				pav_swimpool['entrancepaid'] = daystart
				gs 'stat'
				gt 'pav_pool', 'entrance'
			end
		end
	end
end

			gt 'pav_pool_events', 'locker_clothed_to_nude'

			!! Locker events clothing -> swim
			gt 'pav_pool_events', 'locker_clothed_to_swim'

				$temp_curclothes = 'nude'
			elseif PSwim = 1:
