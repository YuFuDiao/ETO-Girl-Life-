!! ║    'Ninel'     - City Dr. Ninel                                        ║
!! ║    'Petrovich' - Pavlovsk Gyno                                         ║
!! ║                                                                        ║
!! ║  Usage:                                                                ║
!! ║    gs 'clinic_functions', 'request_appointment', $category             ║
!! ║                                                                        ║
!! ║  Example:                                                              ║
!! ║    gs 'clinic_functions', 'request_appointment', 'Gynecologist'        ║
!! ║                                                                        ║

if $ARGS[0] = 'reception_option_label':
	!! ┌────────────────────────────────────────────────────────────────┐
	!! │                    reception_option_label                      │
	!! ├────────────────────────────────────────────────────────────────┤
	!! │ $result = func('clinic_functions', 'reception_option_label',   │
	!! │     $category, $subject)                                       │
	!! │                                                                │
	!! │ Act label for a reception category: a booking prompt if there  │
	!! │ is no appointment yet, or a bolded check-in/attend prompt with │
	!! │ the time once one is already on the books - it is misleading   │
	!! │ to invite the player to "book" a slot they already have. Dr.   │
	!! │ Pavlov is the recurring weekly therapist slot once             │
	!! │ hypnoSchedule = 1, not a one-off booking through the           │
	!! │ appointments module, hence the special case. Usable at any     │
	!! │ clinic.                                                        │
	!! └────────────────────────────────────────────────────────────────┘
	$temp_rol_recur = iif($ARGS[1] = 'Pavlov' and hypnoSchedule = 1, 'therapist_appointment', '')
	$temp_rol_state = $func('appointments', 'get_state', $ARGS[1], $temp_rol_recur)

	if $temp_rol_state = 'none':
		$result = '预约：' + $ARGS[2]
	else
		$temp_rol_id = iif($temp_rol_recur <> '', $temp_rol_recur, func('appointments', 'get_appointment_id', $ARGS[1]))
		$temp_rol_time = func('calendar_query', 'ts_to_str', func('appointments', 'get_event_display_ts', $temp_rol_id))
		if $temp_rol_state = 'pending':
			temp_rol_daystart = iif($temp_rol_recur <> '', daystart, event_daystart[$temp_rol_id])
			if temp_rol_daystart = daystart:
				$result = '查看与' + $ARGS[2] + '的预约（' + $temp_rol_time + '）'
			else
				$result = '查看与' + $ARGS[2] + '的预约（' + func('calendar_query', 'format_relative_day', temp_rol_daystart) + '）'
			end
		else
			$result = '<b>前往与' + $ARGS[2] + '的预约（' + $temp_rol_time + '）</b>'
		end
	end

	killvar 'temp_rol_recur'
	killvar 'temp_rol_state'
	killvar 'temp_rol_id'
	killvar 'temp_rol_time'
	killvar 'temp_rol_daystart'
	exit
end

if $ARGS[0] = 'request_appointment':
	!! ┌────────────────────────────────────────────────────────────────┐
	!! │                      request_appointment                       │
	!! ├────────────────────────────────────────────────────────────────┤
	!! │ gs 'clinic_functions', 'request_appointment', $category        │
	!! │                                                                │
	!! │ The 'appointments' render_acts 'none' handler - the player has │
	!! │ no appointment yet. Checks whether a same-day slot exists, then│
	!! │ hands off to 'request_appointment_offer' to show it alongside  │
	!! │ the regular scheduled slots - free on a small lucky roll, or   │
	!! │ available any time for a fee as an expedited appointment.      │
	!! └────────────────────────────────────────────────────────────────┘
	cla
	menu_off = 1
	gs 'stat'
	$appt_offer_intro = $appt_intro_line

	if $ARGS[1] = 'Pavlov' and hypnoSchedule = 1:
		!! Recurring weekly slot has locked in - no more ad-hoc booking, only Thursdays.
		'"巴甫洛夫医生只在周四接诊，时间是 ' + func('time', 'get_time_string', 18, 0) + ' 到 ' + func('time', 'get_time_string', 19, 0) + '，"前台告诉你。"到时候再来吧。"'
		act '离开': gt $loc, $loc_arg
		exit
	end

	$doctorname = func('clinic_functions', '_category_name', $ARGS[1])
	if $ARGS[1] = 'Pavlov':
		$doctortype = 'Therapist'
	elseif $ARGS[1] = 'Ninel':
		$doctortype = '全科医生'
	elseif $ARGS[1] = 'Petrovich':
		$doctortype = 'Gynecologist'
	else
		$doctortype = $ARGS[1]
	end
	$doc_category = $ARGS[1]
	$doc_loc = iif($region[1] <> 'pav', '城市诊所（City Clinic）', '巴甫洛夫斯克诊所（巴甫洛夫斯克诊所）')
	$doc_title = iif($doctorname = '', $doctortype, $doctorname) + ' Appointment'
	tmp_doc_seed = func('random', 'string_hash', $ARGS[1] + '_' + $region[1])
	tmp_doc_duration = 4
	tmp_doc_range_min = 27
	!!tmp_doc_range_min = iif($start_type['loc'] = 'sg' and gschoolVars['school_diploma'] = 0 and kanikuli = 0, 56, 34)

	!! Same-day slot check: on a small lucky roll it comes free; any day it exists it
	!! can also be booked for a fee as an expedited appointment. 'request_appointment_offer'
	!! folds whichever applies into the regular offer screen below.
	appointment_offer_duration = tmp_doc_duration
	appointment_offer_buffer = 2
	temp_lucky_end = (totminut mod 1440) / 15 + rand(1, 3)
	if temp_lucky_end >= tmp_doc_range_min and temp_lucky_end <= 82 and func('appointments', 'check_conflict', daystart, temp_lucky_end) = 0:
		same_day_window_end = temp_lucky_end
		same_day_free = iif(rand(0,99) < 15, 1, 0)
	else
		same_day_window_end = -1
		same_day_free = 0
	end
	killvar 'temp_lucky_end'

	gs 'clinic_functions', 'request_appointment_offer'
end


if $ARGS[0] = 'request_appointment_offer':
	!! ┌────────────────────────────────────────────────────────────────┐
	!! │                   request_appointment_offer                    │
	!! ├────────────────────────────────────────────────────────────────┤
	!! │ Internal to 'request_appointment'. A self-looping page (same   │
	!! │ pattern as 'wait_for_appointment_wait' below): first entry     │
	!! │ remembers where to return once a slot is booked or the player  │
	!! │ leaves; the slot-selection links in '_offer_row' re-enter via  │
	!! │ 'gt' on every click, which this redraws from scratch. Offers 6 │
	!! │ conflict-free, deterministic candidate slots as a selectable   │
	!! │ table, plus the same-day slot from 'request_appointment' (if   │
	!! │ any) - free of charge if same_day_free = 1, otherwise bookable │
	!! │ as a paid expedited appointment. The scheduled slots are       │
	!! │ picked via the table links and committed with Continue; the    │
	!! │ same-day slot and leaving are separate, immediate-commit acts  │
	!! │ shown below it.                                                │
	!! └────────────────────────────────────────────────────────────────┘
	if $appt_ret_loc = '':
		$appt_ret_loc = $loc
		$appt_ret_arg = $loc_arg
		appointment_selected_index = -1
		!! Rolled once here rather than on every redraw below, so the nurse's line doesn't
		!! change out from under the player each time they pick a different slot.
		appt_offer_flavor_choice = rand(0,4)
	else
		*clr & cla
		'<center><img <<$set_imgh>> src="images/locations/shared/clinic/poli.jpg"></center>'
		$appt_offer_intro
		'"当然可以，让我看看还有哪些空位，"她一边翻着预约簿一边回答。'
	end

	gs 'appointments', 'offer_slots', 6, tmp_doc_seed, 14, tmp_doc_duration, 2, tmp_doc_range_min, 82, 1

	same_day_offer_index = -1
	if same_day_window_end >= 0:
		same_day_offer_index = appointment_offer_count
		appointment_offer_day[same_day_offer_index] = daystart
		appointment_offer_window_end[same_day_offer_index] = same_day_window_end
	end

	$temp_doc_desc = func('clinic_functions', '_category_desc', $doc_category)

	if appt_offer_flavor_choice = 1:
		'"与<<$temp_doc_desc>>的预约，我们有几个可选时间："'
	elseif appt_offer_flavor_choice = 2:
		'"目前与<<$temp_doc_desc>>的预约还有几个空位："'
	elseif appt_offer_flavor_choice = 3:
		'"关于与<<$temp_doc_desc>>的体检，以下是一些可选时间："'
	elseif appt_offer_flavor_choice = 4:
		'护士微笑着说："本周与<<$temp_doc_desc>>的预约还有几个空位："'
	else
		'"与<<$temp_doc_desc>>的预约，我们有以下可选时间："'
	end

	'<table cellspacing=0 cellpadding=4 border=0><tr><th align=left>日期</th><th align=left>时间</th><th align=left>备注</th></tr>'
	temp_roff_i = 0
	:request_offer_loop
	if temp_roff_i < appointment_offer_count:
		if appointment_offer_day[temp_roff_i] >= 0:
			gs 'clinic_functions', '_offer_row', temp_roff_i
		end
		temp_roff_i += 1
		jump 'request_offer_loop'
	end
	killvar 'temp_roff_i'
	'</table>'

	if same_day_offer_index >= 0:
		if same_day_free = 1:
			'"其实今天晚些时候好像也空出了一个位置。如果你不想提前预约，可以免费等候<<$temp_doc_desc>>。"'
		else
			'"如果你不想等这些时间，我们今天也提供付费的加急预约，可以更快见到<<$temp_doc_desc>>。"'
		end
	end
	killvar 'temp_doc_desc'

	!! Commits the table selection above - nags instead if nothing is picked yet.
	if appointment_selected_index >= 0:
		act '继续':
			$temp_appt_dest_loc = $appt_ret_loc
			$temp_appt_dest_arg = $appt_ret_arg
			gs 'clinic_functions', 'book_appointment_confirm', appointment_selected_index
			gt $temp_appt_dest_loc, $temp_appt_dest_arg
		end
	else
		act '继续' + $func('wrap', 'v_neg', '（请选择预约时间）'):
			msg '请选择预约时间。'
		end
	end

	!! Same-day slot: resolved and attended immediately, unlike the scheduled slots above
	if same_day_offer_index >= 0:
		if same_day_free = 1:
			act '免费等候今天的空位':
				temp_wait_minutes = appointment_offer_window_end[same_day_offer_index] * 15 - (totminut mod 1440)
				gs 'clinic_functions', 'book_appointment_confirm', same_day_offer_index
				gs 'appointments', 'resolve', $doc_category
				gs 'clinic_functions', 'wait_for_appointment', $doc_category, temp_wait_minutes
				killvar 'temp_wait_minutes'
			end
		else
			act '支付加急预约费用' + $func('money', 'get_cost_string', 1000):
				if func('money', 'can_afford', 1000) = 0:
					$noMoney
				else
					gs 'money', 'pay', 1000
					temp_wait_minutes = appointment_offer_window_end[same_day_offer_index] * 15 - (totminut mod 1440)
					gs 'clinic_functions', 'book_appointment_confirm', same_day_offer_index
					gs 'appointments', 'resolve', $doc_category
					gs 'clinic_functions', 'wait_for_appointment', $doc_category, temp_wait_minutes
					killvar 'temp_wait_minutes'
				end
			end
		end
	end

	act '返回':
		$temp_appt_dest_loc = $appt_ret_loc
		$temp_appt_dest_arg = $appt_ret_arg
		gs 'clinic_functions', 'book_appointment_cleanup'
		gt $temp_appt_dest_loc, $temp_appt_dest_arg
	end

	!! Not killed here: the same-day acts above read it live at click-time, which is a
	!! later turn than this render - clearing it now would zero it out from under them
	!! before the player ever clicks. 'book_appointment_cleanup' clears it once the flow
	!! actually ends.
	exit
end


if $ARGS[0] = '_offer_row':
	!! ┌────────────────────────────────────────────────────────────────┐
	!! │                          _offer_row                            │
	!! ├────────────────────────────────────────────────────────────────┤
	!! │ Internal to 'request_appointment_offer'. ARGS[1] = slot index. │
	!! │ Prints one <tr> whose day/time cells are both links picking    │
	!! │ that slot - since appointment_selected_index is a single       │
	!! │ value, not a set, picking one always clears any previous       │
	!! │ selection. Clicking re-enters 'request_appointment_offer' via  │
	!! │ 'gt' to redraw the table with the new selection highlighted.   │
	!! └────────────────────────────────────────────────────────────────┘
	temp_or_i = ARGS[1]
	temp_or_offset = appointment_offer_day[temp_or_i] - daystart
	$temp_or_rel = $iif(temp_or_offset = 0, 'Today', $iif(temp_or_offset = 1, 'in 1 day', 'in ' + temp_or_offset + ' days'))
	!! week runs 1-7 (Monday-Sunday); normalize so a multiple-of-7 offset lands on 7, not 0.
	$temp_or_label = $weekName[(week + temp_or_offset - 1) mod 7 + 1] + ' (' + $temp_or_rel + ')'
	$temp_or_time = func('calendar_query', 'ts_to_str', appointment_offer_window_end[temp_or_i]) + '–' + func('calendar_query', 'ts_to_str', appointment_offer_window_end[temp_or_i] + appointment_offer_duration)
	$temp_or_note = $func('clinic_functions', '_offer_row_conflicts', temp_or_i)

	$temp_or_link = '<a href="exec: appointment_selected_index = ' + temp_or_i + ' & gt ''clinic_functions'', ''request_appointment_offer''" style="display:block;">'

	*p '<tr' + $iif(appointment_selected_index = temp_or_i, ' style="font-weight:bold;"', '') + '><td>' + $temp_or_link + $temp_or_label + '</a></td><td>' + $temp_or_link + $temp_or_time + '</a></td><td>' + $temp_or_note + '</td></tr>'

	killvar 'temp_or_i'
	killvar 'temp_or_offset'
	killvar 'temp_or_rel'
	killvar 'temp_or_label'
	killvar 'temp_or_time'
	killvar 'temp_or_note'
	killvar 'temp_or_link'
	exit
end


if $ARGS[0] = '_offer_row_conflicts':
	!! ┌────────────────────────────────────────────────────────────────┐
	!! │                     _offer_row_conflicts                       │
	!! ├────────────────────────────────────────────────────────────────┤
	!! │ $result = func('clinic_functions', '_offer_row_conflicts', i)  │
	!! │                                                                │
	!! │ Notes-column text for slot i: 'Conflict: Title (start-end), …' │
	!! │ or '' if none. Queries the same non-blocking overlaps as       │
	!! │ 'appointments' '_offer_conflict_note', but that one is         │
	!! │ formatted as a sentence fragment for the old one-line-per-slot │
	!! │ description - this is its own format for a standalone column.  │
	!! └────────────────────────────────────────────────────────────────┘
	temp_orc_i = ARGS[1]
	gs 'calendar_query', 'get_events_for_time_range', appointment_offer_day[temp_orc_i], appointment_offer_window_end[temp_orc_i] - appointment_offer_buffer, appointment_offer_buffer + appointment_offer_duration
	copyarr '$temp_orc_events', '$query_events_for_time_range'

	$result = ''
	if arrsize('$temp_orc_events') > 0:
		$result = 'Conflict: '
		temp_orc_k = 0
		:offer_row_conflicts_loop
		if temp_orc_k < arrsize('$temp_orc_events'):
			if temp_orc_k > 0: $result += ', '
			gs 'calendar_query', 'get_event_display_range', $temp_orc_events[temp_orc_k], appointment_offer_day[temp_orc_i]
			$result += $event_vars['title'] + ' (' + $func('calendar_query', 'ts_to_str', result_start_ts) + '-' + $func('calendar_query', 'ts_to_str', result_end_ts) + ')'
			temp_orc_k += 1
			jump 'offer_row_conflicts_loop'
		end
		killvar 'temp_orc_k'
	end

	killvar 'temp_orc_i'
	killvar 'temp_orc_events'
	exit
end


if $ARGS[0] = 'attend_appointment':
	!! ┌────────────────────────────────────────────────────────────────┐
	!! │                       attend_appointment                       │
	!! ├────────────────────────────────────────────────────────────────┤
	!! │ gs 'clinic_functions', 'attend_appointment', $category,        │
	!! │ wait_minutes, $recurring_event_id                              │
	!! │                                                                │
	!! │ The 'appointments' render_acts arrive handler - both on-time   │
	!! │ and late (wait_minutes negative) arrivals. Shows a             │
	!! │ go-to-appointment act; clicking it tells the receptionist and, │
	!! │ if late, rolls via 'check_in' whether the booking still holds -│
	!! │ a lost slot means same-day entry is gone and the player must   │
	!! │ reschedule. Works at any clinic.                               │
	!! └────────────────────────────────────────────────────────────────┘
	$appt_attend_category = $ARGS[1]
	appt_attend_wait = ARGS[2]
	$appt_attend_recur = $ARGS[3]
	$appt_attend_desc = $func('clinic_functions', '_category_desc', $ARGS[1])

	act "去赴与<<$appt_attend_desc>>的预约":
