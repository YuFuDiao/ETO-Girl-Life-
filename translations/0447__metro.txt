	gs 'stat'
	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '地铁站')
	$func('wrap', 'header center b', '瓦西里耶夫斯基岛')

	'瓦西里耶夫斯基岛地铁站'
	*nl
	gs 'transport_functions', 'set_metro_wait_time', 'island'

	act '步行前往瓦西里耶夫斯基岛 (0:05)': minut += 5 & gt 'city_island', ''

	act '查看地铁时刻表':
		*clr & cla
		'<center><h1>地铁时刻表</h1></center>'
		gs 'transport_functions', 'display_metro_schedule', 'is'
		act '别再盯着日程表了': gt 'metro', 'island'
	end

	if transportVars['metro_wait_suburbs'] > 60:
		'现在坐地铁太晚了。下一班将在<<transportVars[''metro_wait_suburbs'']/60>>小时'
	else
		'开往郊区方向的下一班地铁 ' + iif(transportVars['metro_wait_suburbs'] = 0, '正在离开！', '还有<<transportVars[''metro_wait_suburbs'']>>分钟。')

			act '走到售票机前': gt 'metro', 'island_tickets'
		else
			*nl
			$func('transport_functions', 'display_metropass_time')
			act '乘坐地铁前往市中心 (<<func(''transport_functions'', ''display_metro_timecost'', ''island'', ''center'')>>)':				gt 'metro', 'island_center'

		end
	end

	gs 'metro', 'check_events'

elseif $ARGS[0] = 'island_tickets':
	if daystart < transportVars['metropass_day']: gt 'metro', 'island'
	menu_off = 0
	gs 'core_library', 'setloc', 'metro', 'island_tickets'

	$func('wrap', 'header center b', '瓦西里耶夫斯基岛')

	gs 'core_library', 'setloc', 'metro', 'suburbs'
	$location_type = 'public_indoors'

	if sound_settings['environment_off'] = 0: PLAY 'sound/metro' + rand(1, 4) + '.mp3', 30
	gs 'stat'
	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '地铁站')
	$func('wrap', 'header center b', '郊区')
	'<center><img <<$set_imgh>> src="images/locations/city/shared/metro/platisland.jpg"></center>'

	'郊区地铁站'
	*nl
	gs 'transport_functions', 'set_metro_wait_time', 'suburbs'

	act '步行前往郊区 (0:20)': minut += 20 & gt 'city_suburbs', 'start'

	if hour >= 4 and hour < 23: act '步行前往公交车站 (0:15)': minut += 15 & gt 'bus', 'suburbs'

	act '查看地铁时刻表':
		*clr & cla
		gs 'transport_functions', 'display_metro_schedule', 'si'
		act '别再盯着日程表了': gt 'metro', 'suburbs'
	end

	if transportVars['metro_wait_island'] > 60:
		'现在坐地铁太晚了。下一班将在<<transportVars[''metro_wait_island''] / 60>>小时'
	else
		'下一班开往瓦西里岛方向的地铁 ' + iif(transportVars['metro_wait_island'] = 0, '正在离开！', '还有<<transportVars[''metro_wait_island'']>>分钟。')
		if daystart >= transportVars['metropass_day']:
			act '走到售票机前': gt 'metro', 'suburbs_tickets'
			gs 'transport_functions', 'buy_metro_pass'
		else
			*nl
			*pl $func('transport_functions', 'display_metropass_time')
			act '乘坐地铁前往瓦西里耶夫斯基岛 (<<func(''transport_functions'', ''display_metro_timecost'', ''suburbs'', ''island'')>>)':		gt 'metro', 'suburbs_island'
			act '乘坐地铁前往市中心 (<<func(''transport_functions'', ''display_metro_timecost'', ''suburbs'', ''center'')>>)':				gt 'metro', 'suburbs_center'
			!! act 'Take the metro to the Artisan district (<<func(''transport_functions'', ''display_metro_timecost'', ''suburbs'', ''artisan'')>>)':			gt 'metro', 'suburbs_artisan'
			act '乘坐地铁前往住宅区 (<<func(''transport_functions'', ''display_metro_timecost'', ''suburbs'', ''residential'')>>)':	gt 'metro', 'suburbs_residential'
			act '乘坐地铁前往工业区 (<<func(''transport_functions'', ''display_metro_timecost'', ''suburbs'', ''industrial'')>>)':	gt 'metro', 'suburbs_industrial'
		end
	end

	gs 'metro', 'check_events'

elseif $ARGS[0] = 'suburbs_tickets':
	if daystart < transportVars['metropass_day']: gt 'metro', 'suburbs'
	menu_off = 0
	gs 'core_library', 'setloc', 'metro', 'suburbs_tickets'

	$func('wrap', 'header center b', '地铁站')
	$func('wrap', 'header center b', '郊区')
	'<center><img <<$set_imgh>> src="images/locations/city/shared/metro/platisland.jpg"></center>'

	'郊区的售票机'

	act '离开售票处': gt 'metro', 'suburbs'

	act '查看地铁时刻表':
		*clr & cla
		'<center><h1>地铁时刻表</h1></center>'
		gs 'transport_functions', 'display_metro_schedule', 'si'
		act '别再盯着日程表了': gt 'metro', 'suburbs_tickets'
	end

	gs 'transport_functions', 'buy_metro_ticket', 'suburbs', 'si'
	gs 'transport_functions', 'buy_metro_pass'
end
