# state
!! Push/pop the current display state onto a stack.
!!
!! Push:    gs 'state', 'push'
!! Pop:     gs 'state', 'pop'
!!
!! Must always be used in matched push/pop pairs.
!! Supports nesting (e.g. a menu opened from within another menu).

if $ARGS[0] = 'push':
	!! main window
	$main_stack[]       = $maintxt
	!! status window
	$stat_stack[]       = $stattxt
	!! actions window
	$action_stack[]     = $curacts
	!! objects window
	$object_stack[]     = $curobjs
	!! background image
	$background_stack[] = $backimage
	!! return location variables
	$loc_stack[]        = $loc
	$loc_arg_stack[]    = $loc_arg
	!! navigation menu visibility
	menu_off_stack[]    = menu_off
end


if $ARGS[0] = 'pop':
	!! restore the return location variables
	$loc     = $loc_stack[arrsize('$loc_stack')-1]
	$loc_arg = $loc_arg_stack[arrsize('$loc_arg_stack')-1]
	killvar 'loc_stack',     arrsize('$loc_stack')-1
	killvar 'loc_arg_stack', arrsize('$loc_arg_stack')-1

	!! restore the background image
	$BACKIMAGE = $background_stack[arrsize('$background_stack')-1]
	killvar 'background_stack', arrsize('$background_stack')-1

	!! restore the objects/debug window
	killobj
	dynamic $object_stack[arrsize('$object_stack')-1]
	killvar 'object_stack', arrsize('$object_stack')-1

	!! restore the actions window
	cla
	dynamic $action_stack[arrsize('$action_stack')-1]
	killvar 'action_stack', arrsize('$action_stack')-1

	!! restore the status window
	clr
	p $stat_stack[arrsize('$stat_stack')-1]
	killvar 'stat_stack', arrsize('$stat_stack')-1

	!! restore the main window
	*clr
	*p $main_stack[arrsize('$main_stack')-1]
	killvar 'main_stack', arrsize('$main_stack')-1

	!! restore navigation menu visibility
	menu_off = menu_off_stack[arrsize('menu_off_stack')-1]
	killvar 'menu_off_stack', arrsize('menu_off_stack')-1
end
