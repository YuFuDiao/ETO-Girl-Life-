			end
		else
			'<<$npcdesc>>大声笑了起来。'
			'“是啊，我敢打赌那样就能让我脱身了。”'
			gs 'hookup_after', 'later'
		end
	end

	act '把它们送给她':
		cla & *nl
		'“把它们送给她，”你调皮地说，<<$hookup[''dress_describe'']>>。“也许她会穿上。”'
		gs 'hookup_after', 'later'

		gs 'hookup_after', 'dress_loop'
		'“现在它们是你的内裤了，”你得意地补了一句，<<$hookup[''dress_describe'']>>。'
		gs 'hookup_after', 'later'
	end
