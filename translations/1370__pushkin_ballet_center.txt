# pushkin_ballet_center

!! This is where the dance school accommodation is located, just off the city center.
!! mod event. remember to add a act to exit


if $ARGS[0] = '' or $ARGS[0] = 'start':
	menu_off = 0
	gs 'core_library', 'setloc', 'ballet_pushkin_center', $ARGS[0]
	$region = 'puskin'
	$setloc['StageTitle'] = '普希金寄宿学校'
	$setloc['StageImage'] = 'locations/pushkin/ballet_residence/residence.jpg'
	if sound_settings['environment_off'] = 0: PLAY 'sound/street.mp3', 30
	gs 'stat'
	gs 'core_library', 'stage_title'

	'你可以看到这栋被改造成学校宿舍的公寓楼。街上似乎总是停满了汽车，但这个时候街上却没多少人。'

	act '按门铃': gt 'pushkin_ballet_res', 'warden'
	act '回到街上': gt 'pushkin'
end
