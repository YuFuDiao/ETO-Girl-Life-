# autoservisF
if $ARGS[0] = 'start':
	*clr & cla
	menu_off = 0
	gs 'core_library', 'setloc', 'autoservisF', 'start'
	$location_type = 'event'
	gs 'stat'
	gs 'car_funcs', 'avtonorm'
	'<center><b>汽车维修中心</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/industrial/CarServiceSta/autoservis1.jpg"></center>'

	if func('car_funcs', 'is_here') = 0:
		'工头叹了口气。“你的车呢，姑娘？我得看看你的车，才能知道有没有毛病、是什么毛病。把车开来，我来检查检查。”'
		if func('car_funcs', 'has_wreck'):
			'修车厂为损坏的汽车提供拖车服务，把它们拖到这个维修中心。价格：' + $func('money', 'string_price', 2500)

			act '支付拖车服务费 (1:00)' + $func('money', 'get_cost_string', 2500):
				if func('money', 'can_afford', 2500) = 0:
					$noMoney
				else
					*clr & cla
					menu_off = 1
					gs 'money', 'pay', 2500
					gs 'car_funcs', 'setloc', 'autoservisF', 'start', 'city'
					minut += 60
					'<center><img <<$set_imgh>> src="images/locations/city/industrial/CarServiceSta/autoservis2.jpg"></center>'
					'你在服务站坐下等待。大约过了一个小时你的车才到，不过它一到，就几乎立刻被开进了车库，看起来工头已经准备好要检查它了。'

					act '听听工头怎么说': gt 'autoservisF', 'start'
				end
			end
		end
	else
		'你的<a href="exec:gs ''carF'', ''start''"><<$car[''name'']>></a>就停在里面。'
		if func('car_funcs', 'has_wreck'):
			'工头看着你的车——如果这堆残骸也算得上车的话——带着一种职业性的厌倦，那种眼神告诉你，他见过成千上万辆类似甚至更糟状态的车。'
			'他迅速而彻底地检查了一遍，一边低声记下需要做的事情，然后当回到你面前时，只简单地说费用是 ' + $func('money', 'string_price', normrem) + ' 来修复损坏。'

			act '付款并等待他们修好车' + $func('money', 'get_cost_string', normrem):
				if func('money', 'can_afford', normrem) = 0:
					$noMoney
				else
					*clr & cla
					menu_off = 1
					minut += 60
					gs 'money', 'pay', normrem
					car['new_condition'] = car['new_condition'] / 2
					car['current_condition'] = car['new_condition']
					car['wreck'] = 0
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/city/industrial/CarServiceSta/autoservis2.jpg"></center>'
					'他说用不了多长时间——起初你以为他在说反话，但随即意识到他那种一板一眼的态度可没多少幽默的余地——于是你走进了一间带沙发和电视的等候室。翻了大约一个小时的频道后，你被叫回车库，几乎不敢相信自己的眼睛。'
					'车看起来焕然一新！没有一处凹陷，没有一道划痕，你发动引擎确认它还能开时，声音比以往任何时候都好听。你问工头他和他的伙计们是怎么这么快修好这辆车的，但他只是耸了耸肩。'
					'在他让你回办公室支付 ' + $func('money', 'string_price', normrem) + ' 时，他请你以后别再送这种惨不忍睹的车来了——你也不确定他是不是在开玩笑。'

					act '离开': minut += 5 & gt 'city_industrial'
				end
			end
		else
			if car['current_condition'] >= car['new_condition'] * 75 / 100:
				'工头快速而彻底地检查了你的车。'
				'没过几分钟，他回来告诉你，你的车不需要任何维修或保养。'
			else
				'工头快速而彻底地检查了你的车。'
				'他似乎在自言自语地嘀咕着什么，也许是在评估某个缺陷。没过几分钟，他回来向你指出需要修复的地方。'
				'这会花你 ' + $func('money', 'string_price', normrem) + '。'

				act '付款并等待他们修好车' + $func('money', 'get_cost_string', normrem):
					if func('money', 'can_afford', normrem) = 0:
						$noMoney
					else
						*clr & cla
						menu_off = 1
						minut += 60
						gs 'money', 'pay', normrem
						car['current_condition'] = car['new_condition']
						car['wreck'] = 0
						gs 'stat'
						'<center><img <<$set_imgh>> src="images/locations/city/industrial/CarServiceSta/autoservis2.jpg"></center>'
						'他说用不了多长时间，你便走进了一间带沙发和电视的等候室。翻了大约一个小时的频道后，车修好了，你支付了 ' + $func('money', 'string_price', normrem) + '。'

						act '离开': minut += 5 & gt 'city_industrial'
					end
				end
			end
		end
	end


	act '离开': minut += 5 & gt 'city_industrial'
end


if $ARGS[0] = 'is_open':
	RESULT = hour >= 9 and hour <= 20 and week < 7
end
