			'"不然我还能拿它怎么办？" 你耸耸肩。'
			if $npc_cum_pref[$npcID] = 'mouth':
				'"咽下去？"'
				act '哦。对。':
					cla & *nl
					'"哦。" 你眨了眨眼。"对。"'
					act '继续': gt 'date_chill', 'finish_watch'
				end

				act '恶心':
					cla & *nl
					'"恶心。" 你皱了皱鼻子。"是啊，那是不可能的。"'
					act '继续': gt 'date_chill', 'finish_watch'
				end

				act '没兴致':
					cla & *nl
					'"呃。今天没那个兴致。"'
					act '继续': gt 'date_chill', 'finish_watch'
				end
			else
				'"水槽？纸巾？杯子？"'
				act '哦。对。':
					cla & *nl
					'"哦。" 你眨了眨眼。"对。"'
					'你之前根本没想过这些。'
					act '继续': gt 'date_chill', 'finish_watch'
				end

				act '太麻烦了':
					cla & *nl
					'"呃。那些听起来都好麻烦。直接吐地上省事。"'
					'"但待会儿收拾起来可不省事，" 他咕哝道。'
					act '继续': gt 'date_chill', 'finish_watch'
				end
			end
		end

		act '洗手间太远了':
			cla & *nl
			'"洗手间太远了，" 你耸耸肩。'
			act '继续': gt 'date_chill', 'finish_watch'
		end

		act '这是报复':
			cla & *nl
			'"这是报复，" 你不怀好意地坏笑。"你射在我嘴里，我就吐在地板上。"'
			act '继续': gt 'date_chill', 'finish_watch'
		end
	end
end
