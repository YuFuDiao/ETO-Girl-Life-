	!! MiraLoc = 51		Beach

if $ARGS[0] = 'is_here':
	if arrsize('ARGS') = 1: $ARGS[1] = $loc
	if arrsize('ARGS') = 2: $ARGS[2] = $loc_arg

	result = func('miroslava_schedule', 'here_core', $ARGS[1], $ARGS[2], $locat['A60_loc'], $locat['A60_arg'])
	exit
end


if $ARGS[0] = 'was_here':
	if arrsize('ARGS') = 1: $ARGS[1] = $loc
	if arrsize('ARGS') = 2: $ARGS[2] = $loc_arg

	exit

if $ARGS[0] = 'here_core':
	if $ARGS[1] = 'forest':
		result = ($ARGS[3] = 'gad_forest' or $ARGS[3] = 'gad_meadow' or $ARGS[3] = 'gad_swamp')
	elseif $ARGS[1] = 'gp':
		result = ($mid($ARGS[3], 1, 6) = 'gad_gp')
	elseif $ARGS[1] = 'forest_road':
		result = ($ARGS[3] = 'gad_road' or $ARGS[3] = 'mitkabuh_group' or ($ARGS[3] = 'bus' and instr('gadukino;gadukino_tickets', $ARGS[4])))
	elseif $ARGS[1] = 'river':
		result = ($ARGS[3] = 'gad_river' or $ARGS[3] = 'gad_beach')
	else
		result = ($ARGS[3] = $ARGS[1] and $ARGS[4] = $ARGS[2])
	end
	exit
end


if $ARGS[0] = 'get_locat_from_loc':
	if $loc = 'gad_miroslava_home':
		result = 10

	elseif $loc = 'gad_gpyard':
		result = 20
	elseif $loc = 'gad_gphouse':
		result = 21
	elseif $loc = 'gad_gpbath':

	elseif $loc = 'gad_gpbarn':

	elseif $loc = 'gad_church':
		result = 31

		result = 40
	elseif $loc = 'bus' and instr('gadukino;gadukino_tickets', $loc_arg) > 0:
		result = 41
	elseif $loc = 'mitkabuh' or $loc = 'mitkabuh_group':
		result = 42

		result = 50
	elseif $loc = 'gad_beach':

	elseif $loc = 'gad_meadow':

	elseif $loc = 'gad_forest' and $loc_arg = 'forest_outskirts':

	elseif $loc = 'gad_swamp':
		result = 64

		result = 70

	elseif $loc = 'road' and nroad = 10:
		result = 80
	elseif $loc = 'train' and $loc_arg = 'gadukino':
		result = 81
	elseif $loc = 'furi' and nroad = 10:

if $ARGS[0] = 'force_change':
	if $locat['A60_loc'] = 'gad_miroslava_home': exit

	$temp_start_loc = $locat['A60_loc']
	MiraVars['guest'] = 0
	MiraVars['follow_time'] = 0
	MiraVars['follower'] = 0
	gs 'miroslava_schedule'
	mss_i = 0
	:mira_loop_start
	if $locat['A60_loc'] = $temp_start_loc:
		gs 'miroslava_schedule', 'cikl'
		gs 'miroslava_schedule', 'update_locat'
		mss_i += 1
		if mss_i < 10: jump 'mira_loop_start'
	end
	killvar 'mss_i'
	killvar 'temp_start_loc'

if $ARGS[0] = 'update_locat':
	if daystart < MiraVars['emb_day']:
		!! Embarrassed Mira is at Home
		MiraLoc = 10
		jump 'set_locarg'
	end

	if MiraVars['guest'] = 1:
		if MiraVars['guestday'] <= daystart: MiraVars['guestday'] = daystart + 1
		!! check if mira and sveta are in the same location.
		!! if so: $locat['A60_…'] remains the same
		!! else: MiraVars['guest'] = 0
		if $mid($loc, 1, 6) = 'gad_gp' or $loc = 'gad_field':
			$locat['A60_loc'] = $loc
			$locat['A60_arg'] = $loc_arg
			$locat['A60_arg1'] = ''
		else
			MiraVars['guest'] = 0
			gs 'miroslava_schedule', 'update_locat'
		end
		exit
	end

	if MiraVars['follower'] = 1:
		MiraVars['follower'] = 0
		!! Mira is your follower, so set her location to your location.
		MiraLoc = func('miroslava_schedule', 'get_locat_from_loc')
		jump 'set_locarg'
	elseif MiraVars['follow_time'] > 0:
		!! Mira is stationary unless you keep instructing her to follow you.
		$locat['A60_loc'] = $locat['A60_loc_prev']
		$locat['A60_arg'] = $locat['A60_arg_prev']
		$locat['A60_arg1'] = $locat['A60_arg1_prev']
		exit
	end

	MiraLoc = func('miroslava_schedule', 'get_base_schedule')

	if hour <= 7:
		MiraLoc = 11
		jump 'set_locarg'
		exit
	end


	if npc_known['A60'] = 0 and MiraLoc >= 20 : MiraLoc = 30		& !! Sveta and Mira aren''t friends yet.

	if sunWeather = 0:
		if MiraLoc >= 30 and MiraLoc ! 42:
			if MiraVars['invite_day'] >= daystart or npc_known['A60'] = 0 or $locat['A60_loc'] = 'gad_miroslava_home' or npc_rel['A60'] < 15 or (npc_QW['A63'] >= 11 and MiraVars['QW'] < 11 and MiraVars['had_sex'] <= 1):
				MiraLoc = 10
			else
				MiraVars['invite_rand'] = rand(0, 1)
				MiraVars['invite_day'] = daystart
				if MiraVars['invite_rand'] = 0:
					MiraLoc = 20
					MiraVars['follow_time'] = 2
				else
					MiraLoc = 10
				end
			end
		end
	else
		if MiraVars['QW'] > 15 and locat['A60_prost'] and hour >= 16 and hour < 20:			MiraLoc = 80		& !! Prostitution
		if MiraLoc = 61 and (MiraVars['meadow'] ! 2 and MiraVars['meadow'] ! 5):			MiraLoc = 60		& !! If meadow not unlocked -> Forest edge
		if MiraLoc = 51 and (sunWeather = 0 or temper < 15 or month < 5 or month > 9):		MiraLoc = 10		& !! Winter = No Beach
		if MiraLoc = 42 and (MiraVars['QW'] = 0 or npc_drunk['A60'] > 0):					MiraLoc = 10		& !! drunk or doesn''t know the guys
		if (MiraLoc / 10) = 2 and grandpaQW['chore_herd_cattle'] = 1 and $loc = 'gad_field' and npc_rel['A60'] >= 15 and (npc_QW['A63'] < 13 or MiraVars['QW'] >= 10 or MiraVars['had_sex'] > 1):
			MiraLoc = 70		& !! Mira visits in the field if she was planning to visit you either way.
		end
	end

	if MiraVars['guest'] = 0 and (MiraLoc / 10 = 2) and MiraVars['guestday'] > daystart: MiraLoc = 10

	:set_locarg
	gs 'miroslava_schedule', 'set_locarg', MiraLoc
	killvar 'MiraLoc'
end
