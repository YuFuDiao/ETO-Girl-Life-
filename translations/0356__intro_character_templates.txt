	elseif temp_rand = 3:

	elseif temp_rand = 5:

	elseif temp_rand = 6:
		$result = 'runner'
	elseif temp_rand = 7:
		$result = 'football'
	elseif temp_rand = 8:
		$result = 'socialite'
	elseif temp_rand = 9:
		$result = 'beautiful'
	elseif temp_rand = 10:
		$result = 'anorexic'
	elseif temp_rand = 11:
		$result = 'bimbo'
	elseif temp_rand = 12:
		$result = 'gopnikstart'
	elseif temp_rand = 13:
		$result = 'troublemaker'
	elseif temp_rand = 14:
		if $start_type['loc'] <> 'sg':
			$result = $func('intro_character_templates', 'get_random_template')
			exit
		end
		$result = 'vitekgf'
	elseif temp_rand = 15:
		$result = 'alternative'
	elseif temp_rand = 16:
		$result = 'friendless'
	elseif temp_rand = 17:
		$result = 'uglyduckling'
	elseif temp_rand = 18:
		$result = 'goodgirl'
	elseif temp_rand = 19:
		$result = 'slut'
	elseif temp_rand = 20:
		$result = 'goth'
	else
		$result = 'poor'
	end
	killvar 'temp_rand'
	exit
end
