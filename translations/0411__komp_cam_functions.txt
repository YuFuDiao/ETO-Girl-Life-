		nA = <<cam_viewsVars['nA']>>
		nP = <<cam_viewsVars['nP']>>
		"
	end

	if cam_viewsVars['nA'] < 0: cam_viewsVars['nP'] -= cam_viewsVars['nA'] & cam_viewsVars['nA'] = 0
