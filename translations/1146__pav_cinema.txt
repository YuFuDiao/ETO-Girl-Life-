# pav_cinema

if $ARGS[0] = '':
	menu_off = 0
	gs 'core_library', 'setloc', 'pav_cinema', ''
	gs 'stat'
	gs 'themes', 'indoors'

	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/park/theater/kinoteatr.jpg"></center>'
	'只有一个小屏幕的小电影院。一张电影票价格 ' + $func('money', 'string_price', 100) + '。'

	act '离开电影院': minut += 5 & gt 'pav_park', 'start'

	act '去看电影' + $func('money', 'get_cost_string', 100):
		if func('money', 'can_afford', 100) = 0:
			$noMoney
		else
			gt 'pav_cinema', 'watch_film'
		end
	end
end


if $ARGS[0] = 'watch_film':
	*clr & cla
	menu_off = 1
	gs 'money', 'pay', 100
	minut += 120
	gs 'mood', 'raise', rand(30, 50)
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/park/theater/watchmovie.jpg"></center>'
	if fame['pav_slut'] >= 100 and rand(0, 2) > 0:
		cla
		'当你在看电影时，一个家伙坐到你旁边，在你耳边低语："谁都知道你喜欢吃鸡巴。你要是肯来一口，我就付你钱。"'
		'电影院里很暗，但你还能勉强看清他正指着自己的鸡巴。'
		act '拒绝':
			cla
			'你告诉他你更想看这部电影，他嘟囔了一句听起来像是"婊子"的话，然后走开了，让你安静地看完剩下的电影。'
			act '离开电影院': gt 'pav_park', 'start'
		end
		act '同意': gt 'gevent', '2'

		exit
	end

	act '离开电影院': gt 'pav_park', 'start'

	if rand(0, 99) < 90:
		'你安静地看着电影。'
	else
		'你安静地看着电影，但突然身后传来一阵奇怪的声音。'
		act '转身查看':
			*clr & cla
			gs 'stat'
			'<center><video autoplay loop <<$set_imgh>> src="images/locations/pavlovsk/park/theater/sex/kino.mp4"></video></center>'
			'你看到后排有个女孩正在给一个男人口交，整个电影院都能听到那响亮的咂嘴声，却没有人有反应。'
			'你转回头，继续看剩下的电影。'

			gs 'arousal', 'voyeur_sex', 2
			gs 'arousal', 'end'

			act '离开电影院': gt 'pav_park', 'start'
		end
	end
end
