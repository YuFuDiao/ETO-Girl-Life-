if $ARGS[0] = 'get_temp_string':
	!! ── Arguments ───────────────────────────────────────────────────
	!! ARGS[1] = integer part of °C (signed)
	!! ARGS[2] = tenths digit (usually 0-9; may be negative ONLY when ARGS[1] = 0,
	!!           signalling a temperature between -0.9 and -0.1°C that integer
	!!           truncation would otherwise render sign-less)
	!! ARGS[3] = scale override (optional; defaults to stat_cfg['temp_scale'] when absent)
	!!           0=°C  1=°F  2=K  3=°De  4=😡

	temp_sd_scale = iif(arrsize('ARGS') <= 3, stat_cfg['temp_scale'], ARGS[3])
	temp_sd_isneg = (ARGS[1] < 0) or (ARGS[1] = 0 and ARGS[2] < 0)
	temp_sd_frac = iif(ARGS[2] < 0, -ARGS[2], ARGS[2])
	$temp_sd_signprefix = iif(ARGS[1] = 0 and ARGS[2] < 0, "-", "")

	!! ── Build value string ───────────────────────────────────────────
	!! Celsius
	if temp_sd_scale = 0:
		$result = '<<$temp_sd_signprefix>><<ARGS[1]>>.<<temp_sd_frac>>°C'
	!! Fahrenheit
	elseif temp_sd_scale = 1:
		$result = '<<(1625 + 90*ARGS[1] + 9*ARGS[2])/50>>°F'
	!! Kelvin
	elseif temp_sd_scale = 2:
		$result = '<<(ARGS[1]*10 + ARGS[2] + 2732)/10>>.<<(ARGS[1]*10 + ARGS[2] + 2732) mod 10>> K'
	!! Delisle — Goes down as temperature goes up. Water boils at 0°De.
	!! Used in Russia until the 1800s because of course it was.
	elseif temp_sd_scale = 3:
		$result = '<<(1000 - ARGS[1]*10 - ARGS[2])*3/20>>°De'

	!! 😡 Emoji anger scale - a rigorous scientific measurement system
	elseif temp_sd_scale = 4:
		if ARGS[1] < -10:
			$result = '❄️🧊'
		elseif ARGS[1] < -5:
			$result = '🥶😬'
		elseif ARGS[1] < 5:
			$result = '🌨️😐'
		elseif ARGS[1] < 15:
			$result = '🌤️🙂'
		elseif ARGS[1] < 22:
			$result = '☀️😊'
		elseif ARGS[1] < 28:
			$result = '🌞😅'
		elseif ARGS[1] < 33:
			$result = '🔆😤'
		elseif ARGS[1] < 38:
			$result = '🔥😡'
		else
			$result = '☢️🤬'
		end
		!! Emoji scale skips the theme colouring
		killvar 'temp_sd_scale'
		killvar 'temp_sd_isneg'
		killvar 'temp_sd_frac'
		killvar 'temp_sd_signprefix'
		exit
	else
		!! Unknown scale — fall back to Celsius
		$result = '<<$temp_sd_signprefix>><<ARGS[1]>>.<<temp_sd_frac>>°C'
	end

	!! ── Apply temperature theme colouring ───────────────────────────
	if temp_sd_isneg:
		$result = $func('wrap', 'accent', $result)
	elseif ARGS[1] < 10:
		$result = $func('wrap', 'v_pos', $result)
	elseif ARGS[1] < 20:
		$result = $func('wrap', 'pos', $result)
	elseif ARGS[1] < 30:
		$result = $func('wrap', 'neg', $result)
	else
		$result = $func('wrap', 'v_neg', $result)
	end

	killvar 'temp_sd_scale'
	killvar 'temp_sd_isneg'
	killvar 'temp_sd_frac'
	killvar 'temp_sd_signprefix'
	exit
end
