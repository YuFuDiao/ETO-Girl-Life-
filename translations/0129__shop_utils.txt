# shop_utils


	!!====================================================================!!
	!!                                                                    !!
	!!                                INIT                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'is_init':
	result = (shop_display['init'] <> 0)
	exit
end


if $ARGS[0] = 'init':
	if $ARGS[1] = 'start':
		!! $ARGS[2] = 'outfit file' (clothing, shoes, coats, tattoos, etc.)
		!! $ARGS[3] = 'link_type' (shop, cheat, wardrobe, etc.)

		killvar 'outfitfilter'
		killvar 'shop_display'
		killvar 'shop_display_exceptions'
		!! killvar 'shop_display_types'
		killvar 'shop_utils_view'
		killvar 'shop_display_sorted'
		killvar 'shop_display_sorted_titles'

		gs 'shop_utils', 'init', 'set_defaults'
		gs 'shop_utils', 'init', 'set_outfit_file', $ARGS[2]
		gs 'shop_utils', 'init', 'set_link', $ARGS[3]
		exit
	end

	if $ARGS[1] = 'set_defaults':
		shop_display['type_max'] = 0
		shop_display['step'] = 1
		shop_display['offset'] = 0
		shop_display['discount'] = 0
		$shop_display['return_loc'] = $loc
		$shop_display['return_arg0'] = $loc_arg
		$shop_display['return_arg1'] = ''
		$shop_display['link'] = ''
		$shop_display['inclusion_check'] = "1"
		$shop_display['bcolor'] = ''
		gs 'shop_utils', 'filter', 'reset_sorting'
		outfitfilter['number_sort'] = 1
		exit
	end

	if $ARGS[1] = 'add_type':
		!! $ARGS[2]: type to add

		if arrpos('$shop_display_types', $ARGS[2]) < 0:
			$shop_display_types[] = $ARGS[2]
		end
		exit
	end

	if $ARGS[1] = 'add_item':
		!! $ARGS[2]: item type to add
		!! ARGS[3]: item number to add
		if arrpos('$shop_display_items', $ARGS[2] + '-' + ARGS[3]) < 0:
			$shop_display_items[] = $ARGS[2] + '-' + ARGS[3]
		end
		exit
	end

	if $ARGS[1] = 'add_all_types':
		if no loc($shop_display['view_file']):
			msg 'shop_utils ERROR: shop_utils, init, add_all_types called with non-existing view_file: "<<$shop_display[''view_file'']>>"'
			exit
		end

		gs $shop_display['view_file'], 'init', 'add_types', 'all'
	end

	if $ARGS[1] = 'set_outfit_file':
		!! $ARGS[2]: item type (clothing, bras, panties, etc.)
		!! If you pass a custom outfit type you also need to pass:
		!!		$ARGS[3]: func_file
		!!		$ARGS[4]: view_file
		!!		$ARGS[5]: attr_file
		!!		$ARGS[6]: type_suffix
		!!		ARGS[7]: do_sorted

		if instr($ARGS[2], 'clothing'):
			$shop_display['outfit_type'] = 'clothing'
			$shop_display['func_file'] = 'clothing'
			$shop_display['view_file'] = 'clothing_view'
			$shop_display['attr_file'] = 'clothing_attributes'
			$shop_display['type_suffix'] = ''
			shop_display['do_sorted'] = 1
		elseif instr($ARGS[2], 'bra'):
			$shop_display['outfit_type'] = 'bra'
			$shop_display['func_file'] = 'bras'
			$shop_display['view_file'] = 'bra_view'
			$shop_display['attr_file'] = 'underwear_attributes'
			$shop_display['type_suffix'] = '_bras'
			shop_display['do_sorted'] = 1
		elseif instr($ARGS[2], 'panty') or instr($ARGS[2], 'panties'):
			$shop_display['outfit_type'] = 'panty'
			$shop_display['func_file'] = 'panties'
			$shop_display['view_file'] = 'panty_view'
			$shop_display['attr_file'] = 'underwear_attributes'
			$shop_display['type_suffix'] = '_panties'
			shop_display['do_sorted'] = 1
		elseif instr($ARGS[2], 'bodysuit'):
			$shop_display['outfit_type'] = 'bodysuit'
			$shop_display['func_file'] = 'underwear_bodysuits'
			$shop_display['view_file'] = 'underwear_bodysuit_view'
			$shop_display['attr_file'] = 'underwear_attributes'
			$shop_display['type_suffix'] = '_bodysuits'
			shop_display['do_sorted'] = 1
		elseif instr($ARGS[2], 'shoe'):
			$shop_display['outfit_type'] = 'shoe'
			$shop_display['func_file'] = 'shoes'
			$shop_display['view_file'] = 'shoe_view'
			$shop_display['attr_file'] = 'shoe_attributes'
			$shop_display['type_suffix'] = '_shoes'
			shop_display['do_sorted'] = 1
		elseif instr($ARGS[2], 'coat'):
			$shop_display['outfit_type'] = 'coat'
			$shop_display['func_file'] = 'coats'
			$shop_display['view_file'] = 'coat_view'
			$shop_display['attr_file'] = 'coat_attributes'
			$shop_display['type_suffix'] = '_coats'
			shop_display['do_sorted'] = 1
		elseif instr($ARGS[2], 'purse'):
			$shop_display['outfit_type'] = 'purse'
			$shop_display['func_file'] = 'purses'
			$shop_display['view_file'] = 'purse_view'
			$shop_display['attr_file'] = 'purse_attributes'
			$shop_display['type_suffix'] = '_purses'
			shop_display['do_sorted'] = 1
		elseif instr($ARGS[2], 'tattoo'):
			$shop_display['outfit_type'] = 'tattoo'
			$shop_display['func_file'] = 'tattoo_management'
			$shop_display['view_file'] = 'tattoo_view'
			$shop_display['attr_file'] = 'tattoo_attributes'
			$shop_display['type_suffix'] = '_tattoos'
			shop_display['do_sorted'] = 0
		elseif instr($ARGS[2], 'piercing'):
			$shop_display['outfit_type'] = 'piercing'
			$shop_display['func_file'] = 'piercing_management'
			$shop_display['view_file'] = 'piercing_view'
			$shop_display['attr_file'] = 'piercing_attributes'
			$shop_display['type_suffix'] = '_piercings'
			shop_display['do_sorted'] = 0
		else
			$shop_display['outfit_type'] = $ARGS[2]
			$shop_display['func_file'] = $ARGS[3]
			$shop_display['view_file'] = $ARGS[4]
			$shop_display['attr_file'] = $ARGS[5]
			$shop_display['type_suffix'] = $ARGS[6]
			shop_display['do_sorted'] = ARGS[7]
		end
		exit
	end

	if $ARGS[1] = 'set_link':
		$shop_display['link'] = $ARGS[2]

		if $shop_display['link'] = 'shop':
			$shop_display['inclusion_check'] = "func($ARGS[0], 'is_owned', $ARGS[1], ARGS[2]) = 0"
		elseif $shop_display['link'] = 'cheat':
			$shop_display['inclusion_check'] = "1"
		elseif $shop_display['link'] = 'wardrobe':
			$shop_display['inclusion_check'] = "func($ARGS[0], 'in_wardrobe', $ARGS[1], ARGS[2])"
		elseif $shop_display['link'] = 'storage':
			$shop_display['inclusion_check'] = "func($ARGS[0], 'in_storage', $ARGS[1], ARGS[2])"
		elseif $shop_display['link'] = 'unwanted':
			$shop_display['inclusion_check'] = "func($ARGS[0], 'in_unwanted', $ARGS[1], ARGS[2])"
		else
			$shop_display['inclusion_check'] = "func($ARGS[0], 'is_owned', $ARGS[1], ARGS[2])"
		end
		exit
	end

	if $ARGS[1] = 'set_steps':
		!! ARGS[2]: step size
		!! ARGS[3]: offset

		shop_display['step'] = max(1, ARGS[2])
		shop_display['offset'] = max(0, ARGS[3])
		exit
	end

	if $ARGS[1] = 'set_discount':
		!! ARGS[2]: discount

		shop_display['discount'] = min(max(0, ARGS[2]), 100)
		exit
	end

	if $ARGS[1] = 'set_return':
		!! $ARGS[2]: $loc
		!! $ARGS[3]: $loc_arg -> $ARGS[0]
		!! $ARGS[4]: $loc_arg1 -> $ARGS[1]

		$shop_display['return_loc'] = $ARGS[2]
		$shop_display['return_arg0'] = $ARGS[3]
		$shop_display['return_arg1'] = $ARGS[4]
		exit
	end

	if $ARGS[1] = 'set_bcolor':
		!! $ARGS[2]: $temp_bcolor
		$shop_display['bcolor'] = $ARGS[2]
		exit
	end

	if $ARGS[1] = 'set_default_exceptions':
		if loc($shop_display['view_file']):
			gs $shop_display['view_file'], 'init', 'set_exceptions'
		else
			gs 'clothing_view', 'init', 'set_exceptions'
			gs 'bra_view', 'init', 'set_exceptions'
			gs 'panty_view', 'init', 'set_exceptions'
			gs 'underwear_bodysuit_view', 'init', 'set_exceptions'
			gs 'shoe_view', 'init', 'set_exceptions'
			gs 'purse_view', 'init', 'set_exceptions'
			gs 'coat_view', 'init', 'set_exceptions'
			gs 'tattoo_management', 'set_shop_display_exceptions'
			gs 'piercing_management', 'set_shop_display_exceptions'
		end
		exit
	end

	if $ARGS[1] = 'add_exception':
		!! $ARGS[2]: item type
		!! ARGS[3]: item number

		shop_display_exceptions[$ARGS[2] + $shop_display['type_suffix'] + '-' + ARGS[3]] = 1
		exit
	end

	if $ARGS[1] = 'end':
		shop_display['init'] = 1
		shop_display['type_max'] = arrsize('$shop_display_types')
		gs 'shop_utils', 'sorted', 'main'
		exit
	end
	exit
end


if $ARGS[0] = 'sorted':
	!! $ARGS[1]: function name

	if $ARGS[1] = 'main':
		if no loc($shop_display['func_file']):
			msg 'shop_utils ERROR: shop_utils, sorted, main called with non-existing func_file: "<<$shop_display[''func_file'']>>"'
			exit
		end
		if no loc($shop_display['view_file']):
			msg 'shop_utils ERROR: shop_utils, sorted, main called with non-existing view_file: "<<$shop_display[''view_file'']>>"'
			exit
		end

		killvar '$shop_display_sorted_titles'

		shop_display['type_i'] = 0
		:type_loop_sorted
		if shop_display['type_i'] < arrsize('$shop_display_types'):
			$shop_display['type'] = $shop_display_types[shop_display['type_i']]
			gs $shop_display['func_file'], 'totals', $shop_display['type']

			shop_display['number'] = 1 + shop_display['offset']
			:main_loop_sorted
			if shop_display['number'] <= total:
				if shop_display_exceptions[$shop_display['type'] + $shop_display['type_suffix'] + '-' + shop_display['number']] = 0:
					if dyneval('result = ' + $shop_display['inclusion_check'], $shop_display['func_file'], $shop_display['type'], shop_display['number']):
						$shop_display['shop'] = $mid($shop_display['type'], 1, instr($shop_display['type'] + '_', '_') - 1)
						if $shop_display['shop'] <> '':
							if arrpos('$shop_display_sorted_titles', $shop_display['shop']) < 0:
								$shop_display_sorted_titles[] = $shop_display['shop']
								$shop_display_sorted['number_' + shop_display['type_i'] + '_title'] = $shop_display['shop']
							end
						end
						if shop_display['do_sorted']:
							gs $shop_display['view_file'], 'sorted', 'add', $shop_display['type'], shop_display['number'], shop_display['type_i']
						else
							gs 'shop_utils', 'sorted', 'add_to_number', $shop_display['type'], shop_display['number'], shop_display['type_i']
						end
					end
				end

				shop_display['number'] += shop_display['step']
				jump 'main_loop_sorted'
			end

			shop_display['type_i'] += 1
			jump 'type_loop_sorted'
		end

		shop_display['unused_type_i'] = shop_display['type_i']

		shop_display['item_i'] = 0
		:item_loop_sorted
		if shop_display['item_i'] < arrsize('$shop_display_items'):
			$shop_display['item_entry'] = $shop_display_items[shop_display['item_i']]
			shop_display['sep_index'] = instr($shop_display['item_entry'], '-')

			$shop_display['type'] = $mid($shop_display['item_entry'], 1, shop_display['sep_index'] - 1)
			shop_display['number'] = val($mid($shop_display['item_entry'], shop_display['sep_index'] + 1))

			if dyneval('result = ' + $shop_display['inclusion_check'], $shop_display['func_file'], $shop_display['type'], shop_display['number']):
				$shop_display['shop'] = $mid($shop_display['type'] + '_', 1, instr($shop_display['type'] + '_', '_') - 1)
				if arrpos('$shop_display_sorted_titles', $shop_display['shop']) < 0:
					$shop_display_sorted_titles[] = $shop_display['shop']
					$shop_display_sorted['number_' + shop_display['unused_type_i'] + '_title'] = $shop_display['shop']
					shop_display['type_i'] = shop_display['unused_type_i']
					shop_display['unused_type_i'] += 1
				else
					shop_display['type_i'] = arrpos('$shop_display_types', $shop_display['type'])
					if shop_display['type_i'] < 0:
						shop_display['type_i'] = arrsize('$shop_display_types')
						$shop_display_types[] = $shop_display['type']
					end
				end
				if shop_display['do_sorted']:
					gs $shop_display['view_file'], 'sorted', 'add', $shop_display['type'], shop_display['number'], shop_display['type_i']
				else
					gs 'shop_utils', 'sorted', 'add_to_number', $shop_display['type'], shop_display['number'], shop_display['type_i']
				end
			end

			shop_display['item_i'] += 1
			jump 'item_loop_sorted'
		end
		killvar 'shop_display_sorted_titles'
		exit
	end

	if $ARGS[1] = 'add_to_number':
		!! $ARGS[2]: type
		!! ARGS[3]: number
		!! ARGS[4]: index

		$shop_display_sorted['number_<<ARGS[4]>>_type_' + shop_display_sorted['number_<<ARGS[4]>>_index']] = $ARGS[2]
		shop_display_sorted['number_<<ARGS[4]>>_number_' + shop_display_sorted['number_<<ARGS[4]>>_index']] = ARGS[3]
		shop_display_sorted['number_<<ARGS[4]>>_index'] += 1
		shop_display_sorted['max_number'] = max(shop_display_sorted['max_number'], ARGS[4])
		exit
	end

	if $ARGS[1] = 'add_to_quality':
		!! $ARGS[2]: type
		!! ARGS[3]: number
		!! ARGS[4]: quality

		$shop_display_sorted['quality_<<ARGS[4]>>_type_' + shop_display_sorted['quality_<<ARGS[4]>>_index']] = $ARGS[2]
		shop_display_sorted['quality_<<ARGS[4]>>_number_' + shop_display_sorted['quality_<<ARGS[4]>>_index']] = ARGS[3]
		shop_display_sorted['quality_<<ARGS[4]>>_index'] += 1
		shop_display_sorted['max_quality'] = max(shop_display_sorted['max_quality'], ARGS[4])
		exit
	end

	if $ARGS[1] = 'add_to_inhibition':
		!! $ARGS[2]: type
		!! ARGS[3]: number
		!! ARGS[4]: inhibition

		ARGS[4] /= 5

		$shop_display_sorted['inhibition_<<ARGS[4]>>_type_' + shop_display_sorted['inhibition_<<ARGS[4]>>_index']] = $ARGS[2]
		shop_display_sorted['inhibition_<<ARGS[4]>>_number_' + shop_display_sorted['inhibition_<<ARGS[4]>>_index']] = ARGS[3]
		shop_display_sorted['inhibition_<<ARGS[4]>>_index'] += 1
		shop_display_sorted['max_inhibition'] = max(shop_display_sorted['max_inhibition'], ARGS[4])
		exit
	end

	if $ARGS[1] = 'add_to_heelsize':
		!! $ARGS[2]: type
		!! ARGS[3]: number
		!! ARGS[4]: heelsize

		$shop_display_sorted['heelsize_<<ARGS[4]>>_type_' + shop_display_sorted['heelsize_<<ARGS[4]>>_index']] = $ARGS[2]
		shop_display_sorted['heelsize_<<ARGS[4]>>_number_' + shop_display_sorted['heelsize_<<ARGS[4]>>_index']] = ARGS[3]
		shop_display_sorted['heelsize_<<ARGS[4]>>_index'] += 1
		shop_display_sorted['max_heelsize'] = max(shop_display_sorted['max_heelsize'], ARGS[4])
		exit
	end

	if $ARGS[1] = 'add_to_price':
		!! $ARGS[2]: type
		!! ARGS[3]: number
		!! ARGS[4]: price

		ARGS[5] = func('shop_utils', 'get_discount', $ARGS[2], ARGS[3])
		ARGS[4] = ARGS[4] * (100 - ARGS[5]) / 5000 * 50

		$shop_display_sorted['price_<<ARGS[4]>>_type_' + shop_display_sorted['price_<<ARGS[4]>>_index']] = $ARGS[2]
		shop_display_sorted['price_<<ARGS[4]>>_number_' + shop_display_sorted['price_<<ARGS[4]>>_index']] = ARGS[3]
		$shop_display_sorted['discount_<<ARGS[5]>>_type_' + shop_display_sorted['discount_<<ARGS[5]>>_index']] = $ARGS[2]
		shop_display_sorted['discount_<<ARGS[5]>>_number_' + shop_display_sorted['discount_<<ARGS[5]>>_index']] = ARGS[3]

		shop_display_sorted['price_<<ARGS[4]>>_index'] += 1
		shop_display_sorted['discount_<<ARGS[5]>>_index'] += 1

		shop_display_sorted['max_price'] = max(shop_display_sorted['max_price'], ARGS[4])
		shop_display_sorted['max_discount'] = max(shop_display_sorted['max_discount'], ARGS[5])
		exit
	end
	exit
end


if $ARGS[0] = 'display':
	!! $ARGS[1]: function name

	if no loc($shop_display['view_file']):
		msg 'shop_utils ERROR: shop_utils, display called with non-existing view_file: "<<$shop_display[''view_file'']>>"'
		exit
	end
	if no loc($shop_display['func_file']):
		msg 'shop_utils ERROR: shop_utils, display called with non-existing func_file: "<<$shop_display[''func_file'']>>"'
		exit
	end

	if $ARGS[1] = 'init':
		$shop_display['display_func'] = $ARGS[2]
		shop_display['do_shop_title'] = ARGS[3]
		exit
	end

	if $ARGS[1] = 'main':
		if outfitfilter['number_sort']:
			$shop_display['sort_type'] = 'number'
		elseif outfitfilter['quality_sort']:
			$shop_display['sort_type'] = 'quality'
		elseif outfitfilter['inhibition_sort']:
			$shop_display['sort_type'] = 'inhibition'
		elseif outfitfilter['heelsize_sort']:
			$shop_display['sort_type'] = 'heelsize'
		elseif outfitfilter['price_sort']:
			$shop_display['sort_type'] = 'price'
		elseif outfitfilter['discount_sort']:
			$shop_display['sort_type'] = 'discount'
		else
			outfitfilter['number_sort'] = 1
			$shop_display['sort_type'] = 'number'
		end

		gs $shop_display['view_file'], 'display', $shop_display['display_func'], 'header'

		if outfitfilter['sort_direction'] = 0:
			shop_display['j'] = 0
			shop_display['j_inc'] = 1
			shop_display['j_max'] = shop_display_sorted['max_' + $shop_display['sort_type']]
		else
			shop_display['j'] = shop_display_sorted['max_' + $shop_display['sort_type']]
			shop_display['j_inc'] = -1
			shop_display['j_max'] = 0
		end
		:display_filter_loop
			if shop_display['do_shop_title'] and $shop_display['sort_type'] = 'number':
				$shop_display['shop_header'] =  $shop_display_sorted['number_' + shop_display['j'] + '_title']
				if $shop_display['shop_header'] <> '':
					*nl
					*pl '<center><img src="<<$func(''$shop_icon_image'', $shop_display[''shop_header''])>>"></center>'
				end
			end

			shop_display['i'] = 0
			:main_display_loop
				$shop_display['type'] = $shop_display_sorted[$shop_display['sort_type'] + '_' + shop_display['j'] + '_type_' + shop_display['i']]
				shop_display['number'] = shop_display_sorted[$shop_display['sort_type'] + '_' + shop_display['j'] + '_number_' + shop_display['i']]
			if $shop_display['type'] <> '' and shop_display['number'] <> 0:
				if dyneval('result = ' + $shop_display['inclusion_check'], $shop_display['func_file'], $shop_display['type'], shop_display['number']):
					gs $shop_display['view_file'], 'display', $shop_display['display_func'], 'main', $shop_display['link'], $shop_display['type'], shop_display['number'], shop_display['discount']
				end
				shop_display['i'] += 1
				jump 'main_display_loop'
			end
			shop_display['j'] += shop_display['j_inc']
		if 0 <= shop_display['j'] and shop_display['j'] <= shop_display_sorted['max_' + $shop_display['sort_type']]:
			jump 'display_filter_loop'
		end

		gs $shop_display['view_file'], 'display', $shop_display['display_func'], 'footer'
		exit
	end

	if $ARGS[1] = 'wardrobe_storage_unwanted_header':
		*p '<center><table border=1><tr>'
		if $shop_display['link'] <> 'wardrobe':	*p '<th><a href="exec:gs ''shop_utils'', ''cleanup'' & gt ''<<$shop_display[''view_file'']>>'', ''<<$ARGS[2]>>'', ''wardrobe''">衣柜</a></th>'
		if $shop_display['link'] <> 'storage':	*p '<th><a href="exec:gs ''shop_utils'', ''cleanup'' & gt ''<<$shop_display[''view_file'']>>'', ''<<$ARGS[2]>>'', ''storage''">储物</a></th>'
		if $shop_display['link'] <> 'unwanted':	*p '<th><a href="exec:gs ''shop_utils'', ''cleanup'' & gt ''<<$shop_display[''view_file'']>>'', ''<<$ARGS[2]>>'', ''unwanted''">不想要的</a></th>'
		*pl '</tr></table></center>'
		exit
	end

	if $ARGS[1] = 'grid_shop':
		gs 'shop_utils', 'display', 'init', 'grid_shop', 0
		gs 'shop_utils', 'display', 'main'
		exit
	end


	if $ARGS[1] = 'grid_wardrobe':
		gs 'shop_utils', 'display', 'init', 'grid_wardrobe', 1
		gs 'shop_utils', 'display', 'main'
		exit
	end


	if $ARGS[1] = 'list_wardrobe':
		gs 'shop_utils', 'display', 'init', 'list_wardrobe', 0
		gs 'shop_utils', 'display', 'main'
		exit
	end
	exit
end


if $ARGS[0] = 'display_grid_shop':
	gs 'shop_utils', 'display', 'grid_shop'
	exit
end


if $ARGS[0] = 'display_grid_wardrobe':
	gs 'shop_utils', 'display', 'grid_wardrobe'
	exit
end


if $ARGS[0] = 'display_list_wardrobe':
	gs 'shop_utils', 'display', 'list_wardrobe'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                                CORE                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'return':
	dynamic "
		gs 'shop_utils', 'cleanup'
		gt '<<$shop_display['return_loc']>>', '<<$shop_display['return_arg0']>>', '<<$shop_display['return_arg1']>>'
	"
end


if $ARGS[0] = 'cleanup':
