!! DEPRECATED
if $ARGS[0] = 'gym':
	if $regularwornclothingtype = '':
		$regularwornclothingtype = $clothingworntype
		regularwornclothingnumber = clothingwornnumber
	end

	'<center><img src="images/locations/city/citycenter/mall/sports.png"></center>'

	*pl $func('clothing_QV', 'get_filter_header', 'storage')
	!! To filter just this set
	clo_i = 0
	:loopdanilovich_outfits_filter2

	i = 1
	:loopdanilovich2
	if danilovich_outfits[i] = 1:
		gs 'clothing_attributes', 'danilovich_outfits', i
		gs 'clothing_view', 'home_filter'
		!! sort by filter
		if (outfitfilter['quality_sort'] = 1 and CloQuality = clo_i) or (outfitfilter['inhibition_sort'] = 1 and CloInhibit = clo_i) or outfitfilter['number_sort'] = 1:
			if outfitfilter['include'] = 1 and outfitfilter['sport'] >= 0:
				if func('clothing', 'can_wear', 'danilovich', i): *p '<a href="exec:gt ''clothing_QV'', ''change'', ''danilovich_outfits'', <<i>>"><img height="250" src="images/pc/items/danilovich/outfits/<<i>>.jpg"/></a>'
			end
		end
	end
	i += 1
	if i <= arrsize('danilovich_outfits'): jump 'loopdanilovich2'

	clo_i += 1
	if outfitfilter['quality'] = 0 and clo_i <= 7:
		jump 'loopdanilovich_outfits_filter2'
	elseif outfitfilter['inhibition'] = 0 and clo_i <= 50:
		jump 'loopdanilovich_outfits_filter2'
	end

	act '返回': gt $loc, $loc_arg

	if $clothingworntype ! $regularwornclothingtype: gs 'clothing_QV', 'gym2'

	if $clothingworntype ! 'nude':
		act '脱衣': gt 'clothing_QV', 'strip'
	end
end

!! DEPRECATED
if $ARGS[0] = 'change':
	*clr & cla
	$swimwear_description = ''
	gs 'clothing_attributes', $ARGS[1], ARGS[2]
	'<center><img <<$set_imgh>> src="<<$func(''$clothing_image'', $ARGS[1], ARGS[2])>>"></center>'
	if $ARGS[1] = 'misc_outfits' and ARGS[2] = 1:
		'猎人给你的粗麻袋。'
	else
		*pl $func('$short_description', $ARGS[1], ARGS[2], CloStyle2)
		if $swimwear_description = '':
			gs 'clothing_descriptions'
			'<<$description>>'
		else
			$swimwear_description
		end
	end

	if CloBimbo = 1 and cheatVars['bimbo'] = 0: $temp_clo_text[] = 'bimbo'
	if CloGoth = 1: $temp_clo_text[] = 'goth'
	if CloPunk = 1: $temp_clo_text[] = 'punk'

	if arrsize('$temp_clo_text') > 0:
		'这件物品被视为<<func(''string'', ''enumerate_list'', ''$temp_clo_text'')>>服装。'
	end
	killvar 'temp_clo_text'

	if CloStyle = 4: '这套服装可用于卖淫。'

	!! CloStyle2: 1 Maid, 2 Server, 3 Stripper, 4 School, 5 Office, 6 Sport
	if CloStyle2 = 1 or CloMaid = 1: '这是一套女仆装。'
	if CloStyle2 = 2 or CloServer = 1: '这套衣服是服务员制服。'
	if CloStyle2 = 3 or CloStrip = 1: '这套衣服是脱衣舞娘制服。'
	if CloStyle2 = 4 or CloSchool = 1: gs 'clothing', 'school_check', 'check'
	if CloStyle2 = 5 or CloOffice = 1: '这套衣服是办公室制服。'
	if CloStyle2 = 6 or CloSport = 1: '这套衣服是健身服。'


	act '返回': gt 'clothing_QV', 'clothing_list', '<<$ward_list_store>>'

	if $clothingworntype = $ARGS[1] and clothingwornnumber = ARGS[2]:
		'你现在正穿着这套衣服。'
	elseif CloInhibit > pcs_inhib:
		'你觉得自己还没有足够的自信穿上这么暴露的衣服。'
	elseif arrpos('$CloLosTyp', $ARGS[1]) >= 0 and arrpos('CloLosNum', ARGS[2]) >= 0:
		'你在某处丢失了这些衣服，也许你能再找到它们？。'
	else
		if CloInhibit > 10:
			if CloInhibit + 10 > pcs_inhib: '你发现这套衣服比你完全能接受的更暴露，但这反而让它相当刺激。'
		end

		!! Therapist force exhibitionist clothing check
		if hypnoClothes <= 0 or func('clothing', 'is_hypno_approved', $ARGS[1], $ARGS[2]):
			act '穿这套衣服':
				gs 'clothing', 'wear', $ARGS[1], ARGS[2]
				if $regularwornclothingtype = '':
					gt 'wardrobe', 'main'
				else
					gt 'clothing_QV', 'gym'
				end
			end
		else
			act '穿这套衣服':
				*clr & cla
				'你拿起这套衣服，想着试一试……但你实在不喜欢穿这种类型的衣服。'
				gs 'willpower', 'misc', 'resist', 'easy'
				if cheatVars['willpower'] = 0: will_cost += hypnoTime
				if pcs_willpwr < will_cost:
					act '还是穿上它们' + $func('willpower', 'get_willcost_string'): $noWillpower
				else
					act '还是穿上它们' + $func('willpower', 'get_willcost_string'):
						gs 'willpower', 'pay', 'resist'
						gs 'stat'
						gs 'clothing', 'wear', $ARGS[1], ARGS[2]

						if $regularwornclothingtype = '':
							gt 'wardrobe', 'main'
						else
							gt 'clothing_QV', 'gym'
						end
					end
				end
				act '返回':
					if $regularwornclothingtype = '':
						gt 'wardrobe', 'main'
					else
						gt 'clothing_QV', 'gym'
					end
				end
			end
		end
	end
end
