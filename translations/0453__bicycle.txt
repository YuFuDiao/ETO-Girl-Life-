# bicycle

!! BikeLoc - tells us where bicycle is currently located
!! ----------------------------------------------
!! BikeLoc = 1 'gad_gpbarn' ''
!! BikeLoc = 2 'gad_forest' 'forest_edge'
!! BikeLoc = 3 'gad_swamp_yard' 'start'
!! ----------------------------------------------

!! this gs handles biking for Sveta in the various locations of gadukino
!! will only display bicycle actions if the BikeLoc is correct for the location passed in
!! $ARGS[0] - location where this is called from
!! Example: gs 'bicycle', 'gad_gpbarn'
if ($ARGS[0] = 'gad_gpbarn' and BikeLoc = 1) or ($ARGS[0] = 'gad_forest' and BikeLoc = 2) or ($ARGS[0] = 'gad_swamp_yard' and BikeLoc = 3):
	act '骑你的自行车':
		*clr & cla
		menu_off = 1

		if $clothingworntype = 'nude':
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/findvel_nude.jpg"></center>'
		else
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/findvel.jpg"></center>'
		end
		*nl
		'你看到了你的旧自行车。它看起来状况还不错，你应该可以毫不费力地骑上它。'

		if BikeLoc = 1:
			act '别动你的自行车': gt 'gad_gpbarn'
		elseif BikeLoc = 2:
			act '别动你的自行车': gt 'gad_forest', 'forest_edge'
		elseif BikeLoc = 3:
			act '别动你的自行车': gt 'gad_swamp_yard', 'start'
		end
		if BikeLoc = 1 and hour >= 6 and hour < 20 and sunWeather = 1 and month >= 4 and month <= 10 and gadbikeday ! daystart:
			act '骑车环村（1:00）':
				*clr & cla
				menu_off = 1
				minut += 60
				gs 'exp_gain', 'run', rand(0, 2)
				gs 'exp_gain', 'stren', rand(0, 2)
				fat -= 2
				gs 'mood', 'raise', 'tiny'
				gs 'sweat', 'add', 2
				gadbikeday = daystart

				if $clothingworntype = 'nude':
					pcs_horny += 15
					gs 'mood', 'raise', 'small'
					if pcs_inhib < 50: inhib_exp += rand(0, 2)
					gs 'arousal', 'flash', 5
					gs 'arousal', 'end'

					'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike_nude.jpg"></center>'
					*nl
					'你赤身裸体地骑着车，避开人多的地方，为这份新生的勇气而欣喜若狂。很难说是什么让你更兴奋——是恐惧，还是渴望被人注意。'
				elseif PCloSkirt > 0:
					bikerand = rand(0, 10)
					if bikerand >= 8:
						if $pantyworntype ! 'none':
							pcs_horny += 5
							gs 'mood', 'raise', 'tiny'
							if pcs_inhib < 20: inhib_exp += rand(0, 2)
							gs 'arousal', 'flash', 5
							gs 'arousal', 'end'
							gs 'stat'

							'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike_tanga_1.jpg"></center>'
							*nl
							'骑车途中，你几次掀起裙子，向几个路过的幸运村民露出内裤的一角。'
						else
							pcs_horny += 10
							gs 'mood', 'raise', 'tiny'
							if pcs_inhib < 30: inhib_exp += rand(0, 2)
							gs 'arousal', 'flash', 5
							gs 'arousal', 'end'
							gs 'stat'

							'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike_tanga_2.jpg"></center>'
							*nl
							'骑车途中，你几次掀起裙子，向几个路过的幸运村民展示你的光屁股。'
						end
					else
						gs 'mood', 'raise', 'tiny'
						gs 'stat'

						'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike_2.jpg"></center>'
						*nl
						'你骑着自行车，在村子那似乎一望无际的田野边呼吸着新鲜空气。'
					end
				elseif PCloSkirt = 0:
					gs 'mood', 'raise', 'tiny'
					gs 'stat'

					'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike.jpg"></center>'
					*nl
					'你骑着自行车，享受着周围的乡村风光。'
				end
				act '完成你的骑乘': gt 'gad_gpbarn'
			end
		end

		if BikeLoc ! 1:
			if BikeLoc = 2:
				$bikeaction = '骑自行车去爷爷奶奶的谷仓（0:12）'
			elseif BikeLoc = 3:
				$bikeaction = '骑自行车去爷爷奶奶的谷仓（0:44）'
			end
			act '<<$bikeaction>>':
				*clr & cla
				menu_off = 1
				if BikeLoc = 2:
					minut += 12
					gs 'exp_gain', 'run', rand(0, 1)
					gs 'exp_gain', 'stren', rand(0, 1)
					fat -= 1
					gs 'mood', 'raise', 'tiny'
					gs 'sweat', 'add', 1
					if swamp_clothes = 1:
						'你脱掉猎人给你的那身简直算不上衣服的衣服。你确信能在祖父母家里找到更好的衣服穿。'
						gs 'clothing', 'strip', 'gad_forest'
						gs 'clothing', 'gad_swamp_clothes', 0
					end
				elseif BikeLoc = 3:
					minut += 44
					gs 'exp_gain', 'run', rand(0, 3)
					gs 'exp_gain', 'stren', rand(0, 3)
					fat -= 3
					gs 'mood', 'raise', 'small'
					gs 'sweat', 'add', 3
					if swamp_clothes = 1:
						'你脱掉猎人给你的那身简直算不上衣服的衣服。你确信能在祖父母家里找到更好的衣服穿。'
						gs 'clothing', 'strip', 'gad_forest'
						gs 'clothing', 'gad_swamp_clothes', 0
					end
				end
				BikeLoc = 1
				gs 'stat'

				gs 'bicycle', 'clothes'
				*nl
				'你跳上你的自行车，骑往爷爷奶奶的谷仓。'

				act '完成你的骑乘': gt 'gad_gpbarn'
			end
		end

		if BikeLoc ! 2 and pcs_bushcraft > 10:
			if BikeLoc = 1:
				$bikeaction = '骑车去森林边（0:12）'
			elseif BikeLoc = 3:
				$bikeaction = '骑车去森林边（0:32）'
			end
			act '<<$bikeaction>>':
				*clr & cla
				menu_off = 1
				if BikeLoc = 1:
					minut += 12
					gs 'exp_gain', 'run', rand(0, 1)
					gs 'exp_gain', 'stren', rand(0, 1)
					fat -= 1
					gs 'mood', 'raise', 'tiny'
					gs 'sweat', 'add', 1
				elseif BikeLoc = 3:
					minut += 32
					gs 'exp_gain', 'run', rand(0, 2)
					gs 'exp_gain', 'stren', rand(0, 2)
					fat -= 2
					gs 'sweat', 'add', 2
				end
				BikeLoc = 2
				gs 'stat'

				gs 'bicycle', 'clothes'
				*nl
				'你跳上自行车，骑到森林的边缘。'

				act '完成你的骑乘': gt 'gad_forest', 'forest_edge'
			end
		end

		if BikeLoc ! 3 and hunterVars['cabin'] = 1 and hunterVars['were_met'] = 1:
			if BikeLoc = 1:
				$bikeaction = '骑自行车去猎人小屋（0:44）'
			elseif BikeLoc = 2:
				$bikeaction = '骑自行车去猎人小屋（0:32）'
			end
			act '<<$bikeaction>>':
				*clr & cla
				menu_off = 1
				if BikeLoc = 1:
					minut += 44
					gs 'exp_gain', 'run', rand(0, 3)
					gs 'exp_gain', 'stren', rand(0, 3)
					fat -= 3
					gs 'mood', 'raise', 'small'
					gs 'sweat', 'add', 3
				elseif BikeLoc = 2:
					minut += 32
					gs 'exp_gain', 'run', rand(0, 2)
					gs 'exp_gain', 'stren', rand(0, 2)
					fat -= 2
					gs 'mood', 'raise', 'tiny'
					gs 'sweat', 'add', 2
				end
				BikeLoc = 3
				gs 'stat'

				gs 'bicycle', 'clothes'
				*nl
				'你跳上你的自行车，骑到猎人小屋。'

				act '完成你的骑乘': gt 'gad_swamp_yard', 'start'
			end
		end
	end
end

!! this gs handles images for biking for Sveta in the various locations of gadukino
!! Example: gs 'bicycle', 'clothes'
if $ARGS[0] = 'clothes':
	!! need images for biking during snow/rain/winter/nighttime: currently only have summer daytime images
	!! if sunWeather = 0 and (month < 4 or month > 10): snowing
	!! else sunWeather = 0 and month >= 4 and month <= 10: raining
	!! elseif month < 4 or month > 10 and hour < 6 or hour >= 22: winter nighttime
	!! elseif month < 4 or month > 10: winter daytme
	!! elseif hour < 6 or hour >= 22: summer nighttime
	!! else summer daytime
		if $clothingworntype = 'nude':
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike_nude.jpg"></center>'
		elseif PCloSkirt > 0:
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike_2.jpg"></center>'
		else
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/bike.jpg"></center>'
		end
	!! end
end
