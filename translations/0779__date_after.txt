	if date_ev['unique_npc'] = 1:

				gt 'city_center'
			elseif $npc_residence[$npcID] = '':
				!! failsafe for an npc with no residence set
				!! $region is 'pav', 'city', 'gad', 'pushkin'
				if $region = 'pav':
					gt 'pav_residential'
				elseif $region = 'city':

					$hotel = 'city_hotel'
					gs 'date_after', 'ending'
					gt 'HotelRoom', 'best'

					gs 'date_after', 'ending'
					msg '上报安全退出！ (hotel_room, hotel_days > 0, unhandled $region)'
					gt 'HotelRoom', 'better'
				end
			else
				if $region = 'pav':
					gs 'date_after', 'ending'
					gt 'pav_hotel'
				elseif $region = 'city':
					gs 'date_after', 'ending'
					gt 'city_hotel'
