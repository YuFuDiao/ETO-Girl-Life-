# city_house_res_kitch


if $ARGS[0] = 'kitch':
	*clr & cla
	menu_off = 0
	minut += 1
	gs 'core_library', 'setloc', 'city_house_res_kitch', 'kitch'
	$location_type = 'private'
	$locclass = 'kitr'

	gs 'kit_din'
	gs 'stat'

	$func('wrap', 'header center b', '厨房')
	'<center><img <<$set_imgh>> src="images/locations/city/residential/house/crh_kitchen.jpg"></center>'

	act '离开房间': gt 'city_house_res_misc', 'hallw'

	'一个储备充足的厨房包含制作任何饭菜所需的一切。'

	if mc_inventory['dish_plates'] > 0:
		'你还有<b><<mc_inventory[''dish_plates'']>></b>个干净的盘子。'
	else
		$func('wrap', 'neg b', '你已经没有干净的盘子了。')
	end

	if dirttarelka > 0: '水槽里有<b><<dirttarelka>></b>个脏盘子。 <a href="exec:gs ''kit_din'', ''dirtarm''">洗盘子</a>.'

	if mc_inventory['dish_soap'] > 0:
		'水槽下面有一些洗洁精，足够<b><<mc_inventory[''dish_soap'']>></b> ' + iif(mc_inventory['dish_soap'] = 1, '次。', '次。')
	else
		$func('wrap', 'v_neg b', '你没有可用来洗碗的东西。你应该买一些洗洁精。')
	end

	if mc_inventory['food_basic'] > 0:
		if mc_inventory['dish_plates'] = 0 or edahot > 0: $edagot = ''
		if mc_inventory['dish_plates'] > 0 and edahot = 0: $edagot = '<a href="exec:gt ''kit_din'', ''edagotd''">做一顿饭</a>'

		'有足够的食物，够<b><<mc_inventory[''food_basic'']>></b> ' + iif(mc_inventory['food_basic'] = 1, '人份', '人份') + '. <<$edagot>>'
	elseif mc_inventory['food_diet'] = 0 and mc_inventory['food_basic'] = 0:
		$func('wrap', 'v_neg b', '冰箱是空的。你什么吃的都没有。')
	end

	gs 'kit_din', 'edahota'
	gs 'lover_living', 'kitchen'

	gs 'core_library', 'kitchen', 'full'
end
