# gargazel

CLOSE ALL
$loc_arg = $ARGS[0]
$loc = 'gargazel'
$menu_loc = 'gargazel'
$menu_arg = ''
menu_off = 0
$location_type = 'secluded'
fatherdrunk = 0
clr
gs 'family_schedule'
gs 'stat'
'<center><h4>车库</h4></center>'

if locat['Stepdad'] = 9:
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/resident/apartment/garage/gargazel.jpg"></center>'
	'这是一间属于你继父的大车库。每天傍晚和晚上，只要一有空，他就会在里面修理他心爱的嘎斯皮卡车。你能看出门没有上锁，他现在大概就在里面。'
else
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/resident/apartment/garage/gargazel_empty.jpg"></center>'
	'这是一间属于你继父的大车库。每天傍晚和晚上，只要一有空，他就会在里面修理他心爱的嘎斯皮卡车。现在车库关着门，门上挂着一把大挂锁。'
end

if locat['Stepdad'] = 9:
	if fatherdrunk = 1:
		'你的<a href="exec:gt ''garfather''">继父和米沙叔叔</a>正在车库里喝伏特加。'
	else
		'你的<a href="exec:gt ''garfather''">继父</a>正在车库里修理他的Gazelle皮卡车。'
	end
end

act '离开车库': minut += 5 & gt 'pav_complex', 'garages'

if YouCanGar > 0:
	'你已经存放了<<mc_inventory[''trinkets_garage'']>>个小饰品在车库里。你继父允许你在他的车库里存放一些东西，但不想让你占满他所有的空间，他只允许你在这里存放最多100个小饰品。这个车库不太好进出；如果你想在火车站卖掉这些小饰品，你得先把它们带回家。'

	if mc_inventory['trinkets_home'] >= 30:
		'你家里已经存放了太多东西，没有更多空间了。'
	elseif mc_inventory['trinkets_home'] < 30:
		trinkets_can_take = 30 - mc_inventory['trinkets_home']
		'你可以带<<trinkets_can_take>>个小饰品回家并存放起来，直到你有时间收集它们并在火车站出售。'

		if mc_inventory['trinkets_garage'] > 0:
			if mc_inventory['trinkets_garage'] >= trinkets_can_take:
				act '移动<<trinkets_can_take>>个小饰品回家':
					cls
					menu_off = 1
					minut += 60
					mc_inventory['trinkets_home'] += trinkets_can_take
					mc_inventory['trinkets_garage'] -= trinkets_can_take
					gs 'stat'
					'你花了一个小时从各处收集你的小饰品，把它们带回家藏了起来。'

					act '继续': gt $curloc
				end
			end

			if mc_inventory['trinkets_garage'] < trinkets_can_take:
				act '收集<<mc_inventory[''trinkets_garage'']>>个小饰品随身携带':
					cls
					menu_off = 1
					minut += 30
					mc_inventory['trinkets_home'] += mc_inventory['trinkets_garage']
					mc_inventory['trinkets_garage'] = 0
					gs 'stat'
					'你花了半小时在车库里从各处收集你的小饰品，把它们装进袋子里，这样你就可以去火车站把它们卖掉了。'

					act '继续': gt $curloc
				end
			end
		end
	end
end
