!! $ARGS[1] - shoe type name / last worn
!!  ARGS[2] - shoe index
if $ARGS[0] = 'wear':
	if $ARGS[1] = 'last_worn':
		$ARGS[1] = $lastwornshoetype
		ARGS[2] = lastwornshoenumber
	end
	if $ARGS[1] = '' or $ARGS[1] = 'none': exit

	gs 'shoes', 'strip'

	gs 'shoe_attributes', $ARGS[1], ARGS[2]

	if ShoQuality = 0:
		msg '错误：鞋子 "<<$ARGS[1]>>_shoe[<<ARGS[2]>>]" 不存在'
		exit
	end

	if arrpos('$ARGS', 'check') > 0:
		$temp_not_wear_reason = $func('shoes', 'not_wear_reason', $ARGS[1], ARGS[2], 'attributes_set')
		if $temp_not_wear_reason <> '':
			if $temp_not_wear_reason = 'not_owned':
				msg '你没有这双鞋子'
			elseif $temp_not_wear_reason = 'not_in_wardrobe':
				msg '这双鞋子不在你的衣柜里'
			elseif $temp_not_wear_reason = 'low_heels_skill':
				msg '你的高跟鞋技能太低了'
			end
			killvar '$temp_not_wear_reason'
			exit
		end
		killvar '$temp_not_wear_reason'
	end

	$shoeworntype = $ARGS[1]
	shoewornnumber = ARGS[2]

	dynamic "
		<<$shoeworntype>>_shoe_s[<<shoewornnumber>>] = 0
	"

	PShoHeels = ShoHeels
	PShoCut = ShoCut
	PShoStyle = ShoStyle
	PShoStyle2 = ShoStyle2
	PShoBimbo = ShoBimbo

	PShoPunk = ShoPunk

	PShoStrip = ShoStrip
	PShoSport = ShoSport
	PShoPain['severe'] = ShoPain['severe']
	PShoPain['medium'] = ShoPain['medium']
	PShoPain['mild'] = ShoPain['mild']

		PXShoHeels = 0
	elseif PShoHeels = 1:
		PXShoHeels = 25
	elseif PShoHeels = 2:
		PXShoHeels = 50
	elseif PShoHeels = 3:
		PXShoHeels = 100
	elseif PShoHeels = 4:
		PXShoHeels = 150
	elseif PShoHeels = 5:
		PXShoHeels = 200
	elseif PShoHeels = 6:
		PXShoHeels = 300
	else
		PXShoHeels = 400
	end

			if '<<$ARGS[1]>>' = $shoeworntype and <<ARGS[2]>> = shoewornnumber:

			end

			if '<<$ARGS[1]>>' = $shoeworntype and <<ARGS[2]>> = shoewornnumber:
