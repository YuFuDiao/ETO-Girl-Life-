if $ARGS[0] = 'get_multiplied':
	!! returns the multiplier for a given type based on the current difficulty settings.
	!! ARGS[1] = multiplier option variable (e.g. cheatVars['serv_mult_opt'])
	!! ARGS[2] = base value to multiply
	!! ARGS[3] = custom multiplier value (optional, used when ARGS[1] = 99)
	!!
	!! Opt table (symmetric around 0 = Default):
	!! -99 = Off (0%, disabled entirely)
	!!  -4 = ⅓x  (33%)	-3 = ½x  (50%)	-2 = ⅔x  (67%)	-1 = ¾x  (75%)
	!!	0 = 1x  (100%, Default)
	!!	1 = 1¼x (125%)	2 = 1½x (150%)	3 = 2x	(200%)	4 = 3x	(300%)
	!!  99 = Custom (uses ARGS[3] as %)
	if ARGS[1] = -99:
		result = 0
	elseif ARGS[1] = 0:
		result = ARGS[2]
	elseif ARGS[1] = 1:
		result = (ARGS[2] * 5) / 4
	elseif ARGS[1] = 2:
		result = (ARGS[2] * 3) / 2
	elseif ARGS[1] = 3:
		result = ARGS[2] * 2
	elseif ARGS[1] = 4:
		result = ARGS[2] * 3
	elseif ARGS[1] = -1:
		result = (ARGS[2] * 3) / 4
	elseif ARGS[1] = -2:
		result = (ARGS[2] * 2) / 3
	elseif ARGS[1] = -3:
		result = ARGS[2] / 2
	elseif ARGS[1] = -4:
		result = ARGS[2] / 3
	elseif ARGS[1] = 99:
		!! Custom multiplier option
		result = (ARGS[2] * ARGS[3]) / 100
	end
	exit
end

!! following function returns the basic adjustment for max_grades.
!! skill_gain 1 (Very Fast):	185% of intel
!! skill_gain 2 (Fast):			155% of intel
!! skill_gain 3 (Normal):		133% of intel
!! skill_gain 4 (Slow):			115% of intel
!! TODO: migrate to a unique var if we ever actually use this to limit max grades
!! it currently uses skill_gain because that replaced what was previously the general "difficulty" var.
!! use func('_difficulty','getmaxgrades')
if $ARGS[0] = 'getmaxgrades':
	if cheatVars['skill_gain'] = 1 : max_grades = (pcs_intel * 185) / 100
	if cheatVars['skill_gain'] = 2 : max_grades = (pcs_intel * 155) / 100
	if cheatVars['skill_gain'] = 3 : max_grades = (pcs_intel * 133) / 100
	if cheatVars['skill_gain'] = 4 : max_grades = (pcs_intel * 115) / 100

	if max_grades > 100: max_grades = 100
end
