# random_events
!{
	To add new events, copy the following block, modify as needed, and insert it next to the other events:
	!! ── Event Name ──────────────────────────────────────────────
	if <conditions>:
		$rand_events_temp['loc']	= '<location_file>'
		$rand_events_temp['name']	= '<event_name>'
		rand_events_temp['freq']	= <frequency_value>
		rand_events_temp['cooldown'] = <cooldown_value>
		gs 'random_events', 'register'
	end

	- <conditions> are the conditions that must be met for the event to be considered.
	- <location_file> is the file where the event''s code is located (without the .qsrc extension).
	- <rand_events_temp['name']> is the specific event to trigger within that file [via if $ARGS[0] = '<rand_events_temp['name']>'].
	- rand_events_temp['freq'] is the relative chance of the event occurring; higher values increase likelihood.
	An event with a frequency of 30 is ten times more likely to occur than one with a frequency of 3.
	- rand_events_temp['cooldown'] is the cooldown period in hours before the event can occur again.

	Events can occur once every 2 hours.

	- Eva
!}

	if temp_freq = 0: exit

	$rand_events_list_name[] = $rand_events_temp['name']

	killvar 'temp_freq'
	killvar 'rand_events_temp'
	exit
end
