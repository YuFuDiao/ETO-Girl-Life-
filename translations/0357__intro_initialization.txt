if $ARGS[0] = 'init_time':
	!! ------------------------------------------------
	!! Sets daystart, daystart_start, and hour based on
	!! $start_type['loc']. Safe to call whenever loc changes.
	!!
	!! SG: daystart_start is kept as-is (player-chosen in start2).
	!! If not yet set, defaults to end of August.
	!! City: always day 521, hour 13.
	!! Uni: always day 603, hour 7.
	!! ------------------------------------------------
	if $start_type['loc'] = 'city':
		daystart_start = 521
		hour = 13
	elseif $start_type['loc'] = 'uni':
		daystart_start = 603
		hour = 7

		if daystart_start <> 153 and daystart_start <> 183 and daystart_start <> 214: daystart_start = 239
		hour = 9
	end
	daystart = daystart_start
	gs 'time', 'update_date'
	!! SG post-date setup — always re-apply scheduling vars based on chosen date
	if $start_type['loc'] = 'sg':
		kanikuli = 5
		if daystart_start = 239:

			locat['Fam_set_month_inGad'] = 1
		elseif daystart_start = 183:
			locat['Fam_set_month_inGad'] = 1
		elseif daystart_start = 153:
			locat['Fam_set_month_inGad'] = 1
		end
		grandmaQW['last_month_paid'] = month

		kanicont = 245 - daystart_start
	end
	exit
