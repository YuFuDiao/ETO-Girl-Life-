# shop_allure
!! 2022/03/12
if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_allure', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'
	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '魅力')
	if $region = 'pav':
		'<center><img <<$set_imgh>> src="images/locations/city/residential/lake/allure/shop.jpg"></center>'
	else
		'<center><img <<$set_imgh>> src="images/locations/city/residential/lake/allure/shop.jpg"></center>'
	end
	*nl
	'这家商店给人一种海滩夏日的感觉，考虑到它卖的是泳装，这也难怪。'
	*nl

	act '离开':
		if $region = 'pav':
			minut += 6
			gt 'pav_commercial'
		else
			minut += 3
			gt 'city_lake', 'start'
		end
	end
	act '查看泳衣': minut += 5 & gt 'shop_allure', 'swim'
	act '查看比基尼': minut += 5 & gt 'shop_allure', 'bikinis'
end


if $ARGS[0] = 'swim':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_allure', 'swim'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '泳衣 - 均码')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'allure_swimsuit'
			gs 'shop_utils', 'init', 'set_steps', 3, Clothingstock mod 3
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_allure', 'start'
	end
end


if $ARGS[0] = 'bikinis':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_allure', 'bikinis'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '比基尼 - 均码')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'allure_bikinis'
			gs 'shop_utils', 'init', 'set_steps', 4, (Clothingstock / 3) mod 4
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_allure', 'start'
	end
end
