$location_type = 'public_outdoors'

if $ARGS[0] = 'start':
	menu_off = 0
	gs 'core_library', 'setloc', 'city_suburbs', 'start'
	gs 'stat'

	'<center><h4>精英定居点<h4></center>'
	'<left><img <<$set_imgh>> src="images/locations/city/suburb/settlement.jpg" ></left>'

	if func('car_funcs', 'is_here'):
		'你的<a href="exec:gs ''carF'', ''start''"><<$car[''name'']>></a>停在街上。'
	end

	' <a href="exec: minut += 20 & gt ''metro'', ''suburbs''">地铁</a>站距这里有20分钟步行路程。'

	gs 'taxi'

	if AlexandriaQW > 6:
		if hour > 7 and hour < 20: '如果你想见见那位脾气古怪的巫师，可以去拜访<a href="exec: minut += 5 & gt ''AlexandriaHome'', ''intercom''">Aleksei家</a>。'
	end

	act '步行前往主干道 (0:01)': minut += 15 & gt 'road', '1'
	act '步行前往地铁站 (0:20)': minut += 20 & gt 'metro', 'suburbs'
	act '步行前往公交车站 (0:05)': minut += 5 & gt 'bus', 'suburbs'
	act '步行前往运河 (0:15)': minut + 15 & gt 'city_canals', 'start'

	if bdsmclub['unlocked'] = 1:
		act '去BDSM俱乐部': gt 'bdsm_mansion', 'start'
	end

	if func('homes_properties', 'has_access', 'matryona_mansion') = 0:
		'这里有一块待售的空地，可以建一栋大宅邸。'
		act '去看看那块待售的土地 (0:10)': minut += 10 & gt 'city_mansion_entrance'
	elseif func('homes_properties', 'is_property_of_status', 'owned', 'matryona_mansion'):
		constructionstatus = func('homes_properties', 'get_property_construction_status', 'matryona_mansion')
		if constructionstatus = 0:
			'你那块空地就在附近。'
		elseif constructionstatus = 1:
			'你的宅邸已经建了一部分，就在附近。'
		else
			'你的宅邸离这里不远。'
		end
		act '去看看你的宅邸 (0:10)': minut += 10 & gt 'city_mansion_entrance'
	else
		'这里最引人注目的建筑之一，就是你新建成的宅邸。'
		act '参观玛特廖娜宅邸': gt 'city_mansion_entrance'
	end
	act '看看疯人院 (0:02)': minut += 2 & gt 'city_suburbs', 'ps1'
	act '看看那栋阴森的旧楼 (0:02)': minut += 2 & gt 'city_suburbs', 'ps2'

	if AlexandriaQW = 6:
		gs 'willpower', 'misc', 'self', 'easy'
		if pcs_willpwr < will_cost:
			act '寻找亚历山德拉' + $func('willpower', 'get_willcost_string'): $noWillpower
		else
			act '寻找亚历山德拉' + $func('willpower', 'get_willcost_string'):
				gs 'willpower', 'pay', 'easy'
				gt 'alexandriaEv', 'interview1'
			end
		end
	end
end


if $ARGS[0] = 'ps1':
	*clr & cla
	$menu_loc = 'city_suburbs'
	$menu_arg = 'ps1'
	menu_off = 0
	gs 'stat'
	'<center><h4>疯人院<h4></center>'
	'路边标牌上写着："不对公众开放"。'
	if psiklik = 0: '<center><img <<$set_imgh>> src="images/locations/city/suburb/asylum0.jpg" ></center>'
	if psiklik = 1: '<center><img <<$set_imgh>> src="images/locations/city/suburb/asylum1.jpg" ></center>'

	act '离开': minut += 2 & gt 'city_suburbs', 'start'
end


if $ARGS[0] = 'ps2':
	*clr & cla
	$menu_loc = 'city_suburbs'
	$menu_arg = 'ps2'
	menu_off = 0
	gs 'stat'
	'<center><h4>一栋废弃的旧建筑<h4></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/suburb/abandoned.jpg" ></center>'
	'一栋废弃的建筑，破坏了当地社区的面貌，也让附近的居民感到不安。'

	act '离开': minut += 2 & gt 'city_suburbs', 'start'
end
