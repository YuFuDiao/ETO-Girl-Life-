# jobs

!! Jobs System v4 - Unified Architecture with Separate Single-Level Arrays
!! Inspired by the original employment system, created by Hooded Silence
!! Main API for employment management, work tracking, payment, and schedule queries
!!
!! ==========================
!! REFERENCE
!! ==========================
!! Usage:
!!   - Procedures: gs 'jobs', 'function', ...
!!   - Return values: result = func('jobs', 'function', …)
!!	 String returns use: $result = $func('jobs', 'function', …)
!!
!! Employment state
!! - set_employed($job_id): Employ player, init job data, create calendar events.
!!   Output: sets employment_conflict = 1 if schedule conflict detected (warning only).
!! - set_terminated($job_id): End employment on good terms, remove events/bookings.
!! - set_fired($job_id): End employment on bad terms, remove events/bookings.
!! - suspend_job($job_id): Suspend (keep employed) and remove events, clear bookings.
!! - resume_job($job_id): Resume suspended job and restore events if configured.
!! - set_rank($job_id, new_rank): Update job_rank.
!! - terminate_all_jobs(): Force-terminate all employed jobs (cleanup helper).
!!
!! Work tracking / penalties
!! - clock($job_id): Record completed shift; increments worked_count and job_shifts_total.
!! - clock_in($job_id): Start a dynamic shift (sets job_clocked_in if not already set).
!! - clock_out($job_id): End a dynamic shift; adds minutes to worked_count.
!! - missed_shift($job_id): Increment missed count; apply penalties/auto-fire if configured.
!! - daily_check(): Daily miss detection for all employed jobs; handles booking expiry.
!!
!! Payment
!! - paycheck($job_id[, $dest]): Compute pay, reset counters, pay via money system.
!!   Returns: result = amount paid. $dest optional ('cash'/'bank') to override income route.
!! - bonus_pay($job_id, amount): Add bonus to next paycheck.
!! - dock_pay($job_id, amount): Deduct from next paycheck.
!!
!! Schedule & time queries
!! - resolve_schedule_idx($job_id): Returns active schedule index (int).
!! - is_work_day($job_id, daystart): Returns 1 if scheduled, 0 otherwise.
!!     For booking mode: also returns 1 if yesterday''s shift crosses midnight into today.
!! - is_work_time($job_id): Returns 1 if current time falls in work window, 0 otherwise.
!!     For booking mode: iterates per-slot results + yesterday''s cross-midnight overflow.
!! - is_arrival_time($job_id): Returns 1 if player can arrive now, 0 otherwise.
!!     For booking mode: iterates per-slot arrival windows (no cross-midnight arrival).
!! - get_shift_for_day($job_id, daystart): Sets result_arrival/result_start/result_duration/result_end.
!!     For booking mode: scans booked slots, uses per-slot timing (job_slot_arrival/start/shift).
!!     Also populates per-slot arrays: result_slot_count, result_slot_arrival[i],
!!     result_slot_start[i], result_slot_end[i]. Envelope values derived from min/max of slots.
!! - next_work_day($job_id, start_day): Returns next daystart in next 14 days, or 0.
!! - matches_recurring_schedule($job_id, daystart): Returns 1 if weekday matches.
!! - get_shift_from_recurrence($job_id, daystart): Sets result_arrival/start/duration from recur.
!!
!! Booking mode
!! - clear_all_bookings($job_id): Cancel all bookings (all slots) and remove their events.
!! - booking_count($job_id): Returns active booking count.
!! - has_booking_for_day($job_id, daystart): Returns 1 if any slot has a booking record
!!   for that day, 0 otherwise. Pure data check — does NOT detect cross-midnight overflow
!!   from the previous day. Callers needing cross-midnight awareness should additionally
!!   check has_booking_for_day(daystart-1) and inspect per-slot results.
!! - get_booking_data_for_day($job_id, daystart): Returns the booking data string from
!!   the first occupied slot on that day, or '' if no booking exists.
!! - get_next_booking_day($job_id, start_day, max_days): Returns the daystart of the
!!   first booked day starting from start_day (inclusive), or -1 if none found.
!!   max_days defaults to 50. Scans using has_booking_for_day.
!! - book_slot($job_id, daystart, slot_idx, $data): Returns 1 if booked, 0 if rejected
!!   (at max, duplicate slot, out of window, gap violation, or calendar conflict).
!! - get_booking_data($job_id, daystart, slot_idx): Returns booking metadata string or ''.
!! - check_booking_conflict($job_id, daystart[, slot_idx]): Returns 1 if conflict, 0 if clear.
!!   Optional slot_idx uses per-slot timing (job_slot_arrival/shift) if defined.
!! - cancel_booking($job_id, daystart, slot_idx): Cancel a specific booking and remove event.
!!
!! Schedule switching
!! - change_schedule($job_id, schedule_idx): Update active schedule and rebuild calendar event.
!!
!! Custom job definitions
!! - set_job_file($job_id, $file): Register a custom definition file for a job.
!!   Call once (e.g. at init). The mapping persists across fire/re-hire cycles.
!!   get_job_definition will use this file instead of 'jobs_list' for that job_id.
!! - set_job_icon($job_id, $icon_path): Register a custom path for job icons
!!   Call once (e.g. at init), the mapping is persistent.
!!
!! Job data access
!! - get_job_definition($job_id): Populate job definition arrays.
!!   Delegates to $job_definition_source[$job_id] if set, otherwise 'jobs_list'.
!! - get_property($job_id, key): Returns definition property (use result or $result).
!! - check_employment_possible($job_id[, schedule_idx]): Returns 1 if no conflicts, else 0.
!!   Output: $employment_conflict_reason for diagnostics.
!!
!! Calendar helpers
!! - build_event_vars_for_job($job_id[, schedule_idx_override]): Populates event_vars{}.
!! - create_recurring_event($job_id): Create calendar event for fixed/windowed jobs.
!! - create_availability_events($job_id): Create availability reminders (on_demand only).
!!
!! Cleanup
!! - cleanup_job($job_id): Remove all arrays and state for a job (full data wipe).
!!
!! ARCHITECTURE: Uses separate single-level arrays for data storage
!! (Multi-dimensional arrays can''t be properly cleaned up in QSP - see architecture doc)
!!
!! Data structures:
!!
!!   All job data uses separate single-level arrays with the job_id as the
!!   string key.  This is the only approach compatible with QSP''s killvar
!!   limitations (multi-dimensional string keys cannot be individually deleted).
!!   Access pattern:  array_name['job_id']
!!   Schedule arrays: array_name['job_id, schedule_index']
!!
!!   -----------------------------------------------------------------------
!!   EMPLOYMENT STATE  (runtime, per-job, persisted in saves)
!!   -----------------------------------------------------------------------
!!   $job_status['id']               - '' (never employed / cleaned up),
!!                                     'employed', 'terminated', 'fired'
!!   job_rank['id']                  - Integer rank/position level. 0 = base.
!!                                     Used by location code for progression.
!!   job_added['id']                 - daystart value when set_employed was called.
!!   job_hiring_step['id']           - 0 = unaware of job, 1+ = hiring quest
!!                                     progress. Managed by location files,
!!                                     NOT by the jobs engine. Cleared on cleanup.
!!   job_termination_day['id']       - daystart when terminated/fired (0 if active).
!!   $job_termination_reason['id']   - Free-text reason set by caller before
!!                                     set_terminated/set_fired, e.g. 'no_show',
!!                                     'quit', 'retired', 'blacklisted', ''.
!!
!!   --- Payment tracking ---
!!   job_pay_rate['id']              - Active payment amount per interval. Copied
!!                                     from job_pay_rate_def on hire; can be
!!                                     modified at runtime (e.g. promotions).
!!   $job_pay_interval['id']         - Active payment mode. Copied from
!!                                     $job_pay_interval_def on hire.
!!                                     Values: 'event', 'shift', 'hourly',
!!                                             'weekly', 'monthly', 'none'
!!   job_worked_count['id']          - Accumulated work units since last payday.
!!                                     For 'shift': number of shifts (clock increments by 1).
!!                                     For 'hourly': minutes worked (clock adds shift duration;
!!                                       clock_in/clock_out adds elapsed minutes).
!!                                     Reset to 0 by paycheck.
!!   job_bonus_pay['id']             - Bonus ₽ accumulated via bonus_pay. Added to
!!                                     next paycheck, then reset to 0.
!!   job_docked_pay['id']            - Amount ₽ to deduct via dock_pay. Subtracted
!!                                     from next paycheck, then reset to 0.
!!   job_shifts_total['id']          - Lifetime total shifts worked. Incremented
!!                                     by clock/clock_out. Never reset by
!!                                     set_employed — only by cleanup_job.
!!
!!   --- Work tracking ---
!!   job_last_work_day['id']         - daystart of the last day the player worked.
!!                                     Set by clock/clock_out.
!!   job_clocked_in['id']            - totminut when clock_in was called (0 if not
!!                                     currently clocked in). clock_out computes
!!                                     elapsed minutes and adds to worked_count.
!!   job_missed_total['id']          - Lifetime total shifts missed. Incremented
!!                                     by missed_shift. Never reset by set_employed.
!!   job_miss_acknowledged['id']     - Last job_missed_total value the player was
!!                                     warned about. Location code uses this to
!!                                     avoid repeating scolding dialogue.
!!   job_paid_total['id']            - Lifetime total paid out via paycheck
!!                                     (net, after docks and bonuses). Never
!!                                     reset by set_employed — only by cleanup_job.
!!   job_shifts_this_period['id']    - Shifts completed since last paycheck.
!!                                     Incremented by clock and clock_out.
!!                                     Reset to 0 by paycheck.
!!
!!   --- Suspension ---
!!   job_suspended['id']             - 0 = active, 1 = temporarily suspended.
!!                                     suspend_job sets to 1 and removes calendar
!!                                     events. resume_job sets to 0 and restores them.
!!                                     Player remains 'employed' while suspended.
!!
!!   --- Calendar link ---
!!   $job_event_id['id']             - Primary calendar event ID. Automatically set
!!                                     to 'job_' + job_id for recurring events.
!!                                     Used to remove the event on termination.
!!
!!   --- Schedule tracking ---
!!   $job_active_schedule['id']      - '0', '1', etc. Index of the currently active
!!                                     schedule group for multi-schedule jobs.
!!                                     Set by change_schedule.
!!
!!   --- Booking system (only for schedule_mode = 'booking') ---
!!   job_bookings_active['id']       - Number of currently active bookings. Incremented
!!                                     by book_slot, decremented by cancel_booking
!!                                     and daily_check expiry.
!!   job_booking_debt['id']          - Accumulated penalty debt in ₽. Increased by
!!                                     missed_shift (penalty_per_miss) or by custom
!!                                     miss handlers. Auto-fire if > penalty_max_debt.
!!   $job_booking['id, daystart, slot'] - Booking data for a specific day and slot.
!!                                     Contains metadata string (e.g. film type).
!!                                     Non-empty means a booking exists for that
!!                                     day+slot combination.
!!
!!   --- Job list ---
!!   $job_list[]                     - Integer-indexed array of active job IDs.
!!                                     Entries added by set_employed, removed by
!!                                     cleanup_job. Iterated by daily_check,
!!                                     terminate_all_jobs.
!!
!!   --- Custom definition routing ---
!!   $job_definition_source['id']    - Location file name registered via set_job_file.
!!                                     get_job_definition delegates to this file
!!                                     instead of 'jobs_list'. Persists across
!!                                     fire/re-hire cycles.
!!
!!   -----------------------------------------------------------------------
!!   JOB DEFINITIONS  (static templates, loaded on demand from jobs_list)
!!   -----------------------------------------------------------------------
!!   Populated by  gs 'jobs_list', $job_id  (or a custom definition file).
!!   Lazy-loaded: only fetched when $job_title[$id] = ''.
!!
!!   --- Core info ---
!!   $job_title['id']                - Display name shown in UI/calendar.
!!   $job_location['id']             - Location name shown in calendar events.
!!   $job_desc['id']                 - Optional description text.
!!   job_refresh_desc['id']         - Set to 1 to re-call the definition source each time
!!                                     the card is rendered, so jobs_list can recompute
!!                                     $job_desc from runtime state (e.g. weekly counters).
!!
!!   --- Schedule mode ---
!!   $job_schedule_mode['id']        - Determines how the job''s schedule works.
!!                                     'fixed'    — set days/times, player must show up.
!!                                     'windowed' — flexible arrival within a window.
!!                                     'on_demand'— player chooses when to work.
!!                                     'booking'  — player books specific days ahead.
!!
!!   --- Fixed/Windowed schedule properties (per schedule group) ---
!!   $job_work_days['id, N']         - Space-separated weekday numbers for group N.
!!                                     '1'=Mon .. '7'=Sun. E.g. '1 2 3 4 5'.
!!   job_arrival['id, N']            - Earliest arrival time in minutes from midnight.
!!                                     For 'fixed': arrival window opens here.
!!                                     For 'windowed': earliest the player can show up.
!!   job_start['id, N']              - Shift start / latest arrival deadline, minutes.
!!                                     For 'fixed': shift begins here (arrival window closes).
!!                                     For 'windowed': latest time to start work.
!!   job_shift['id, N']              - Shift duration in minutes.
!!   $job_recurrence_pattern['id, N']- Complex recurrence string (advanced usage).
!!                                     If set, overrides work_days/arrival/start/shift.
!!   $job_work_dates['id']           - Date range 'MMDD-MMDD' when the job is active.
!!                                     Maps directly to the yearly- recurrence pattern.
!!                                     E.g. '0601-0831' for summer (Jun–Aug).
!!
!!   --- On-demand properties ---
!!   $job_available_days['id']       - Space-separated weekday numbers when work is
!!                                     available. E.g. '1 2 3 4 5 6 7'.
!!   job_available_start['id']       - Earliest availability, minutes from midnight.
!!   job_available_end['id']         - Latest availability, minutes from midnight.
!!   job_create_availability_events['id'] - 1 = create calendar reminders for
!!                                     availability windows, 0 = don''t.
!!
!!   --- Booking-mode properties ---
!!   job_booking_window_days['id']   - How many days ahead the player can book.
!!   job_booking_slots_per_day['id'] - Maximum bookable slots per day.
!!   job_booking_max_concurrent['id']- Maximum simultaneously active bookings.
!!   job_booking_min_gap_days['id']  - Minimum days required between any two
!!                                     bookings (0 = no gap required).
!!   job_arrival_start['id']         - Booking arrival window start, minutes.
!!                                     Overrides job_arrival for booking-mode
!!                                     get_shift_for_day / is_arrival_time.
!!   job_arrival_end['id']           - Booking arrival window end, minutes.
!!                                     Overrides job_start for booking-mode.
!!   job_slot_arrival['id, slot']    - Per-slot arrival time, minutes from midnight.
!!                                     When defined, book_slot and check_booking_conflict
!!                                     use this instead of job_arrival_start for the given slot.
!!   job_slot_start['id, slot']      - Per-slot shift start (arrival deadline), minutes from
!!                                     midnight. Marks when work actually begins — the window
!!                                     from slot_arrival to slot_start is the arrival window.
!!                                     Falls back to job_arrival_end or slot_arrival + 60.
!!   job_slot_shift['id, slot']      - Per-slot shift duration, minutes (start to end of work).
!!                                     End time = slot_start + slot_shift. If end > 1440,
!!                                     the shift crosses midnight.
!!                                     Calendar event duration = (slot_start - slot_arrival) + slot_shift.
!!                                     Cross-midnight conflict detection is handled natively
!!                                     by the calendar event system (start_ts + duration_ts > 96).
!!
!!   --- Flags ---
!!   job_blocking['id']              - 0 = non-blocking (can coexist with other jobs),
!!                                     1 = blocks the schedule (check_employment_possible
!!                                     will reject overlapping blocking jobs).
!!   job_add_to_calendar['id']       - 1 = create calendar events on hire, 0 = don''t.
!!   job_show_stat_icon['id']        - 1 = show work reminder icon in stat bar,
!!                                     0 = don''t. For on_demand jobs, shows a
!!                                     static white/black icon ("Available today.")
!!                                     and suppresses it after working that day or
!!                                     while the job is suspended.
!!   job_schedule_switchable['id']   - 1 = player can switch between schedule groups
!!                                     via change_schedule, 0 = cannot.
!!
!!   --- Payment defaults (definition) ---
!!   $job_pay_interval_def['id']     - Default payment interval. Copied to
!!                                     $job_pay_interval on hire.
!!                                     Values: 'event', 'shift', 'hourly',
!!                                             'weekly', 'monthly', 'none'
!!   job_pay_rate_def['id']          - Default payment amount. Copied to
!!                                     job_pay_rate on hire.
!!
!!   --- Penalty / miss handling ---
!!   job_penalty_per_miss['id']      - Amount added to job_booking_debt per missed
!!                                     shift. 0 = no automatic penalty.
!!   job_penalty_max_debt['id']      - If job_booking_debt exceeds this value,
!!                                     the player is auto-fired by missed_shift.
!!                                     0 = no auto-fire threshold.
!!   $job_on_miss_handler['id']      - Location file for custom miss handler.
!!                                     If set, daily_check calls this instead of
!!                                     the default missed_shift logic.
!!   $job_on_miss_handler_func['id'] - Function name within the handler location.
!!                                     Called as: gs $handler, $handler_func, $job_id
!!
!! Schedule modes: 'fixed', 'windowed', 'on_demand', 'booking'

!! ==========================================
!! Custom job definitions
!! ==========================================

!! Job definitions are located in 'jobs_list' by default. To define jobs in a
!! separate file (e.g. for mods or standalone content), use set_job_file:
!!
!!   gs 'jobs', 'set_job_file', 'my_job_id', 'my_jobs_file'
!!
!! Call this once (e.g. at init). The mapping persists across fire/re-hire
!! cycles, so every subsequent jobs call for that job_id will
!! automatically load the definition from 'my_jobs_file'.
!!
!! The custom file should define jobs the same way jobs_list does —
!! see that file for the format and required arrays.




	!!====================================================================!!
	!!                                                                    !!
	!!                    EMPLOYMENT STATE MANAGEMENT                     !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'set_employed':
	!! Fully employ player, create calendar events based on schedule_mode
	!! gs 'jobs', 'set_employed', $job_id
	!! Sets employment_conflict = 1 if conflict detected (warning only, doesn''t block)
	!! check_employment_possible returns 1=possible, 0=blocked; inverted for employment_conflict


	!! Guard: If already employed, do nothing (must terminate first)
	if $job_status[$ARGS[1]] = 'employed':
		exit
	end

	!! Check for conflicts before employment
	!! This is a warning check - we proceed anyway, but set a flag
	!! check_employment_possible returns 1=possible, 0=blocked
	!! employment_conflict uses opposite sense: 1=conflict, 0=no conflict
	employment_conflict = 1 - func('jobs', 'check_employment_possible', $ARGS[1], -1)

	!! Add to active jobs list if not already there
	if arrpos('$job_list', $ARGS[1]) = -1:
		$job_list[] = $ARGS[1]
	end

	!! Load job definition into working memory
	gs 'jobs', 'get_job_definition', $ARGS[1]

	!! Initialize employment data using separate arrays
	$job_status[$ARGS[1]] = 'employed'
	job_added[$ARGS[1]] = daystart
	job_pay_rate[$ARGS[1]] = job_pay_rate_def[$ARGS[1]]
	$job_pay_interval[$ARGS[1]] = $job_pay_interval_def[$ARGS[1]]
	job_worked_count[$ARGS[1]] = 0
	job_bonus_pay[$ARGS[1]] = 0
	job_docked_pay[$ARGS[1]] = 0
	!! job_shifts_total and job_paid_total are NOT reset — lifetime counters
	job_last_work_day[$ARGS[1]] = 0
	job_clocked_in[$ARGS[1]] = 0
	job_missed_total[$ARGS[1]] = 0
	job_shifts_this_period[$ARGS[1]] = 0

	!! Create calendar events based on schedule mode
	if $job_schedule_mode[$ARGS[1]] = 'fixed' or $job_schedule_mode[$ARGS[1]] = 'windowed':
		!! Build recurring event (only if job uses calendar)
		if job_add_to_calendar[$ARGS[1]] = 1:
			gs 'jobs', 'create_recurring_event', $ARGS[1]
		else
			$job_event_id[$ARGS[1]] = ''
		end

	elseif $job_schedule_mode[$ARGS[1]] = 'on_demand':
		!! Optionally create availability reminder events
		if job_create_availability_events[$ARGS[1]] = 1:
			gs 'jobs', 'create_availability_events', $ARGS[1]
		else
			!! No calendar events
			$job_event_id[$ARGS[1]] = ''
		end

	elseif $job_schedule_mode[$ARGS[1]] = 'booking':
		!! No events yet - created when bookings are made
		$job_event_id[$ARGS[1]] = ''
		job_bookings_active[$ARGS[1]] = 0
		job_booking_debt[$ARGS[1]] = 0
	end

	!! Initialize active_schedule for switchable multi-schedule jobs
	!! Check if job is switchable AND has multiple schedules
	if job_schedule_switchable[$ARGS[1]] = 1 and $job_work_days[$ARGS[1] + ', 1'] <> '':
		$job_active_schedule[$ARGS[1]] = '0'
	end
	exit
end


if $ARGS[0] = 'set_terminated':
	!! Terminate employment (good terms)
	!! gs 'jobs', 'set_terminated', $job_id
	gs 'jobs', '_end_employment', $ARGS[1], 'terminated'
	exit
end


if $ARGS[0] = 'set_fired':
	!! Terminate employment (bad terms, blacklisted)
	!! gs 'jobs', 'set_fired', $job_id
	gs 'jobs', '_end_employment', $ARGS[1], 'fired'
	exit
end


if $ARGS[0] = '_end_employment':
	!! Shared implementation for set_terminated / set_fired
	!! gs 'jobs', '_end_employment', $job_id, $status ('terminated' or 'fired')

	!! Update status
	$job_status[$ARGS[1]] = $ARGS[2]
	job_termination_day[$ARGS[1]] = daystart

	!! Remove calendar events
	if $job_event_id[$ARGS[1]] <> '':
		gs 'calendar_events', 'remove_event', $job_event_id[$ARGS[1]]
		$job_event_id[$ARGS[1]] = ''
	end

	!! Cancel all active bookings
	if job_bookings_active[$ARGS[1]] > 0:
		gs 'jobs', 'clear_all_bookings', $ARGS[1]
	end

	!! Remove from active job list
	temp_pos = arrpos('$job_list', $ARGS[1])
	if temp_pos >= 0:
		killvar 'job_list', temp_pos
	end

	killvar 'temp_pos'
	exit
end


if $ARGS[0] = 'suspend_job':
	!! Temporarily suspend employment (keeps status as 'employed', stays in job_list)
	!! Removes calendar events but preserves all work counters and penalty state
	!! gs 'jobs', 'suspend_job', $job_id

	!! Only suspend if currently employed and not already suspended
	if $job_status[$ARGS[1]] = 'employed' and job_suspended[$ARGS[1]] = 0:
		job_suspended[$ARGS[1]] = 1
		job_clocked_in[$ARGS[1]] = 0

		!! Remove calendar events (will be restored on resume)
		if $job_event_id[$ARGS[1]] <> '':
			gs 'calendar_events', 'remove_event', $job_event_id[$ARGS[1]]
			$job_event_id[$ARGS[1]] = ''
		end

		!! Cancel all active bookings (player can''t attend while suspended)
		if $job_schedule_mode[$ARGS[1]] = 'booking' or job_bookings_active[$ARGS[1]] > 0:
			gs 'jobs', 'clear_all_bookings', $ARGS[1]
		end
	end
	exit
end


if $ARGS[0] = 'resume_job':
	!! Resume a suspended job (restore calendar events)
	!! Does NOT reset any counters (missed_total, booking_debt, worked_count, etc.)
	!! gs 'jobs', 'resume_job', $job_id

	!! Only resume if currently employed and suspended
	if $job_status[$ARGS[1]] = 'employed' and job_suspended[$ARGS[1]] = 1:
		job_suspended[$ARGS[1]] = 0

		!! Restore calendar events based on schedule mode
		gs 'jobs', 'get_job_definition', $ARGS[1]

		if $job_schedule_mode[$ARGS[1]] = 'fixed' or $job_schedule_mode[$ARGS[1]] = 'windowed':
			if job_add_to_calendar[$ARGS[1]] = 1:
				gs 'jobs', 'create_recurring_event', $ARGS[1]
			end
		elseif $job_schedule_mode[$ARGS[1]] = 'on_demand':
			if job_create_availability_events[$ARGS[1]] = 1:
				gs 'jobs', 'create_availability_events', $ARGS[1]
			end
		end
	end
	exit
end


if $ARGS[0] = 'set_rank':
	!! Update rank/position
	!! gs 'jobs', 'set_rank', $job_id, new_rank

	job_rank[$ARGS[1]] = ARGS[2]
	exit
end


		!!------------------------------------------------------------!!
		!!           HELPER: RESOLVE ACTIVE SCHEDULE INDEX            !!
		!!------------------------------------------------------------!!


if $ARGS[0] = 'resolve_schedule_idx':
	!! Get the active schedule group index for a job
	!! result = func('jobs', 'resolve_schedule_idx', $job_id)
	!! Returns: active schedule index (from $job_active_schedule), or 0 if not set
	!!
	!! For switchable jobs: returns whichever group is currently active
	!! For non-switchable jobs: returns 0 (but callers may iterate all groups)
	!! For single-schedule jobs: returns 0

	if $job_active_schedule[$ARGS[1]] <> '':
		result = val($job_active_schedule[$ARGS[1]])
	else
		result = 0
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                           WORK TRACKING                            !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'clock':
	!! Record a completed shift/work session
	!! gs 'jobs', 'clock', $job_id
	!!
	!! For 'shift' pay_interval: adds 1 to job_worked_count (one shift)
	!! For 'hourly' pay_interval: adds shift duration in minutes to job_worked_count
	!!   (paycheck divides by 60 to convert minutes → hours for hourly pay)

	!! Load job definition
	if job_shift[$ARGS[1] + ', 0'] = 0 and $job_title[$ARGS[1]] = '':
		gs 'jobs', 'get_job_definition', $ARGS[1]
	end

	!! Increment worked count based on pay interval
	if $job_pay_interval[$ARGS[1]] = 'hourly' or $job_pay_interval_def[$ARGS[1]] = 'hourly':
		!! Hourly jobs: add shift duration in minutes (paycheck divides by 60)
		job_worked_count[$ARGS[1]] += job_shift[$ARGS[1] + ', ' + $str(func('jobs', 'resolve_schedule_idx', $ARGS[1]))]
	else
		!! Shift/weekly/monthly/etc: add 1 per clock call
		job_worked_count[$ARGS[1]] += 1
	end

	!! Update last work day
	job_last_work_day[$ARGS[1]] = daystart

	!! Record clock-in time (always overwrite — each clock call is a new shift)
	job_clocked_in[$ARGS[1]] = totminut

	!! Increment lifetime and per-period shift counters
	job_shifts_total[$ARGS[1]] += 1
	job_shifts_this_period[$ARGS[1]] += 1
	exit
end

if $ARGS[0] = 'daily_check':
	!! Check all v4 jobs for missed shifts (called from cikl.qsrc at start of each day)
	!! Detects: yesterday was a work day AND player didn''t work AND job is employed + not suspended
	!! Uses masseuse pattern: check yesterday, compare job_last_work_day
	!! For booking-mode jobs: miss detection checks yesterday (daystart-1), but expiry
	!!   is delayed to daystart-2. This keeps yesterday''s booking data alive so that
	!!   cross-midnight query functions (is_work_day, is_work_time) can detect overnight
	!!   overflow from yesterday''s shift into today.
	!!   Calls custom miss handler if defined ($job_on_miss_handler).

	temp_dc_idx = 0
	:daily_check_v4_loop
	if temp_dc_idx < arrsize('$job_list'):
		$temp_dc_job = $job_list[temp_dc_idx]

		!! Only check employed, non-suspended jobs
		if $job_status[$temp_dc_job] = 'employed' and job_suspended[$temp_dc_job] = 0:
			gs 'jobs', 'get_job_definition', $temp_dc_job

			!! For booking-mode jobs: miss detection (yesterday) + expiry (day before yesterday).
			!! Expiry is delayed by one day so that yesterday''s booking data remains
			!! available for cross-midnight query functions (is_work_day, is_work_time).
			if $job_schedule_mode[$temp_dc_job] = 'booking':
				temp_dc_max_slot = job_booking_slots_per_day[$temp_dc_job]
				if temp_dc_max_slot < 1: temp_dc_max_slot = 1

				!! --- Miss detection: yesterday (daystart - 1) ---
				temp_dc_yesterday = daystart - 1
				temp_dc_slot = 0
				:daily_check_booking_slot_loop
				if temp_dc_slot < temp_dc_max_slot:
					$temp_dc_booking_data = $job_booking[$temp_dc_job + ', ' + $str(temp_dc_yesterday) + ', ' + $str(temp_dc_slot)]

					if $temp_dc_booking_data <> '':
						!! Booking existed for yesterday — check if player attended.
						!! Days up to and including the hire day are not counted.
						if job_last_work_day[$temp_dc_job] < temp_dc_yesterday and temp_dc_yesterday > job_added[$temp_dc_job]:
							!! MISSED — call custom handler if defined, else generic missed_shift
							if $job_on_miss_handler[$temp_dc_job] <> '':
								job_missed_total[$temp_dc_job] += 1
								gs $job_on_miss_handler[$temp_dc_job], $job_on_miss_handler_func[$temp_dc_job], $temp_dc_job, $temp_dc_booking_data
							else
								gs 'jobs', 'missed_shift', $temp_dc_job
							end
						end
					end
					killvar 'temp_dc_booking_data'
					temp_dc_slot += 1
					jump 'daily_check_booking_slot_loop'
				end

				!! --- Expiry: day before yesterday (daystart - 2) ---
				!! Bookings live one extra day so cross-midnight detection works.
				temp_dc_daybefore = daystart - 2
				temp_dc_slot = 0
				:daily_check_expire_loop
				if temp_dc_slot < temp_dc_max_slot:
					if $job_booking[$temp_dc_job + ', ' + $str(temp_dc_daybefore) + ', ' + $str(temp_dc_slot)] <> '':
						$job_booking[$temp_dc_job + ', ' + $str(temp_dc_daybefore) + ', ' + $str(temp_dc_slot)] = ''
						job_bookings_active[$temp_dc_job] -= 1
						if job_bookings_active[$temp_dc_job] < 0: job_bookings_active[$temp_dc_job] = 0

						$temp_dc_evt = 'job_booking_' + $temp_dc_job + '_' + $str(temp_dc_daybefore) + '_' + $str(temp_dc_slot)
						gs 'calendar_events', 'remove_event', $temp_dc_evt
						killvar 'temp_dc_evt'
					end
					temp_dc_slot += 1
					jump 'daily_check_expire_loop'
				end
				killvar 'temp_dc_daybefore'
				killvar 'temp_dc_max_slot'
				killvar 'temp_dc_slot'
			else
				!! Fixed/windowed/on_demand: standard miss detection
				temp_dc_yesterday = daystart - 1
				if func('jobs', 'is_work_day', $temp_dc_job, temp_dc_yesterday) = 1:
					!! Days up to and including the hire day are never counted.
					if job_last_work_day[$temp_dc_job] < temp_dc_yesterday and temp_dc_yesterday > job_added[$temp_dc_job]:
						gs 'jobs', 'missed_shift', $temp_dc_job
					end
				end
			end
		end

		!! Only advance index if the job wasn''t removed from the list
		!! (set_fired/set_terminated remove from $job_list, shifting remaining items down)
		if arrpos('$job_list', $temp_dc_job) >= 0:
			temp_dc_idx += 1
		end
		jump 'daily_check_v4_loop'
	end

	killvar 'temp_dc_idx'
	killvar 'temp_dc_job'
	killvar 'temp_dc_yesterday'
	exit
end

	!! result = func('jobs', 'paycheck', $job_id)
	!! result = func('jobs', 'paycheck', $job_id, 'cash')
	!! $ARGS[2] = optional payment destination override: 'cash', 'bank'
	!!			When omitted, the money system routes according to the
	!!			player''s preferred income method (cfg_vars['income_opt']).
	!!			Only pass an explicit value when the narrative requires it

	elseif $job_pay_interval[$ARGS[1]] = 'hourly':

if $ARGS[0] = 'is_work_day':
	!! Check if today matches work schedule (any mode)
	!! result = func('jobs', 'is_work_day', $job_id, check_daystart)
	!! Returns: 1 if scheduled to work, 0 otherwise
	!! For booking mode: also returns 1 if yesterday has a cross-midnight shift
	!!   that is still active (overflow into check_day).
	!! Note: Works for both employed and non-employed jobs (useful for planning/interviews)

	temp_check_day = ARGS[2]
	if temp_check_day = 0: temp_check_day = daystart

	!! Load job definition (only if not already loaded)
	if $job_title[$ARGS[1]] = '':
		gs 'jobs', 'get_job_definition', $ARGS[1]
	end

	!! Mode-specific logic
	if $job_schedule_mode[$ARGS[1]] = 'fixed' or $job_schedule_mode[$ARGS[1]] = 'windowed':
		!! Check recurring schedule
		result = func('jobs', 'matches_recurring_schedule', $ARGS[1], temp_check_day)

	elseif $job_schedule_mode[$ARGS[1]] = 'on_demand':
		!! Check if available today
		if instr(' ' + $job_available_days[$ARGS[1]] + ' ', ' ' + $str(func('time', 'get_week_from_daystart', temp_check_day)) + ' ') > 0:
			result = 1
		else
			result = 0
		end

	elseif $job_schedule_mode[$ARGS[1]] = 'booking':
		!! For non-employed jobs, bookings don''t exist yet, return 0
		!! For employed jobs, check if booking exists for this day
		if $job_status[$ARGS[1]] = 'employed':
			result = func('jobs', 'has_booking_for_day', $ARGS[1], temp_check_day)
			!! Cross-midnight: if no booking today, check yesterday for overflow
			if result = 0:
				if func('jobs', 'has_booking_for_day', $ARGS[1], temp_check_day - 1) = 1:
					gs 'jobs', 'get_shift_for_day', $ARGS[1], temp_check_day - 1
					!! Check if any slot from yesterday crosses midnight
					temp_iwd_si = 0
					:iwd_crossmid_loop
					if temp_iwd_si < result_slot_count:
						if result_slot_end[temp_iwd_si] > 1440:
							result = 1
							jump 'iwd_crossmid_done'
						end
						temp_iwd_si += 1
						jump 'iwd_crossmid_loop'
					end
					:iwd_crossmid_done
					killvar 'temp_iwd_si'
				end
			end
		else
			result = 0
		end
	end

	killvar 'temp_check_day'
	exit
end

	!! Returns: 1 if current time falls anywhere within any slot''s time span, 0 otherwise
	!! For fixed/windowed: arrival <= now < start + shift (full span including arrival window)
	!! For on_demand: within availability hours
	!! For booking: iterates per-slot results; also checks yesterday''s cross-midnight overflow
	!! Use is_arrival_time to check if player can START work now

	!! For fixed/windowed: arrival <= now < start (within arrival window only)
	!! For on_demand: same as is_work_time (can start anytime during availability)
	!! For booking: iterates per-slot arrival windows; no cross-midnight arrival
	!!   (arrival window is always before midnight; cross-midnight applies to shifts only)

if $ARGS[0] = '_is_time_check':
	!! Shared implementation for is_work_time / is_arrival_time
	!! gs 'jobs', '_is_time_check', $job_id, $mode
	!! $mode = 'work'	— full span: arrival → end (including cross-midnight overflow)
	!! $mode = 'arrival' — arrival window only: arrival → start (no cross-midnight)
	!! Sets: result = 1 if current time matches, 0 otherwise

	gs 'jobs', 'get_job_definition', $ARGS[1]
	temp_itc_now = hour * 60 + minut

	if $job_schedule_mode[$ARGS[1]] = 'booking':
		!! Booking mode: check per-slot time windows for today
		result = 0

		!! Check today''s slots
		if func('jobs', 'has_booking_for_day', $ARGS[1], daystart) = 1:
			gs 'jobs', 'get_shift_for_day', $ARGS[1], daystart
			temp_itc_i = 0
			:itc_today_loop
			if temp_itc_i < result_slot_count:
				if $ARGS[2] = 'work':
					!! Work mode: arrival through end of shift
					if temp_itc_now >= result_slot_arrival[temp_itc_i] and temp_itc_now < result_slot_end[temp_itc_i]:
						result = 1
						jump 'itc_done'
					end
				else
					!! Arrival mode: arrival window only (arrival through start)
					if temp_itc_now >= result_slot_arrival[temp_itc_i] and temp_itc_now < result_slot_start[temp_itc_i]:
						result = 1
						jump 'itc_done'
					end
				end
				temp_itc_i += 1
				jump 'itc_today_loop'
			end
		end

		!! Cross-midnight overflow: only for work mode
		!! Arrival windows are always before midnight, so arrival mode skips this.
		if $ARGS[2] = 'work' and $job_status[$ARGS[1]] = 'employed' and func('jobs', 'has_booking_for_day', $ARGS[1], daystart - 1) = 1:
			gs 'jobs', 'get_shift_for_day', $ARGS[1], daystart - 1
			temp_itc_i = 0
			:itc_yest_loop
			if temp_itc_i < result_slot_count:
				if result_slot_end[temp_itc_i] > 1440:
					!! This slot crosses midnight — check if now is in the overflow
					if temp_itc_now < result_slot_end[temp_itc_i] - 1440:
						result = 1
						jump 'itc_done'
					end
				end
				temp_itc_i += 1
				jump 'itc_yest_loop'
			end
		end

		:itc_done
		killvar 'temp_itc_now'
		killvar 'temp_itc_i'
		exit
	end

	!! Non-booking modes: gate on is_work_day first
	if func('jobs', 'is_work_day', $ARGS[1], daystart) = 0:
		result = 0
		killvar 'temp_itc_now'
		exit
	end

	!! Get shift details for today
	gs 'jobs', 'get_shift_for_day', $ARGS[1], daystart

	if $job_schedule_mode[$ARGS[1]] = 'fixed' or $job_schedule_mode[$ARGS[1]] = 'windowed':
		if $ARGS[2] = 'work':
			!! Work mode: full span = arrival through start + duration
			if temp_itc_now >= result_arrival and temp_itc_now < result_start + result_duration:
				result = 1
			else
				result = 0
			end
		else
			!! Arrival mode: arrival window only = arrival through start
			if temp_itc_now >= result_arrival and temp_itc_now < result_start:
				result = 1
			else
				result = 0
			end
		end

	elseif $job_schedule_mode[$ARGS[1]] = 'on_demand':
		!! On-demand: both modes use same availability window
		if temp_itc_now >= job_available_start[$ARGS[1]] and temp_itc_now <= job_available_end[$ARGS[1]]:
			result = 1
		else
			result = 0
		end
	end

	killvar 'temp_itc_now'
	exit
end


if $ARGS[0] = 'get_shift_for_day':
	!! Get shift details for specific day (fixed/windowed/booking)
	!! gs 'jobs', 'get_shift_for_day', $job_id, check_day
	!! Sets: result_arrival, result_start, result_duration, result_end (in minutes)
	!!
	!! For booking mode: scans booked slots and returns per-slot result arrays.
	!!   Per-slot outputs (booking mode only):
	!!     result_slot_count          = number of booked slots found
	!!     result_slot_arrival[i]     = arrival time for slot i (minutes from midnight)
	!!     result_slot_start[i]       = shift start / arrival deadline for slot i
	!!     result_slot_end[i]         = shift end for slot i (may exceed 1440 for cross-midnight)
	!!   Envelope outputs (all modes, derived from per-slot arrays for booking):
	!!     result_arrival  = earliest slot arrival (min of result_slot_arrival[])
	!!     result_start    = earliest slot start (min of result_slot_start[])
	!!     result_end      = latest slot end (max of result_slot_end[])
	!!     result_duration = result_end - result_start

	temp_check_day = ARGS[2]
	if temp_check_day = 0: temp_check_day = daystart

	!! Load job definition (only if not already loaded)
	if $job_title[$ARGS[1]] = '':
		gs 'jobs', 'get_job_definition', $ARGS[1]
	end

	if $job_schedule_mode[$ARGS[1]] = 'booking':
		!! Booking mode: scan booked slots, build per-slot result arrays and envelope.
		!! Uses per-slot timing (job_slot_arrival/job_slot_start/job_slot_shift) when
		!! available, falls back to job-level arrival_start/arrival_end otherwise.

		!! Clear any stale per-slot results from a previous call
		killvar 'result_slot_arrival'
		killvar 'result_slot_start'
		killvar 'result_slot_end'
		result_slot_count = 0

		temp_gsfd_max_slot = job_booking_slots_per_day[$ARGS[1]]
		if temp_gsfd_max_slot < 1: temp_gsfd_max_slot = 1

		temp_gsfd_slot = 0
		:gsfd_booking_slot_loop
		if temp_gsfd_slot < temp_gsfd_max_slot:
			!! Check if this slot is booked for the requested day
			if $job_booking[$ARGS[1] + ', ' + $str(temp_check_day) + ', ' + $str(temp_gsfd_slot)] <> '':
				!! Resolve timing for this slot (sets temp_resolved_arrival/start/end)
				gs 'jobs', '_resolve_slot_timing', $ARGS[1], temp_gsfd_slot

				!! Store per-slot results
				result_slot_arrival[result_slot_count] = temp_resolved_arrival
				result_slot_start[result_slot_count] = temp_resolved_start
				result_slot_end[result_slot_count] = temp_resolved_end
				result_slot_count += 1
				killvar 'temp_resolved_arrival'
				killvar 'temp_resolved_start'
				killvar 'temp_resolved_end'
			end
			temp_gsfd_slot += 1
			jump 'gsfd_booking_slot_loop'
		end

		if result_slot_count > 0:
			!! Derive envelope from per-slot arrays
			result_arrival = result_slot_arrival[0]
			result_start = result_slot_start[0]
			result_end = result_slot_end[0]

			temp_gsfd_i = 1
			:gsfd_envelope_loop
			if temp_gsfd_i < result_slot_count:
				if result_slot_arrival[temp_gsfd_i] < result_arrival:
					result_arrival = result_slot_arrival[temp_gsfd_i]
				end
				if result_slot_start[temp_gsfd_i] < result_start:
					result_start = result_slot_start[temp_gsfd_i]
				end
				if result_slot_end[temp_gsfd_i] > result_end:
					result_end = result_slot_end[temp_gsfd_i]
				end
				temp_gsfd_i += 1
				jump 'gsfd_envelope_loop'
			end

			result_duration = result_end - result_start
			if result_duration < 0: result_duration = 0
		else
			!! No booked slots found — return zeros
			result_arrival = 0
			result_start = 0
			result_duration = 0
			result_end = 0
		end

		killvar 'temp_check_day'
		killvar 'temp_gsfd_max_slot'
		killvar 'temp_gsfd_slot'
		killvar 'temp_gsfd_i'
		exit
	end

	!! Get weekday
	temp_weekday = func('time', 'get_week_from_daystart', temp_check_day)

	!! Resolve schedule: switchable checks only active group; non-switchable iterates all
	temp_schedule_idx = func('jobs', 'resolve_schedule_idx', $ARGS[1])
	temp_found = 0
	temp_idx = temp_schedule_idx
	:find_schedule_loop
		$temp_work_days = $job_work_days[$ARGS[1] + ', ' + $str(temp_idx)]
		if $temp_work_days <> '':
			if instr(' ' + $temp_work_days + ' ', ' ' + $str(temp_weekday) + ' ') > 0:
				result_arrival = job_arrival[$ARGS[1] + ', ' + $str(temp_idx)]
				result_start = job_start[$ARGS[1] + ', ' + $str(temp_idx)]
				result_duration = job_shift[$ARGS[1] + ', ' + $str(temp_idx)]
				temp_found = 1
			end
		end
		!! Non-switchable: continue iterating remaining groups for a match
		!! Switchable (or single-schedule): stop after checking the active group
		if temp_found = 0 and $temp_work_days <> '' and job_schedule_switchable[$ARGS[1]] = 0:
			temp_idx += 1
			jump 'find_schedule_loop'
		end

	!! If not found, set to 0
	if temp_found = 0:
		result_arrival = 0
		result_start = 0
		result_duration = 0
	end

	killvar 'temp_check_day'
	killvar 'temp_schedule_idx'
	killvar 'temp_weekday'
	killvar 'temp_found'
	killvar 'temp_idx'
	killvar 'temp_work_days'
	exit
end


if $ARGS[0] = 'next_work_day':
	!! Find next work day (fixed/windowed)
	!! result = func('jobs', 'next_work_day', $job_id, start_day)
	!! Returns: next daystart, or 0 if none in next 14 days

	temp_nwd_start_day = ARGS[2]
	if temp_nwd_start_day = 0: temp_nwd_start_day = daystart

	!! Extract shift details for specific day from complex recurrence pattern
	!! gs 'jobs', 'get_shift_from_recurrence', $recur_pattern, check_day
	!! Sets: result_arrival, result_start, result_duration (in minutes)
	!!
	!! This is jobs-specific because it extracts work shift details (arrival/start/duration).
	!! Calendar only needs to know IF an event occurs (check_complex_recurrence).

	check_day = ARGS[2]
	temp_weekday = func('time', 'get_week_from_daystart', check_day)
	temp_pos = 1

		$temp_shift = mid($ARGS[1], temp_pos, temp_semi - temp_pos)

		temp_pipe1 = instr($temp_shift, '|')

				result_start = temp_start

	result_arrival = 0
	result_start = 0

if $ARGS[0] = '_resolve_slot_timing':
	!! Resolve timing for a single booking slot using the 3-tier fallback:
	!!   1. Per-slot timing (job_slot_arrival/start/shift) if defined for slot
	!!   2. Job-level arrival window (job_arrival_start/end + job_available_end)
	!!   3. Last resort: job_available_start/end with 1h defaults
	!!
	!! gs 'jobs', '_resolve_slot_timing', $job_id, slot_index
	!! Sets: temp_resolved_arrival, temp_resolved_start, temp_resolved_end
	!! Caller must killvar these when done.
	!! Assumes job definition is already loaded.

	if job_slot_arrival[$ARGS[1] + ', ' + $str(ARGS[2])] > 0:
		!! Per-slot timing: arrival, start (deadline), shift (start-to-end)
		temp_resolved_arrival = job_slot_arrival[$ARGS[1] + ', ' + $str(ARGS[2])]
		if job_slot_start[$ARGS[1] + ', ' + $str(ARGS[2])] > 0:
			temp_resolved_start = job_slot_start[$ARGS[1] + ', ' + $str(ARGS[2])]
		else
			temp_resolved_start = temp_resolved_arrival
		end
		temp_resolved_end = temp_resolved_start + job_slot_shift[$ARGS[1] + ', ' + $str(ARGS[2])]
	elseif job_arrival_start[$ARGS[1]] > 0:
		!! Fallback: job-level arrival window
		temp_resolved_arrival = job_arrival_start[$ARGS[1]]
		if job_arrival_end[$ARGS[1]] > 0:
			temp_resolved_start = job_arrival_end[$ARGS[1]]
		else
			temp_resolved_start = temp_resolved_arrival + 60
		end
		if job_available_end[$ARGS[1]] > 0:
			temp_resolved_end = job_available_end[$ARGS[1]]
		else
			temp_resolved_end = temp_resolved_start + 60
		end
	else
		!! Last resort: available_start with 1h defaults
		temp_resolved_arrival = job_available_start[$ARGS[1]]
		temp_resolved_start = temp_resolved_arrival + 60
		temp_resolved_end = iif(job_available_end[$ARGS[1]] > 0, job_available_end[$ARGS[1]], temp_resolved_start + 60)
	end
	exit
end

if $ARGS[0] = 'build_event_vars_for_job':
	!! Build event_vars{} structure for a job to check conflicts or create events
	!! gs 'jobs', 'build_event_vars_for_job', $job_id[, schedule_idx_override]
	!! Populates event_vars{} with job schedule data
	!!
	!! Optional schedule_idx_override (ARGS[2]):
	!!   -1 = use resolve_schedule_idx (default behavior)
	!!   0, 1, 2, … = use that specific schedule group
	!!   NOTE: omitted ARGS[2] = 0 in QSP, so callers wanting resolve_schedule_idx
	!!   must explicitly pass -1. Omitting the arg checks schedule 0.

	!! Clear event_vars
	killvar 'event_vars'

	!! Set event ID (use job_id prefixed with 'job_')
	$event_vars['id'] = 'job_' + $ARGS[1]
	$event_vars['title'] = $job_title[$ARGS[1]]
	$event_vars['loc'] = $job_location[$ARGS[1]]
	$event_vars['desc'] = '工作班次' + $job_title[$ARGS[1]]

	!! Set event daystart to today (for recurring events, this is the start date)
	!! Use max(1, daystart) to avoid issues in test environment
	event_vars['daystart'] = max(1, daystart)

	!! Not all-day events
	event_vars['all_day'] = 0

	!! Blocking and priority from job definition
	!! Blocking jobs get priority 2 (high), non-blocking get priority 1 (normal)
	event_vars['blocking'] = job_blocking[$ARGS[1]]
	event_vars['priority'] = iif(job_blocking[$ARGS[1]] = 1, 2, 1)

	!! Color — 0 means assign_color will pick a random color
	event_vars['color'] = 0

	!! Not a holiday
	event_vars['holiday'] = 0

	!! Build recurrence pattern
	if $job_schedule_mode[$ARGS[1]] = 'fixed' or $job_schedule_mode[$ARGS[1]] = 'windowed':
		!! Both fixed and windowed use flex_type=1 (flexible) in the calendar
		!! All scheduled jobs have arrival windows, so calendar always shows flexible block
		!! The fixed/windowed distinction affects gameplay (end time behavior), not calendar display
		!!
		!! Calendar fields:
		!!   window_start_ts = arrival / 15  (earliest player can arrive)
		!!   window_end_ts   = start / 15	(latest player can start work)
		!!   duration_ts	 = shift / 15	(work duration)
		!!
		!! Calendar conflict span = window_start_ts → window_end_ts + duration_ts
		!!   Fixed:	9:00 arrival, 10:00 start, 8h shift → span 9:00-18:00 ✓
		!!   Windowed: 6:00 arrival, 8:00 start, 8h shift → span 6:00-16:00 ✓
		event_vars['flex_type'] = 1

		!! Get active schedule group (or use override if provided)
		if ARGS[2] >= 0:
			temp_bev_sched_idx = ARGS[2]
		else
			temp_bev_sched_idx = func('jobs', 'resolve_schedule_idx', $ARGS[1])
		end

		!! Convert to timestamps (15-minute intervals)
		event_vars['window_start_ts'] = job_arrival[$ARGS[1] + ', ' + $str(temp_bev_sched_idx)] / 15
		event_vars['window_end_ts'] = job_start[$ARGS[1] + ', ' + $str(temp_bev_sched_idx)] / 15
		event_vars['duration_ts'] = job_shift[$ARGS[1] + ', ' + $str(temp_bev_sched_idx)] / 15

		!! Build recurrence string; seasonal jobs use yearly date range instead of weekday list
		if $job_work_dates[$ARGS[1]] <> '':
			$event_vars['recur'] = 'yearly-' + $job_work_dates[$ARGS[1]]
		else
			$event_vars['recur'] = $job_work_days[$ARGS[1] + ', ' + $str(temp_bev_sched_idx)]
		end

	elseif $job_schedule_mode[$ARGS[1]] = 'on_demand':
		!! On-demand jobs create availability events (if enabled)
		event_vars['flex_type'] = 1
		event_vars['blocking'] = 0  & !! Override - availability never blocks
		event_vars['priority'] = 0   & !! Low priority

		!! Availability window
		event_vars['window_start_ts'] = job_available_start[$ARGS[1]] / 15
		event_vars['window_end_ts'] = job_available_end[$ARGS[1]] / 15
		event_vars['duration_ts'] = 0  & !! No fixed duration

		$event_vars['recur'] = $job_available_days[$ARGS[1]]

	elseif $job_schedule_mode[$ARGS[1]] = 'booking':
		!! Booking jobs - individual events created per booking
		!! For conflict checking, we don''t have a schedule to check
		!! So just set minimal data (conflict check will fail gracefully)
		event_vars['flex_type'] = 0
		event_vars['start_ts'] = 0
		event_vars['duration_ts'] = 0
		$event_vars['recur'] = ''
	end

	!! No recurrence end date (ongoing employment)
	event_vars['recur_end'] = 0

	!! Clean up temp variables
	killvar 'temp_bev_sched_idx'
	exit
end

	job_booking_debt[$ARGS[1]] = 0
	$job_recurrence_pattern[$ARGS[1]] = ''

	!! Clear booking slots (iterate over all days × all slots per day)
	!! Use the larger of 7 or the job''s booking window size
	temp_booking_range = job_booking_window_days[$ARGS[1]]
	if temp_booking_range < 7: temp_booking_range = 7
	temp_cleanup_max_slot = job_booking_slots_per_day[$ARGS[1]]
	if temp_cleanup_max_slot < 1: temp_cleanup_max_slot = 1
	temp_day_idx = daystart - temp_booking_range
	:cleanup_bookings_loop
	if temp_day_idx <= daystart + temp_booking_range:
		temp_cleanup_slot = 0
		:cleanup_bookings_slot_loop
		if temp_cleanup_slot < temp_cleanup_max_slot:
			if $job_booking[$ARGS[1] + ', ' + $str(temp_day_idx) + ', ' + $str(temp_cleanup_slot)] <> '':
				$job_booking[$ARGS[1] + ', ' + $str(temp_day_idx) + ', ' + $str(temp_cleanup_slot)] = ''
				!! Also remove calendar event for this booking
				gs 'calendar_events', 'remove_event', 'job_booking_' + $ARGS[1] + '_' + $str(temp_day_idx) + '_' + $str(temp_cleanup_slot)
			end
			temp_cleanup_slot += 1
			jump 'cleanup_bookings_slot_loop'
		end
		temp_day_idx += 1
		jump 'cleanup_bookings_loop'
	end
	killvar 'temp_cleanup_max_slot'
	killvar 'temp_cleanup_slot'

	!! Reset booking counter
	job_bookings_active[$ARGS[1]] = 0

	!! Clear job definition arrays (set to empty)
	$job_title[$ARGS[1]] = ''
	$job_location[$ARGS[1]] = ''
	$job_schedule_mode[$ARGS[1]] = ''
	$job_pay_interval_def[$ARGS[1]] = ''
	job_pay_rate_def[$ARGS[1]] = 0
	$job_work_dates[$ARGS[1]] = ''
	$job_available_days[$ARGS[1]] = ''
	job_available_start[$ARGS[1]] = 0
	job_available_end[$ARGS[1]] = 0
	job_create_availability_events[$ARGS[1]] = 0
	job_booking_window_days[$ARGS[1]] = 0
	job_booking_slots_per_day[$ARGS[1]] = 0
	job_booking_max_concurrent[$ARGS[1]] = 0
	job_booking_min_gap_days[$ARGS[1]] = 0
	job_arrival_start[$ARGS[1]] = 0
	job_arrival_end[$ARGS[1]] = 0
	!! Clear per-slot timing (for up to max slots)
	temp_cleanup_slot_idx = 0
	:cleanup_slot_timing_loop
	if temp_cleanup_slot_idx < 4:
		job_slot_arrival[$ARGS[1] + ', ' + $str(temp_cleanup_slot_idx)] = 0
		job_slot_start[$ARGS[1] + ', ' + $str(temp_cleanup_slot_idx)] = 0
		job_slot_shift[$ARGS[1] + ', ' + $str(temp_cleanup_slot_idx)] = 0
		temp_cleanup_slot_idx += 1
		jump 'cleanup_slot_timing_loop'
	end
	killvar 'temp_cleanup_slot_idx'
	job_blocking[$ARGS[1]] = 0
	job_add_to_calendar[$ARGS[1]] = 0
	job_show_stat_icon[$ARGS[1]] = 0
	job_refresh_desc[$ARGS[1]] = 0
	job_penalty_per_miss[$ARGS[1]] = 0
	job_penalty_max_debt[$ARGS[1]] = 0
	$job_on_miss_handler[$ARGS[1]] = ''
	$job_on_miss_handler_func[$ARGS[1]] = ''
	job_schedule_switchable[$ARGS[1]] = 0

	!! Clear schedule properties (for up to 5 schedules - should cover all jobs)
	temp_schedule_idx = 0
	:cleanup_schedules_loop
	if temp_schedule_idx < 5:
		$job_work_days[$ARGS[1] + ', ' + $str(temp_schedule_idx)] = ''
		job_arrival[$ARGS[1] + ', ' + $str(temp_schedule_idx)] = 0
		job_start[$ARGS[1] + ', ' + $str(temp_schedule_idx)] = 0
		job_shift[$ARGS[1] + ', ' + $str(temp_schedule_idx)] = 0
		$job_recurrence_pattern[$ARGS[1] + ', ' + $str(temp_schedule_idx)] = ''
		temp_schedule_idx += 1
		jump 'cleanup_schedules_loop'
	end

	!! Remove from job_list (integer index - killvar works here!)
	temp_list_idx = arrpos('$job_list', $ARGS[1])
	if temp_list_idx >= 0: killvar 'job_list', temp_list_idx

	killvar 'temp_schedule_idx'
	killvar 'temp_day_idx'
	killvar 'temp_list_idx'
	killvar 'temp_booking_range'
	exit
end


if $ARGS[0] = 'terminate_all_jobs':
	!! Terminate and clean up ALL employed/fired/terminated v4 jobs at once.
	!! Used for major life events (prison, etc.) that strip all employment.
	!! gs 'jobs', 'terminate_all_jobs'
	!!
	!! cleanup_job removes entries from $job_list, so iterate backwards to
	!! avoid index shifting issues.

	temp_term_idx = arrsize('$job_list') - 1
	:terminate_all_loop
	if temp_term_idx >= 0:
		$temp_term_job = $job_list[temp_term_idx]
		gs 'jobs', 'cleanup_job', $temp_term_job
		temp_term_idx -= 1
		jump 'terminate_all_loop'
	end

	killvar 'temp_term_idx'
	killvar 'temp_term_job'
	exit
end


!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ format_days — convert a space-separated day-number string (1=Mon…7=Sun)│
!! │ to readable text. Common patterns use shorthand; falls back to a       │
!! │ comma-separated list.                                                  │
!! │ ARGS[1] = day string (e.g. '1 2 3 4 5')                                │
!! └────────────────────────────────────────────────────────────────────────┘

	if $ARGS[1] = '1 2 3 4 5':			$result = '周一至周五'				& exit

	$fd_names[1] = '周一' & $fd_names[2] = '周二' & $fd_names[3] = '结婚'
	$fd_names[4] = '周四' & $fd_names[5] = '周五' & $fd_names[6] = '周六' & $fd_names[7] = '周日'

	:fd_loop

		if $result <> '': $result += ', '
		$result += $fd_names[fd_i]
	end
	fd_i += 1
	if fd_i <= 7: jump 'fd_loop'
	killvar 'fd_names'
	killvar 'fd_i'
	exit
end

!! ┌────────────────────────────────────────────────────────────────────────┐
!! │ card — return an HTML card for a single jobs_list job.                 │
!! │ ARGS[1] = job_id                                                       │
!! │ Returns HTML in $result. Reads and updates $temp_bcolor (caller-       │
!! │ managed) for alternating row backgrounds.                              │
!! │ Hidden + not employed → mystery card with hidden.png (like traits).    │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'card':
	$jc_id = $ARGS[1]
	gs 'jobs', 'get_job_definition', $jc_id

	!! Hidden jobs — render a mystery card unless employed or discovered
	if job_card_hidden[$jc_id] = 1 and $job_status[$jc_id] <> 'employed' and job_hiring_step[$jc_id] = 0 and cheatVars['show_hidden_jobs'] = 0:
		$temp_bcolor = $func('themes', 'alt_color', $temp_bcolor)
		$card_in['icon'] = 'images/system/icons/traits/hidden.png'
		$card_in['title'] = '???'
		$card_in['body'] = '<div style="font-size:0.9em; margin-top:4px;">这份工作目前尚未解锁。继续探索即可解锁。</div>'
		$card_in['bg'] = $temp_bcolor
		$card_in['opacity'] = '0.4'
		gs 'cards', 'shell'
		killvar 'card_in'
		killvar 'jc_id'
		exit
	end

	!! ── Employment state ──────────────────────────────────────────────
	if $job_status[$jc_id] = 'employed':
		$jc_opac = '1.0'
		$jc_border = $theme_hex['accent']
		$jc_status = $func('wrap', 'v_pos', '&#9679; 已就业')
	elseif job_hiring_step[$jc_id] > 0:
		$jc_opac = '0.85'
		$jc_border = $theme_hex['goth']
		$jc_status = $func('wrap', 'pos', '&#9675; 目前未就业')
	else
		$jc_opac = '0.6'
		$jc_border = iif(theme['is_dark'] = 1, '#555555', '#aaaaaa')
		$jc_status = $func('wrap', 'goth', '&#9675; 无业')
	end

	!! ── Icon ──────────────────────────────────────────────────────────
	$jc_icon = 'images/system/icons/' + iif($job_icons_source[$jc_id] <> '', $job_icons_source[$jc_id], 'jobs') + '/' + $jc_id + '.png'

	!! ── Schedule info ─────────────────────────────────────────────────
	jc_slot = iif($job_active_schedule[$jc_id] <> '', val($job_active_schedule[$jc_id]), 0)
	$jc_skey = $jc_id + ', ' + str(jc_slot)
	if $job_schedule_mode[$jc_id] = 'booking':
		$jc_sched = '基于预订 —报名可用时段'
	elseif $job_schedule_mode[$jc_id] = 'on_demand':
		$jc_sched = '按需，随时'
	else
		$jc_days = $func('jobs', 'format_days', $job_work_days[$jc_skey])
		$jc_t_arr = $func('time', 'get_time_string', job_arrival[$jc_skey] / 60, job_arrival[$jc_skey] mod 60)
		$jc_t_st = $func('time', 'get_time_string', job_start[$jc_skey] / 60, job_start[$jc_skey] mod 60)
		jc_end_m = (job_start[$jc_skey] + job_shift[$jc_skey]) mod 1440
		$jc_t_end = $func('time', 'get_time_string', jc_end_m / 60, jc_end_m mod 60)
		if $job_schedule_mode[$jc_id] = 'windowed':
			jc_shift_h = job_shift[$jc_skey] / 60
			jc_shift_m = job_shift[$jc_skey] mod 60
			$jc_dur = iif(jc_shift_m = 0, str(jc_shift_h) + 'h', str(jc_shift_h) + 'h ' + str(jc_shift_m) + 'min')
			$jc_sched = $jc_days + ', anytime ' + $jc_t_arr + '-' + $jc_t_st + ', ' + $jc_dur + ' shift'
		else
			$jc_sched = $jc_days + ', arrive ' + $jc_t_arr + '-' + $jc_t_st + ', until ' + $jc_t_end
		end
	end

	!! ── Pay info ──────────────────────────────────────────────────────
	if $job_pay_interval_def[$jc_id] = 'none':
		$jc_pay = '薪酬：单独管理'
	elseif job_pay_rate_def[$jc_id] = 0:
		$jc_pay = '薪酬：视表现而定'
	else
		$jc_pay = '报酬： ' + $func('money', 'string_profit', job_pay_rate_def[$jc_id]) + '每' + $job_pay_interval_def[$jc_id]
	end

	!! ── Body text: hire hint (not employed) or description ──────────────
	if job_refresh_desc[$jc_id] = 1:
		!! Re-call definition to let jobs_list recompute $job_desc from runtime state
		if $job_definition_source[$jc_id] <> '' and loc($job_definition_source[$jc_id]):
			gs $job_definition_source[$jc_id], $jc_id
		else
			gs 'jobs_list', $jc_id
		end
	end
	$jc_body = ''
	if $job_status[$jc_id] <> 'employed':
		$jc_body = $job_hire_hint[$jc_id + ', ' + str(job_hiring_step[$jc_id])]
	end
	if $jc_body = '': $jc_body = $job_desc[$jc_id]

	!! ── Alternating background (caller sets $temp_bcolor = '' to begin) ─
	$temp_bcolor = $func('themes', 'alt_color', $temp_bcolor)

	!! ── Render ────────────────────────────────────────────────────────
	$jc_body_html = '<div style="font-size:0.85em; margin-top:3px;"><<$jc_status>></div>'
	if $jc_sched <> '':
		$jc_body_html += '<div style="font-size:0.85em; margin-top:2px; opacity:0.8;"><<$jc_sched>></div>'
	end
	$jc_body_html += '<div style="font-size:0.85em; margin-top:2px; opacity:0.8;"><<$jc_pay>></div>'
	if $jc_body <> '':
		$jc_body_html += '<div style="margin-top:6px; font-size:0.9em;"><<$jc_body>></div>'
	end

	$card_in['icon'] = $jc_icon
	$card_in['title'] = $job_title[$jc_id]
	$card_in['title_sub'] = 'at ' + $job_location[$jc_id]
	$card_in['body'] = $jc_body_html
	$card_in['border'] = $jc_border
	$card_in['bg'] = $temp_bcolor
	$card_in['opacity'] = $jc_opac
	gs 'cards', 'shell'
	killvar 'card_in'
	killvar 'jc_body_html'

	killvar 'jc_id'		& killvar 'jc_opac'		& killvar 'jc_border'
	killvar 'jc_icon'	& killvar 'jc_slot'		& killvar 'jc_skey'
	killvar 'jc_days'	& killvar 'jc_t_arr'	& killvar 'jc_t_st'
	killvar 'jc_end_m'	& killvar 'jc_shift_h'	& killvar 'jc_shift_m'
	killvar 'jc_pay'	& killvar 'jc_body'		& killvar 'jc_status'
	killvar 'jc_sched'	& killvar 'jc_t_end'	& killvar 'jc_dur'
	exit
end

!! │   employed → preemployed → terminated → fired → unemployed → hidden    │
!! └────────────────────────────────────────────────────────────────────────┘
if $ARGS[0] = 'show_all':
	gs 'jobs', 'ensure_registry'

	show_all_ji = 0
	:show_all_sort
	if show_all_ji < arrsize('$job_registry'):
		$sa_id = $job_registry[show_all_ji]
		gs 'jobs', 'get_job_definition', $sa_id

			$sa_hidden[] = $sa_id

		elseif $job_status[$sa_id] = 'preemployed':
			$sa_pre[] = $sa_id
		elseif $job_status[$sa_id] = 'terminated':
			$sa_term[] = $sa_id
		elseif $job_status[$sa_id] = 'fired':

		else
			$sa_unemp[] = $sa_id
		end
		show_all_ji += 1
		jump 'show_all_sort'
	end
	!! Render buckets in order
	$temp_bcolor = ''
	show_all_ji = 0
	:show_all_r_emp
	if show_all_ji < arrsize('$sa_emp'):
		*p $func('jobs', 'card', $sa_emp[show_all_ji])

		jump 'show_all_r_emp'

	:show_all_r_pre
	if show_all_ji < arrsize('$sa_pre'):
		*p $func('jobs', 'card', $sa_pre[show_all_ji])
		show_all_ji += 1
		jump 'show_all_r_pre'
	end
	show_all_ji = 0
	:show_all_r_term
	if show_all_ji < arrsize('$sa_term'):
		*p $func('jobs', 'card', $sa_term[show_all_ji])
		show_all_ji += 1
		jump 'show_all_r_term'
	end
	show_all_ji = 0
	:show_all_r_fired
	if show_all_ji < arrsize('$sa_fired'):
		*p $func('jobs', 'card', $sa_fired[show_all_ji])
		show_all_ji += 1
		jump 'show_all_r_fired'
	end
	show_all_ji = 0
	:show_all_r_unemp
	if show_all_ji < arrsize('$sa_unemp'):
		*p $func('jobs', 'card', $sa_unemp[show_all_ji])
		show_all_ji += 1
		jump 'show_all_r_unemp'
	end
	show_all_ji = 0
	:show_all_r_hidden
	if show_all_ji < arrsize('$sa_hidden'):
		*p $func('jobs', 'card', $sa_hidden[show_all_ji])
		show_all_ji += 1
		jump 'show_all_r_hidden'
	end
	killvar 'temp_bcolor' & killvar 'show_all_ji' & killvar 'sa_id'
	killvar 'sa_emp' & killvar 'sa_pre' & killvar 'sa_term' & killvar 'sa_fired' & killvar 'sa_unemp' & killvar 'sa_hidden'
	exit
end
