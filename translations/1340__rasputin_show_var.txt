# rasputin_show_var

if $ARGS[0] = '':
!! First half of the show
	if hour = 21:
		gt 'rasputin_show_var', 'first_half'
	else
		gt 'rasputin_show_var', 'second_half'
	end
end


if $ARGS[0] = 'first_half':
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/rasputin_show_var_1.jpg"></center>'
	'服务员把你带到座位上。还没有其他观众到场。'
	act	'离开':
		minut += 1
		rasputin['variety_ticket'] = 1
		gt 'rasputin_walkway'
	end

	act	'等待演出开始': gt 'rasputin_show_var', 'first_half_start'
	!! if minut < 50:
		act	'要一杯免费香槟':
			minut += 10
			gs 'mood', 'raise', 'tiny'
			gs 'beverage', 'champagne_stats'
			gs 'stat'
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/drinks_free_4.jpg"></center>'
			'服务员微笑着递给你一杯香槟享用。'
			act '继续': gt 'rasputin_show_var', 'first_half_start'
		end
	!! end
end


if $ARGS[0] = 'first_half_start':
	minut += 60 - minut
	rasputin['variety_ticket'] = 2
	*clr & cla
	gs 'stat'
	if sound_settings['music_off'] = 0:
		$track_loop = 'sound/rasputin/music_var_1.mp3'
		volume = 100
		music_loop = 1
	end

	!! if month <= 6:
		'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/1.jpg"></center>'
		'帷幕缓缓升起，一群舞者优雅地旋转着来到舞台中央。'
		'你觉得流行音乐配上华丽服装的搭配很有意思。'
		act '好的':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/2.jpg"></center>'
			'舞者们配合着排练纯熟的舞步，随着音乐舞动。'
			act '好的':
				minut += 5
				*clr & cla
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/3.jpg"></center>'
				'就在观众欣赏演出时，穿红衣的舞者绕到后面，你可以看到她们在帮穿黑衣的舞者解开上衣。'
				act '好的':
					minut += 10
					*clr & cla
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/4.jpg"></center>'
					'一名穿黑衣的舞者举起双臂，展示她赤裸的上身，以艺术与女性之美的完美结合进一步撩拨观众，'
					'尽显艺术与女性的魅力。'
					act '好的':
						minut += 25
						*clr & cla
						gs 'stat'
						'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/5.jpg"></center>'
						'其余的舞者也很快加入进来，你周围的观众兴奋得沸腾起来。'
						'上半场演出在口哨声甚至一些大声的下流叫喊声中落幕。'
						act '离开':
							minut += 20
							gs 'mood', 'raise', 'small'
							gt 'rasputin_walkway'
						end
					end
				end
			end
		end
	!! end
end


if $ARGS[0] = 'second_half':
	!! Second half of the show
	menu_off = 1
	*clr & cla
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/1.jpg"></center>'
	'服务员把你带到座位上。许多人已经就座等待。'
	act	'离开':
		minut += 1
		rasputin['variety_ticket'] = 2
		gt 'rasputin_walkway'

	!! if minut < 50:
		act	'要一杯免费香槟':
			minut += 10
			gs 'mood', 'raise', 'tiny'
			gs 'beverage', 'champagne_stats'
			gs 'stat'

			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/drinks_free_4.jpg"></center>'
			'服务员微笑着递给你一杯香槟享用。'
			act '继续': gt 'rasputin_show_var', 'second_half_start'
		end
	!! end
end


if $ARGS[0] = 'second_half_start':
	minut += 60 - minut
	rasputin['variety_ticket'] = 0
	if sound_settings['music_off'] = 0:
		$track_loop = 'sound/rasputin/music_var_2.mp3'
		volume = 100
		music_loop = 1
	end

	!! if month <= 6:
		*clr & cla
		'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/6.jpg"></center>'
		'滑稽歌舞秀的下半场开始了。'
		'舞台重新亮起，舞者们从雾气效果中现身。'
		'她们的服装比上半场更加大胆豪放。'
		act '好的':
			*clr & cla
			'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/7.jpg"></center>'
			'舞者们舞姿性感撩人，诱惑着全场观众。'
			act '好的':
				minut += 10
				*clr & cla
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/8.jpg"></center>'
				'舞者们以椅子为支点进行表演，逐渐解开彼此的上衣。'
				'这些俏皮挑逗的动作很快吸引了观众的全部注意力。'
				act '好的':
					minut += 10
					*clr & cla
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/9.jpg"></center>'
					'舞者们轮流蹲下身子脱去内裤，向观众若隐若现地展示她们的私处。'
					act '好的':
						minut += 10
						*clr & cla
						gs 'stat'
						'<center><img <<$set_imgh>> src="images/locations/pushkin/rasputin/shows/var_1/10.jpg"></center>'
						'舞者们继续表演，赤裸的身体上只覆着一层薄薄的网纱。'
						'这场情色滑稽歌舞秀的下半场已经结束。'
						act '好的':
							gs 'mood', 'raise', 'small'
							if pcs_inhib >= 80:
								inhib_exp += rand(2, 6)
							elseif pcs_inhib >= 50 and pcs_inhib < 80:
								inhib_exp += rand(1, 3)
							end
							gs 'arousal', 'erotic', 15
							gs 'arousal', 'end'
							gt 'rasputin_walkway'
						end
					end
				end
			end
		end
	!! end
end
