			'你在市中心花了一个小时寻找客人。'
			act '回到埃里克那里':
				*clr & cla
				'<center><img <<$set_imgh>> src="images/locations/city/citycenter/nightclub/club11.jpg"></center>'
				if pcs_apprnc > rand (80, 120):
					gs 'money', 'earn', 300
					'你超出了埃里克的预期，找到了很多客人。他付给你 ' + $func('money', 'string_profit', 300) + '。'
				elseif rand (0, 1) = 0:
					gs 'money', 'earn', 200
					'埃里克似乎相当满意，付给你 ' + $func('money', 'string_profit', 200) + '。'
				else
					gs 'money', 'earn', 100
					'你几乎没找到什么人。埃里克看起来不太高兴，但还是为你的努力付给你 ' + $func('money', 'string_profit', 100) + '。'
				end
				if hour < 1 or hour >= 18:
					act '从埃里克那里再领一份活': gt 'city_nightclub', 'work'
				end

				act '去舞池': gt 'city_nightclub', 'inside'
			end
		end
	elseif clubr = 1:
		'"听着，<<$pcs_nickname>>, 我要你在俱乐部里转转，让大家尝尝这些酒，最好能让他们换喝这些。这活儿不会出错，你只要微笑，发发免费酒水就行。"'
		'"不用舍不得，反正供应商买单。一小时后我在这儿等你。"'
		act '开始工作':
			*clr & cla
			minut += 60
			gs 'stat'
			'<center><img <<$set_imgh>> src="images/locations/city/citycenter/nightclub/club6.jpg"></center>'
			'你花了一个小时派发这些味道不佳的酒水。'

				*clr & cla
				gs 'money', 'earn', 200
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/city/citycenter/nightclub/club11.jpg"></center>'
				'你递给埃里克空瓶子，他给了你 ' + $func('money', 'string_profit', 200) + '。'
				if hour < 1 or hour >= 18: act '再向埃里克要一份工作': gt 'city_nightclub', 'work'

				act '去舞池': gt 'city_nightclub', 'inside'
			end
		end
	elseif clubr = 2:
		'"我们缺一个舞者，你进那边的笼子里跳舞吧。要是表演精彩，还有奖金。"'
		act '进笼子':
			if pcs_inhib < 60: inhib_exp += rand(1, 2)
			*clr & cla
			minut += 60
			gs 'exp_gain', 'danc', rand (1, 3)
			'<center><img <<$set_imgh>> src="images/locations/city/citycenter/nightclub/club12.jpg"></center>'
			if pcs_danc > rand (30, 100):
				gs 'money', 'earn', 300
				'你奉献了一场精彩的表演，吸引了许多目光，还带动人们一起跳舞。'
				'埃里克非常高兴，付给你 ' + $func('money', 'string_profit', 300) + '。'
			else
				gs 'money', 'earn', 200
				'你很努力，但客人大多无视你。'
				'埃里克感谢你的帮助，付给你 ' + $func('money', 'string_profit', 200) + ', 但对你的舞技只字未提。'
			end
			gs 'stat'
			if hour < 1 or hour >= 18: act '再向埃里克要一份工作': gt 'city_nightclub', 'work'

			act '去舞池': gt 'city_nightclub', 'inside'
		end
	end
	killvar 'clubr'
end
