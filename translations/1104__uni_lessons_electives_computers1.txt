	elseif trait_vars['academic'] >= 3:
		will_cost = 0
	end

	if pcs_willpwr < will_cost:
		act '认真听布拉戈夫教授讲课' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '认真听布拉戈夫教授讲课' + $func('willpower', 'get_willcost_string'):
			if will_cost > 0: gs 'willpower', 'pay', 'self', 'chore'
			gt 'uni_lessons_electives_computers1', 'computers_101_listen'
		end
	end
