# nichCook

if $ARGS[0] = 'desc':
	*clr
	'<center><img <<$set_imgh>> src="images/characters/city/jegor/cook.jpg"></center>'
	if nichKnowsCook = 0:
		'这位厨师看起来相当老了。你猜他至少六十好几了。'
		'他注意到你时，从手头的活计上抬起头来。'
		if nichOutfitState = 0:
			'"你好。我叫叶戈尔。你是谁呢？"你还没来得及回答，他就大笑起来。"开玩笑的。我知道你是新来的女仆。难道没人告诉过你在这里要穿制服吗？这对尼古拉斯主人来说很重要。总之，我想我还没问你的名字。"'
		else
			'"你好。你一定就是我听说过的那个新女仆吧。我叫叶戈尔。"'
		end
		act '自我介绍':
			cla
			nichKnowsCook = 1
			if $pcs_firstname = $pcs_nickname:
				'"很高兴认识你，叶戈尔。我叫<<$pcs_nickname>>。"'
			else
				'"很高兴认识你，叶戈尔。我的名字是<<$pcs_firstname>>。但你可以叫我<<$pcs_nickname>>。"'
			end
			'"很高兴见到你，<<$pcs_nickname>>。我很高兴这里来了个新女仆。自从卡廷卡离开后，这里就有点无聊了。"'
			'他转回去继续干活。"她是个好姑娘。不过我想她和加拉夫人相处得不太好。"'
			'"不过我想你不是来跟一个老家伙闲聊的吧？你饿了吗？"他指了指冰柜。"这里的冷冻餐你可以随便吃。只要放进微波炉或烤箱加热一下就行。"'
			'"我做这些冷冻餐就是给这个用的。家里人总是吃新鲜现做的。但既然你肯定和我一样忙，我想你也不会愿意等我专门再给你做一份吧。"'
			'他指了指冰箱。"冰箱里的东西你都可以喝。毕竟补满冰箱也是你的职责。"他朝你眨了眨眼。'
			act '走开':
				gt $loc
			end
		end
	else
		if nichOutfitState = 0:
			'叶戈尔正忙着准备一些饭菜。他注意到你时，朝你笑了笑。"很高兴见到你，<<$pcs_nickname>>。你是被解雇了？还是只是忘了穿制服？"他冲你眨了眨眼，然后继续干活。'
		else
			'叶戈尔正忙着准备一些饭菜。他注意到你时朝你笑了笑，然后继续干活。'
		end
		cla
		act '交谈':
			'"你好，叶戈尔"'
			'"嘿，<<$pcs_nickname>>。"'
			cla
			act '聊聊尼古拉':
				'"你能跟我说说尼古拉斯主人的事吗？"'
				'"也就比你知道的多一点点。他是个商人，而且非常成功。他的父母在三十年前被人杀害。他三年前认识了加拉，没过多久他们就结婚了。"'
				'"如果你想了解更多，应该去问他本人。我可没资格散布关于我雇主的闲话。"'
			end
			act '聊聊加拉':
				'"你能跟我说说加拉夫人的事吗？"'
				'"我不想说任何对她不利的话。所以恐怕我不能告诉你太多。只有一个善意的提醒：我敢惹她不高兴，是因为我在尼古拉斯主人认识她之前很久，就一直是他忠实的仆人了。你可要更加小心。"'
			end
			act '聊聊塔尼娅':
				'"你能跟我说说塔尼娅的事吗？"'
				'"我第一次见到她，是她跟她母亲搬进来的时候。她看起来是个不错的年轻姑娘，虽然我们不怎么交谈。她和尼古拉斯主人似乎相处得格外融洽。"'
			end
			act '聊聊这套公寓':
				'"关于这套公寓，你知道些什么？"'
				'"其实知道得不多。我知道在苏联解体前，这里曾是一栋政府大楼。尼古拉斯主人在五年前买下并翻新了它。如果你去看看书房，还能看到那些旧家具。除了他的电脑，那间房间基本上没动过。"'
			end
			act '聊聊叶戈尔':
				'"你能跟我说说你自己的事吗？"'
				'"当然可以。我曾经是尼古拉斯主人父亲的保镖。他是个有权势的人，就像他儿子今天一样。但他有仇家。很多仇家。他们最终决定派杀手来杀他。我成功击退了他们两次。但当他们第三次来的时候，我不得不做一个艰难的决定。我当时只能救下一位家人。"'
				'他犹豫了几秒钟，才继续讲他的故事。'
				'"那是我一生中最艰难的选择。我受雇去救那位老人，但我知道他宁愿自己去死，也不愿看到儿子被杀。所以我救了尼古拉。那时他还年轻。"'
				'"我成功救下了他。但付出了巨大的代价。我被刺客的几发子弹击中。医护人员发现我的时候，以为我已经死了。但我的求生意志很强。"'
				'"在医院里躺了几个月后，我康复了。但我的反应、力量和耐力却没有恢复。我再也无法担任保镖了。"'
				'"我迷失了。一个破碎的人，除了使用这副已经没用的身体，我什么都不会。"'
				'"但尼古拉斯主人没有忘记我。他知道我为保护他的家人尽了一切努力。所以他把唯一能给我的闲职给了我：当他的厨师。"'
				'"起初我是个糟糕透顶的厨师。你知道尼古拉斯主人在各方面都追求完美。唯一的例外就是救过他命的人。"'
				'"他给了我时间来提升我的手艺。如今，即使他不讲情面，我也足够优秀到能被他雇用。他亲口这么跟我说过。"'
				'"他是个很好的雇主，我欠他一切。"'
			end
			act '聊聊不寻常的事？':
				'"你在这里见过什么''不寻常''的事吗？"'
				'"当然没有。尼古拉斯主人、他的家人、他的员工或他的客人做什么，都不是我该过问的。当然，他本人不透露的事，我也绝不会透露。在我看来，这套公寓里发生的一切都是''寻常''的。"'
			end
			act '走开':
				gt $loc
			end
		end
		act '调情':
			'"你好，叶戈尔。"你故意俯身靠在台面上，把乳沟展示给他看。"你填那只鹅的时候，我有了个主意。你想听听吗？"'
			'他朝你咧嘴一笑。"你是个好姑娘，我很感激你的用心。但在这里你是在浪费时间。"'
			cla
			act '走开':
				gt $loc
			end
		end
		act '走开':
			gt $loc
		end
	end

end
