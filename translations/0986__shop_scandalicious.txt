!! 2022/06/08

if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_scandalicious', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '惹火')
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/scandalicious/shop.jpg"></center>'
	*nl
	'这家店又便宜又俗气。不过衣服倒是名副其实，你猜他们在布料上省了不少钱。'
	'这不是你会和妈妈一起去的地方，也不会让她知道你甚至看过橱窗——但如果你想炫耀身材，没有比这里更好的购物地点了。店员不会评判任何人，但顾客中既有掩饰尴尬的，也有散发着自信的。'
	*nl

	act '离开': minut += 3 & gt 'city_mall'
	act '查看服装': minut += 5 & gt 'shop_scandalicious', 'clothes'
	act '查看连衣裙': minut += 5 & gt 'shop_scandalicious', 'dresses'
	act '查看鞋子': minut += 5 & gt 'shop_scandalicious', 'shoes'
	act '查看泳衣': minut += 5 & gt 'shop_scandalicious', 'swim'
	act '查看比基尼': minut += 5 & gt 'shop_scandalicious', 'bikinis'
	act '查看连体衣': minut += 5 & gt 'shop_scandalicious', 'bodysuit'
end


if $ARGS[0] = 'clothes':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_scandalicious', 'clothes'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看绯闻女郎套装')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'scandalicious_outfits'
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_scandalicious', 'start'
	end
end
