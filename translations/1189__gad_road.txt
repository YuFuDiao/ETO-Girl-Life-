# gad_road
!! 2024/05/05 - Mira schedule introduced.

if $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'gad_road', 'start'
	$location_type = 'public_outdoors'
	gs 'schedule', 'A60'
	gs 'gadukino_event', 'sound'
	gs 'stat'

	'<center><h4>林间小路</h4></center>'

	if month >= 4 and month <= 10:
		if hour >= 6 and hour < 22:
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadroad.jpg"></center>'
		else
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadroad_night.jpg"></center>'
		end
	else
		if hour >= 8 and hour < 18:
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadroad_winter.jpg"></center>'
		else
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadroad_winter_night.jpg"></center>'
		end
	end
	*nl
	'一条土路通向加杜基诺的村外。你被麦田包围着，麦子高得仿佛一直延伸到天际。'
	'森林 ' + iif(gad_meadow_found = 1, '和草地 ', '') + '可以在道路更远处，远离村庄找到。'
	*nl
	!! 10% chance of meeting Mitka once per day on the road when Sveta is not the village boys slut or whole villages whore
	if rand(0, 9) = 0 and GadBoy['first_drink'] > 0 and GadBoy['mitka_day'] ! daystart and hour >= 17 and hour < 20 and (npc_QW['A63'] < 11 or GadBoy['river_gang'] > 2):
		'<a href="exec:gt ''mitka''">米特卡</a>正靠在路边抽烟，你经过时他朝你微笑。'
	!! Meet the boys to drink with them at 20 if Sveta has been invited by Mitka already
	elseif GadBoy['first_drink'] > 0 and hour = 20 and alko < 10 and GadBoy['drinkday'] ! daystart:
		'<a href="exec:gt ''gad_road'', ''drinking_invite''">米特卡，科良巴，和瓦西扬</a>正沿着路朝森林走去，一边喝着私酿酒一边大声交谈。'
	!! If Sveta is late to go drink with the boys if she already knows where the trailer is she can still get there
	elseif GadBoy['first_drink'] = 2 and hour > 20 and alko < 10 and GadBoy['drinkday'] ! daystart:
		!! if Sveta did not bring Mira with her and Mira has met the boys already then there is a 50% chance she will show up too

		if $locat['A60_loc'] = 'mitkabuh_group':
			if MiraVars['QW'] >= 16 and GadBoy['river_gang'] = 2:
				'你可以去和米拉，米特卡，科良巴和瓦夏在他们的<a href="exec:gt ''mitkabuh_group''">拖车</a>藏在树林里。'
			else
				'你可以去和米拉，米特卡，科良巴和瓦夏在他们的<a href="exec:gt ''mitkabuh''">拖车</a>藏在树林里。'
			end
		else
			'你可以去和米特卡，科良巴和瓦西扬一起喝酒，在他们的<a href="exec:gt ''mitkabuh''">拖车</a>藏在树林里。'
		end
	end

	act '步行前往加杜基诺 (0:10)': minut += 10 & gt 'gadukino'
	act '步行前往河边 (0:10)': minut += 10 & gt 'gad_river', 'start'
	act '步行前往田野 (0:10)': minut += 10 & gt 'gad_field', 'field'
	act '步行前往森林 (0:15)': minut += 15 & gt 'gad_forest', 'forest_edge'
	act '步行前往公交车站 (0:01)': minut += 1 & gt 'bus', 'gadukino'

	if hour >= 6 and hour < 22:
		if PSport = 0:
			'如果你穿着运动服和运动鞋，你可以在这里跑一跑。'
		else
			if pcs_stam < stammax / 5:
				act '去跑步 (1:00)': $func('wrap', 'neg', '<br>你太累了，做不了这个。在尝试跑步之前，先恢复体力。')
			elseif pcs_energy < 20:
				act '去跑步 (1:00)': $func('wrap', 'neg', '<br>你饿得无法面对去跑步。')
			elseif pcs_hydra < 20:
				act '去跑步 (1:00)': $func('wrap', 'neg', '<br>你渴得无法面对去跑步。')
			else
				act '去跑步 (1:00)':
					*clr & cla
					menu_off = 1
					gs 'exercise', 'tier1', 60, 'run'

					if month >= 11 or month <= 3:
						'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadbeg4.jpg"></center>'
					elseif month >= 9 or month <= 5:
						'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadbeg3.jpg"></center>'
					else
						if strenbuf >= pcs_mass['body']:
							'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadbeg1.jpg"></center>'
						else
							'<center><img <<$set_imgh>> src="images/locations/gadukino/village/gadbeg2.jpg"></center>'
						end
					end
					'你跑了一个小时，享受着沿途的风景。'

					act '完成': gt 'gad_road', 'start'
				end
			end
		end
	end
end


if $ARGS[0] = 'drinking_invite':
	gs 'core_library', 'setloc', 'gad_road', 'drinking_invite'
	$location_type = 'secluded'
	menu_off = 1
	gs 'miroslava_schedule'
	minut += 5
	gs 'stat'

	'<center><img <<$set_imgh>> src="images/locations/gadukino/village/drunk.jpg"></center>'
	'你很快就追上了已经喝起来的男孩们。'
	!! Sveta has never been drinking with Mitka before
	if GadBoy['first_drink'] = 1:
		'米特卡第一个向你打招呼，"嘿，<<$pcs_nickname>>，很高兴你今天决定来加入我们！"'
		'科良巴和瓦西扬打量着你并点头表示同意。'
		'"我们知道树林里有个聚会的好地方。从来没有人打扰过我们。"'
	else
		'米特卡第一个向你打招呼，"嘿，<<$pcs_nickname>>，很高兴你今天决定来加入我们！"'
		'科良巴和瓦西扬打量着你并点头表示同意。'
	end
	'米特卡问道："你准备好了吗？"'

	!! if Sveta came here to kick off Miras quest then she cannot leave otherwise Sveta is free to decline the offer
	if MiraVars['QW'] > 0 or $locat['A60_loc'] ! 'mitkabuh_group':
		gs 'willpower', 'drink', 'resist'
		if pcs_willpwr < will_cost:
			act '不了，谢谢，今晚不行。' + $func('willpower', 'get_willcost_string'): $noWillpower
		else
			act '不了，谢谢，今晚不行。' + $func('willpower', 'get_willcost_string'):
				gs 'willpower', 'drink', 'resist'
				gs 'willpower', 'pay', 'resist'
				gs 'stat'

				gt 'gad_road', 'start'
			end
		end
	end

	!! if Sveta did not bring Mira with her and Mira has met the boys already then there is a 50% chance she will show up too
	if $locat['A60_loc'] = 'mitkabuh_group' and $locat['A60_loc_prev'] ! 'mitkabuh_group' and MiraVars['follow_time'] = 0:
		'米特卡话音未落，米拉就从拐角处蹦蹦跳跳地出现了。'
		'"嘿，伙计们，谢谢你们的邀请！<<$pcs_nickname>>也在这儿，太棒了！我们走吧！"'
	end

	if $locat['A60_loc'] = 'mitkabuh_group':
		!! Mira is here and both her and Sveta are the town whores
		!! mitkabuh_group has the drinking events for group sex for 2 girls and 3 boys
		if MiraVars['QW'] >= 16 and GadBoy['river_gang'] = 2:
			act '和他们一起去': minut += 15 & gt 'mitkabuh_group'

		!! Mira is here and neither or only 1 out 2 of them is the town whore
		!! mitkabuh has the drinking events for group sex for 1 girl and 3 boys or no sex at all if quest is not far enough along
		else
			act '和他们一起去': minut += 15 & gt 'mitkabuh'
		end
	!! Mira is not here so Sveta goes by herself
	else
		act '和他们一起去': minut += 15 & gt 'mitkabuh'
	end
end
