if nerd_game['game_day'] = daystart and hour = 20 and yearstart = 1: act '<b>加入书呆子们</b>': gt 'nerd_game_night', 'game'

if hour >= 12 and hour < 20:
	'这个时间段，社区中心会开放各种兴趣小组。'

	if krumonth = month and kruyear = year:
		act '去缝纫社': gt 'krupo'
	else
		act '去缝纫社':
			cls
			menu_off = 1
			gs 'stat'
			$func('wrap', 'header center b', '缝纫小组')
			'<center><img <<$set_imgh>> src="images/locations/pavlovsk/community/sew/sewing_class.jpg"></center>'
			'你来到缝纫小组的活动室。走进去时，一位老师看着你说：“一个月的课程费用是<<$func(''money'', ''string_price'', 500)>>，你可以想参加多少次就参加多少次。一个月结束后，没来上课的日子是不退款的。”'
			if func('money', 'can_afford', 500) = 1:
				act '报名':
					cls
					gs 'money', 'pay', 500
					krumonth = month
					kruyear = year
					gs 'stat'
					'你支付学费。'
					act '离开': gt 'pav_commclubs'
				end
			end

			act '离开': gt 'pav_commclubs'
		end
	end

	if chessmonth = month and chessyear = year:
		act '去国际象棋社': gt 'kruchess'
	else
		act '去国际象棋社':
			cls
			menu_off = 1
			gs 'stat'
			$func('wrap', 'header center b', '国际象棋社')
			'<center><img <<$set_imgh>> src="images/locations/pavlovsk/community/chess/kruchess.jpg"></center>'
			'你来到国际象棋社的活动室。走进去时，一位老师看着你说：“入会费用是<<$func(''money'', ''string_price'', 500)>>一个月，你想来多少天都可以。一个月到期后，没来的日子是不退款的。”'
			if func('money', 'can_afford', 500) = 1:
				act '报名':
					cls
					gs 'money', 'pay', 500
					chessmonth = month
					chessyear = year
					gs 'stat'
					'你支付学费。'
					act '离开': gt 'pav_commclubs'
				end
			end

			act '离开': gt 'pav_commclubs'
		end
	end

	if ml_guitarlesson['advertisement'] = 1 and ml_guitarlesson['enquired'] = 0:
		act '咨询吉他课': gt 'music_guitarlesson', 'enquiry'
	end

	if ml_guitarlesson['enquired'] = 1 and ml_guitarlesson['lessonday'] = 8:
		act '约定上课日期': gt 'music_guitarlesson', 'setlessonday'
	end

	if (ml_guitarlesson['nextlesson'] = daystart):
		if (ml_guitarlesson['lessonhour']-1 = hour):
			act '等待你的吉他课' + $func('money', 'get_cost_string', ml_guitarlesson['lessoncost']):
				if func('money', 'can_afford', ml_guitarlesson['lessoncost']) = 0:
					$noMoney
				else
					minut += (60 - minut)
					gt 'music_guitarlesson', 'lesson'
				end
			end
		end
		if (ml_guitarlesson['lessonhour'] = hour):
			act '去上你的吉他课' + $func('money', 'get_cost_string', ml_guitarlesson['lessoncost']):
				if func('money', 'can_afford', ml_guitarlesson['lessoncost']) = 0:
					$noMoney
				else
					gt 'music_guitarlesson', 'lesson'
				end
			end
		end
	end
else
	'俱乐部关门了。'
end
