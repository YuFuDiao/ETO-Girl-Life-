				if class['<<$class_list_institution[i]>>_<<$class_list_name[i]>>_optional_weekly_grade_gain'] < class['<<$class_list_institution[i]>>_<<$class_list_name[i]>>_optional_weekly_max']:

					'你这周不用再为你的<<$class_list_name[i]>>课程学习了。'
