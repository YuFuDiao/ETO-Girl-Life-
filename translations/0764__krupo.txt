# krupo
$loc_arg = $ARGS[0]
$loc = 'krupo'
$location_type = 'event'
menu_off = 1

frost = 0
gs 'stat'

gs 'themes', 'indoors'

$func('wrap', 'header center b', '缝纫小组')
'<center><img <<$set_imgh>> src="images/locations/pavlovsk/community/sew/sewing_lesson.jpg"></center>'
if pcs_sewng < 40:
	gs 'exp_gain', 'sewng', rand(pcs_intel/20, pcs_intel/10)
	minut += 60
	'你花了一个小时学习不同的缝纫技巧。'
end

if pcs_sewng >= 40:
	if tkanfirst = 0:
		minut += 60
		gs 'exp_gain', 'sewng', rand(pcs_intel/20, pcs_intel/10)
		'因为这是你第一次需要布料，她匀了一些给你，但她告诉你下次要自己带。'
		msg '老师告诉你，以后学习需要自己带布料来，你可以在超市的五金区买到。'
		tkanfirst = 1
	elseif mc_inventory['sewing_fabric'] <= 0:
		msg '老师提醒你，要想进一步提高技能，你需要买些布料来练习。'
		'没有布料就没法上课，所以你现在在这里无事可做。'
	else
		minut += 60
		mc_inventory['sewing_fabric'] -= 1
		gs 'exp_gain', 'sewng', rand(pcs_intel/20, pcs_intel/10)
		'你花了一个小时学习更高级的技巧，练习裁剪和缝纫你买来的布料。'

		if pcs_sewng >= 60 and poTalk = 0:
			poTalk = 1
			*nl
			'老师满意地笑着对你说："干得好，现在你已经能把衣服改得完全合身了！不过，这并不意味着继续学习就没用了。再上几次课，我觉得就能教你绣出精美的挂毯，那可是很抢手的东西，虽然制作起来非常费工夫。"'
		end

		if pcs_sewng >= 80 and poTalk = 1:
			*nl
			'老师轻轻地点了点头说："你已经掌握了简单的挂毯。虽然还有些瑕疵，但已经好到可以拿去卖了，而且随着你的技艺不断提高，质量也会越来越好。"'
		end
	end
end

*nl
'你的缝纫技能现在是<<pcs_sewng>>。'
*nl
if mc_inventory['sewing_fabric'] >= 0: '你的布料还够上<<mc_inventory[''sewing_fabric'']>>节课。'

gs 'stat'
act '离开': gt 'pav_commclubs'
