		Yakov['dildo'] = -4
		Yakov['mood'] = 10
		shared_apt['step'] = 0
		act '离开': gt 'pav_shared_apt', 'start'

	!! may attempt again
	elseif shared_apt['step'] = 120:
		shared_apt['step'] = 0
		'<font color=#<<$Yakov[''font'']>>>"我明白了。至少还留着一丝希望。回头见，<<$pcs_firstname>>。"</font>'
		Yakov['mood'] = 8
		act '离开': gt 'pav_shared_apt', 'start'

	!! paygate
	elseif shared_apt['step'] = 200:
		Yakov['dildo'] += 1
		shared_apt['step'] = 0
		gs 'arousal', 'end'
		gs 'pav_shared_apt', 'payService'
	end
end
