# onobjsel
!! the dynamics can all be found in obj_din and cheatmenu_din

if $selobj = '<center>Refresh</center>':
	gs 'stat'
	if stat_cfg['android']: clr & pl $stat_msg
end
if $selobj = '<center>Console</center>':
	$dynamicCommand = input("输入要执行的命令。")
	if $dynamicCommand = '': exit
	dynamic $dynamicCommand
	killvar 'dynamicCommand'
end
if $selobj = '<center>切换调试变量</center>':
	if objectWindowDebug = 0:
		objectWindowDebug = 1
	else
		objectWindowDebug = 0
	end
	gs 'stat'
end

if $selobj = '<center>添加调试变量</center>':
	$tmpVar = input('输入变量名')
	if $tmpVar <> '':
		gs 'obj_din', 'AddDebugVar', $tmpVar
		gs 'stat'
	end
	killvar 'tmpVar'
end

if $selobj = '<center>删除调试变量</center>':
	$tmpVar = input('输入要删除的变量名')
	if $tmpVar <> '':
		gs 'obj_din', 'DeleteDebugVar', $tmpVar
		gs 'stat'
	end
	killvar 'tmpVar'
end

if $selobj = '<center>切换 HTML</center>':
	usehtml = iif(usehtml, 0, 1)
end



if instr($selobj, ':') > 0:
	tmpPos = instr($selobj, ':') - 1
	$tmpVar = mid($selobj, 1, tmpPos)
	$tmpVal = input('输入一个值给<<$tmpVar>>')
	if $tmpVal <> '':
		if instr($tmpVar, '$') = 1:
			dyneval('<<$tmpVar>> = "<<$tmpVal>>"')
		else
			dyneval('<<$tmpVar>> = <<VAL($tmpVal)>>')
		end
	end
	gs 'stat'
	killvar 'tmpVar'
	killvar 'tmpPos'
	killvar 'tmpVal'
end

if $selobj = '<center>-1 Hour</center>': gs 'obj_din', '-1 Hour'
if $selobj = '<center>+1 Hour</center>': gs 'obj_din', '+1 Hour'
if $selobj = '<center><font color = red>Emergency</font></center>': gs 'obj_din', 'emergency'
if $selobj = '<center>移除调试信息</center>':
	msg '加载存档时调试窗口将不再自动打开。你可以在设置中调整。'
	debug_warning_closed = 1
	cfg_vars['debug'] = 0
	showobjs cfg_vars['debug']
	gs 'stat'
end

if $selobj = '<center>切换调用跟踪</center>':
	if debug['trace_shown'] = 0:
		debug['trace_shown'] = 1
		gs 'stat_display'
	else
		debug['trace_shown'] = 0
		gs 'stat'
	end
end

unselect
