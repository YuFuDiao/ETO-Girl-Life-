!! This file only runs once

!! Add image path $adv_imgPath['function'] to every advert

!! Example
!! $adv_imgPath['model'] = 'images/locations/shared/adverts/flyer_modelling.jpg'

!! Give the advert a name and a hidden name
!! Name is visible when the Advert is known
!! HiddenName is visible, when the advert is unknown

!! Example
!! $adv_name['model'] = 'Modelling'
!! $adv_nameHidden['model'] = 'Job offer'

!! You can set a chance for advert to appear 0% to 100%
!! "100" and above will always appear
!! "50" is about a 50/50 chance
!! "0" and below will never appear

!! Examples
!! adv_chance['model'] = 100
!! adv_chance['model'] = 60

!! You can add allowed advert locations like this adv_&loc['&curloc'] = 1
!! Remember to put new unknown locations also into 'advert_manager' 'check_location'

!! Examples
!! adv_pav_commercial['model'] = 1
!! adv_pav_commcenter['model'] = 1
!! adv_city_residential['model'] = 1
!! adv_city_center['model'] = 1
