if $ARGS[0] = 'init':
	$loc_id = 'beta_journal_work'
	$hot_link = $loc_id
	gs 'beta_journal', 'nav_construct'

	!! Ad hoc events and contracts
	if arrsize('evt_event') ! 0: gs 'jobs_gigs', 'disp_evt', 2

	!! jobs_list job cards
	gs 'jobs', 'show_all'

	!! ── Non-registry jobs ──────────────────────────────────────────────
	*nl
	if teacher['job'] > 2: '你在 Lycée 当老师，工作时间是周一至周六，授课时间为 '+func('time', 'get_time_string', 14, 0)+' 至 '+func('time', 'get_time_string', 16, 0)+'。你的薪水是每天<<$func(''money'', ''string_profit'', 300)>>。'
	if bumtolik = 4: '你被请去城市工业区的仁爱诊所（Mercy Clinic）帮助无家可归的人。'
	if bumtolik >= 5: '你在城市工业区的仁爱诊所（Mercy Clinic）做志愿者，帮助无家可归的人。工作时间为周六和周日的 '+func('time', 'get_time_string', 9, 0)+' 至 '+func('time', 'get_time_string', 17, 0)+'。'

	*nl

	!! ── Upcoming events ────────────────────────────────────────────────
	if audition['day'] = daystart:
		'你试演的角色<<$role>>就在今天！'
	elseif audition['day'] - daystart = 1:
		'你明天要试镜角色<<$role>>。'
	elseif audition['day'] - daystart > 1:
		'你将在<<audition[''day''] - daystart>>天后试镜角色<<$role>>。'
	end

	if actress_cast = 1:
		if acting['shoot_day'] = daystart:
			'你是 '
		elseif acting['shoot_day'] - daystart = 1:
			'你被安排在<<acting[''shoot_day''] - daystart>>天后扮演<<$role>>。'
		elseif acting['shoot_day'] - daystart > 1:
			'你被安排在<<acting[''shoot_day''] - daystart>>天后扮演<<$role>>。'
		end
	end

end
