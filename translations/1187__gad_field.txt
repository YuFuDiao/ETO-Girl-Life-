if $ARGS[0] = 'set_nomira_acts':
	if $locat['A60_loc'] ! 'gad_field':
		if ARGS[1] = 1:
			act '在田野里跑跑步 (1:00)':
				*clr & cla
				menu_off = 1
				'<center><img <<$set_imgh>> src="images/locations/gadukino/village/graze_cow_run.jpg"></center>'
				'既然牛群都很安分，你便在田野里慢跑了一会儿。'

				frost = 0
				gs 'exercise', 'tier1', 60, 'run'
				if temper >= 25 and sunWeather = 1:
					pcs_hydra -= 6
					gs 'sweat', 'add', 2
				end
				gs 'stat'

				act '继续': gt $loc, $loc_arg
			end
			if temper >= 15 and sunWeather = 1:
				if pcs_sleep < 40:
					act '小睡一会儿 (1:00)':
						*clr & cla
						menu_off = 1
						'<center><img <<$set_imgh>> src="images/locations/gadukino/village/graze_cow_lying.jpg"></center>'
						'看着牛群站在周围，你感到昏昏欲睡，于是躺倒在草地上。你打了个盹，大约一小时后醒来。'

						if temper >= 25: pcs_hydra -= 12
						gs 'sleep_simple', 'nap', 1

						act '继续': gt $loc, $loc_arg
					end
				end
				act '采花 (1:00)':
					*clr & cla
					menu_off = 1
					'<center><img <<$set_imgh>> src="images/locations/gadukino/village/graze_cow_flowers.jpg"></center>'
					'你一边留神着牛群，一边在田野里逛了一个小时，采摘花朵。'
					minut += 60
					gs 'mood', 'raise', 'small'
					pcs_stam -= 4
					if temper >= 25:
						pcs_hydra -= 8
						gs 'sweat', 'add', 5
					else
						pcs_hydra -= 5
						gs 'sweat', 'add', 3
					end
					gs 'stat'

					act '继续': gt $loc, $loc_arg
				end
			end
		elseif ARGS[1] = 2:
			if temper >= 15 and sunWeather = 1:
				if pcs_stam >= (5 * (10 - sport_clothes_exercise_bonus)) / 2:
					act '脱衣游泳 (0:15)':
						*clr & cla
						menu_off = 1
						'<center><img <<$set_imgh>> src="images/locations/gadukino/village/graze_cow_river.jpg"></center>'
						'你飞快地脱掉衣服，在凉爽宜人的河水里扑腾了几分钟。'

						cumspclnt = 4 & gs 'cum_cleanup'
						pcs_stam -= 3
						gs 'mood', 'raise', 'tiny'

						gs 'exercise', 'tier1', 15, 'stren'
						gs 'stat'

						act '继续': gt 'gad_field', 'cow2'
					end
				end
			end
		end

			*clr & cla
			menu_off = 1
			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/graze_cow_sun.jpg"></center>'
			'你在河边躺下晒日光浴，爷爷则照看着牛群。'

			gs 'sweat', 'add', 10
			gs 'mood', 'raise', 'tiny'
			pcs_tan += 1
			gs 'stat'

			act '继续': gt $loc, $loc_arg
		end

			*clr & cla
			menu_off = 1
			gs 'library_functions', 'read_book'

			'<center><img <<$set_imgh>> src="images/locations/gadukino/village/graze_cow_book'+rand(1, 2)+'.jpg"></center>'

			if trait_vars['bookworm'] > 0:
				gs 'mood', 'raise', 'medium'
				if pcs_stren > 1: gs 'exp_deg', 'stren', 1
				if pcs_vital > 5: gs 'exp_deg', 'vital', 1
				'你开始读随身带来的书。你完全沉浸在故事里，忘记了时间。'
			else
				gs 'mood', 'raise', 'tiny'
				'读着书，时间不知不觉就过去了，因为你完全被情节吸引住了。'
			end
			gs 'stat'

			act '合上书': gt $loc, $loc_arg
		end
	end
