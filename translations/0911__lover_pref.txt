if $ARGS[0] = '':

	killvar 'temp_trait'
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'body_bmi'
	if $ngpPrefResult['HasPos'] ! '':
		$temp_trait = $ngpPrefResult['HasPos']
	elseif $ngpPrefResult['NotPos'] ! '':
		$temp_trait = $ngpPrefResult['NotPos']
	end

	if $temp_trait = 'body_bmi_starving' or $temp_trait = 'body_bmi_underweight':
		figurePrefMin = 0
		figurePrefMax = 18
	elseif $temp_trait = 'body_bmi_normal':
		figurePrefMin = 19
		figurePrefMax = 25
	elseif $temp_trait = 'body_bmi_overweight' or $temp_trait = 'body_bmi_obese':

	!! This is to find out if they want you to wear piercings.
	!! This is to set the max min
	killvar 'temp_trait'
	gs 'npc_get_preference', $npcID, 'randomPosIndNeg', 'cosmetics_piercings'

		$temp_trait = $ngpPrefResult['NotPos']

		pierPrefTmpMax = 2
	elseif $temp_trait = 'cosmetics_piercings_multiple':
		pierPrefTmpMin = 3
		pierPrefTmpMax = 5
	elseif $temp_trait = 'cosmetics_piercings_lot':
		pierPrefTmpMin = 6
		pierPrefTmpMax = 99
	else
		pierPrefTmpMin = 0
		pierPrefTmpMax = 99

	else
		lipsPrefTmpMin = 0
		lipsPrefTmpMax = 99
	end

	if npc_rel[$npcID] <= 40:
		$giveNick = '<<$pcs_firstname>>'
	elseif npc_rel[$npcID] <= 80:
		if rand(0, 1) = 0:
			$giveNick = '宝贝'
		else
			$giveNick = '<<$pcs_nickname>>'
		end
	else
		temp_rand = rand(0, 2)

		elseif temp_rand <= 1:
			$giveNick = '美人'
		else
			$giveNick = '亲爱的'
		end
		killvar 'temp_rand'
	end

	exit
