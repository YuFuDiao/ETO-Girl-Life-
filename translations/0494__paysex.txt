!! Since there is no way of knowing how big $boy is, before using the array use: killvar 'boy'
!! It does not need the array size
!! Custom text can be entered for each kind of sex type, in $paysextext[0] = '', $paysextext[1] = '' format

menu_off = 1

if ARGS[1] > 0:
	knwn = ARGS[1]
	:knownloop
		knownboy[knwn] = 1
		knwn -= 1
	if knwn > 0: jump 'knownloop'
end


if $ARGS[0] = 'start':
	stat['gangbang_count'] += 1
	gs 'sweat', 'add', 10
	picrand = rand(0, 1)
	gt 'paysex', 'var'
end

	if temp_randpicture = 1: $randpicture = 'anal'

			end

			if knownboy[j] = 1: gs 'cum_call', '', $temp2[j], 1 else gs 'cum_call', '', $temp2[j]

			if paysxanal = 0: paysxanal = 1

			if $paysextext[j] = '':
				'<<$boydesc>>操你的屁股。'
			else
				'<<$paysextext[j]>>'
			end

			if paysxbj = 0: paysxbj = 1
			gs 'arousal', 'bj', -5, 'prostitution', 'gangbang'
			if $paysextext[j] = '':
				'<<$boydesc>>操你的嘴。'
			else
				'<<$paysextext[j]>>'
			end
		elseif j = 3:
			if paysxhj = 0: paysxhj = 1
			gs 'arousal', 'hj', -5, 'prostitution', 'gangbang'
			if $paysextext[j] = '':
				'<<$boydesc>>用你的一只手撸他的鸡巴。'
			else
				'<<$paysextext[j]>>'
			end
		elseif j = 4:
			gs 'arousal', 'hj', -5, 'prostitution', 'gangbang'
			if $paysextext[j] = '':
				'<<$boydesc>>用你的另一只手撸他的鸡巴。'
			else
				'<<$paysextext[j]>>'
			end
		elseif j > 4:
			if $paysextext[j] = '':
				'<<$boydesc>>站在你面前撸着自己，等着轮到他。'
			else
				'<<$paysextext[j]>>'
			end
		end
		j += 1
	if j < i: jump 'arrayloop'

	$boy[0] = $boyP

	gs 'arousal', 'end'
	gs 'stat'

	if sexvar > 1:
		act '继续': gt 'paysex', 'var'
	else
		act '完成': gt 'paysex', 'end'
	end
end


if $ARGS[0] = 'end':

	*clr
	sexvar -= 1

	copyarr '$temp1', '$boy'

			end

				if knownboy[j] = 1: gs 'cum_call', 'mouth', $temp2[j], 1 else gs 'cum_call', 'mouth', $temp2[j]

				'<<$boydesc>>正在往你的脸上射精。'
				if knownboy[j] = 1: gs 'cum_call', 'face', $temp2[j], 1 else gs 'cum_call', 'face', $temp2[j]

		elseif j > 2:

				if knownboy[j] = 1: gs 'cum_call', 'breasts', $temp2[j], 1 else gs 'cum_call', 'breasts', $temp2[j]

				if knownboy[j] = 1: gs 'cum_call', 'stomach', $temp2[j], 1 else gs 'cum_call', 'stomach', $temp2[j]

				'<<$boydesc>>正在往你的头发上射精。'

			elseif cumrand = 3:
				'<<$boydesc>>正在往你的脸上射精。'
				if knownboy[j] = 1: gs 'cum_call', 'face', $temp2[j], 1 else gs 'cum_call', 'face', $temp2[j]
			elseif cumrand = 4:
				'<<$boydesc>>正在往你的屁股上射精。'
				if knownboy[j] = 1: gs 'cum_call', 'butt', $temp2[j], 1 else gs 'cum_call', 'butt', $temp2[j]
			elseif cumrand = 5:
				'<<$boydesc>>正在往你的阴唇上射精。'
				if knownboy[j] = 1: gs 'cum_call', 'labia', $temp2[j], 1 else gs 'cum_call', 'labia', $temp2[j]
			elseif cumrand = 6:
				'<<$boydesc>>正在往你的小穴里射精。'
				if knownboy[j] = 1: gs 'cum_call', '', $temp2[j], 1 else gs 'cum_call', '', $temp2[j]
			elseif cumrand = 7:
				'<<$boydesc>>正在往你的菊花里射精。'
				if knownboy[j] = 1: gs 'cum_call', 'anus', $temp2[j], 1 else gs 'cum_call', 'anus', $temp2[j]

	gs 'stat'
	killvar 'knownboy'
	killvar 'knwn'
	'所有男人都在你身上发泄完之后，你终于可以喘口气了。'

	act '离开':
		paysxbj = 0
		paysxsex = 0
		paysxanal = 0
		paysxhj = 0
		killvar 'boy'
		killvar 'temp2'
		killvar 'paysextext'
		firstact = 0
		if picrand = 16 or picrand = 17:
			gt 'office', 'work'
		elseif picrand = 18:
			gt 'husbSex', 'dolgend'
		else
			gt $sexloc, 'start'
		end
	end
end
