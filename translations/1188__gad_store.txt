# gad_store


*clr & cla
menu_off = 0
gs 'core_library', 'setloc', 'gad_store', ''
$location_type = 'public_indoors'

frost = 0

$func('wrap', 'header center h1', '乡村商店')
*pl '<center><img <<$set_imgh>> src="images/locations/gadukino/village/market.jpg"></center>'

'你走进这家小商店，四条狭窄的过道两旁是齐胸高的货架，上面摆满了食品和其他日用品。商品种类不多，只有一些最基本的必需品。'
'你慢慢逛着过道，想找点要买的东西。'
*nl
'当你走近收银台时，你注意到一个架子上摆满了杂志、香烟和一些卫生用品。'
*nl
act '逛逛过道': gt 'gad_store', 'cart'

act '离开商店': minut += 5 & gt 'gadukino'
gs 'gp_elene', 'check_for_chores', 'store'


if $ARGS[0] = 'cart':
	$loc_s = 'gad_store'
	$args_s = 'cart'
	*clr & cla

	if hour < 8 or hour > 20:
		'商店目前关门了。'

		act '离开': gt 'gadukino'
		exit
	end

	gs 'themes', 'indoors'
	gs 'item_cart', 'shopping_aisle', 'gad_store'
	gs 'stat'
	$backimage = 'images/locations/gadukino/village/market.jpg'

	act '退出购物车': $backimage = '' & gt 'gad_store'
	act '离开商店': minut += 5 & gt 'gadukino'
end
