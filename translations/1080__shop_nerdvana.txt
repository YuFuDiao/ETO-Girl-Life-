if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_nerdvana', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '书呆天堂')
	'<center><img <<$set_imgh>> src="images/locations/city/island/nerdvana/shop.jpg"></center>'
	*nl
	'如果你愿意在游客中挤来挤去，书呆天堂 有年轻人可能喜欢穿的一切。它只比 G&M 高一个档次，但除非你买得起老城区的精品店，否则你只能在这里买衣服了。'

	if nerdvana_staff_day ! daystart:
		nerdvana_staff = rand(0, 1)
		nerdvana_staff_day = daystart
	end

	act '离开书呆天堂': minut += 3 & gt 'city_island'
	act '去柜台': minut += 3 & gt 'shop_nerdvana', 'counter'
	act '浏览服装': gt 'shop_nerdvana', 'browse'

end

		gs 'npcStat', 'A290'
	else
		gs 'npcStat', 'A291'
	end
	gs 'stat'

	'<center><img <<$set_imgh>> src="images/characters/shared/headshots_main/big<<$mid($npcID, 2)>>.jpg"></center>'
	'你走到柜台前，开始和那位<<iif(nerdvana_staff = 0, ''瘦高'', ''活泼开朗'')>>的店员交谈。<<$Xyrc>>的名牌上写着<<$npcdesc>>，<<$xyr>>的目光短暂停留在你的' + iif(pcs_buttsize > pcs_cupsize, '臀部。', '胸部。')

	act '返回': minut += 1 & gt 'shop_nerdvana', 'start'


	gs 'blackmailer', 'set_nerdvana_talk_act'
end

	act '查看角色扮演服装': minut += 5 & gt 'shop_nerdvana', 'cosplay'
	act '查看其他服装': minut += 5 & gt 'shop_nerdvana', 'outfits'
	act '查看泳衣': minut += 5 & gt 'shop_nerdvana', 'swim'
	act '查看比基尼': minut += 5 & gt 'shop_nerdvana', 'bikinis'
	act '查看钱包': minut += 5 & gt 'shop_nerdvana', 'purses'
end


if $ARGS[0] = 'cosplay':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_nerdvana', 'cosplay'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看书呆天堂角色扮演服装')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'nerdvana_cosplay'
			gs 'shop_utils', 'init', 'set_steps', 2, Clothingstock mod 2
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_nerdvana', 'browse'
	end
end
