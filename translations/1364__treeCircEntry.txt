# treeCircEntry
!! Use this to insert an Entry Point to the Tree Cicle
!!	ARGS[0] = the EntryPoint ID name from $tpLocations[]

if $start_type['magic'] ! 'nomagic':
	!! Setup for teleport circle
	if tpKnown[$ARGS[0]] = 1:
		dynamic "act '走进隐秘的树林圆环': gt 'treeCircle', '<<$ARGS[0]>>'"
	else
		dynamic "act '查看一片神秘的树丛': gt 'treeCircle', '<<$ARGS[0]>>'"
	end
end
