	act '放下你的日记': gt $menu_loc, $menu_arg
	act '返回日志页面顶部': gt 'journal', 'records'
	$jumploc = 'worktab'
	gs 'journal', 'journalmenu'

	!! Ad hoc events and contracts
	if arrsize('evt_event') ! 0: gs 'jobs_gigs', 'disp_evt', 2

	!! jobs_list job cards
	gs 'jobs', 'show_all'

	!! ── Non-registry jobs ──────────────────────────────────────────────
	*nl
	if bumtolik = 4: '你被请去城市工业区的仁爱诊所（Mercy Clinic）帮助无家可归的人。'
	if bumtolik >= 5: '你在城市工业区的仁爱诊所做志愿者，帮助无家可归者。工作时间从 '+func('time', 'get_time_string', 9, 0)+' 到 '+func('time', 'get_time_string', 17, 0)+' 周六和周日。'

	*nl

	!! ── Upcoming events ────────────────────────────────────────────────
	if audition_ev['day'] = daystart:
		'你试演的角色<<$role>>就在今天！'
	elseif audition_ev['day'] - daystart = 1:
		'你明天要试镜角色<<$role>>。'
	elseif audition_ev['day'] - daystart > 1:
		'你将在<<audition_ev[''day''] - daystart>>天后试镜角色<<$role>>。'
	end

	if actress_cast = 1:
		if acting_ev['shoot_day'] = daystart:
			'你是 '
		elseif acting_ev['shoot_day'] - daystart = 1:
			'你被安排在<<acting_ev[''shoot_day''] - daystart>>天后扮演<<$role>>。'
		elseif acting_ev['shoot_day'] - daystart > 1:
			'你被安排在<<acting_ev[''shoot_day''] - daystart>>天后扮演<<$role>>。'
		end
	end

end
