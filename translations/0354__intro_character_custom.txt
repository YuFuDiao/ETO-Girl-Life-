!! This is used to set value, is used to avoid any values that go below 5.

	nstat = 0
	nstat = input '请输入 5 的倍数，范围在 10 到 150 之间。比如 10, 15, 60 等。'
	if nstat > 150:
		msg '请输入小于或等于 150 的值'
		gs 'intro_character_custom', 'setval'
	elseif nstat < 10:
		msg '请输入大于或等于 5 的值'
		gs 'intro_character_custom', 'setval'
	elseif nstat mod 5 ! 0:
		msg '请输入5的倍数的值'
		gs 'intro_character_custom', 'setval'
	else
		gs 'stat'
		exit
	end
end
