# intro_game_start


if $ARGS[0] = 'start':
	gs 'themes', 'indoors'
	*clr & cla

	!! All variable initialization is handled inside intro_initialization/init.
	gs 'intro_initialization', 'init'

	if $start_type['loc'] = 'sg':
		if $start_type['magic'] = 'nomagic':
			$temp_image = '2_sg/start_sg.jpg'
			$stat_explanation = 'sg'
		elseif $start_type['magic'] = 'magic':
			$temp_image = '2_sg/start_csg.jpg'
			$stat_explanation = 'sg_m'
		else
			$temp_image = '2_sg/start_csg.jpg'
			$stat_explanation = 'sg_tg'
		end
	elseif $start_type['loc'] = 'uni':
		$temp_image = '6_uni/uni_start.jpg'
		$stat_explanation = 'uni'
	else
		$temp_image = '5_city/city_start.jpg'
		$stat_explanation = 'city'
	end

	!! --- Welcome image ---
	'<center><img <<$set_imgh>> src="images/system/1_openings/<<$temp_image>>"></center>'
	killvar 'temp_image'

	!! --- Welcome text ---
	if $start_type['loc'] = 'sg':
		if $start_type['magic'] = 'nomagic':
			'欢迎来到女学生版本！你需要继续你的校园生活，处理好你的人际关系，并应对一路上出现的各种状况！'
		elseif $start_type['magic'] = 'magic':
			'欢迎来到被诅咒的女学生版本！引导<<$pcs_nickname>>度过她的校园生活、人际关系和各种可能出现的问题，再加上魔法的调剂，让一切更加有趣！'
		else
			'欢迎来到转变的开始！虽然你已经失去了男子气概和旧生活，但新的可能性正在你眼前展开！引导<<$pcs_nickname>>度过她的校园生活、人际关系和各种可能出现的问题。'
		end
	elseif $start_type['loc'] = 'uni':
		if $start_type['magic'] = 'nomagic':
			'欢迎来到 Girl Life。'
		elseif $start_type['magic'] = 'magic':
			'欢迎来到被诅咒的大学生版本！'
		else
			'欢迎来到转变的开始！虽然你已经失去了男子气概和旧生活，新的可能性正在你眼前展开，尤其是得益于你对世界魔法本质的新认识！'
		end
		'引导<<$pcs_nickname>>应对大学生活中的种种挑战。管理好学业、工作和旅途中遇到的人之间的关系。'
	else
		if $start_type['magic'] = 'nomagic':
			'欢迎来到 Girl Life。你终于摆脱了父母，准备好面对随之而来的一切自由与挑战。你的选择几乎无穷无尽，现在是时候走出去，把它们一一发掘出来。'
		elseif $start_type['magic'] = 'magic':
			'欢迎来到 Girl Life 的魔法少女版本！你终于摆脱了父母，准备好面对随之而来的一切自由与挑战。引导<<$pcs_nickname>>度过她的人生、人际关系和各种可能出现的问题，再加上魔法的调剂，让一切更加有趣！'
		else
			'欢迎来到转变的开始！虽然你已经失去了男子气概和旧生活，新的可能性正在你眼前展开，尤其是得益于你对世界魔法本质的新认识！'
		end
		'别忘了，如果你想上大学，记得在八月报名入学。'
	end

	!! --- Start button ---
	act '开始游戏':
		music_loop = 0
		CLOSE ALL
		if $start_type['loc'] = 'sg':
			if start_location = 0:
				gt 'intro_sg', 'intro_pavlovsk'
			elseif start_location = 1:
				gt 'intro_sg', 'intro_gadukino'
			elseif start_location = 2:
				gt 'intro_sg', 'intro_pushkin'
			else
				gt 'intro_sg', 'intro_sharing'
			end
		elseif $start_type['loc'] = 'uni':
			gt 'uni_grounds'
		else
			gt 'korr'
		end
	end
	!! --- Stat explanation ---
	act '角色属性和状态栏介绍': gt '$menu_settings', 'explanation_start'

	savegame 'autosave_newgame.sav'
end


if $ARGS[0] = 'quick_start':
	gs 'intro_initialization', 'init'
	music_loop = 0
	CLOSE ALL
	savegame 'autosave_newgame.sav'
	act '开始游戏':
		if $start_type['loc'] = 'sg':
			gt 'intro_sg', 'quick_start', start_location
		elseif $start_type['loc'] = 'uni':
			gt 'uni_grounds'
		else
			gt 'korr'
		end
	end
end


--- intro_game_start ---------------------------------
