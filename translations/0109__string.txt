!! Parse delimited string into an array
!!	$ARGS[1] =		ArrayName (has to include $ for string values)
!!	$ARGS[2] =		Delimited string to parse
!!	$ARGS[3] =		Delimiter used (Default = '|' pipe)
!!  Example 1: gs 'string', 'parse_string', '$my_array', 'alpha|beta|gamma|delta|epsilon|zeta'
!!  Example 2: gs 'string', 'parse_string', 'my_array', '1-12-123-1234-12345-123456', '-'
if $ARGS[0] = 'parse_string':
	!! Some failsafes (blank array name, blank string or no delimiters in string)
	if $ARGS[1] = '':
		msg '<b>错误：未找到数组名称！</b>在字符串中，parse_string'
		exit
	end

	!! Set delimiter with default if needed
	if $ARGS[3] = '': $ARGS[3] = '|'

	if $ARGS[2] = '' or instr($ARGS[2], $ARGS[3]) <= 0:
		msg '<b>错误：未找到分隔字符串！</b>在字符串中，parse_string'
		exit
	end


	$ParserTmpDelim = $ARGS[3]
	!! Clear temp array if needed
	killvar 'ParserTmpArr'
	$ParserTmpStr = $ARGS[2]
	!! Loop through string until no more delimiters found
	:StringParser01
	ParserTmpIdx = instr($ParserTmpStr, $ParserTmpDelim)
	if ParserTmpIdx > 0:
		if mid($ARGS[1], 1, 1) = '$':
			$ParserTmpArr[] = $mid($ParserTmpStr, 1, ParserTmpIdx-1)
			$ParserTmpStr = $mid($ParserTmpStr, ParserTmpIdx + len($ParserTmpDelim))
		else
			ParserTmpArr[] = val($mid($ParserTmpStr, 1, ParserTmpIdx-1))
			$ParserTmpStr = $mid($ParserTmpStr, ParserTmpIdx + len($ParserTmpDelim))
		end
		jump 'StringParser01'
	!! Get last part since no more delimiters were found and copy to the final array
	elseif mid($ARGS[1], 1, 1) = '$':
		$ParserTmpArr[] = $ParserTmpStr
		copyarr $ARGS[1], '$ParserTmpArr'
	else
		ParserTmpArr[] = val($ParserTmpStr)
		copyarr $ARGS[1], 'ParserTmpArr'
	end
	killvar 'ParserTmpDelim'
	killvar 'ParserTmpStr'
	killvar 'ParserTmpIdx'
	killvar 'ParserTmpArr'
	exit
end



!! func('string', 'enumerate_list', '$arr', {start}, {number}, ', ', ' and ', ', and ')
!! returns a string with the array entries listed:
!! $arr[0] = 'Sidisi'	-> returns: 'Sidisi'
!! $arr[1] = 'Eriette'	-> returns: 'Sidisi and Eriette'
!! $arr[2] = 'Dynaheir'	-> returns: 'Sidisi, Eriette, and Dynaheir'
!!
!!  ARGS[2] indicates where the enumeration start (Defaults to the first item)
!!  ARGS[3] indicates how many items are enumerated (Defaults to the end of the list)
!! $ARGS[4] is the small separator (defaults to ', ')
!! $ARGS[5] is the final separator for 2 items (defaults to ' and ')
!! $ARGS[6] is the final separator for more items (defaults to ', and ')
!!
!! If you want to list the entries of an array, you can use
!! $func('string', 'enumerate_list', '$arr', 0, 0, '<br>', '<br>', '<br>')


if $ARGS[0] = 'enumerate_list':
	$temp_stringVars['text'] = ''
	$temp_stringVars['name'] = $ARGS[1]

	if ARGS[2] <= 0:
		temp_stringVars['i'] = 0
	else
		temp_stringVars['i'] = ARGS[2]
	end
	if ARGS[3] <= 0:
		temp_stringVars['i_max'] = arrsize($temp_stringVars['name'])
	else
		temp_stringVars['i_max'] = temp_stringVars['i'] + ARGS[3]
		if temp_stringVars['i_max'] > arrsize($temp_stringVars['name']): temp_stringVars['i_max'] = arrsize($temp_stringVars['name'])
	end

	if $ARGS[4] = '':
		$temp_stringVars['small_sep'] = ', '
	else
		$temp_stringVars['small_sep'] = $ARGS[4]
	end
	if $ARGS[5] = '':
		$temp_stringVars['duo_sep'] = ' and '
	else
		$temp_stringVars['duo_sep'] = $ARGS[5]
	end
	if $ARGS[6] = '':
		$temp_stringVars['final_sep'] = ', 而且 '
	else
		$temp_stringVars['final_sep'] = $ARGS[6]
	end

	temp_stringVars['count'] = temp_stringVars['i_max'] - temp_stringVars['i']

	if temp_stringVars['count'] <= 0:
		!! empty on purpose
	elseif temp_stringVars['count'] = 1:
		dynamic "$temp_stringVars['text'] = <<$temp_stringVars['name']>>[<<temp_stringVars['i']>>]"
	elseif temp_stringVars['count'] = 2:
		dynamic "$temp_stringVars['text'] = <<$temp_stringVars['name']>>[<<temp_stringVars['i']>>] + '<<$temp_stringVars['duo_sep']>>' + <<$temp_stringVars['name']>>[<<temp_stringVars['i'] + 1>>]"
	else
		:sgs_enum_list
			dynamic "$temp_stringVars['text'] += <<$temp_stringVars['name']>>[<<temp_stringVars['i']>>]"
			temp_stringVars['i'] += 1
		if temp_stringVars['i'] < temp_stringVars['i_max'] - 1: $temp_stringVars['text'] += $temp_stringVars['small_sep'] & jump 'sgs_enum_list'

		dynamic "$temp_stringVars['text'] += '<<$temp_stringVars['final_sep']>>' + <<$temp_stringVars['name']>>[<<temp_stringVars['i']>>]"
	end

	$result = $temp_stringVars['text']

	killvar 'temp_stringVars'
	exit
end


!! Function to convert number to text
!! func('string', 'parse_number', 6276345) -> 'six million two hundred and seventy-six thousand three hundred and forty-five'
!! func('string', 'parse_number', -65) -> 'negative sixty-five'
!! func('string', 'parse_number', 123, 'rank') -> one hundred and twenty-third
if $ARGS[0] = 'parse_number':
	if $ARGS[1] = '':
		temp_stringVars['num'] = ARGS[1]
	else
		if isnum($ARGS[1]):
			temp_stringVars['num'] = val($ARGS[1])
		else
			msg '错误：string, parse_number 收到无效输入'
			exit
		end
	end

	temp_stringVars['rank'] = iif(instr('|ordinal|rank|', '|' + $ARGS[1] + '|' + $ARGS[2] + '|') > 0, 1, 0)
	temp_stringVars['n'] = temp_stringVars['num']
	$result = ''
	if temp_stringVars['n'] < 0:
		$result = '负'
		temp_stringVars['n'] = -temp_stringVars['n']
	end
	if temp_stringVars['n'] = 0:
		$result += iif(temp_stringVars['rank'] = 0, '零', '第零')
		killvar 'temp_stringVars'
		exit
	end

	$dg[0] = '零'	& $dg[1] = '一'	& $dg[2] = '二'
	$dg[3] = '三'	& $dg[4] = '四'	& $dg[5] = '五'
	$dg[6] = '六'	& $dg[7] = '七'	& $dg[8] = '八'
	$dg[9] = '九'

	$result += iif(temp_stringVars['rank'] = 1, '第', '')

	if temp_stringVars['n'] >= 100000000:
		temp_stringVars['w'] = temp_stringVars['n'] / 100000000
			if temp_stringVars['w'] >= 1000:
				$result += $dg[temp_stringVars['w'] / 1000] + '千'
				temp_stringVars['w'] = temp_stringVars['w'] mod 1000
				if temp_stringVars['w'] > 0 and temp_stringVars['w'] < 100: $result += '零'
			end
			if temp_stringVars['w'] >= 100:
				$result += $dg[temp_stringVars['w'] / 100] + '百'
				temp_stringVars['w'] = temp_stringVars['w'] mod 100
				if temp_stringVars['w'] > 0 and temp_stringVars['w'] < 10: $result += '零'
			end
			if temp_stringVars['w'] >= 10:
				if temp_stringVars['w'] / 10 > 1: $result += $dg[temp_stringVars['w'] / 10]
				$result += '十'
				temp_stringVars['w'] = temp_stringVars['w'] mod 10
			end
			if temp_stringVars['w'] > 0: $result += $dg[temp_stringVars['w']]
		$result += '亿'
		temp_stringVars['n'] = temp_stringVars['n'] mod 100000000
		if temp_stringVars['n'] > 0 and temp_stringVars['n'] < 1000: $result += '零'
	end
	if temp_stringVars['n'] >= 10000:
		temp_stringVars['w'] = temp_stringVars['n'] / 10000
			if temp_stringVars['w'] >= 1000:
				$result += $dg[temp_stringVars['w'] / 1000] + '千'
				temp_stringVars['w'] = temp_stringVars['w'] mod 1000
				if temp_stringVars['w'] > 0 and temp_stringVars['w'] < 100: $result += '零'
			end
			if temp_stringVars['w'] >= 100:
				$result += $dg[temp_stringVars['w'] / 100] + '百'
				temp_stringVars['w'] = temp_stringVars['w'] mod 100
				if temp_stringVars['w'] > 0 and temp_stringVars['w'] < 10: $result += '零'
			end
			if temp_stringVars['w'] >= 10:
				if temp_stringVars['w'] / 10 > 1: $result += $dg[temp_stringVars['w'] / 10]
				$result += '十'
				temp_stringVars['w'] = temp_stringVars['w'] mod 10
			end
			if temp_stringVars['w'] > 0: $result += $dg[temp_stringVars['w']]
		$result += '万'
		temp_stringVars['n'] = temp_stringVars['n'] mod 10000
		if temp_stringVars['n'] > 0 and temp_stringVars['n'] < 1000: $result += '零'
	end
	temp_stringVars['w'] = temp_stringVars['n']
		if temp_stringVars['w'] >= 1000:
			$result += $dg[temp_stringVars['w'] / 1000] + '千'
			temp_stringVars['w'] = temp_stringVars['w'] mod 1000
			if temp_stringVars['w'] > 0 and temp_stringVars['w'] < 100: $result += '零'
		end
		if temp_stringVars['w'] >= 100:
			$result += $dg[temp_stringVars['w'] / 100] + '百'
			temp_stringVars['w'] = temp_stringVars['w'] mod 100
			if temp_stringVars['w'] > 0 and temp_stringVars['w'] < 10: $result += '零'
		end
		if temp_stringVars['w'] >= 10:
			if temp_stringVars['w'] / 10 > 1: $result += $dg[temp_stringVars['w'] / 10]
			$result += '十'
			temp_stringVars['w'] = temp_stringVars['w'] mod 10
		end
		if temp_stringVars['w'] > 0: $result += $dg[temp_stringVars['w']]

	killvar 'temp_stringVars'
	exit
end

	temp_stringVars['outer_jump'] = len($temp_stringVars['pair_sep'])
	temp_stringVars['inner_jump'] = len($temp_stringVars['kv_sep'])

	killvar $temp_stringVars['keys_array']
	killvar $temp_stringVars['values_array']

	:split_key_value_outer_loop
		temp_stringVars['outer_index'] = instr($temp_stringVars['key-value-string'], $temp_stringVars['pair_sep'])
		if temp_stringVars['outer_index'] = 0: temp_stringVars['outer_index'] = len($temp_stringVars['key-value-string']) + 1
		$temp_stringVars['sub_string'] = $mid($temp_stringVars['key-value-string'], 1, temp_stringVars['outer_index'] - 1)

		temp_stringVars['inner_index'] = instr($temp_stringVars['sub_string'], $temp_stringVars['kv_sep'])
		if temp_stringVars['inner_index'] = 0:
			msg "String.split_key_value 错误：传入的键值字符串不正确。<br>'<<$temp_stringVars['sub_string']>>' 不包含键值分隔符 '<<$temp_stringVars['kv_sep']>>'。<br>原始传入字符串：'<<$ARGS[1]>>'"
			exit
		end

		$temp_stringVars['key'] = $trim($mid($temp_stringVars['sub_string'], 1, temp_stringVars['inner_index'] - 1))
		$temp_stringVars['value'] = $trim($mid($temp_stringVars['sub_string'], temp_stringVars['inner_index'] + temp_stringVars['inner_jump']))

		dynamic "
			$<<$temp_stringVars['keys_array']>>[] = $temp_stringVars['key']
		"
		if temp_stringVars['type_flag'] = 1:
			!! Force to string
			dynamic "
				$<<$temp_stringVars['values_array']>>[$temp_stringVars['key']] = $temp_stringVars['value']
			"
		elseif isnum($temp_stringVars['value']):
			dynamic "
				<<$temp_stringVars['values_array']>>[$temp_stringVars['key']] = val($temp_stringVars['value'])
			"
		else
			dynamic "
				$<<$temp_stringVars['values_array']>>[$temp_stringVars['key']] = $temp_stringVars['value']
			"
		end

		$temp_stringVars['key-value-string'] = $mid($temp_stringVars['key-value-string'], temp_stringVars['outer_index'] + temp_stringVars['outer_jump'])
	if len($temp_stringVars['key-value-string']) > 0: jump 'split_key_value_outer_loop'

	killvar 'temp_stringVars'
	exit
end
