# Nudelake
!! 2021/03/27

if $ARGS[0] = '':
	menu_off = 0
	gs 'core_library', 'setloc', 'Nudelake', ''
	$location_type = 'secluded'
	$locclass = 'beach'
	gs 'stat'

	'<center><b>裸体海滩</b></center>'
	'<center><img <<$set_imgh>> src="images/locations/city/residential/lake/nudelake.jpg"></center>'
	'裸体海滩，衣服是可选的。'

	if (PSwim = 1 or $clothingworntype = 'nude') and $lastwornclothingtype['swim'] <> '' and $lastwornclothingtype['swim'] <> 'nude':
		act '穿好衣服':
			*clr & cla
			minut += 5
			gs 'outfit', 'restore', 'swim'
			'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/secluded_beach/dress/dress_nudebeach.jpg"></center>'
			'你重新穿上衣服。'
			act '继续': gt 'Nudelake'
		end
	end


	if hour < 6 or hour > 20:
		'裸体海滩已经关闭。'
	elseif temper < 15 or month < 5 or month > 9:
		'天气太冷了，不适合任何裸体活动。'
	else
		if func('changingroom', 'count_swim_item') > 0 and PSwim = 0:
			act '换上你的泳衣':
				menu_off = 1
				*clr & cla
				minut += 5
				if pcs_horny >= 90:
					'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/secluded_beach/self/wetpussypants1.jpg"></center>'
					'你想换上泳衣，但这里没有更衣室，你的阴道汁液顺着腿流下来，再加上想到会被人看到，让你感到害怕。你放弃了这个念头。'
					'"我应该先处理好自己的欲望。"你对自己说。'

					act '继续': gt 'Nudelake'
				else
					'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/secluded_beach/dress/swimsuit_nudebeach.jpg"></center>'
					'你用毛巾把自己裹起来，准备换上泳衣。'

					act '选择衣服': gt 'changingroom'
				end
			end
		end

		if $clothingworntype <> 'nude':
			act '脱掉衣服':
				menu_off = 1
				if pcs_inhib > 30:
					*clr & cla
					minut += 5
					if pcs_horny >= 80 and trait_vars['exhibitionist'] > 1 and (PCloPanties = 1 or $pantyworntype <> 'none'):
						'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/secluded_beach/self/wetpussypants3.jpg"></center>'
						'你脱光了衣服。几个路过的男人先是一愣，然后停下来盯着你湿透的内裤看，大概在想是什么让你这么兴奋。你不在乎他们会说什么或怎么想你。'
					elseif pcs_inhib >= 60:
						'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/secluded_beach/dress/undressnudebeach.jpg"></center>'
						'不管有没有人在看你，你慢慢脱下你的衣服。你像脱衣舞娘一样一件件脱掉。很快，你完全赤裸。'
					elseif pcs_inhib < 60:
						'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/secluded_beach/dress/undressnudebeach.jpg"></center>'
						'你迅速环顾四周，确认没有人特别关注你之后，脱下了衣服。现在你一丝不挂。'
					end
					if PSwim = 0 and $clothingworntype <> 'nude':
						gs 'outfit', 'backup', 'swim'
					end
					gs 'outfit', 'strip_all'
					gs 'stat'
				else
					*clr & cla
					'<center><img <<$set_imgh>> src="images/locations/pavlovsk/lake/secluded_beach/self/shyness.jpg"></center>'
					'即使周围没有人，你也没有足够的信心真的把衣服脱下来。'
				end
				act '继续': gt 'Nudelake'
			end
		end

		if $clothingworntype = 'nude':
			'在这里你可以<a href="exec:gt ''Nudelake'', ''zagarat''">晒日光浴</a>和<a href="exec:gt ''Nudelake'', ''swim''">游泳</a>'
			if pcs_stam >= (20 * (10 - sport_clothes_exercise_bonus)) / 2:
				act '去游泳': gt 'Nudelake', 'swim'
			end
			act '晒日光浴': gt 'Nudelake', 'zagarat'
			'你可以玩<a href="exec:gt ''Nudelake'', ''voleybol''">沙滩排球</a>。'
		end

		if katday <> daystart and kat >= 1:
			'你在海滩上看到了凯特。'
			act '走向凯特': gt 'Katlake', 'kat'
		end
	end

	if $clothingworntype = 'nude':
		act '穿好衣服':
			gs 'outfit', 'restore', 'swim'
			gt 'Nudelake'
		end
	else
		act '回到主海滩': gt 'city_lake', 'start'
	end
end


if $ARGS[0] = 'voleybol':
	*clr & cla
	menu_off = 0
	$location_type = 'event_outdoors'
	$menu_loc = 'Nudelake'
	$menu_arg = 'voleybol'
	minut += 1
	'<center><img <<$set_imgh>> src="images/locations/city/residential/lake/volleyball.jpg"></center>'
	'你来到排球场。'

	act '观看': gt 'Nudelake', 'smotr'
	act '离开': gt 'Nudelake'

	act '玩':
		menu_off = 1
		*clr & cla
		gs 'exp_gain', 'agil', 1
		gs 'exp_gain', 'react', 1
		gs 'exp_gain', 'vball', rand(1, 3)
		minut += 30
		inhib_exp += rand(1, 3)
		'<center><img <<$set_imgh>> src="images/locations/city/residential/lake/volleyball.jpg"></center>'
		'你走到球员们面前，问能不能加入。他们很高兴让你加入一个队伍。'

		act '完成': gt 'Nudelake'
	end
end


if $ARGS[0] = 'zagarat':
	*clr & cla
	$location_type = 'event_outdoors'
	minut += 30
	if pcs_sweat < 35 : gs 'sweat', 'add', 5
	inhib_exp += rand(1, 3)
	gs 'mood', 'raise', 'tiny'
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/shared/lake/tanning_nude.jpg"></center>'

	if mc_inventory['suncream'] = 0:
		pcs_tan += 1
		'你来到海滩，晒日光浴。'
	else
		mc_inventory['suncream'] -= 1
		pcs_tan += 3
		'你在身上涂好防晒霜，然后晒日光浴。'
	end

	nzagrand = rand(0, 100)

	if nzagrand < 60:
		!! Intentionally empty
	elseif nzagrand < 70:
		if npc_rel['A219'] > 30 and katkey = 0: gt 'Katlake', 'key'
	elseif nzagrand < 80:
		if kat = 0 and katday <> daystart: gt 'Katlake'
	elseif nzagrand < 90:
		gt 'Vnlake1'
	else
		gt 'Vnlake2'
	end

	act '完成': gt 'Nudelake'
end


if $ARGS[0] = 'swim':
	*clr & cla
	menu_off = 0
	$location_type = 'event_outdoors'
	pcs_makeup = 1
	cumspclnt = 4
	gs 'cum_cleanup'
	gs 'exercise', 'tier1', 60, 'stren', 'vital'

	if $clothingworntype = 'nude':
		gs 'exercise', 'tier1', -5, 'stren', 'vital'
		inhib_exp += rand(1, 3)
	else
		if pcs_inhib < 50: inhib_exp += rand(1, 3)
	end
	pcs_sweat = 10 + rand(0, 4)

	'<center><img <<$set_imgh>> src="images/pc/activities/swim/lake/swim_nude_<<rand(0,10)>>.jpg"></center>'
	'你去游了个泳。'
	if deodorant_on = 1: gs 'sweat', 'remove_deo' & '<br>你身上的除臭剂被水冲掉了。'

	act '完成': gt 'Nudelake'
end


if $ARGS[0] = 'smotr':
	*clr & cla
	$location_type = 'event_outdoors'
	menu_off = 0
	minut += 30
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/residential/lake/volleyball.jpg"></center>'
	'你看着其他人玩。'

	act '离开': gt 'Nudelake'
end
