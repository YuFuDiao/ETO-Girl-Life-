	temp_abd_thresh = (func('math', 'int_clamp', slut_compare / 3, 0, 250)) + (func('archetypes', 'get_percentage', 'bimbo', 1) / 2) + (trait_vars['doormat'] * 30) + (alko * 8) - (min(pcs_stren - 50, 0) * 2) - (min(pcs_run - 50, 0) * 2) - (min(prcptn_lvl - 50, 0) * 2)
	if cheatVars['abduction_chance'] = 1:
		!! Disabled
		temp_abd_thresh = 0
	elseif cheatVars['abduction_chance'] = 2:
		!! Low chance
		temp_abd_thresh /= 5
	elseif cheatVars['abduction_chance'] = 3:
		!! High chance
		temp_abd_thresh = (temp_abd_thresh * 3) / 2
	elseif cheatVars['abduction_chance'] = 4:
		!! Very high chance
		temp_abd_thresh *= 2
	end
	temp_abd_thresh = func('math', 'int_clamp', temp_abd_thresh, 0, 500)
	if rand(1, 1000) <= temp_abd_thresh:
		'然而，这个施暴者还不打算放过你。'
		'他又抓住了你，把你拖到他的车旁，塞进后备箱……'
		'你有一种预感，这对你来说不会有什么好结果……'
		act '大声呼救': gt 'abduction', 'start'
	else
		'施暴者站起来，只是看着你痛苦地爬走。'
		'他转过身，平静地走开了，留下你一个人。'
		act '逃跑': gt 'fuelstation', 'start'
	end
end
