# city_house_res_misc


!! Small frontyard
if $ARGS[0] = 'fronty':
	menu_off = 0
	gs 'core_library', 'setloc', 'city_house_res_misc', 'fronty'
	$location_type = 'public_outdoors'

	minut += 1
	gs 'stat'

	if sound_settings['environment_off'] = 0:
		if month >=11 and month <= 12 or month >=1 and month <=3:
			if hour >= 8 and hour <= 23:
				PLAY 'sound/street2.mp3', 30
				PLAY 'sound/street7.mp3', 30
				PLAY 'sound/zima.mp3', 30
			end
			if hour >= 0 and hour <= 7: PLAY 'sound/street_night.mp3', 30 & PLAY 'sound/zima.mp3', 30
		else
			if hour >= 8 and hour <= 23:
				PLAY 'sound/street2.mp3', 30
				PLAY 'sound/street7.mp3', 30
				PLAY 'sound/street_leto.mp3', 30
			end
			if hour >= 0 and hour <= 7: PLAY 'sound/street_night_leto.mp3' & PLAY 'sound/street_night.mp3', 30
		end
	end

	*clr & cla
	$func('wrap', 'header center b', '你的城市住宅')
	'<center><img <<$set_imgh>> src="images/locations/city/residential/house/crh_fronty.jpg"></center>'
	'你的两层小楼坐落在住宅区的边缘。尽管如此，邻居们离得很近，街道上的声音充斥着空气。'
	*nl
	if func('car_funcs', 'is_here'):
		'<a href="exec:gs ''carF'', ''start''">你的<<$car[''name'']>></a>停在车道上。'
	end

	act '走进去': gt 'city_house_res_misc', 'hallw'
	act '去市区':
		minut += 10
		gt 'city_residential'
	end
	if func('homes_properties', 'is_current_home', $loc) = 0:
		act '把这栋房子设为主居所': gs 'homes_properties', 'set_home' & gt 'city_house_res_misc', 'fronty'
	end

end


if $ARGS[0] = 'hallw':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'city_house_res_misc', 'hallw'
	$location_type = 'private'

	minut += 1
	gs 'stat'
	gs 'themes', 'indoors'
	if frost > 0: frost = 0

	*clr & cla
	$func('wrap', 'header center b', '门厅')
	'<center><img <<$set_imgh>> src="images/locations/city/residential/house/crh_hallway.jpg"></center>'

	gs 'courtletter'

	act '去前院':
		if $clothingworntype ! 'nude':
			minut += 1
			gt 'city_house_res_misc', 'fronty'
		else
			msg '<b><font color = red>你需要穿好衣服。</font></b>'
			gt 'city_house_res_misc', 'hallw'
		end
	end

	'这是你家的主门厅。你可以从这里前往各个房间。'
	'墙上挂着一面<a href="exec:gt ''mirror'', ''start''">镜子</a>.'

	if mc_inventory['umbrella'] > 0: '门边的挂钩上挂着一把雨伞。'

	act '去你的卧室': gt 'city_house_res_bedr', 'bedro'
	act '去你的浴室': gt 'city_house_res_bathr', 'bathr'
	act '去你的厨房': gt 'city_house_res_kitch', 'kitch'
	act '去你的客厅': gt 'city_house_res_misc', 'livroom'
	act '去你的后院': gt 'city_house_res_misc', 'backy'

	gs 'daily_routine', 'offer_here'
end


!! Small backyard for outdoor working out or maybe gardening?
if $ARGS[0] = 'backy':
	menu_off = 0
	gs 'core_library', 'setloc', 'city_house_res_misc', 'backy'
	$location_type = 'secluded'
	minut += 1
	gs 'stat'

	*clr & cla
	$func('wrap', 'header center b', '客厅')
	'<center><img <<$set_imgh>> src="images/locations/city/residential/house/crh_backy.jpg"></center>'
	'你的小后院用栅栏围了起来，里面有几块用于种植的菜地。'

	act '走进去': gt 'city_house_res_misc', 'hallw'

	if sunWeather = 1:
		if pcs_stam >=15 or (pcs_stam >= 10 and mc_inventory['book_yoga'] + mc_inventory['hula_hoop'] > 0):
			*nl
			if PSport = 0:
				'如果你穿着运动服，你可以在这里做些锻炼。'
			else
				'你可以在这里做些锻炼。'
				gs 'wardrobe', 'default_clothing_options'
				act '锻炼': gt 'exercise', 'workout'
			end
		end
	end

end


if $ARGS[0] = 'livroom':
	menu_off = 0
	gs 'core_library', 'setloc', 'city_house_res_misc', 'livroom'
	$location_type = 'private'
	$locclass = 'livingr'
	minut += 1
	gs 'stat'
	gs 'music_actions', 'clear_restrictions'

	*clr & cla
	$func('wrap', 'header center b', '客厅')
	'<center><img <<$set_imgh>> src="images/locations/city/residential/house/crh_living.jpg"></center>'

	act '离开房间': gt 'city_house_res_misc', 'hallw'

	'你可以在这里观看<a href="exec:gt ''TV'',''start''">电视</a>, 或者在你的<a href="exec:gt ''divan'', ''start''">沙发</a>上放松。'
	if $ml_guitar['location'] = $loc: '你的吉他就靠在沙发旁边的架子上。'
	!! Computer and internet use.
	if mc_inventory['tech_computer'] = 1:
		gs 'internet_mobile', 'get_access'
		!! TODO: Need to build a check for husband being at home or something.
		'你的<a href="exec:gt ''stol'', ''start''">书桌</a>上放着一台<a href="exec:gt ''komp'', ''start''">电脑</a>.'
	else
		'靠墙放着一张小小的<a href="exec:gt ''stol'', ''start''">书桌</a>.'
	end

	'这里有足够的空间可以锻炼。'
	gs 'exercise', 'start'

	gs 'subkid'

	!! IMPORTANT: if there is anything added later that could mean that others are in the room, gs 'music_actions', 'not_alone' should be added with the correct conditions.
	gs 'music_actions', 'start'

	if mc_inventory['sewing_kit'] = 1:
		'你的<a href="exec:gt ''sewing'',''start''">针线盒</a>放在你最喜欢的椅子旁边。'
		*nl
		if pcs_sewng >= 80:
			if newgobelen = 0 and mc_inventory['sewing_fabric'] > 0:
				act '开始一幅新的挂毯': gt 'sewing', 'tapestry'
			elseif newgobelen >= 1:
				'你的挂毯已完成<<newgobelen/10>>%.'
				act '织挂毯': gt 'sewing', 'tapestry'
			end
		end
		if mc_inventory['tapestry'] > 0: '你有<<mc_inventory[''tapestry'']>>条已完成的挂毯。'
	end
end
