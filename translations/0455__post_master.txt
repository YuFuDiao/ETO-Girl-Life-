# post_master

!! variables used:
!! job_hiring_step['pav_mailgirl']: (0 = never asked, 1 = trial offered but declined, 2 = trial accepted, 3 = job offered after trial but deferred)
!! $job_status['pav_mailgirl']: '' (not hired), 'employed' (active or suspended), 'fired' (permanent)
!! job_suspended['pav_mailgirl']: 1 if skipping this Saturday (still employed, no calendar)
!! job_missed_total['pav_mailgirl']: total missed shifts (never reset)
!! job_miss_acknowledged['pav_mailgirl']: last missed_total value player was scolded about
!! job_booking_debt['pav_mailgirl']: penalty strikes (1 per miss, fired at 4)
!! $job_termination_reason['pav_mailgirl']: 'no_show' if auto-fired for missing too many shifts
!! post_vars['late']	= (0-15), how often you come back to the post office after 15:15. You get fired if this counter hits 15
!! post_vars['burn']	= (0-1), whether hoodlums tampered with your mail. Gets you fired if yes

if $ARGS[0] = 'start':
	menu_off = 1
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'
	'邮局经理，也就是邮政局长，刚刚帮完一位顾客，正要走回他的办公室时，他看到了你。'
	'“你看起来想问点什么。如果问的是我们的常规服务，那你应该去排队。我没时间再接待更多顾客了，”他说。'
	act '离开': gt 'post_office', 'start'

	gs 'jobs', 'get_job_definition', 'pav_mailgirl'

	if $job_status['pav_mailgirl'] = '':
		if job_hiring_step['pav_mailgirl'] = 0:
			act '找工作': gt 'post_master', 'ask_for_work'
		elseif job_hiring_step['pav_mailgirl'] = 1:
			!! if you refused the job initially
			gt 'post_master', 'visit_after_refusing'
		elseif job_hiring_step['pav_mailgirl'] = 2:
			if job_last_work_day['pav_mailgirl'] <> daystart:
				if func('jobs', 'is_arrival_time', 'pav_mailgirl') = 0:
					!! if you''re at the post office when it''s not working time
					gt 'post_master', 'visit_outside_work_hours'
				else
					!! when you go to work on Saturday, first time
					gt 'post_master', 'start_work'
				end
			end
		elseif job_hiring_step['pav_mailgirl'] = 3:
			!! if you did the trial and wasn''t sure about taking the job
			gt 'post_master', 'visit_after_refusing2'
		end
	elseif $job_status['pav_mailgirl'] = 'employed':
		!! Check for unacknowledged missed shifts (scolding)
		if job_missed_total['pav_mailgirl'] > job_miss_acknowledged['pav_mailgirl']:
			gt 'post_master', 'scolding'
		end

		if job_suspended['pav_mailgirl'] = 1:
			!! Player suspended (skipping Saturday) — offer to resume
			act '告诉他你其实还能来这个周六上班':
				*clr & cla
				gs 'jobs', 'resume_job', 'pav_mailgirl'
				minut += 2
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'
				'你在一个合适的时机走进邮政局长的办公室。他看起来现在没什么事做。'
				'“<<$pcs_firstname>>，有什么需要帮忙的吗？”他一看到你就问道。'
				'“我来是想告诉你，我这个周六其实还是能来的！”你回答。'
				'“啊，太好了！我会告诉阿列克谢他可以留在家里陪女儿了，”他一边快速记下一边说。“还有什么我能帮你的吗？”'
				'“没有了，就这些，”你告诉他。“周六见！”'

				act '别管他': gt 'post_office', 'start'
			end
		elseif func('jobs', 'is_work_day', 'pav_mailgirl') = 0 or (func('jobs', 'is_work_day', 'pav_mailgirl') = 1 and job_last_work_day['pav_mailgirl'] = daystart):
			!! Not a work day or after work — offer to skip next Saturday
			act '告诉他你下周六没法来上班':
				*clr & cla
				gs 'jobs', 'suspend_job', 'pav_mailgirl'
				minut += 2
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'
				'邮政局长看起来正忙得不可开交。他的办公桌上堆满了邮件。'
				'“打扰一下？”你试着引起他的注意。“我下周六可能来不了了。”'
				'邮政局长有些失望地抬头看着你。“哦，好吧。没问题。我会告诉阿列克谢他周六得来上班。至少你还知道告诉我一声，谢了。”'
				'他在一张纸上潦草地写了点什么，又埋头工作了。'

				act '别管他': gt 'post_office', 'start'
			end
		else
			if func('jobs', 'is_arrival_time', 'pav_mailgirl') = 1:
				*clr & cla
				'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'
				'“嘿，<<$pcs_firstname>>！”你走进邮局时，邮政局长笑着说。“来得正好。请跟我来。”'
				act '跟着他': gt 'post_deliveries', 'start'
			else
				if hour < 11:
					!! if it''s Saturday but you''re early
					*clr & cla
					minut += 2
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'
					'“你来得有点早，<<$pcs_firstname>>！”看到你走进他的办公室，邮政局长说。“你的邮件还没准备好，不过我喜欢你的热情。在 '+func('time', 'get_time_string', 11, 0)+' 到 '+func('time', 'get_time_string', 12, 0)+' 之间再来找我吧。到时候应该就准备好了。”'

					act '别管他': gt 'post_office', 'start'
				elseif hour > 11 and job_last_work_day['pav_mailgirl'] <> daystart:
					!! if it''s Saturday but you''re late
					*clr & cla
					minut += 2
					job_last_work_day['pav_mailgirl'] = daystart
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'
					'“你迟到了，<<$pcs_firstname>>！”看到你走进办公室，邮政局长嗤笑道。“我已经叫阿列克谢来顶你的班了。他不高兴，我也不高兴。别养成这种习惯……”'
					'你试图道歉，但他不感兴趣，让你别烦他。'

					act '离开他的办公室': gt 'post_office', 'start'
				end
			end
		end

		act $func('wrap', 'v_neg', '辞去工作'):
			*clr & cla
			minut += 5
			gs 'jobs', 'set_terminated', 'pav_mailgirl'
			gs 'stat'
			'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'
			'“<<$pcs_nickname>>，”邮政局长从办公桌上抬起头说。“有什么可以帮你的吗？”'
			'你叹了口气，告诉他你想辞掉这份工作：“嗯，我来就是想辞职的。谢谢你给我这个机会，但我的情况变了，我没办法继续做我的班次了。”'
			'“你确定吗？阿列克谢真的很享受和家人在一起的时光。”'
			'你点头确认：“是的，我确定。”'
			'“那好吧；祝你未来的计划顺利，<<$pcs_nickname>>”'
			'你向他道谢后离开了。'
			act '离开': gt 'post_office', 'start'
		end

	elseif $job_status['pav_mailgirl'] = 'fired':
		*clr & cla
		if $job_termination_reason['pav_mailgirl'] = 'no_show':
			!! Fired for missing too many shifts
			'“<<$pcs_firstname>>……”看到你，邮政局长叹了口气，摇了摇头。“我真的很想让这事顺利下去，但你让我别无选择。你一直不来上班，阿列克谢也没法总是临时替你顶班。很抱歉，我不得不让你走。”'
			'他看起来真的很失望。“我真希望你至少提前告诉我你来不了。我会理解的。但就这么不来？这可不是这么玩的。”'
		else
			!! Fired for burning mail or other serious offence
			'“你？！滚出去！”你走进他的办公室时，他愤怒地喊道。“这里没有你的工作！现在没有，永远都没有！你那次搞砸的事毁了我们完美的记录！”'
			'你试图解释发生了什么，但他对你所说的话完全不感兴趣。看来他再考虑雇用你的可能性微乎其微。'
		end

		act '别管他': gt 'post_office', 'start'
	end
end


if $ARGS[0] = 'scolding':
	!! Postmaster scolds player for unacknowledged missed shifts
	!! Escalates based on total misses (booking_debt tracks penalty strikes)
	*clr & cla
	menu_off = 1
	minut += 5
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postmaster.jpg"></center>'

	if job_booking_debt['pav_mailgirl'] = 1:
		'“<<$pcs_firstname>>，”邮政局长皱着眉从办公桌上抬起头说。“你周六没来上班。我最后一刻才把阿列克谢叫来顶班，他对此<i>非常</i>不高兴。”'
		'他叹了口气。“听着，我理解总会有些突发状况。但如果你来不了，<b>请提前告诉我</b>，这样我好做安排。别就这么把我晾着，好吗？”'
	elseif job_booking_debt['pav_mailgirl'] = 2:
		'“<<$pcs_firstname>>！”看到你，邮政局长厉声说道。“这是你第二次不打声招呼就不来了！你知道临时找人来替你有多难吗？”'
		'他沮丧地摇了摇头。“我给你另一次机会，是因为你真正来上班时工作做得不错。但我警告你——别养成这种习惯。”'
	elseif job_booking_debt['pav_mailgirl'] >= 3:
		'“<<$pcs_firstname>>。”邮政局长的声音冰冷。“你没来。又没来。连句话都没有。”'
		'他身体前倾，表情极其严肃。“这是你最后一次警告。再放一次鸽子，我就把你的工作给真正想要它的人。阿列克谢一直在要求拿回他的周六班次，而我现在很难想出不让他拿回的理由。”'
	end

	!! Mark misses as acknowledged
	job_miss_acknowledged['pav_mailgirl'] = job_missed_total['pav_mailgirl']
	act '道歉并离开': gt 'post_office', 'start'
end
