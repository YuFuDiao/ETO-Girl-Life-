if $ARGS[0] = 'get_total':
	if $mid($ARGS[1], 1, 4) = 'cats':
		result = 72
	elseif $mid($ARGS[1], 1, 10) = 'danilovich':
		result = 8
	elseif $mid($ARGS[1], 1, 5) = 'eroto':
		result = 43
	elseif $mid($ARGS[1], 1, 11) = 'fashionista':
		result = 79
	elseif $mid($ARGS[1], 1, 2) = 'gm':
		result = 37
	elseif $mid($ARGS[1], 1, 5) = 'lusso':
		result = 82
	elseif $mid($ARGS[1], 1, 9) = 'salacious':
		result = 59
	end
	exit
end


if $ARGS[0] = 'totals':

if $ARGS[0] = 'sum':
	result = 0
	result += func('panties', 'sum_inner', 'lusso')
	result += func('panties', 'sum_inner', 'cats')
	result += func('panties', 'sum_inner', 'salacious')
	result += func('panties', 'sum_inner', 'fashionista')
	result += func('panties', 'sum_inner', 'gm')
	result += func('panties', 'sum_inner', 'eroto')
	result += func('panties', 'sum_inner', 'danilovich')
	if $ARGS[1] = 'dresser':
		if $pantyworntype <> 'none' and result > 0: result -= 1
	end
	exit
end


if $ARGS[0] = 'sum_inner':
	result = 0
	panties_i = 1
	panties_max_i = arrsize('<<$ARGS[1]>>_panties')
	:loopsumpanties
		if func('panties', 'in_wardrobe', $ARGS[1], panties_i): result += 1
		panties_i += 1
	if panties_i < panties_max_i : jump 'loopsumpanties'
	killvar 'panties_i'
	killvar 'panties_max_i'
	exit
end

	if arrsize('ARGS') = 2: ARGS[2] = pantywornnumber

		exit

		exit
	end

		$result = 'low_strength'
		exit
	end

		exit
	end

end

if $ARGS[0] = 'wear':
	if $ARGS[1] = '' or $ARGS[1] = 'last_worn':
		if lastwornunderwear = 2:
			gs 'underwear_bodysuits', 'wear', 'last_worn'
			exit
		end
		$ARGS[1] = $lastwornpantytype
		ARGS[2] = lastwornpantynumber
	end
	if $ARGS[1] = '' or $ARGS[1] = 'none': exit

	gs 'panties', 'strip'

	gs 'underwear_attributes', $ARGS[1] + '_panties', ARGS[2]

	if PanQuality = 0:
		msg '错误：内裤 "<<$ARGS[1]>>_panties[<<ARGS[2]>>]" 不存在'
		exit
	end

	if arrpos('$ARGS', 'check') > 0:
		$temp_not_wear_reason = $func('panties', 'not_wear_reason', $ARGS[1], ARGS[2], 'no_init')
		if $temp_not_wear_reason <> '' and $temp_not_wear_reason <> 'hypno':
			if $temp_not_wear_reason = 'not_owned':
				msg '你没有这条内裤。'
			elseif $temp_not_wear_reason = 'not_in_wardrobe':
				msg '这些内裤不在你的衣柜里'
			elseif $temp_not_wear_reason = 'low_strength':
				msg '这些内裤已经穿旧了'
			elseif $temp_not_wear_reason = 'hypno':
				msg '你不想穿这些内裤。'
			end
			killvar '$temp_not_wear_reason'
			exit
		end
	end

	$pantyworntype = $ARGS[1]
	pantywornnumber = ARGS[2]
	underwear['type'] = 0

	dynamic "<<$pantyworntype>>_pantiesS[<<pantywornnumber>>] = 0"

	!! PPanStyle = PanStyle

	PPanFun = PanFun

	PPanCoverBack = PanCoverBack
	PPanSport = PanSport
	!! PPanSwim = PanSwim

	PPanDirt = PanDirt
	PPanStrength = PanStrength
	PPanMaxStrength = PanMaxStrength

		PPanBorrowed = 1
		PPanDirt = 0
		PPanStrength = PPanMaxStrength
	end

	exit
