# street_events_pav


!! ------------- Fame Events -------------
if $ARGS[0] = 'fame_events':
	gs 'fame_events', 'select', 'pav'
	gt $loc, $loc_arg
end


!! ------------- Prostitutes -------------
if $ARGS[0] = 'minor_prostitutes':
	*clr & cla
	menu_off = 1
	minut += 5
	gs 'stat'
	$streetev_title
	'<center><img <<$set_imgh>> src="images/locations/shared/street/town/hookers.jpg"></center>'
	'你沿街走着，看到两个半裸的女人站在街角。她们会凑近放慢车速的汽车，或是路过的男人。很明显，她们是在招揽嫖客的妓女。'
	act '离开': gt $loc
	exit
end


!! ------------- Naked hawker -------------
if $ARGS[0] = 'minor_naked_hawker':
	*clr & cla
	menu_off = 1
	minut += 5
	gs 'stat'
	$streetev_title
	'<center><img <<$set_imgh>> src="images/locations/shared/street/town/nakedmarket.jpg"></center>'
	'你沿街走着，路过一个年轻女孩，她站在一所房子前，面前摆着一张堆满各种物品的桌子。尽管天这么冷，她除了靴子和帽子之外全身赤裸。她正招呼人们过来看看她卖的物品。她的法子似乎很管用；她和她的东西都吸引了不少注意力。'
	act '无视她': gt $loc
	act '看看她卖的货':
		*clr & cla
		$streetev_title
		'<center><img <<$set_imgh>> src="images/locations/shared/street/town/nakedmarket.jpg"></center>'
		'你走过去，看了看她摆出来卖的几样东西。'

		tovpay1 = rand(3, 8) * 100
		'她有一些化妆品，售价 ' + $func('money', 'string_price', tovpay1) + '。'
		act '买化妆品' + $func('money', 'get_cost_string', tovpay1, 'cash'):
			if func('money', 'can_afford', tovpay1, 'cash') = 0:
				$noMoney
			else
				cls
				minut += 15
				gs 'money', 'pay', tovpay1, 'cash'
				mc_inventory['cosmetics'] += rand(10, 40)
				gs 'stat'
				gt $loc
			end
		end

		tovpay2 = rand(3, 9) * 10
		'她有润唇膏，售价 ' + $func('money', 'string_price', tovpay2) + '。'
		act '买润唇膏' + $func('money', 'get_cost_string', tovpay2, 'cash'):
			if func('money', 'can_afford', tovpay2, 'cash') = 0:
				$noMoney
			else
				cls
				minut += 15
				mc_inventory['lipbalm'] += 20
				gs 'money', 'pay', tovpay2, 'cash'
				gt $loc
			end
		end

		tovpay3 = rand(2, 5) * 100
		'她有一些维生素，售价 ' + $func('money', 'string_price', tovpay3) + '。'
		act '买维生素' + $func('money', 'get_cost_string', tovpay3, 'cash'):
			if func('money', 'can_afford', tovpay3, 'cash') = 0:
				$noMoney
			else
				cls
				minut += 15
				mc_inventory['vitamins'] += 20
				gs 'money', 'pay', tovpay3, 'cash'
				gt $loc
			end
		end

		'她有一本爱情小说 (' + $func('money', 'string_price', 300) + ')'
		if func('money', 'can_afford', 300, 'cash') = 1:
			gs 'library_functions', 'set_single_book_buy_act', '一本爱情小说', 'romance', 300
		end
		act '离开': gt $loc
	end
end

!! ------------- Older Women Shaming PC -------------
if $ARGS[0] = 'minor_older_women_shaming':
	*clr & cla
	menu_off = 1
	minut += 5
	if trait_vars['exhibitionist'] = 0: gs 'mood', 'lower', 'small'
	if trait_vars['exhibitionist_exp'] >= 30: pcs_horny += trait_vars['exhibitionist_exp']/5
	gs 'stat'
	$streetev_title
	'<center><img <<$set_imgh>> src="images/locations/shared/street/oldladies.jpg"></center>'
	'你走过一群坐在一栋楼旁长椅上的老太太，无意中听到她们正朝着你的方向大声议论。'
	'“看看那个婊子，穿着短裙还不穿内裤到处走，好让所有人都能看到她的屁股！”其中一个嚷道。'
	'“她该为自己感到羞耻！”第二个附和道。“就连婊子都比她更懂廉耻。”'
	if $loc ! 'pav_residential' and $loc ! 'pav_commercial':
		'“该感到羞耻的是她妈！”第三个嗤笑道。“什么样的母亲才会养出一个像</i>那样</i>的女儿？”'
	else
		'“对娜塔莎的女儿，你还能指望什么？”第三个嗤笑道。“看来有其母必有其女。”'
		'其他人低声附和，表示赞同。'
	end
	
	if pcs_inhib >= 50:
		'<i>难道喜欢不穿内裤也是罪过吗？</i>你一边继续往前走，一边恼火地想。'
	elseif trait_vars['exhibitionist'] = 0:
		'她们刻薄的话让你羞愧得满脸通红，你加快脚步走开了。'
	end
	act '继续': gt $loc
end
