# post_deliveries

!! variables used:
!! post_vars['load']	= (0-2)	whether there is a little, medium or lots of mail to deliver. Affects how much time you have to do other things
!! post_vars['round']	= (0-1)	used to set first and second round of events
!! post_vars['qw_X']	= (0+)	sets quest X stage for events with after delivery stages
!! post_vars['danger']	= (0-1)	whether the "dangerous streets" are used to save time. Enables encounters with bandits
!! post_vars['bandit'] = (0-6), quest stages for the "bandits" story line
!! post_vars['exhib'] = (0-4), how often you''ve shown your boobs during post_vars['bandit'] = 2
!! post_vars['hj']	= how many times you''ve given the guys a handjob
!! post_vars['bj']	= how many times you''ve given the guys a blowjob
!! post_vars['late']	= (0-15), how often you come back to the post office after 15:15. You get fired if this counter hits 15
!! post_vars['burn']	= (0-1), whether hoodlums tampered with your mail. Gets you fired if yes

menu_off = 1

if $ARGS[0] = 'start':
	cla
	gs 'jobs', 'clock', 'pav_mailgirl'
	*nl
	'他领你到了后面的房间，几个小包裹和信件已经装进了一只挎包里。'

	!! 20% little to do, 40% normal, 40% heavy load
	temp_rand = rand(1, 5)
	post_vars['load'] = iif(temp_rand = 1, 0, iif(temp_rand <= 3, 1, 2))
	killvar 'temp_rand'
	minut += 60 - minut
	minut += 30
	gs 'stat'

	if post_vars['load'] = 0:
		'"今天邮件不多，小菜一碟！"他笑了笑。"别紧张。"'
		'你拎起包，点了点头。确实轻得很！'
	elseif post_vars['load'] = 1:
		'"数量是不少，不过你肯定应付得来，"他说。"照平时的做法干就行，出不了事。"'
		'你拿起包。是有点分量，但拎着还是相当轻。'
	else
		'"今天活儿堆成山了，希望你穿着跑鞋！"他咧嘴一笑。你看得出来，派给你这么多活他心里也有点过意不去，但还是盼着你按时送完。'
		'你的包太重了，两只手都得用上。想赶在 ' + func('time', 'get_time_string', 15, 0) + ' 之前送到，就得抓紧点时间了！'
	end
	'你花了点时间为这趟派送做准备，核对地址、规划路线，然后中午前后离开邮局开始干活。'

	act '开始巡逻':
		*clr & cla
		post_vars['round'] = 0

		'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postgirl.jpg"></center>'
		'头半个小时很轻松，你在邮局附近几条街上穿行。越往市中心外走，你越忍不住看看包里剩下的东西，琢磨今天这趟该怎么送。'
		if post_vars['load'] = 0:
			minut += 25
			'你的包感觉挺轻，里面剩下的东西也不多了。你完全来得及赶在 ' + func('time', 'get_time_string', 15, 0) + ' 之前回去，甚至还有时间歇歇脚！'
		elseif post_vars['load'] = 1:
			minut += 35
			'你的包还挺满，你得一路快走才能按时回去。不过你觉得还没到非要在那些危险街区冒险的地步。'
		else
			minut += 45
			'你的包还是很满，你意识到要是照着既定路线走，根本别想按时赶回去！'
			'你可以只走安全的路线，但邮政局长可不喜欢你晚归。要不就赌一把，穿过那些容易惹麻烦的地段，指望能赶在 ' + func('time', 'get_time_string', 15, 0) + ' 之前把邮件全送完。'
		end
		gs 'stat'

		act '走安全路线':
			*clr & cla
			minut += 15
			gs 'stat'
			post_vars['danger'] = 0

			'<center><img <<$set_imgh>> src="images/locations/shared/postoffice/postgirl.jpg"></center>'
			'你决定照着既定路线走，继续送包里的邮件。'
			if post_vars['load'] = 0:
				'你的时间绰绰有余，就算慢慢送也来得及。'
			elseif post_vars['load'] = 1:
				'你觉得按既定路线走时间够用，能按时送完，于是就这么办了。你没空停下来做别的事，也没法跟人聊太久，不过会照预计那样在 ' + func('time', 'get_time_string', 15, 0) + ' 左右刚好送完。'
			else
				'又过了半小时，你又看了眼自己的包，发现还有一大堆没送！照这样下去，你根本别想按时送完。'
			end
			act '在你送信途中……': gt 'post_deliveries', 'pick_event'
		end

		gs 'willpower', 'misc', 'self'
		if pcs_willpwr < will_cost:
			act '走更危险的捷径抢时间' + $func('willpower', 'get_willcost_string'): $noWillpower
		else
			act '走更危险的捷径抢时间' + $func('willpower', 'get_willcost_string'):

				if post_vars['load'] = 0:
					'你的包很快就见了底，你觉得再花个半小时左右就能送完。'
				elseif post_vars['load'] = 1:
					'照这个速度，只要不出什么岔子，送完还能剩下大约半小时……'
				else
					'又过了半小时，你又看了眼自己的包。还有不少没送，不过只要保持现在的速度，差不多能按时送完。'
				end

				act '在你送信途中……': gt 'post_deliveries', 'pick_event'
			end
		end
	end
	exit
end
