# food
!! 2021/05/06
!! Snacks: +20 pcs_energy, +7 fat (most cases has water available separately)
!! Fast Food: +70 pcs_energy +70 pcs_hydra, +14 fat (hamburgers, fried food, like the downtown diner)
!! Small Meal: +40 pcs_energy +40 pcs_hydra, +4 fat (soup and sandwiches, cafe light lunches, etc)
!! Medium Meal and Hotel Food: +60 pcs_energy +60 pcs_hydra, +7 fat (home cooked dinners and room service equivalent)
!! Large Meal: +100 pcs_energy +100 pcs_hydra, +14 fat (steak dinners, new years party dinner, etc)
!! Bagged Lunch: +40 pcs_energy +40 pcs_hydra, +5 fat (sandwich, cold leftovers, etc)
!! Health Meal: +50 pcs_energy +50 pcs_hydra, +1 fat (Diet food, Salads, rice cakes with peanut butter)
!! 16/10/2022 - Added watercooler - HS


if $ARGS[0] = 'aftermeal':
	if pcs_energy > 100:
		$mtxt = '你不得不硬撑着才吃完，现在肚子撑得满满的。'
	elseif pcs_energy >= 80:
		$mtxt = '分量刚好让你吃得饱足而满足。'
	elseif pcs_energy >= 60:
		$mtxt = '你虽然已经不觉得特别饿了，但也并不觉得特别满足。'
	elseif pcs_energy >= 40:
		$mtxt = '你虽然不再觉得自己饿得要命，但仍相当饿——事实上，你大概还能再吃掉同样多的分量，而且还有肚子再吃一份甜点。'
	else
		$mtxt = '它没能缓解你的饥饿，反而让你更想多吃一些。'
	end
end


if $ARGS[0] = 'family_meals':
	frost = 0
	minut += 20
	pcs_health += 10
	gs 'mood', 'raise', 'tiny'
	fat += iif($ARGS[1] = 'dinner', 8, 6)
	pcs_energy += 60
	if pcs_hydra < 20:
		pcs_hydra += 80
	elseif pcs_hydra < 100:
		pcs_hydra += 60
	else
		pcs_hydra += 30
	end
	gs 'cum_cleanup', 2
	pcs_breath = 0
	gs 'stat'
end


if $ARGS[0] = 'set_exit':
	!! if you dont killvar eat and drink, the next time whatever you eat or drink has the shop price from the first item
	!! As you only eat here, but you came from the order menu, the drink variable is never cleared
	killvar '_drink'
	killvar '_eat'
	if $loc = 'uni_cafe':
		uni_lunch_bought = 1
		gs 'uni_cafe', 'sitting_choice'
	elseif food_loc = 1 and $loc ! 'brothel':
		act '完成': gt 'food_menu'
	else
		!! finish_step returns to the daily-routine hub when a routine is active,
		!! and to the current room (gt $loc, $loc_arg) otherwise.
		!! This wires every standard meal into the routine system without duplicating it.
		act '完成': gs 'daily_routine', 'finish_step', $loc, $loc_arg
	end
end
