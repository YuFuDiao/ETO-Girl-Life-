# wakeup_events
$sleep_loc = 'wakeup_events'
menu_off = 1

!! wakeup events trigger during when waking up
!!
!! Moved here from wakeup
!! When done: Move to bed2


	!!====================================================================!!
	!!                                                                    !!
	!!                           EVENT HANDLERS                           !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'start' or $ARGS[0] = '':
	menu_off = 1
	killvar 'sleep_events'
	killvar 'sleep_events_priority'

	if sleepVars['events_active'] = 1:
		sleepVars['events_done'] = 0

		if $start_type['loc'] = 'sg' and locat['Fam_inGad'] = 0 and motherQW['bathroom_dildos'] > 0 and MarishaQW["marisha_sleepover"] = 0:
			$sleep_events[] = "gs 'wakeup_events', 'mother_sextalk'"
		end

		gs 'city_experimental_trials_events', 'check_for_events'

		if week = 1 and $home_name[$loc] = 'parents_home':
			$sleep_events[] = "gs 'wakeup_events', 'mother_laundry'"
		end

		gs 'wakeup_events', 'vomiting_check'

		if MarishaQW['marisha_sleepover'] = 1: $sleep_events_priority[] = 'gs ''wakeup_events'', ''marisha_ev_morning'' '
		if $start_type['magic'] = 'tg' and daystart - daystart_start < 9:
			$sleep_events_priority[] = "gs 'wakeup_events', 'trans_early_shocked'"
		elseif $start_type['magic'] = 'tg' and daystart - daystart_start < 29:
			$sleep_events_priority[] = "gs 'wakeup_events', 'trans_little_shocked'"
		end

		if therapistQW['hotel_sleep_flag'] = 1:
			therapistQW['hotel_sleep_flag'] = 0
			$sleep_events_priority[] = "gs 'therapist_home', 'wakeup'"
		end

		if NatbelQW['FriendLover'] >= 8 and $loc = 'natbelapt' and $loc_arg = 'natroom':
			$sleep_events_priority[] = "gs 'wakeup_events', 'nat_wakeup_sex'"
		end

		if $start_type['loc'] = 'sg' and sleepVars['slept_in'] = 1 and kanikuli = 0 and gschoolVars['school_diploma'] = 0 and gschoolVars['block'] = 0 and week < 6 and suspended['on'] ! 1:
			if (hour = 7 or hour = 8):
				$sleep_events_priority[] = "gs 'wakeup_events', 'sg_go_school'"
			elseif hour >= 9 and hour < 14:
				$sleep_events_priority[] = "gs 'wakeup_events', 'sg_slept_in'"
			end
		end

		xgt 'wakeup_events', 'mod_sleepevents'
	end

	xgt 'wakeup_events', 'continue'
end


if $ARGS[0] = 'mod_sleepevents':
	!! This location is here to allow mods to hook into the system.
	!! Check for: if $ARGS[0] = 'wakeup_events' and $ARGS[1] = 'mod_sleepevents'
	!!
	!! You can add events as: $sleep_events[] = 'gs ''mod_loc'' ''mod_args0'' '
	!! A random event will be chosen from amongst the added ones.
	!! If an event MUST trigger (due to quest timing), add it tp $sleep_events_priority[] = 'gs ''mod_loc'' ''mod_args0'' '
	!!
	!! If the event does not break the sleep cycle (merely interrupts it) it needs to end with gs 'wakeup_events', 'event_end'
	!! If the event removed Sveta from her location (aka her sleep is cut short and/or she leaves her bed), you must call gs 'wakeup_events', 'exit'
	!!
	!! This is to cleanup all necessary variables.
	!!
	!! DO NOT DIRECTLY GOTO/GOSUB OUTOF THIS LOCATION. ALWAYS ADD IT TO THE QUEUEs!!!!
	!! IT WILL ALMOST CERTIANLY END BADLY WHEN MORE THAN 1 MOD IS INVOLVED.

	gs 'mod_system', 'sleep', 'wakeup_events', 'mod_sleepevents'
	xgt 'wakeup_events', 'event_handler'
end


if $ARGS[0] = 'event_handler':
	if sleepVars['events_done'] < 10:
		if arrsize('$sleep_events_priority') > 0:
			xgt 'wakeup_events', 'event_handler2', 'priority'
		elseif arrsize('$sleep_events') > 0:
			xgt 'wakeup_events', 'event_handler2'
		end
	end

	xgt 'wakeup_events', 'continue'
end


if $ARGS[0] = 'event_handler2':
	sleepVars['events_done'] += 1

	if $ARGS[1] = 'priority':
		temp_slev_id = rand(0, arrsize('$sleep_events_priority')-1)
		$temp_sleep_event_chosen = $sleep_events_priority[temp_slev_id]
		killvar 'sleep_events_priority', temp_slev_id
	else
		temp_slev_id = rand(0, arrsize('$sleep_events')-1)
		$temp_sleep_event_chosen = $sleep_events[temp_slev_id]
		killvar 'sleep_events', temp_slev_id
	end

	killvar 'temp_slev_id'
	dynamic $temp_sleep_event_chosen

	xgt 'wakeup_events', 'event_end'
end


if $ARGS[0] = 'event_end':
	killvar 'temp_sleep_event_chosen'

	if arrsize('$sleep_events_priority') > 0:
		xgt 'wakeup_events', 'event_handler2', 'priority'
	end

	xgt 'wakeup_events', 'event_handler'
end


if $ARGS[0] = 'exit':
	sleepVars['events_done'] = 0
	killvar 'sleep_loc'
	killvar 'sleep_events'
	killvar 'sleep_events_priority'
	killvar 'temp_slev_id'
	killvar 'temp_sleep_event_chosen'
	inSleep = 0
	gs 'wakeup', 'wear_bed_clothes'
end


if $ARGS[0] = 'continue':
	sleepVars['events_done'] = 0
	killvar 'sleep_loc'
	killvar 'sleep_events'
	killvar 'sleep_events_priority'
	killvar 'temp_slev_id'
	killvar 'temp_sleep_event_chosen'

	xgt 'wakeup', 'get_out'
end




	!!====================================================================!!
	!!                                                                    !!
	!!                               EVENTS                               !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'mother_sextalk':
	gt 'mother_sextalk', 'dildo_wakeup1'
end


if $ARGS[0] = 'mother_laundry':
	xgt 'wakeup_events', 'mother_laundry2'
end


if $ARGS[0] = 'mother_laundry2':
	'你妈妈已经洗好了衣服，你看到衣柜里放着一叠干净的衣服。'

	act '继续':
		*nl
		gs 'wakeup_events', 'event_end'
	end
end


if $ARGS[0] = 'vomiting_check':
	!! at high alko_points, diminish this events as it''ll fire often
	if vomit['hangover'] = 1 and rand(50, 125) > trait_vars['drinking_exp'] or vomit['unlucky'] = 1 or vomit['morning_sick'] = 1:
		!! if not one of these locations, event will not fire
		if $loc = $func('homes_properties_attr', 'get_property_attribute', '$bedroom') and func('homes_properties', 'is_at_a_home'):
			$sleep_events[] = 'gs ''wakeup_events'', ''vomiting_event'' '
			exit
		end
	end

	vomit['morning_sick'] = 0
	vomit['hangover'] = 0
	vomit['unlucky'] = 0
end


if $ARGS[0] = 'vomiting_event':
	gs 'wakeup_events', 'exit'

	gs 'home_activity', 'vomiting_images'
	gt 'home_activity', 'morning_vomit'
end


if $ARGS[0] = 'nat_wakeup_sex':
	gt 'wakeup_events', 'nat_wakeup_sex1'
end


if $ARGS[0] = 'nat_wakeup_sex1':
	gs 'natbel_kissinggames', 'wakeup_event_text'

	act '起床': gs 'wakeup_events', 'event_end'
	act '和娜塔莎来一次晨间性爱':
		gs 'wakeup_events', 'exit'
		gt 'natbel_kissinggames', 'wakeup_sex'
	end
end


if $ARGS[0] = 'marisha_ev_morning':
	*clr
	'<center><img <<$set_imgh>> src="images/characters/pavlovsk/resident/marisha/marisha.jpg"></center>'
	'你醒来时感到浑身放松，转头一看，玛丽莎已经起床了。她对你微微一笑说："谢谢你做的一切，<<$pcs_firstname>>。回头见。"说完她便离开了，而你继续起身。'
	MarishaQW['marisha_sleepover'] = 0
	gs 'wakeup_events', 'event_end'
end


if $ARGS[0] = 'trans_early_shocked':
	'你醒来时脑海里闪过的第一个念头是："该死！这不是梦，我现在真的是个女人了。"'
	gs 'wakeup_events', 'event_end'
end


if $ARGS[0] = 'trans_little_shocked':
	'看着自己女性的身体，你仍然有点惊讶这真的是你。'
	gs 'wakeup_events', 'event_end'
end


if $ARGS[0] = 'sg_go_school':
	gs 'mood', 'lower', 'small'
	gs 'stat'
	msg '<center>该死，这<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>，你睡过头了！你得赶紧，不然上学就要迟到了！</center>'

	gs 'wakeup_events', 'event_end'
end


if $ARGS[0] = 'sg_slept_in':
	if gschoolVars['absence_count'] <= 2:
		gs 'mood', 'lower', 'medium'
		gs 'stat'
		msg '<center>该死，这<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>. 现在去学校已经太迟了！<br><<$npc_nickname[''A29'']>>如果我不是每天都去学校，她会非常生气。</center>'
	elseif gschoolVars['expell_stage'] = 0:
		gs 'mood', 'lower', 'large'
		gs 'stat'
		msg '<center>该死，这<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>. 现在去学校已经太迟了！<br><<$npc_nickname[''A29'']>>会发现我缺了一天的课而生气。</center>'
	elseif gschoolVars['expell_stage'] = 1:
		gs 'mood', 'lower', 'large'
		gs 'stat'
		msg '<center>该死，这<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>. 现在去学校已经太迟了！<br><<$npc_nickname[''A29'']>>如果我再缺一天课会非常生气。</center>'
	else
		gs 'mood', 'lower', 'large'
		gs 'stat'
		msg '<center>该死，这<b><<func(''time'', ''get_time_string'', hour, minut, cheatVars[''time_format''])>></b>. 现在去学校已经太迟了！<br><<$npc_nickname[''A29'']>>会大发雷霆。我不知道她会做出什么事来……</center>'
	end
	gs 'wakeup_events', 'event_end'
end
