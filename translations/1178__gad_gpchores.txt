!! Task initiator for the grandparents chores whilst Sveta is staying with grandparents.

if $ARGS[0] = 'init':
	if gp_chores['active'] > 1:
		gt 'gad_gpchores', 'active'
	end
	gs 'gad_gpchores', 'convo'
end


if $ARGS[0] = 'active':
end


!! Conversations
if $ARGS[0] = 'convo':
	if gp_chores = 0:
		'你走近你的<relation>，问问他们今天有没有什么活儿要你干。'
		if gp_chores['mood'] < 0:
			gp_chores['bonus'] = 0
			'你太懒了，{mc}，我不确定能不能信任你把该干的活儿干完。'
		elseif gp_chores['mood'] < 10:
			'你的 <relation 狠狠地瞪了你一眼，你就是个游手好闲的懒鬼。滚出我的视线！'
		elseif gp_chores['mood'] > 10:
			gp_chores['bonus'] = 10
			'你帮了我们很多忙，{mc}，有你在真是太好了。这里有点额外的小心意，是给你的。 '
		elseif gp_chores['mood'] > 20:
			gp_chores['bonus'] = 20
			'你的<relation>一看到你，脸上就露出笑容。你真是个勤奋的姑娘，你妈妈一定会为你变得这么勤快而感到骄傲。这里有点额外的零花钱，奖励你的努力。'

			!! potential for helping in village with high rep?
		end
	end
	if gp_chores['active'] = -1:
		gp_chores['failed'] += 1
		gp_chores['mood'] -= 1
	end
	if gp_chores['active'] = 5:
		gp_chores['completed'] += 1
		gp_chores['mood'] += 1
	end
end
