# albina_starlets_aftermath
!! Sveta encounters Albina a few days after speaking with Chernov, with different dialogue depending on whether Sveta gave a statement or not.

frost = 0
menu_off = 1
AlbinaQW['RideDaystart'] = 0
AlbinaQW['albride'] = 1
gs 'stat'
'<center><img <<$set_imgh>> src="images/characters/pavlovsk/school/girl/albina/bmw.jpg"></center>'
'一辆看起来很贵的SUV在你面前停下，前车窗摇了下来。阿尔比娜坐在驾驶座上，冲你笑了笑。"去我家吧，我们得谈谈。"'

act '进来':
	*clr & cla
	'<center><img <<$set_imgh>> src="images/characters/pavlovsk/school/girl/albina/drive.jpg"></center>'
	minut += 5
	gs 'stat'
	'你钻进车里，坐到阿尔比娜旁边。她挂上挡，开车穿过小镇。'
	!! If Sveta helped Chernov
	if AlbinaQW['Chernov'] = 1:
		'"这是你的车？你什么时候会开车了？"你问道。'
		'她笑着回答："不是，这是我爸的车，我……借来的。而且别问，我没有驾照，不过这一带的警察根本不管。"'
		'她在红灯前停下来，转过身看着你，显得相当高兴。"你不知道你帮了多大的忙，<<$pcs_nickname>>。我没想到Chernov会拿着那些线索行动得这么快。"'
		'"等等，<i>你</i>就是他那个揭露新星的''匿名线人''？为什么？"你惊讶地问。'
		'"我？不。呃，不完全是……"'
		'"那是谁？你到底瞒着我什么？"你问道，但她怯生生地把目光移开。绿灯亮了，她继续沉默地开着车。'
		*nl
		'开了一会儿车后，你来到了她家。她把车开上车道，在前门外停下，然后叫你跟她走，随后下了车。'
	end
	!! If Sveta did not help Chernov
	if AlbinaQW['Chernov'] = 2:
		'"这是你的车？你什么时候会开车了？"你问道。'
		'她笑着回答："不是，这是我爸的车，我……借来的。而且别问，我没有驾照，不过这一带的警察根本不管。"'
		'她在红灯前停下来，转向你。"我真希望你把那个人想知道的一切都告诉他，<<$pcs_nickname>>，"她说，"那对我帮助会非常大。"'
		'"什么叫''对你帮助很大''？他是想让我站出来指证新星！"你反驳道。'
		*nl
		'是啊，因为他拿到了他需要的证据。"'
		'"什么？怎么会？谁给的？"你困惑地问。'
		'她没有回答，只是怯怯地把目光移开。绿灯亮了，她继续沉默地开着车。'
		*nl
		'开了一会儿车后，你来到了她家。她把车开上车道，在前门外停下，然后叫你跟她走，随后下了车。'
	end

	act '跟阿尔比娜走': gt 'albina_house_events', 'start'
end
