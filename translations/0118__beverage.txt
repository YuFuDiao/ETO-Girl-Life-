if $ARGS[0] = 'semen_stats':
	!! 5ml -> 0 energy/hydra
	!! 40ml (avg npc) -> 4 energy/hydra
	!! 73ml (max npc) -> 11
	!! 150ml (abd override 4) -> 36
	!! 250ml (abd override 5) -> 75
	temp_semen_gain = min(75, ARGS[1] * func('math', 'int_sqrt', ARGS[1]) / 50)
