# nichBathServant

if $ARGS[0] = '' or $ARGS[0] = 'start' or $ARGS[0] = 'return':
	CLOSE ALL
	*clr & cla
	menu_off = 0
	gs 'core_library', 'setloc', 'nichBathServant', ''
	$nichLoc = 'servantBathroom'

	gs 'stat'
	gs 'themes', 'indoors'
	!! Argument return: PC has been in the room before, dont perform actions meant to be executed before or as she enters the room.
	$func('wrap', 'header center b', '仆人浴室')
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/nichApartment/bathroomServant.jpg"></center>'
	'这是佣人的浴室，里面有淋浴、马桶和带<a href="exec:gt ''mirror'',''start''">镜子</a>的洗手池，房间角落里还放着一台洗衣机。'

	if nichWork = 2:
		if mc_inventory['shampoo'] = 0:
			'既然你被允许使用家里的一些洗发水，你打开一瓶新的，放进自己的淋浴间里。'
			mc_inventory['shampoo'] = 15
		end

		act '<b>去你的卧室</b>': gt 'nichBedroomServant'

		gs 'nichChore', 'inspect', 'bathServant'
		act '检查待洗衣物': gt 'nichChore', 'inspect', 'laundry', 1
	else
		act '去佣人的卧室': gt 'nichBedroomServant'
	end

	gs 'din_van', 'private'
	gs 'home_events', 'bathroom'
end
