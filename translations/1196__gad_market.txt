	gs 'core_library', 'setloc', 'gad_market', 'start'
	$region = 'gad'
	$location_type = 'public_outdoors'
	if sound_settings['environment_off'] = 0:
		if month >=11 and month <= 12 or month >=1 and month <=3:
			PLAY 'sound/street2.mp3', 30 & PLAY 'sound/zima.mp3', 30
		else
			PLAY 'sound/street2.mp3', 30 & PLAY 'sound/street_leto.mp3', 30
		end
	end
	gs 'stat'

	'<center><h2>农夫市场</h2></center>'
	'<center><img <<$set_imgh>> src="images/locations/gadukino/market/main.jpg"></center>'
	'一个简陋朴实的农夫市场，几间小店铺占满了整个场地。'
	*nl

	if hour < 8 or hour > 20:
		gs 'gad_market', 'close'
		exit
	end

	'在一侧你看到一个摊位，那里' + iif(hunterVars['were_met'] = 1, ' 猎人们正在 ', ' ') + '卖<a href="exec:minut += 1 & gt ''gad_market'', ''hunter_stand''">干货</a>。'
	act '浏览<<iif(hunterVars[''were_met''] = 1, ''猎人'', ''干货'')>>摊位': minut += 1 & gt 'gad_market', 'hunter_stand'

	'你看到一个摆满了<a href="exec:minut += 1 & gt ''gad_market'', ''clothing_stand''">廉价衣物</a>的摊位。'
	act '浏览服装摊位': minut += 1 & gt 'gad_market', 'clothing_stand'

	if gad_stand = 1:
		if hour <= 20:
			act '去你自己的摊位 (0:01)': minut += 1 & gt 'gad_market', 'your_stand'
		end

		act '拆掉你的摊位 (0:30)': gt 'gad_market', 'breakdown_stand'
		act $func('wrap', 'neg', '返回'): *nl $func('wrap', 'neg', '摊位还开着的时候你不能离开。')
	else
		if hour < 18 and (boletus_stored + bilberry_stored + fish_stored) > 0:
			act '搭起你自己的摊位 (1:00)': gt 'gad_market', 'build_stand'
		elseif hour >= 18:
			*nl
			'现在搭摊位太晚了。'
		else
			*nl
			'你没有什么可以卖的。'
		end
		act '返回': gt 'gad_market', 'leave'
	end
end
