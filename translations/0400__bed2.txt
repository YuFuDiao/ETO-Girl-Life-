# bed2

!! Lying in bed hub. From here PC can take various actions, including going to sleep.
!!
!! Moved here from bed_events
!! Going to sleep: move to pre_sleep_events

if $ARGS[0] = 'bed2' or $ARGS[0] = 'start' or $ARGS[0] = '':
	$menu_loc = 'bed2'
	$menu_arg = 'start'
	menu_off = 0

	if clo_flag['bed'] = 1:
		clo_flag['bed'] = 0
		gs 'outfit', 'restore', 'bed'
	end
	gs 'stat'

	gt 'bed2', 'mod_sleeptriggers'
end


if $ARGS[0] = 'mod_sleeptriggers':
	!! This location is here to allow mods to hook into the system.
	!! Either check for: if $ARGS[0] = 'bed2' and $ARGS[1] = 'mod_sleeptriggers'
	!!
	!! This is NOT for events!!

	gs 'mod_system', 'sleep', 'bed2', 'mod_sleeptriggers'
	gt 'bed2', 'main'
end


if $ARGS[0] = 'main':
	$menu_loc = 'bed2'
	$menu_arg = 'main'
	menu_off = 0

	if pcs_makeup ! 1 and cosmetic_tattoo = 0: *nl $func('wrap', 'neg b', '睡觉前你需要卸掉脸上的妆。')

	if $clothingworntype ! 'nude':
		'<center><img <<$set_imgh>> src="images/shared/home/bed.jpg"></center>'
	else
		'<center><img <<$set_imgh>> src="images/shared/home/bedroom/son.jpg"></center>'
	end
	*nl
	$func('alarmclock', 'base_alarmclock_text')
	*nl
	gs 'bed2', 'wear_panties_decider'
	*nl
	gs 'bed2', 'wear_buttplug_decider'
	gs 'bed2', 'wear_vibe_decider'


	if pcs_sleep <= 80:
		if $clothingworntype = 'nude':
			act '去睡觉': gs 'shortgs', 'autosave' & gt 'pre_sleep', 'start'
		else
			act '脱衣服上床睡觉': gs 'shortgs', 'autosave' & gt 'pre_sleep', 'start'
		end
	end

	act '起床': gt 'bed_get_out', 'start'
	act '小睡一会 (1:00)': gs 'sleep_simple', 'nap_bed'

	gs 'library_functions', 'set_home_read_acts'

	if $loc ! 'bedrPar' or locat['Anya_inroom'] = 0 or locat['Anya'] = 1 or locat['Anya'] = 26:
		gs 'library_functions', 'set_read_porn_act'
		if (pcs_horny > 75 and pcs_inhib >= 60) or (pcs_horny > 50 and (fame['pav_slut'] >= 150 or (fame['pav_slut'] >= 50 and pcs_inhib >= 60))):
			act '自慰':
				inSleep = 0
				gt 'selfplay', 'start'
			end
		end
	end
end


if $ARGS[0] = 'wear_panties_decider':
	if hypnoPanty <= 0:
		if sleepVars['bedPanty'] = 0:
			'你总是穿着内衣睡觉吗？目前 - <a href="exec:sleepVars[''bedPanty''] = 1 & gs ''stat'' & gt ''bed2'', ''main''">按原样</a>.'
		elseif sleepVars['bedPanty'] = 1:
			'你总是穿着内衣睡觉吗？目前 - <a href="exec:sleepVars[''bedPanty''] = 2 & gs ''stat'' & gt ''bed2'', ''main''">是</a>.'
		elseif sleepVars['bedPanty'] = 2:
			'你总是穿着内衣睡觉吗？目前 - <a href="exec:sleepVars[''bedPanty''] = 0 & gs ''stat'' & gt ''bed2'', ''main''">NO</a>.'
		end
	end
end


if $ARGS[0] = 'wear_buttplug_decider':
	if mc_inventory['buttplug'] = 1:
		if sleepVars['bedAnal'] = 0:
			'你应该总是在睡前取下肛塞吗？当前 - <a href="exec:sleepVars[''bedAnal''] = 1 & gs ''stat'' & gt ''bed2'', ''main''">NO</a>.'
		elseif sleepVars['bedAnal'] = 1:
			'你应该总是在睡前取下肛塞吗？当前 - <a href="exec:sleepVars[''bedAnal''] = 0 & gs ''stat'' & gt ''bed2'', ''main''">是</a>.'
		end
	end
end


if $ARGS[0] = 'wear_vibe_decider':
	if mc_inventory['vibe'] = 1 and cheatVars['sleep_vib'] = 1:
		if sleepVars['bedVibrator'] = 0:
			'你应该总是在睡前取下振动棒吗？当前 - <a href="exec:sleepVars[''bedVibrator''] = 1 & gs ''stat'' & gt ''bed2'', ''main''">NO</a>.'
		elseif sleepVars['bedVibrator'] = 1:
			'你应该总是在睡前取下振动棒吗？当前 - <a href="exec:sleepVars[''bedVibrator''] = 0 & gs ''stat'' & gt ''bed2'', ''main''">是</a>.'
		end
	end
end


if $ARGS[0] = 'fin':
	if instr('bedr;korr', $loc): gt 'bedr'
	if instr('bedr2x;korr2x', $loc): gt 'bedr2x'
	if $loc = 'uni_dorm': gt 'uni_dorm', 'dorm_room'
	if $loc = 'nichBedroomServant': gt 'nichBedroomServant'
end
