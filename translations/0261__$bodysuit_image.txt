# $bodysuit_image
!! get the image for bodysuits
!! ARGS 0 - bodysuits shop name
!! ARGS 1 - bodysuits index

if $ARGS[0] = 'scandalicious':
	$RESULT = 'images/pc/items/scandalicious/bodysuits/<<ARGS[1]>>.jpg'
elseif $ARGS[0] = 'moncheri':
	$RESULT = 'images/pc/items/moncheri/bodysuits/<<ARGS[1]>>.jpg'
elseif $ARGS[0] = 'salacious':
	$RESULT = 'images/pc/items/salacious/bodysuits/<<ARGS[1]>>.jpg'
elseif $ARGS[0] = 'lusso':
	$RESULT = 'images/pc/items/lusso/bodysuits/<<ARGS[1]>>.jpg'
end
