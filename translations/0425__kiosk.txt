# kiosk

!! WD:"SUB >= 15" for act - Will need balancing / playtesting

if $kioskloc = 'counter' and $args[0] ! 'counter': gt 'kiosk', 'counter'


if $ARGS[0] = 'start':
	if hour >= 14 and hour <= 16:
		*clr & cla
		menu_off = 1
		gs 'stat'
		'<center><img <<$set_imgh>> src="images/locations/city/shared/kiosk/kiosk21.jpg"></center>'
		'目前人太多，看来你想结账先要排队了。'

		act '离开': minut += 1 & gt $loc

		act '排队等候 (0:20)': minut += 20 & gt 'kiosk', 'counter'

		if trait_vars['exhibitionist'] > 1:
			act '露出你的奶子和小穴跳过排队环节':
				*clr & cla
				minut += 1
				gs 'flash', 'full', 'outdoors', 5
				gs 'stat'

				act '去柜台': gt 'kiosk', 'counter'
			end
		end

		if trait_vars['exhibitionist'] > 0:
			act '露出你的阴部跳过排队环节':
				*clr & cla
				minut += 1
				gs 'flash', 'pussy', 'outdoors', 5
				gs 'stat'

				act '去柜台': gt 'kiosk', 'counter'
			end
		end

		if pcs_inhib >= 40:
			act '露出你的乳头跳过排队环节':
				*clr & cla
				minut += 1
				gs 'flash', 'tits', 'outdoors', 5
				gs 'stat'

				act '去柜台': gt 'kiosk', 'counter'
			end
		end
	else
		gt 'kiosk', 'counter'
	end
end


if $ARGS[0] = 'counter':
	*clr & cla
	$kioskloc = 'counter'
	$args_s = 'counter'
	$loc_s = 'kiosk'
	$menu_loc = 'kiosk'
	$menu_arg = 'counter'
	menu_off = 0
	gs 'stat'

	gs 'themes', 'outdoors'
	gs 'item_cart', 'shopping_aisle', 'kiosk'
	gs 'stat'

	$backimage = 'images/locations/city/shared/kiosk/kiosk123.jpg'
	act '离开': minut += 1 & gt $loc

	if func('money', 'can_afford', 100) = 0:
		'<center><table><TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你没有足够的钱买零食。</td></tr></table></center>'
	else
		act '买零食 (' + $func('money', 'string_price', 100) + ') (0:05)':
			*clr & cla
			minut += 5
			gs 'money', 'pay', 100
			gs 'food', 'snack_stats'
			'<center><img <<$set_imgh>> src="images/shared/food/food.jpg"></center>'
			'<center><table><TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你享用了一份小巧可口、但有点发胖的零食。</td></tr></table></center>'

			act '返回': gt 'kiosk', 'start'
		end
	end

	if func('money', 'can_afford', 40) = 0:
		'<center><table><TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你没有足够的钱买水。</td></tr></table></center>'
	else
		act '买些水 (' + $func('money', 'string_price', 40) + ') (0:05)':
			*clr & cla
			minut += 5
			gs 'money', 'pay', 40
			gs 'beverage', 'water_stats'
			'<center><img <<$set_imgh>> src="images/locations/shared/brothel/barorderwater.jpg"></center>'
			'<center><table><TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你享受喝水</td></tr></table></center>'

			act '返回': gt 'kiosk', 'start'
		end
	end

	if func('money', 'can_afford', 100) = 0 and mc_inventory['newspaper'] = 0:
		'<center><table><TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你没有足够的钱买报纸。</td></tr></table></center>'
	elseif mc_inventory['newspaper'] = 1:
		'<center><table><TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你已经买了今天的报纸。</td></tr></table></center>'
	else
		act '买报纸 (' + $func('money', 'string_price', 100) + ') (0:05)':
			*clr & cla
			mc_inventory['newspaper'] = 1
			minut += 5
			gs 'money', 'pay', 100
			gs 'stat'
			'<center><img <<$set_imgh>> src="images/system/image_needed.png"></center>'
			'<center><table><TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你买了周报</td></tr></table></center>'
			act '返回': gt 'kiosk', 'start'
		end
	end

	act '浏览手机主题': $kioskloc = '' & gt 'kiosk', 'themes'
end


if $ARGS[0] = 'themes':
	*clr & cla
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/city/shared/kiosk/kiosk123.jpg"></center>'
	'<center><img WIDTH = 245 HEIGHT = 56 src="images/locations/city/shared/kiosk/ruch.jpg"></center>'
	'所有主题花费 ' + $func('money', 'string_price', 500) + '。'

	if func('money', 'can_afford', 500):
		i = 1
		:phonethemeloop
			if i = 1: $phoneNameTemp = 'Bimbo' & $phoneNameTempZh = '花瓶（Bimbo）'
			if i = 2: $phoneNameTemp = 'Aluminus' & $phoneNameTempZh = '铝（Aluminus）'
			if i = 3: $phoneNameTemp = 'Bottlecapped' & $phoneNameTempZh = '瓶盖（Bottlecapped）'
			if i = 4: $phoneNameTemp = 'Galactic' & $phoneNameTempZh = '银河（Galactic）'
			if i = 5: $phoneNameTemp = 'Vectored' & $phoneNameTempZh = '矢量（Vectored）'
			if i = 6: $phoneNameTemp = 'Veneer' & $phoneNameTempZh = '贴面（Veneer）'
			if i = 7: $phoneNameTemp = 'Voidwalker' & $phoneNameTempZh = '虚空行者（Voidwalker）'
			if i = 8: $phoneNameTemp = 'Voidwalker Red' & $phoneNameTempZh = '虚空行者·红（Voidwalker Red）'
			if i = 9: $phoneNameTemp = 'Voidwalker Toxic' & $phoneNameTempZh = '虚空行者·毒（Voidwalker Toxic）'
			if i = 10: $phoneNameTemp = 'Gopnik' & $phoneNameTempZh = '戈普尼克（Gopnik）'
			if i = 11: $phoneNameTemp = 'Sports' & $phoneNameTempZh = '运动（Sports）'
			if i = 12: $phoneNameTemp = 'Succubus' & $phoneNameTempZh = '魅魔（Succubus）'
			if i = 13: $phoneNameTemp = 'Woodshop' & $phoneNameTempZh = '木工房（Woodshop）'
			if phonetheme[i] = 0: '<a href="exec: VIEW ''images/system/phone/previews/<<$lcase($phoneNameTemp)>>.png''">预览</a> <a href="exec: gs ''money'', ''pay'', 500 & phonetheme[<<i>>] = 1 & gt ''kiosk'', ''themes''">购买</a>一个<b><<$phoneNameTempZh>></b>的主题。'
			i += 1
		if i < 14: jump 'phonethemeloop'
		killvar 'phoneNameTemp' & killvar 'phoneNameTempZh'
	else
		'<TR BGCOLOR="' + $theme_hex['table_bg'] + '"><td><b></b>你没有足够的钱买手机主题。</td></tr>'
	end

	act '返回': gt 'kiosk', 'counter'
end
