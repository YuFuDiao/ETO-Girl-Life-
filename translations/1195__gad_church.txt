	'你走近祈祷长椅，跪下祈祷。'
	act '祈祷':
		*clr & cla
		menu_off = 1
		minut += 10
		gs 'fame', 'church_reduction', 'village', 1
		gs 'archetypes', 'faith_gain', 'pray', 10

			gs 'mood', 'raise', 'tiny'
			gs 'fame', 'church_reduction', 'village', 1
		end
		'<center><img <<$set_imgh>> src="images/locations/pavlovsk/church/ch_praying.jpg"></center>'
		if hour > 6 and hour < 20 and church_moral < 30:
			pcs_horny -= 10
			gs 'mood', 'raise', 'tiny'
			church_moral += 1
			'你决定祈祷一会儿。你并不特别虔诚，但这份平和与宁静给了你思考的时间。'
		elseif hour > 6 and hour < 20 and church_moral >= 30 and church_moral < 60:
			pcs_horny -= 20
			gs 'mood', 'raise', 'small'
			church_moral += 1
			minut += 15
			'你决定祈祷一会儿。虽然你并不认为自己十分虔诚，但当你看着周围的圣像时，你能感受到与上帝之间切实的联系。你任由自己深入审视灵魂，思考人生的方向。'
		elseif hour >6 and hour < 20 and church_moral >= 60:
			pcs_horny -= 30
			gs 'mood', 'raise', 'medium'
			church_moral += 1
			minut += rand(30, 60)
			'你决定祈祷。置身于上帝殿宇之中的美好与荣耀令你心醉神迷，你完全沉浸在祈祷中。当你终于重新注意到周围的一切时，已不确定过了多久，不过这并不重要，因为你已与上帝和世界和平相处。'
		elseif (hour >= 20 or hour <= 6) and church_moral < 30:
			pcs_horny -= 40
			gs 'mood', 'raise', 'medium'
			church_moral += 1
			minut += rand(30, 60)
			'中殿只点着几支小蜡烛，四周一片昏暗。你开始感到一种说不清道不明的奇异氛围。你突然涌起一股强烈的祈祷冲动，不知不觉间，一个小时过去了，你感觉自己重获新生。'
		elseif (hour >= 20 or hour <= 6) and church_moral >= 30 and church_moral < 60:
			pcs_horny = 0
			gs 'mood', 'raise', 'medium'
			church_moral += 1
			minut += rand(30, 60)
			if drugVars['cocaine_system'] > 1: 3 * drugVars['cocaine_system'] = drugVars['cocaine_system'] / 4
			'中殿只点着几支小蜡烛，四周一片昏暗，但你反而觉得安心。你涌起一股强烈的祈祷冲动，不知不觉间，一个小时过去了，你感觉自己得到了净化。'
		else
			pcs_horny = 0
			gs 'mood', 'raise', 'large'
			church_moral += 1
			minut += 60
			if drugVars['cocaine_system'] > 1: drugVars['cocaine_system'] = drugVars['cocaine_system'] / 2
			'中殿只点着蜡烛和几盏小夜灯，黑暗让人感到非常安心，就像回到母体中一样。你涌起一股强烈的祈祷冲动，不知不觉间，一个多小时过去了，你感觉自己重获新生、得到了净化。'
		end
		gs 'stat'
		act '返回前厅': gt 'gad_church', 'Narthex'
	end

			gs 'flash', 'butt', 'indoors', 2, 2

			gs 'stat'
			act '返回前厅': gt 'gad_church', 'Narthex'
