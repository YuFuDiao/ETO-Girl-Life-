if $ARGS[0] = 'calc':
	temppain['total'] = 0
	temppain['i'] = 0
	temppain['maxi'] = arrsize('$painParts')
	:pain_calc_loop
		temppain['total'] += pain[$painParts[temppain['i']]] * pain[$painParts[temppain['i']]]
		temppain['i'] += 1
	if temppain['i'] < temppain['maxi']: jump 'pain_calc_loop'
	temppain['total'] += 4 * vgape * vgape + 16 * agape * agape + 576 * spanked * spanked

	pain['total'] = min(func('math', 'int_sqrt', temppain['total']), 100)
	if pain['total'] < 0: pain['total'] = 0
	pain['damage'] = pain['total']

	killvar 'temppain'

	if pain['total'] <= 0: exit

	gs 'drugs', 'pain_relief'
	if pain['relief'] > 0:
		pain['total'] = pain['total'] * (100 - pain['relief']) / 100
		if pain['total'] < 0: pain['total'] = 0
	end
	exit
end


if $ARGS[0] = 'manage':
	!! first initialization
	if pain['timer'] <= 0: pain['timer'] = totminut + 10
	gs 'pain', 'calc'

	if totminut < pain['timer']: exit
	if (pain['damage'] + pain['total']) <= 0: exit

	temppain['tmp_0'] = 1 + (totminut - pain['timer']) / 10

	temppain['i'] = 0
	temppain['maxi'] = arrsize('$painParts')
	:pain_manage_loop
		$temppain['loc'] = $painParts[temppain['i']]
		if pain[$temppain['loc']] > 0:
			temppain['tmp_1'] = rand(temppain['tmp_0'] / 3, temppain['tmp_0'])
			temppain['tmp_2'] = rand(temppain['tmp_0'] / 2, temppain['tmp_0'])
			temppain['tmp_3'] = iif(inSleep, rand(0, pain[$temppain['loc']]), 0)
			pain[$temppain['loc']] -= max(temppain['tmp_1'], temppain['tmp_2'] + temppain['tmp_3'])
		end
		temppain['i'] += 1
	if temppain['i'] < temppain['maxi']: jump 'pain_manage_loop'

	temppain['i'] = 0
	temppain['maxi'] = arrsize('pain')
	:statpainloop
		if pain[temppain['i']] < 0: pain[temppain['i']] = 0
		temppain['i'] += 1
	if temppain['i'] < temppain['maxi']: jump 'statpainloop'
	killvar 'temppain'

	gs 'pain', 'calc'
	gs 'pain', 'stat_changes'

	!! 10 min here, but 30 when just received pain
	pain['timer'] = totminut + 10
	exit
end


if $ARGS[0] = 'stat_changes':
	if pain['total'] <= 0: exit

	if pain['total'] <= 20:
		if (pcs_health * 10 / 9)	>= healthmax: pcs_health = pcs_health * 9 / 10
	elseif pain['total'] <= 40:
		if (pcs_health * 5 / 4)		>= healthmax: pcs_health = pcs_health * 8 / 10
	elseif pain['total'] <= 60:
		if (pcs_health * 5 / 3)		>= healthmax: pcs_health = pcs_health * 7 / 10
	elseif pain['total'] <= 80:
		if (pcs_health * 5 / 2)		>= healthmax: pcs_health = pcs_health * 5 / 10
	else
		if (pcs_health * 5)			>= healthmax: pcs_health = pcs_health * 4 / 10
	end

	temppain['mood_loss'] = max(1, pain['total'] / 10)
	if trait_vars['pain_tolerance'] = -1:
		temppain['mood_loss'] = (temppain['mood_loss'] * 12) / 10
	elseif trait_vars['pain_tolerance'] = 1:
		temppain['mood_loss'] = max(1, temppain['mood_loss'] * 8 / 10)
	end

	!! Gates shift based on pain tolerance.
	!! Low: can push below disp at low pain, no floor at moderate pain.
	!! High: protected at low pain, floor of 30 at moderate pain.
	!! Above 80: everyone loses mood without limit.
	if pain['total'] <= 40:
		if trait_vars['pain_tolerance'] = -1:
			if pcs_mood > moodVars['disp'] - 10: gs 'mood', 'lower', temppain['mood_loss']
		elseif trait_vars['pain_tolerance'] = 1:
			if pcs_mood > moodVars['disp'] + 5: gs 'mood', 'lower', temppain['mood_loss']
		else
			if pcs_mood > moodVars['disp']: gs 'mood', 'lower', temppain['mood_loss']
		end
	elseif pain['total'] <= 80:
		if trait_vars['pain_tolerance'] = -1:
			gs 'mood', 'lower', temppain['mood_loss']
		elseif trait_vars['pain_tolerance'] = 1:
			if pcs_mood > 30: gs 'mood', 'lower', temppain['mood_loss']
		else
			if pcs_mood > 20: gs 'mood', 'lower', temppain['mood_loss']
		end
	else
		gs 'mood', 'lower', temppain['mood_loss']
	end

	killvar 'temppain'
	exit
end
