# kafeM

if $ARGS[0] = 'start':
	minut += 30
	'你到达咖啡馆，在一张桌子旁坐下。 <<$npcdesc>>点了食物和酒。在等待食物上桌的时候， <<$npcdesc>>给你讲了一件轶事：<br>'
	gs 'anekdot'
	*nl
	if npc_gentle[$npcID] = 1:
		if npc_dates[$npcID] > 25: gs 'boylove', 'tits'
		if npc_dates[$npcID] > 30: gs 'boylove', 'figure'
	elseif npc_rough[$npcID] = 0:
		if npc_dates[$npcID] > 20: gs 'boylove', 'tits'
		if npc_dates[$npcID] > 25: gs 'boylove', 'figure'
	else
		if npc_dates[$npcID] > 15: gs 'boylove', 'tits'
		if npc_dates[$npcID] > 20: gs 'boylove', 'figure'
	end

	act '笑': gs 'npc_relationship', 'modify', $npcID, rand(0, 1) & xgt 'kafeM', 'a'
	act '微笑': xgt 'kafeM', 'a'
	act '不好笑': gs 'npc_relationship', 'modify', $npcID, rand(-1, 0) & xgt 'kafeM', 'a'
end


if $ARGS[0] = 'a':
	minut += 30
	gs 'beverage', 'wine_stats'
	'最后，食物上桌了。 <<$npcdesc>>给你倒上酒，提议为你干杯。'

	act '吃': xgt 'kafeM', 'b'
end


if $ARGS[0] = 'b':
	gs 'food', 'medium_meal_stats'

	'咖啡馆里开始播放音乐，并且<<$npcdesc>>问你是否愿意跳舞。'

	act '跳舞': gs 'npc_relationship', 'modify', $npcID, rand(0, 1) & gt 'kafeM', 'dance'
	gs 'willpower', 'misc', 'resist', 'easy'
	if pcs_willpwr < will_cost:
		act '拒绝' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '拒绝' + $func('willpower', 'get_willcost_string'):
			gs 'willpower', 'pay', 'resist'
			gs 'stat'
			gs 'npc_relationship', 'modify', $npcID, rand(-1, 0)
			gt 'kafeM', 'end'
		end
	end
end


if $ARGS[0] = 'dance':
	minut += 5
	'<center><img <<$set_imgh>> src="images/pc/activities/dance.jpg"></center>'
	'你和<<$npcdesc>>去舞池。他搂住你，你们开始跳舞。'

	if npc_gentle[$npcID] = 1:
		'<<$npcdesc>>跳舞时非常拘谨。他有点笨拙地抱着你，并道歉。'

		act '跳完舞': gs 'npc_relationship', 'modify', $npcID, rand(0, 1) & xgt 'kafeM', 'end'
	elseif npc_rough[$npcID] = 0:
		'<<$npcdesc>>在舞蹈中自信地引导你。'

		act '跳完舞': gs 'npc_relationship', 'modify', $npcID, rand(0, 1) & xgt 'kafeM', 'end'
	else
		'<<$npcdesc>>跳舞时抓着你的屁股，紧贴着你。你能透过衣服感觉到他硬挺的阴茎顶着你的肚子。'
		'他低声对你说："我们去厕所吧。"'
		gs 'willpower', 'sex', 'resist', 'medium'
		if pcs_willpwr < will_cost:
			act '跳完舞' + $func('willpower', 'get_willcost_string'): $noWillpower
		else
			act '跳完舞' + $func('willpower', 'get_willcost_string'):
				gs 'willpower', 'pay', 'resist'
				gs 'stat'
				gs 'npc_relationship', 'modify', $npcID, rand(-1, 0)
				xgt 'kafeM', 'end'
			end
		end

		act '去厕所': xgt 'kafeM', 'tualet'
	end
end


if $ARGS[0] = 'tualet':
	'<<$npcdesc>>把你拉进厕所，开始揉捏你的胸部，为你做好准备。'
	gs 'arousal', 'foreplay', 5
	gs 'stat'

	act '口交': gt 'kafeM', 'bj'
	act '趴在马桶上': gt 'kafeM', 'sex'
end


if $ARGS[0] = 'sex':
	sexa += 1
	gs 'npc_relationship', 'modify', $npcID, 1
	'<center><img <<$set_imgh>> src="images/characters/city/boyfriend/sex/event/cafe/kafe.jpg"></center>'
	'你掏出他的阴茎，开始为他手淫。 <<$npcdesc>>戴上避孕套，然后让你站在马桶上，推着他的<<dick>>厘米鸡巴插进你颤抖的小穴里。'
	gs 'arousal', 'vaginal', 15, 'sub'
	gs 'arousal', 'end'

	'很快<<$npcdesc>>再也忍不住，射在了避孕套里。'

	act '下一步': gt 'kafeM', 'end'
end


if $ARGS[0] = 'bj':
	bja += 1
	gs 'cum_call', 'mouth_swallow', $npcID, 1
	gs 'npc_relationship', 'modify', $npcID, 1
	'<center><img <<$set_imgh>> src="images/characters/city/boyfriend/sex/event/cafe/kafe1.jpg"></center>'
	'你把他的阴茎从裤子里拉出来，开始给他手淫，然后蹲在他面前，把他含进嘴里，慢慢吮吸他那<<dick>>厘米的鸡巴。'
	'很快<<$npcdesc>>再也忍不住，在你嘴里射了出来，充满你口中的是他的酸涩精液。你试图全部咽下去，但精液还是从你嘴里溢出，顺着嘴唇和下巴流下。'
	'你满脸精液，忠诚地望着<<$npcdesc>>的眼睛。'
	gs 'arousal', 'bj', 15, 'sub'
	gs 'arousal', 'end'

	act '下一步': gt 'kafeM', 'end'
end


if $ARGS[0] = 'end':
	gs 'stat'
	'<<$npcdesc>>提出要抱住你。'

	act '回家': gt 'sexm', 'start'
end
