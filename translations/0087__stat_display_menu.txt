if $ARGS[0] = 'render_cell_opt':
	!! ARGS[1] = header label
	!! ARGS[2] = variable expression whose current value is matched (e.g. "stat_cfg['font_size']")
	!! Caller must populate before calling:
	!!   $temp_values[]        — 'value|Label' pairs. Use 'custom|<any>' for a free-input slot.
	!!                           value is compared as integer if numeric, otherwise as string.
	!!   $temp_actions_left[]  — exec string for left arrow at matching index (falls back to [0],
	!!                           then auto-cycle to previous value with wrapping if still empty)
	!!   $temp_actions_right[] — exec string for right arrow at matching index (falls back to [0],
	!!                           then auto-cycle to next value with wrapping if still empty)
	!! All three arrays are killvared after rendering.

	!! Read current variable value (numeric and string)
	!! rco[]  = numeric workspace  |  $rco[] = string workspace
	dynamic 'rco[''cur_num''] = ' + $ARGS[2]
	dynamic '$rco[''cur_str''] = $str(' + $ARGS[2] + ')'

	!! Scan entries, splitting each 'value|Label' pair
	rco['size'] = arrsize('$temp_values')
	rco['idx'] = -1
	rco['custom_idx'] = -1
	rco['i'] = 0
	:rco_search
	if rco['i'] < rco['size']:
		rco['pipe'] = instr($temp_values[rco['i']], '|')
		if rco['pipe'] > 0:
			$rco['entry_val'] = mid($temp_values[rco['i']], 1, rco['pipe'] - 1)
			$rco['entry_lbl'] = mid($temp_values[rco['i']], rco['pipe'] + 1)
		else
			$rco['entry_val'] = $temp_values[rco['i']]
			$rco['entry_lbl'] = $temp_values[rco['i']]
		end
		if $rco['entry_val'] = 'custom':
			rco['custom_idx'] = rco['i']
		elseif $rco['entry_val'] = $rco['cur_str'] or val($rco['entry_val']) = rco['cur_num']:
			rco['idx'] = rco['i']
		end
		rco['i'] += 1
		jump 'rco_search'
	end

	!! If no direct match: fall back to 'custom' slot if present, otherwise first entry
	if rco['idx'] = -1:
		if rco['custom_idx'] >= 0:
			rco['idx'] = rco['custom_idx']
		else
			rco['idx'] = 0
		end
	end

	!! Re-parse the matched entry to get its label
	rco['pipe'] = instr($temp_values[rco['idx']], '|')
	if rco['pipe'] > 0:
		$rco['entry_val'] = mid($temp_values[rco['idx']], 1, rco['pipe'] - 1)
		$rco['entry_lbl'] = mid($temp_values[rco['idx']], rco['pipe'] + 1)
	else
		$rco['entry_val'] = $temp_values[rco['idx']]
		$rco['entry_lbl'] = $temp_values[rco['idx']]
	end

	!! Build display label
	if $rco['entry_val'] = 'custom':
		$rco['label'] = '自定义：' + $rco['cur_str']
	else
		$rco['label'] = $rco['entry_lbl']
	end

	!! Resolve actions — fall back to index 0, then auto-cycle if still empty
	$rco['action_left'] = iif($temp_actions_left[rco['idx']]  <> '', $temp_actions_left[rco['idx']],  $temp_actions_left[0])
	$rco['action_right'] = iif($temp_actions_right[rco['idx']] <> '', $temp_actions_right[rco['idx']], $temp_actions_right[0])

	if $rco['action_left'] = '' or $rco['action_right'] = '':
		!! Auto-cycle: compute neighbour indices with wrapping
		rco['idx_prev'] = (rco['idx'] - 1 + rco['size']) mod rco['size']
		rco['idx_next'] = (rco['idx'] + 1) mod rco['size']
		!! Parse the value portion of each neighbour entry
		rco['pipe'] = instr($temp_values[rco['idx_prev']], '|')
		$rco['prev_val'] = iif(rco['pipe'] > 0, mid($temp_values[rco['idx_prev']], 1, rco['pipe'] - 1), $temp_values[rco['idx_prev']])
		rco['pipe'] = instr($temp_values[rco['idx_next']], '|')
		$rco['next_val'] = iif(rco['pipe'] > 0, mid($temp_values[rco['idx_next']], 1, rco['pipe'] - 1), $temp_values[rco['idx_next']])
		!! Assign: use numeric assignment if value looks like a number, else string
		if $rco['action_left'] = '':
			$rco['action_left'] = iif(val($rco['prev_val']) <> 0 or $rco['prev_val'] = '0', $ARGS[2] + ' = ' + $rco['prev_val'], $ARGS[2] + ' = ''' + $rco['prev_val'] + '''')
		end
		if $rco['action_right'] = '':
			$rco['action_right'] = iif(val($rco['next_val']) <> 0 or $rco['next_val'] = '0', $ARGS[2] + ' = ' + $rco['next_val'], $ARGS[2] + ' = ''' + $rco['next_val'] + '''')
		end
	end

	!! Render the cell
	$result =	'<table width="100%" cellpadding="0" cellspacing="0" border="0">'
	$result +=		'<tr>'
	$result +=			'<td align="center" valign="middle" width="1%" nowrap rowspan="3" style="text-align: center; vertical-align:middle">'
	$result +=				'<a style="text-decoration: none" href="exec:<<$rco[''action_left'']>> & gt ''$menu_settings'', ''status''">'
	$result +=					'<big>◀</big>'
	$result +=				'</a>'
	$result +=			'</td>'
	$result +=			'<td align="center" valign="middle" width="98%" style="text-align: center; vertical-align:middle">'
	$result +=				'<b><<$ARGS[1]>></b>'
	$result +=			'</td>'
	$result +=			'<td align="center" valign="middle" width="1%" nowrap rowspan="3" style="text-align: center; vertical-align:middle">'
	$result +=				'<a style="text-decoration: none" href="exec:<<$rco[''action_right'']>> & gt ''$menu_settings'', ''status''">'
	$result +=					'<big>▶</big>'
	$result +=				'</a>'
	$result +=			'</td>'
	$result +=		'</tr>'
	$result +=		'<tr>'
	$result +=			'<td align="center" valign="middle" rowspan="2" style="text-align: center; vertical-align:middle">'
	$result +=				$rco['label']
	$result +=			'</td>'
	$result +=		'</tr>'
	$result +=	'</table>'

	killvar 'rco'
	killvar 'temp_values'
	killvar 'temp_actions_left'
	killvar 'temp_actions_right'
	exit
end


if $ARGS[0] = 'render_cell_num':
	!! Renders a numeric stepper cell.
	!! ARGS[1] = header label
	!! ARGS[2] = variable expression (e.g. "stat_cfg['icon_height']")
	!! ARGS[3] = default value (integer)
	!! ARGS[4] = min value (integer)
	!! ARGS[5] = max value (integer)
	!! $ARGS[6] = unit label (e.g. "px" or "%"), can be empty
	!! Left/right arrows step by 1 (clamped). Value is a link that opens input().
	dynamic 'rcn[''cur''] = ' + $ARGS[2]
	rcn['def'] = ARGS[3]
	rcn['min'] = ARGS[4]
	rcn['max'] = ARGS[5]

	$rcn['act_left'] = $ARGS[2] + ' = max(' + $str(rcn['min']) + ', ' + $ARGS[2] + ' - 1)'
	$rcn['act_right'] = $ARGS[2] + ' = min(' + $str(rcn['max']) + ', ' + $ARGS[2] + ' + 1)'
	!! On input: clamp result; if player cancelled (0 or ''), restore default
	$rcn['act_input'] = '$rcn_inp = input(''' + $ARGS[1] + ' [默认值：' + $str(rcn['def']) + ', 最小值：' + $str(rcn['min']) + ', 最大值：' + $str(rcn['max']) + ']'')'
	$rcn['act_input'] += ' & ' + $ARGS[2] + ' = iif($rcn_inp = '''' or val($rcn_inp) = 0 and $rcn_inp <> ''0'', ' + $str(rcn['def']) + ', min(' + $str(rcn['max']) + ', max(' + $str(rcn['min']) + ', val($rcn_inp))))'
	$rcn['act_input'] += ' & killvar ''rcn_inp'''

	$result =	'<table width="100%" cellpadding="0" cellspacing="0" border="0">'
	$result +=		'<tr>'
	$result +=			'<td align="center" valign="middle" width="1%" nowrap rowspan="3" style="text-align:center; vertical-align:middle">'
	$result +=				'<a style="text-decoration:none" href="exec:<<$rcn[''act_left'']>> & gt ''$menu_settings'', ''status''">'
	$result +=					'<big>◀</big>'
	$result +=				'</a>'
	$result +=			'</td>'
	$result +=			'<td align="center" valign="middle" width="98%" style="text-align:center; vertical-align:middle">'
	$result +=				'<b><<$ARGS[1]>></b>'
	$result +=			'</td>'
	$result +=			'<td align="center" valign="middle" width="1%" nowrap rowspan="3" style="text-align:center; vertical-align:middle">'
	$result +=				'<a style="text-decoration:none" href="exec:<<$rcn[''act_right'']>> & gt ''$menu_settings'', ''status''">'
	$result +=					'<big>▶</big>'
	$result +=				'</a>'
	$result +=			'</td>'
	$result +=		'</tr>'
	$result +=		'<tr>'
	$result +=			'<td align="center" valign="middle" rowspan="2" style="text-align:center; vertical-align:middle">'
	$result +=				'<a href="exec:<<$rcn[''act_input'']>> & gt ''$menu_settings'', ''status''">'
	$result +=					'<<rcn[''cur'']>><<$ARGS[6]>>'
	$result +=				'</a>'
	$result +=			'</td>'
	$result +=		'</tr>'
	$result +=	'</table>'

	killvar 'rcn'
	exit
end

!! ── Row layout helpers ──────────────────────────────────────────────
!! _row_cell ARGS[1]: returns HTML for one cell (with surrounding <tr> as needed).
!!   Caller must do: $result += $func('stat_display_menu', '_row_cell', content)
if $ARGS[0] = '_row_cell':
	if temp_rc_count mod 3 = 0: $result = '<tr>'
	$result += $temp_st_td['cell'] + $ARGS[1] + '</td>'
	temp_rc_count += 1
	if temp_rc_count mod 3 = 0: $result += '</tr>'
	exit
end

!! _row_flush: returns HTML padding the last incomplete row and closes it.
!!   Caller must do: $result += $func('stat_display_menu', '_row_flush')
if $ARGS[0] = '_row_flush':
	if temp_rc_count mod 3 <> 0:
		:_row_flush_pad
		$result += '<td></td>'
		temp_rc_count += 1
		if temp_rc_count mod 3 <> 0: jump '_row_flush_pad'
		$result += '</tr>'
	end
	killvar 'temp_rc_count'
	exit
end
