# havana_dance

menu_off = 1
minut += 5
gs 'stat'
gs 'themes', 'indoors'
$func('wrap', 'header center b', '舞蹈工作室')
'<center><img <<$set_imgh>> src="images/locations/city/citycenter/gym/dance/danceclass.jpg"></center>'
'房间宽敞开阔，采光很好，还有巨大的镜子，非常适合提升你的舞蹈技巧。'
if pcs_stam < stammax / 5:
	*nl
	'你累得连看一眼有哪些活动可做的力气都没有了。先恢复体力，再想着跳舞吧。'
elseif pcs_energy < 10:
	*nl
	'你饿得连看一眼有哪些活动可做的力气都没有了。'
elseif pcs_hydra < 10:
	*nl
	'你渴得连看一眼有哪些活动可做的力气都没有了。'
else
	act '现代舞课 (0:30)':
		*clr & cla
		gs 'exercise', 'tier2', 30, 'agil', 'stren', 'danc'
		abonement -= 1
		if pcs_inhib < 40: inhib_exp += rand(1, 2)

		'<center><img <<$set_imgh>> src="images/locations/city/citycenter/gym/dance/moderndance.jpg"></center>'
		if pcs_danc < 100:
			'你学习现代舞，这种舞蹈在夜店里非常流行。'
		else
			'你已从这门课学到所有能学的东西，掌握了现代舞的精髓。'
		end

		gs 'stat'
		act '离开': gt 'havana_dance'
	end

	act '艳舞课 (0:30)':
		*clr & cla
		gs 'exercise', 'tier2', 15, 'agil', 'stren', 'danc', 'dancero'
		abonement -= 1
		if pcs_inhib < 60: inhib_exp += rand(1, 2)

		'<center><img <<$set_imgh>> src="images/locations/city/citycenter/gym/dance/burlesque.jpg"></center>'
		if pcs_danc < 50:
			gs 'exercise', 'tier2', 15, 'danc'
			'这门课可不仅仅是脱衣服那么简单——这是一种很难的舞蹈，而你的基础还不够。于是老师转而帮你巩固现代舞的基本步伐。'
		else
			gs 'exercise', 'tier2', 15, 'dancero'
			'你学习艳舞的技巧，想象着自己随着音乐一件件褪去衣衫。'
			if pcs_dancero >= 100: '你终于掌握了艳舞的精髓。这门课已经没有别的可学了。'
		end

		gs 'stat'
		act '离开': gt 'havana_dance'
	end

	if pcs_stren >= 40 or needstrength = 0:
		killvar 'needstrength'
		act '钢管舞课 (0:30)':
			*clr & cla
			gs 'exercise', 'tier2', 15, 'agil', 'stren', 'dancpol'
			abonement -= 1
			if pcs_inhib < 60: inhib_exp += rand(1, 2)
			'<center><img <<$set_imgh>> src="images/locations/city/citycenter/gym/dance/poledance.jpg"></center>'

			if pcs_dancero < 40 or pcs_stren < 40:
				minut += 15
				'你的动作谈不上性感，脑袋还重重地撞到了钢管上。这比看起来难多了。如果你能提升一下现代舞或艳舞的水平会大有帮助，不过你总归还是学到了一点。'
				if pcs_stren < 40: needstrength = 1 & '教练努力对你保持耐心，但很明显你的力气应付不了这种舞蹈。你应该先增强力量。'
			else
				gs 'exercise', 'tier2', 15, 'dancpol'
				'你学习并练习各种钢管上的杂技动作。'
				if pcs_dancpol >= 100: '你已经掌握了钢管舞的精髓。这门课已经没有别的可学了。'
			end

			gs 'stat'
			act '离开': gt 'havana_dance'
		end
	end
end

act '离开': gt 'havana', 'dressing_room'
