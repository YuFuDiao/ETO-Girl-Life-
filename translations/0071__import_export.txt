!! Import game settings via input box
!! NOTE: don''t use 'dynamic', the input box is sometimes limited to 511 chars
if $ARGS[0] = 'import':
	*clr & cla
	'把你保存的游戏设置粘贴到输入框里（通常位于屏幕底部），然后点击"继续"。'
	showinput 1

	act '继续':
		$initext = str($usrtxt)
		cmdclear
		showinput 0
		if $initext = '': exit
		dynamic $initext
		killvar 'initext'

		!! Re-apply settings that need more than a plain variable assignment

		!! Theme: restore visual colours from imported name/type
		gs 'themes', 'set_theme', $theme['name'], $theme['type']

		!! Calendar: sync event packs with imported calendar_show flags
		if calendar_show['disco_party'] = 1:
			gs 'calendar', 'add', 'disco_party'
		else
			gs 'calendar', 'remove', 'disco_party'
		end
		if calendar_show['church'] = 1:
			gs 'calendar', 'pack', 'add', 'church'
		else
			gs 'calendar', 'pack', 'remove', 'church'
		end
		if calendar_show['intercity_trains'] = 1:
			gs 'calendar', 'pack', 'add', 'intercity_trains'
		else
			gs 'calendar', 'pack', 'remove', 'intercity_trains'
		end
		gs 'calendar', 'cycle_rebuild', 1

		!! Debug panel: apply imported visibility flag
		showobjs cfg_vars['debug']

		gs 'stat'
		msg '导入完成'
		gt '$menu_settings'
	end
end
