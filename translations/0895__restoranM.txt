# restoranM

$location_type = 'public_indoors'
menu_off = 1

if $ARGS[0] = 'start':
	minut += 30

	'你到达餐厅，在桌边坐下。 <<$npcdesc>>点了一份餐和好酒。你不得不等一会儿，直到食物端上来。'
	'<<$npcdesc>>给你讲了一件轶事：<br>'
	gs 'anekdot'
	*nl

	if npc_gentle[$npcID] = 1:
		if npc_dates[$npcID] > 25: gs 'boylove', 'tits'
		if npc_dates[$npcID] > 30: gs 'boylove', 'figure'
	elseif npc_gentle[$npcID] = 0:
		if npc_dates[$npcID] > 20: gs 'boylove', 'tits'
		if npc_dates[$npcID] > 25: gs 'boylove', 'figure'
	else
		if npc_dates[$npcID] > 15: gs 'boylove', 'tits'
		if npc_dates[$npcID] > 20: gs 'boylove', 'figure'
	end


	act '笑': gs 'npc_relationship', 'modify', $npcID, rand(0, 1) & xgt 'restoranM', 'a'
	act '微笑': xgt 'restoranM', 'a'
	act '不好笑': gs 'npc_relationship', 'modify', $npcID, rand(-1, 0) & xgt 'restoranM', 'a'
end


if $ARGS[0] = 'a':
	minut += 30
	gs 'beverage', 'wine_stats'
	'当食物端上来时，<<$npcdesc>>给你倒了一杯酒，提议为爱情干杯。'

	act '吃': xgt 'restoranM', 'b'
end


if $ARGS[0] = 'b':
	$loc_arg = 'b'
	$loc = 'restoranM'

	gs 'food', 'medium_meal'
	cla

	gs 'beverage', 'wine_stats'
	gs 'stat'
	'当你吃完饭、喝光酒后，<<$npcdesc>>提议带你去他家。'

	act '回家': gt 'sexm', 'start'
end
