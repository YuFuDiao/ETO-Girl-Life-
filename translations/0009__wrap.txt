# wrap
!! Moved from themes to reduce call length
!! $ARGS[0]: color key with optional space-separated modifiers (b, i, u, s, center, h1-h6)
!! $ARGS[1]: text to wrap
$temp_wkey = iif(instr($ARGS[0], ' ') > 0, $mid($ARGS[0], 1, instr($ARGS[0], ' ') - 1), $ARGS[0])
$temp_wmods = ' ' + $ARGS[0] + ' '
if $temp_wkey = 'none' or $temp_wkey = '':
	$temp_color = func('shortgs', 'rgb_to_hex', theme['fcolor'])
elseif $temp_wkey = 'preppy':
	$temp_color = $theme_hex['accent']
elseif $temp_wkey = 'prude':
	$temp_color = $theme_hex['neutral']
else
	$temp_color = $theme_hex[$temp_wkey]
end
$result = '<font color="' + $temp_color + '">' + $ARGS[1] + '</font>'
if instr($temp_wmods, ' b ') > 0: $result = '<b>' + $result + '</b>'
if instr($temp_wmods, ' i ') > 0: $result = '<i>' + $result + '</i>'
if instr($temp_wmods, ' u ') > 0: $result = '<u>' + $result + '</u>'
if instr($temp_wmods, ' s ') > 0: $result = '<s>' + $result + '</s>'
if instr($temp_wmods, ' center ') > 0: $result = '<center>' + $result + '</center>'
if instr($temp_wmods, ' h1 ') > 0: $result = '<h1>' + $result + '</h1>'
if instr($temp_wmods, ' h2 ') > 0: $result = '<h2>' + $result + '</h2>'
if instr($temp_wmods, ' h3 ') > 0: $result = '<h3>' + $result + '</h3>'
if instr($temp_wmods, ' h4 ') > 0: $result = '<h4>' + $result + '</h4>'
if instr($temp_wmods, ' h5 ') > 0: $result = '<h5>' + $result + '</h5>'
if instr($temp_wmods, ' h6 ') > 0: $result = '<h6>' + $result + '</h6>'
killvar 'temp_color'
killvar 'temp_wkey'
killvar 'temp_wmods'

--- wrap ---------------------------------
