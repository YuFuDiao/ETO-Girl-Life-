# shop_fancy_pancy
!! 2021/05/12

if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_fancy_pancy', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	gs 'stat'

	gs 'themes', 'indoors'

	$func('wrap', 'header center b', '花哨')
	'<center><img <<$set_imgh>> src="images/locations/pushkin/fancypancy/shop.jpg"></center>'
	*nl
	'这家小精品店与其说是服装店，不如说更像附近艳舞俱乐部的更衣室。缎面、皮革、荷叶边和华丽感都清楚地表明，这里可不是买实用衣服的地方。'
	'这是城里唯一能买到适合艳舞俱乐部的服装的地方，或许还能给卧室里的情趣加点料。'
	*nl

	'这家精品店出售设计师款冬季外套、艳舞服装、漂亮的鞋子和精美的手袋。'

	act '离开精品店': minut += 3 & gt 'pushkin_sq'
	act '查看服装': minut += 5 & gt 'shop_fancy_pancy', 'clothes'
end


if $ARGS[0] = 'clothes':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_fancy_pancy', 'clothes'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看花哨的服装')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'fancy_burlesque'
			gs 'shop_utils', 'init', 'set_steps', 2, Clothingstock mod 2
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_fancy_pancy', 'start'
	end
end
