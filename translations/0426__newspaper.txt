# newspaper


	!!====================================================================!!
	!!                                                                    !!
	!!                                CIKL                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'cikl':
	mc_inventory['newspaper'] = 0
	gs 'newspaper_pages', 'init'


	i = 0
	:newspaper_frontpage_loop
		idx = rand(0, arrsize('$np_front_pages') - 1)
		$newspaperVars['page_<<i>>'] = $np_front_pages[idx]
		killvar 'np_front_pages', idx
		i += 1
	if arrsize('$np_front_pages') > 0: jump 'newspaper_frontpage_loop'
	killvar 'np_front_pages'

	:newspaper_shuffle_loop
		idx = rand(0, arrsize('$np_pages') - 1)
		$newspaperVars['page_<<i>>'] = $np_pages[idx]
		killvar 'np_pages', idx
		i += 1
	if arrsize('$np_pages') > 0: jump 'newspaper_shuffle_loop'
	killvar 'np_pages'

	newspaperVars['max_page_num'] = i - 1

	killvar 'i'
end




	!!====================================================================!!
	!!                                                                    !!
	!!                                VIEW                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'start':
	temp_np_page = 0
	menu_off = 1
	gs 'stat'
	gs 'newspaper', 'view'
end


if $ARGS[0] = 'view':
	*clr & cla
	'<center><h2>圣彼得堡商报</h2></center>'

	act '收起报纸':
		killvar 'temp_np_page'
		if newspaperVars['dbag'] = 0:
			gt $menu_loc, $menu_arg
		else
			newspaperVars['dbag'] = 0
			gs 'din_bad', 'd_bag'
		end
	end

	if temp_np_page < newspaperVars['max_page_num']:
		act '翻到下一篇报道':
			temp_np_page += 1
			gs 'newspaper', 'view'
		end
	end
	if temp_np_page > 0:
		act '翻到上一篇报道':
			temp_np_page -= 1
			gs 'newspaper', 'view'
		end
	end

	dynamic $newspaperVars['page_<<temp_np_page>>']
end
