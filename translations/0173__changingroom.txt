!!  ARGS 0 - view_swim_item
!! $ARGS 1 - clothing type
!!  ARGS 2 - clothing index
!!  ARGS 3 - ignore inhibition
!!           0: No
!!           1: Partially, overcome through willpower
!!           2: Fully
if $ARGS[0] = 'view_swim_item':
	*clr & cla
	gs 'clothing_attributes', $ARGS[1], ARGS[2]

	'<center><img <<$set_imgh>> src="<<$func(''$clothing_image'', $ARGS[1], ARGS[2])>>"></center>'
	'<<$ARGS[1]>>编号<<ARGS[2]>>'

	temp_changingroomVars['strength'] = CloStrength

	*pl '（耐久 ' + temp_changingroomVars['strength'] + ')'

	act '返回': gt $menu_loc, $menu_arg

	!! if the clothing is worn out
	if temp_changingroomVars['strength'] <= 0:
		'这件物品已经穿旧了，不适合再穿了。'
		act '扔掉它':
			gs 'clothing', 'dispose'
			gt $menu_loc, $menu_arg
		end
	elseif pcs_inhib < CloInhibit:
		if ARGS[3] = 0:
			'你觉得自己还没有胆量穿这么暴露的泳衣。'
		elseif ARGS[3] = 1:
			'你没有足够的胆量穿这么暴露的泳衣，但如果你想下水，就得鼓起勇气。'

			gs 'willpower', 'exhib', 'self', 'easy'
			if pcs_willpwr < will_cost:
				act '穿上它' + $func('willpower', 'get_willcost_string'): $noWillpower
			else
				act '穿上它' + $func('willpower', 'get_willcost_string'):
					gs 'willpower', 'pay', 'self'

					gs 'underwear', 'remove'
					gs 'shoes', 'strip'
					gs 'clothing', 'wear', $ARGS[1], ARGS[2]

					gs 'changingroom', 'leave'
				end
			end

		elseif ARGS[3] = 2:
			'你没有足够的胆量穿这么暴露的泳衣，但如果你想下水，就得鼓起勇气。'
			act '穿上':
				gs 'underwear', 'remove'
				gs 'shoes', 'strip'
				gs 'clothing', 'wear', $ARGS[1], ARGS[2]

				gs 'changingroom', 'leave'
			end
		end

	else
		if pcs_inhib < (CloInhibit + 10):
			'你觉得这件泳衣比你完全能接受的程度更暴露，但这反而让它相当刺激。'
		end

		act '穿上':
			gs 'underwear', 'remove'
			gs 'shoes', 'strip'
			gs 'clothing', 'wear', $ARGS[1], ARGS[2]

			gs 'changingroom', 'leave'
		end
	end
end
