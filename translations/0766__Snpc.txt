$func('wrap', 'header center b', '<<$npc_firstname[''A<<numnpc>>'']>> <<$npc_lastname[''A<<numnpc>>'']>>')

if npc_rel['A<<numnpc>>'] < 20: '他们真的很不喜欢你。'

if npc_rel['A<<numnpc>>'] >= 40 and npc_rel['A<<numnpc>>'] < 60: '你们俩相处得相当不错，就像普通朋友。'
if npc_rel['A<<numnpc>>'] >= 60 and npc_rel['A<<numnpc>>'] < 80: '你和他们相处得很好，他们是你朋友中的一员。'
if npc_rel['A<<numnpc>>'] >= 80: '他们是你最好的朋友之一。'


!! act 'Move away':gt $loc, $loc_arg

gs 'cumreaction'


if $loc ! 'pav_disco':
	act '离开': gt 'gschool_lessons', 'short_break'
else
	act '走开': gt $loc, $loc_arg
end

!! Chat. You come here eg from school lunch.
act '聊天':
	*clr & cla

	if $loc = 'pav_disco': minut += 5

	gs 'stat'
	$func('wrap', 'header center b', '<<$npc_firstname[''A<<numnpc>>'']>> <<$npc_lastname[''A<<numnpc>>'']>>')
	'<center><img <<$set_imgh>> src="images/characters/shared/headshots_main/big<<numnpc>>.jpg"></center>'

	if $loc ! 'pav_disco':
	'你和<<$npc_firstname[''A<<numnpc>>'']>> <<$npc_lastname[''A<<numnpc>>'']>>'
		if npc_rel['A<<numnpc>>'] < 20:
			'"我得走了，再见。"'
		elseif npc_rel['A<<numnpc>>'] >= 20 and npc_rel['A<<numnpc>>'] < 40:
			'"很高兴和你聊天，我们课堂上见。"'
		elseif npc_rel['A<<numnpc>>'] >= 40 and npc_rel['A<<numnpc>>'] < 60:
			'"和你聊天真的很开心，我们课堂上见。"'
		elseif npc_rel['A<<numnpc>>'] >= 60 and npc_rel['A<<numnpc>>'] < 80:
			'"和你聊天真的非常愉快，我们课堂上见。"'
		elseif npc_rel['A<<numnpc>>'] >= 80:
			'"和你聊天真的很开心，但我真的得走了。你知道的，还得备课。哦对了，顺便说一句，你今天看起来真棒！"'
		end

		act '继续': gt 'gschool_lessons', 'short_break'
	else
		act '走开': gt $loc, $loc_arg
	end
end
