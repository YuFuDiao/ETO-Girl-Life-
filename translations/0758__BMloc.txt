# BMloc

menu_off = 1
gs 'stat'

act '离开': gt 'pav_disco'
act '跳舞':
	*clr & cla
	gs 'stat'
	gs 'boystat', $bmNane[Tboynum]
	$hookupboy = $bmNane[Tboynum]
	$npc_usedname[$npclastcalled] = $npc_nickname[$npclastcalled]

	if bmFrend[Tboynum] = 0:
		if bmHJ[Tboynum] > 0 or bmBJ[Tboynum] > 0 or bmSEX[Tboynum] > 0 or bmANAL[Tboynum] > 0 or bmGANG[Tboynum] > 0:
			if bmHJ[Tboynum] > 0: $text2 = '你给我打过手枪。'
			if bmBJ[Tboynum] > 0: $text2 = '你给我口交过。'
			if bmSEX[Tboynum] > 0: $text2 = '你让我操过你。'
			if bmANAL[Tboynum] > 0: $text2 = '你让我操过你的屁股。'
			if bmGANG[Tboynum] > 0: $text2 = '你让我和我朋友一起操过你。'

			'<<$niknameboy>>对你说 "还记得我吗，<<$text2>>。 我是<<$boydesc>>。"'
		else
			if bmKISS[Tboynum] > 0 or bmTITS[Tboynum] > 0:
				'<<$niknameboy>>对你说 "还记得我吗，我们以前一起玩过，还一起喝过啤酒。我是<<$boydesc>>。"'
			else
				'<<$niknameboy>>对你说 "还记得我吗，我们一起跳过舞。我是<<$boydesc>>。"'
			end
		end
	end

	if bmTip[Tboynum] = 0: '舞会结束后，<<$boydesc>>提议出去喝啤酒。'
	if bmTip[Tboynum] = 1: '舞会结束后，<<$boydesc>>提议一起去逛逛。'
	if bmTip[Tboynum] = 2: '舞会结束后，<<$boydesc>>提议去他的酒店。'

	gs 'willpower', 'drink', 'resist'
	if pcs_willpwr < will_cost:
		act '拒绝' + $func('willpower', 'get_willcost_string'): $noWillpower
	else
		act '拒绝' + $func('willpower', 'get_willcost_string'):
			gs 'exp_gain', 'sprt', rand(0, 1)
			gs 'willpower', 'drink', 'resist'
			gs 'willpower', 'pay'
			gs 'stat'
			gt 'pav_disco'
		end
	end

	act '同意':
		*clr & cla
		boynumBlock = 1

		if bmTip[Tboynum] = 0:
			gdk_nice_guy = 1
			gt 'pav_disco_sex', 'outside'
		end
		if bmTip[Tboynum] = 1: gt 'pav_disco_sex', 'outside'
		if bmTip[Tboynum] = 2:
			DPtipe = 2
			gt 'pav_disco_sex', 'hotel'
		end
	end
end
