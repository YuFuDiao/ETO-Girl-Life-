# initg

if $ARGS[0] = '':
	$initext = str($usrtxt)
	cmdclear
	showinput 0
	*clr
	:initg_loop
		$t = $mid($initext, 1, instr($initext, '\n') - 1)
		$inittext[] = $t
		$t
		dynamic $t
		$initext = $mid($initext, instr($initext, '\n') + 1)
		if instr($initext, '\n') < 4 : $initext = $mid($initext, instr($initext, '\n') + 1)
		if instr($initext, '\r') < 4 : $initext = $mid($initext, instr($initext, '\r') + 1)
	if instr($initext, '\n') > 0 : jump 'initg_loop'
	$t = $initext
	$inittext[] = $t
	$t
	dynamic $t
	killvar 't'
	gs 'stat'
	gt '$menu_settings'
end

!! DEPRECATED, use import_export instead
if $ARGS[0] = 'get_game_set':
	gs 'import_export', 'export'
end


!! DEPRECATED, use import_export instead
if $ARGS[0] = 'set_game_set':
	gs 'import_export', 'import'
end
