		gs 'dina', 'prezik'
		condomPrice = 0
	end
	if forced = 2:
		payment = (basePay * useCondom) - condomPrice
	else
		payment = (basePay / 2) - condomPrice
	end
	'<center><img <<$set_imgh>> src="images/locations/city/residential/sauna/givemoney.jpg"></center>'

	if workDolg > 0:
		gs 'money', 'debt_pay', 'workDolg', payment, 'none'
		'完事后，客户穿上衣服，离开房间，并付给桑拿房 ' + $func('money', 'string_debt_reduction', payment) + ' 作为你的服务费。'

		gs 'arousal', 'end'

		act '回到主区域': gt 'city_saunawhore', 'saunawork'
	elseif nichTarSauna = 1:
		nichTarSaunaPay += payment
		'完事后，客户穿上衣服，离开房间，并付给桑拿房 ' + $func('money', 'string_profit', payment) + ' 作为你的服务费。这笔钱稍后由塔拉斯收走。'
		'你赚到了 ' + $func('money', 'string_profit', nichTarSaunaPay) + ' 给塔拉斯。'
		gs 'arousal', 'end'
		act '回到主区域': gt 'city_saunawhore', 'saunawork'
	else
		gs 'money', 'earn', payment
		'和你完事之后，客户穿上衣服，拿出钱包，递给你 ' + $func('money', 'string_profit', payment) + '。'

		gs 'arousal', 'end'

		act '收下钱并送他出去': gt 'city_saunawhore', 'saunawork'
	end
end
