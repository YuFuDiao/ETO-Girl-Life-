!! St Petersburg Canals
!!
!! Sub-location in St Petersburg for dates, boat rides, transport and other events

if $ARGS[0] = 'init':
	$loc_history = $loc
	$loc_args_history = $loc_arg
end


if $ARGS[0] = 'setup':
	*clr & cla
	menu_off = 0
	$region = 'city'
	if $loc ! 'city_canals': gs 'city_canals', 'init'

	gs 'core_library', 'setloc', 'city_canals', $ARGS[1]
	$location_type = 'public_outdoors'
	gs 'stat'

	$setloc['imagepath'] = 'locations/city/canals/'

	!! Set seasonal image
	if month >= 3 and month < 9:
		!! Spring & Summer
		if daystage = 2 or daystage = 3:
			$setloc['StageImage'] = $setloc['imagepath'] + 'city_canals_day_<<rand(1, 2)>>.jpg'
		else
			$setloc['StageImage'] = $setloc['imagepath'] + 'city_canals_night_<<rand(1, 2)>>.jpg'
		end
	else
		!! Autumn & Winter
		if daystage = 2 or daystage = 3:
			$setloc['StageImage'] = $setloc['imagepath'] + 'w_city_canals_day_1.jpg'
		else
			$setloc['StageImage'] = $setloc['imagepath'] + 'w_city_canals_night_1.jpg'
		end
	end
	gs 'core_library', 'stage_title'
	*nl
end


if $ARGS[0] = 'exit':
	!! Clean up any used vars and other code.
	killvar 'loc_history'
	killvar 'loc_args_history'
	menu_off = 0
	!! Now leave the location to destination
	if $ARGS[1] = 'arts':
		msg '尚未实现'
		act '返回': gt $loc, $loc_arg
		!! gt 'city_arts', 'start'
	elseif $ARGS[1] = 'island':
		gt 'city_island'
	elseif $ARGS[1] = 'industrial':
		gt 'city_industrial', ''
	elseif $ARGS[1] = 'residential':
		gt 'city_suburbs', 'start'
	else
		gt $ARGS[1], $ARGS[2]
	end
end


if $ARGS[0] = 'content' or $ARGS[0] = 'start':
	!! Main location content.
	*clr & cla
	$setloc['StageTitle'] = '圣彼得堡运河'
	gs 'city_canals', 'setup', 'start'
	!! temp navigation
	!! if events_active = 1: trigger event
	!! if $meet_date['loc'] = $loc_history: act 'Meet your date': gt 'city_canals', 'dates'
	!! Navigation
	if month >= 3 and month < 12:
		'你站在那里，眺望着圣彼得堡著名的北方运河，街道上人来人往、熙熙攘攘，背后是优雅的俄罗斯建筑。你看到船夫们在下面的水面上经营生意，在观光船上忙碌着。'
		!! Spring & Summer
		!! if canals_ticket < daystart: act 'Buy a ticket for a boat ride (100 <b>₽</b>)':
		!! gt 'city_canals', 'tickets'
		!! else
				if $loc_history ! 'city_suburbs':
					act '乘船前往城市郊区 (0:20, ' + $func('money', 'string_price', 100) + ')':
						*clr & cla
						menu_off = 1
						minut += 20
						gs 'money', 'pay', 100
						gs 'stat'
						'<center><img <<$set_imgh>> src="images/<<$setloc[''imagepath'']>>neva_river.jpg"></center>'
					'你乘上小船，沿着涅瓦河航行，再沿 Malaya 涅瓦上行，抵达举世闻名的瓦西里岛海岬。船靠上码头时，你能看到河流继续流入涅瓦湾，以及 Dekabristov 和 Petrosky 两座岛屿。'
					act '继续': gt $loc, 'exit', 'residential'
				end
			end
			if $loc_history ! 'city_island':
				act '乘船前往瓦西里耶夫斯基岛 (0:20, ' + $func('money', 'string_price', 100) + ')':
					*clr & cla
					menu_off = 1
					minut += 20
					gs 'money', 'pay', 100
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/<<$setloc[''imagepath'']>>island_canal.jpg"></center>'
					'你乘上小船，沿着涅瓦河航行——它是构成圣彼得堡生命线的主要河流之一。你从一座座桥下经过，凝视着河岸两旁优雅的建筑，沉浸在思绪之中。'
					act '继续': gt $loc, 'exit', 'island'
				end
			end
			if $loc_history ! 'city_industrial':
				act '乘船前往工业区 (0:20, ' + $func('money', 'string_price', 100) + ')':
					*clr & cla
					menu_off = 1
					minut += 20
					gs 'money', 'pay', 100
					gs 'stat'
					'<center><img <<$set_imgh>> src="images/<<$setloc[''imagepath'']>>chernaya_rechka.jpg"></center>'
					'你登上小船，沿着黑河前往城市的工业区。你陷入沉思，想起历史课上学到的内容：著名诗人亚历山大普希金在这条河岸边的一场决斗中丧生。'
					act '继续': gt $loc, 'exit', 'industrial'
				end
			end
			act '乘船游览 (1:20, ' + $func('money', 'string_price', 150) + ')':
				*clr & cla
				menu_off = 1
				minut += 120
				gs 'money', 'pay', 150
				gs 'stat'
				'<center><img <<$set_imgh>> src="images/<<$setloc[''imagepath'']>>tour_boat_<<rand(1, 2)>>.jpg"></center>'
				!! TODO: add the boat journey here
				'你乘船游览。此处为乘船游览的描述'
				act '继续': gt $loc, $loc_arg
			end
			act '离开运河': gt $loc, 'exit', $loc_history, $loc_args_history
		!! end
	else
		'冬天不能乘船游览。运河都结冰了。'
		act '离开运河': gt $loc, 'exit', $loc_history, $loc_args_history
	end
end


if $ARGS[0] = 'tickets':
	!! Buy a ticket for a boat ride.
	act '付钱给船夫' + $func('money', 'get_cost_string', 100):
		if func('money', 'can_afford', 100) = 0:
			$noMoney
		else
			gs 'money', 'pay', 100
			canals_ticket = daystart
			'你付钱给船夫 ' + $func('money', 'string_price', 100) + ' 买了一张船票。'
			gt 'city_canals', 'start'
		end
	end
end


if $ARGS[0] = 'dates':
	!! For all dating related activities.
	msg '尚未实现'
		!! $meet_date['loc']
		!! $meet_date['npc']
	gt $loc, $loc_arg
end
