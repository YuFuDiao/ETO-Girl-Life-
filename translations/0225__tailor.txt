if $ARGS[0] = 'set_resize_current_act':
	act '改身上穿的衣服的尺寸 (0:10)' + $func('money', 'get_cost_string', 500):
		*clr & cla
		if PCloStyle = 5 or PCloStyle2 = 6 or PCloSport = 1:
			'裁缝恼火地看着你，"为什么要用运动服浪费我的时间？那种衣服不需要改尺寸……"'
		elseif dyneval('result = (<<$clothingworntype>>_b[<<clothingwornnumber>>] = pcs_hips)'):
			'裁缝恼火地看着你，"这是在开玩笑吗？你的衣服就像为你量身定做的一样合身。我可没时间陪你玩这种愚蠢的把戏。"'
		elseif dyneval('result = (<<$clothingworntype>>_b[<<clothingwornnumber>>] > pcs_hips + 4 or <<$clothingworntype>>_b[<<clothingwornnumber>>] < pcs_hips - 4)'):
			'裁缝恼火地看着你，"为什么要浪费我的时间？你的衣服不需要改尺寸。虽然不是完美合身，但也还行。"'
		elseif func('money', 'can_afford', 500) = 0:
			$noMoney
		else
			minut += 10
			gs 'money', 'pay', 500
			gs 'stat'
			dynamic '<<$clothingworntype>>_b[<<clothingwornnumber>>] = pcs_hips'
			'你脱下衣服，把衣服交给裁缝修改，同时领到一件T恤和一条短裤。'
			'他迅速量了量你的腰围和臀围，只花了几分钟就改好了你的衣服。'
			'他把改好的衣服递还给你，你迅速穿上。你对裁缝的手艺很满意。'
		end

		gs 'tailor', 'set_leave_act'
	end
end

	gs 'tailor', 'set_resize_cost'

		'你目前没有任何需要修改的衣服，'
	elseif func('money', 'can_afford', tailor_total_resize_cost) = 0:
		'你没有足够的钱来调整你所有的衣服。总费用为<<$func(''money'', ''string_price'', tailor_total_resize_cost)>>。'
	else
		act '调整你所有的衣服（0:10，<<$func(''money'', ''string_price'', tailor_total_resize_cost)>>）':
			cla
			minut += 10
			gs 'money', 'pay', tailor_total_resize_cost
			'你调整了所有衣服，费用为<<$func(''money'', ''string_price'', tailor_total_resize_cost)>>。'

			gs 'tailor', 'set_leave_act'
		end
	end

$tailor_funcs['clothing_resize_cost'] = {
	dynamic "
		if <<$ARGS[0]>>[<<ARGS[1]>>]:
			if ((<<$ARGS[0]>>_b[<<ARGS[1]>>] > pcs_hips + 8) or (<<$ARGS[0]>>_b[<<ARGS[1]>>] < pcs_hips - 8)):

if $ARGS[0] = 'core_loop':
	temp_tailorVars['j_max'] = arrsize('$shop_display_types')
	if temp_tailorVars['j_max'] = 0:
		killvar 'temp_tailorVars'
		exit
	end

	temp_tailorVars['j'] = 0
	:core_loop_outer
		$temp_tailorVars['type'] = $shop_display_types[temp_tailorVars['j']]

		if $temp_tailorVars['type'] <> '':
			temp_tailorVars['total'] = arrsize($temp_tailorVars['type'] + $temp_tailorVars['array_postfix'])
			temp_tailorVars['number'] = 1
			:core_loop_inner
				dynamic $loop_func, $temp_tailorVars['type'], temp_tailorVars['number'], ARGS[1]
				temp_tailorVars['number'] += 1
			if temp_tailorVars['number'] < temp_tailorVars['total']: jump 'core_loop_inner'
		end

		temp_tailorVars['j'] += 1
	if temp_tailorVars['j'] < temp_tailorVars['j_max']: jump 'core_loop_outer'

	killvar 'temp_tailorVars'
	killvar 'shop_display_types'
	exit
end
