if $ARGS[0] = 'venue_open':
	if $ARGS[2] = 'casual_meal' and $npc_residence[$ARGS[1]] = 'pav_residential':
		!! Del Parco: open 14:00-21:00 (cafe_parco.qsrc)
		result = iif(ARGS[3] >= 14 and ARGS[3] < 21, 1, 0)
	elseif $ARGS[2] = 'movie_date' and $npc_residence[$ARGS[1]] = 'pav_residential':
		!! Pavlovsk park cinema: open 9:00-18:00 (pav_park.qsrc)
		result = iif(ARGS[3] > 8 and ARGS[3] < 19, 1, 0)
	elseif $ARGS[2] = 'movie_date':
		!! City mall cinema: open 8:00-20:00 (city_cinema.qsrc)
		result = iif(ARGS[3] >= 8 and ARGS[3] <= 20, 1, 0)
	else
		!! netflix_chill (at the NPC''s own place) and city_residential casual_meal have no fixed hours
		result = 1
	end
	exit
end

if $ARGS[0] = 'exit':
	minut += 2
	!! exit should be based on type of apartment and location
	if date_ev['unique_npc'] = 1:
		gs 'date_ev', 'ending'
		gt $date_ev_exit['exit_file'], $date_ev_exit['exit_arg']
	else
		if $date_ev['locat'] = 'npc_home':
			gs 'date_ev', 'ending'
			if $npc_residence[$npcID] = 'pav_residential':
				!! Pav residential exit
				gt 'pav_residential'
			elseif $npc_residence[$npcID] = 'uni_grounds':
				!! Student dorms exit
				gt 'uni_grounds'
			elseif $npc_residence[$npcID] = 'city_residential':
				!! St. Petersburg Residential exit
				gt 'city_residential'
			elseif $npc_residence[$npcID] = 'city_center':
				!! St. Petersburg City Center exit
				gt 'city_center'
			elseif $npc_residence[$npcID] = '':
				!! failsafe for an npc with no residence set
				!! $region is 'pav', 'city', 'gad', 'pushkin'
				if $region = 'pav':
					gt 'pav_residential'
				elseif $region = 'city':
					gt 'city_residential'
				else
					msg '上报安全退出！'
					gt 'pav_residential'
				end
			end
		elseif $date_ev['locat'] = 'hotel_room':
			if date_ev['hotel_days'] > 0:
				if $region = 'pav':
					$hotel = 'pav_hotel'
					gs 'date_ev', 'ending'
					gt 'HotelRoom', 'better'
				elseif $region = 'city':
					$hotel = 'city_hotel'
					gs 'date_ev', 'ending'
					gt 'HotelRoom', 'best'
				end
			else
				if $region = 'pav':
					gs 'date_ev', 'ending'
					gt 'pav_hotel'
				elseif $region = 'city':
					gs 'date_ev', 'ending'
					gt 'city_hotel'
				end
			end
	!!		gt $date_ev_exit['exit_file'], $date_ev_exit['exit_arg']
		elseif $date_ev['locat'] = 'pc_home':
			gs 'date_ev', 'ending'
			if $home['current'] = 'parents_home':
				if $date_ev_exit['loc'] = 'bedroom':
					gt 'bedrPar'
				elseif $date_ev_exit['loc'] = 'hallway':
					gt 'korrPar'
				end
			else
				if $date_ev_exit['loc'] = 'bedroom':
					gt 'bedr'
				elseif $date_ev_exit['loc'] = 'hallway':
					gt 'korr'
				end
			end
		else
			gs 'date_ev', 'ending'
			!! $region is 'pav', 'city', 'gad', 'pushkin'
			if $region = 'pav':
				gt 'pav_residential'
			elseif $region = 'city':
				gt 'city_residential'
			end
		end
	end
end
