# telefon_schedules


if $ARGS[0] = 'cikl':
	gs 'telefon_schedules', 'cikl_calls'
	gs 'telefon_schedules', 'cikl_sms'
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                                Call                                !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'cikl_calls':
	gs 'lover_call', 'scheduler'

	gs 'telefon', 'ClearInCallSchedule', 'A160'
	if leonidSlave = 1 and $job_status['pav_leonid_slave'] = 'employed':
		if arrpos('$contact', 'A160') < 0:
			gs 'telefon', 'AddContact', 'A160', 'images/characters/pavlovsk/resident/leonid/leonidphone.jpg', 0
		end

		if leonidVisit = 0 and rand(1, 4) = 1:
			gs 'jobs', 'book_slot', 'pav_leonid_slave', daystart, 0, 'leonid slave appointment'
			gs 'telefon', 'SetInCallSchedule', 'A160', "gs 'leonid', 'call'", "totminut > <<totminut + (8 - hour) * 60 + rand(0, 600) - minut>> and hour >= 8 and hour < 19 and leonidVisit = 0"
		end
	end
	exit
end




	!!====================================================================!!
	!!                                                                    !!
	!!                                SMS                                 !!
	!!                                                                    !!
	!!====================================================================!!


if $ARGS[0] = 'cikl_sms':
	!! set the time when the SMS will be send
	gs 'booty_call', 'scheduler'
	booty_call_time['daystart'] = daystart


	if nerd_game['invite_day'] < daystart and week = 1:
		if nerd_game['first_SMS_received'] = 0: if arrpos('$contact', 'A152') = -1: gs 'telefon', 'AddContact', 'A152', 'icon_na', 1
		gs 'telefon', 'SetInSMSSchedule', 'A152', "gs 'nerd_game_night', 'game_invite_sms_switch'", "totminut > <<totminut + (11 - hour) * 60 + rand(20, 280) - minut>>"
	end

	if year = 2017 and month = 6 and gopnikbandQW['on_tour'] = 0:
		gopnikbandQW['on_tour'] = 1
		if npc_rel['A144'] >= 50 and arrpos('$contact', 'A144') < 0: gs 'telefon', 'AddContact', 'A144', 'icon_na', 1
	elseif gopnikbandQW['on_tour'] = 1 and year = 2017 and month = 8 and day >= 19:
		gopnikbandQW['on_tour'] = 0
	end

	if gopnikbandQW['on_tour'] = 1 and npc_rel['A144'] >= 50 and anushkaQW['tour_SMS_send'] < daystart and anushkaQW['num_tour_SMS_send'] < 11:
		if anushkaQW['num_tour_SMS_send'] <= 9 and week = 6:
			gs 'telefon', 'SetInSMSSchedule', 'A144', "gs 'band_tour_anushka_SMS', 'start'", "totminut > <<totminut + (12 - hour) * 60 + rand(20, 100) - minut>>"

		elseif anushkaQW['num_tour_SMS_send'] = 10 and week = 2:
			gs 'telefon', 'SetInSMSSchedule', 'A144', "gs 'band_tour_anushka_SMS', 'start'", "totminut > <<totminut + (19 - hour) * 60 + rand(20, 100) - minut>>"
		end
	end

	if mainQW >= 1:
		gs 'tatiana_lab', 'stage_SMS'
	end
end
