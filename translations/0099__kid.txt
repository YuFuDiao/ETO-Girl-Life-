$breastfeed = {
	cla
	*clr
	gs 'stat'
	$func('wrap', 'header center b', '<<$kidname[ks]>> <<$pcs_lastname>>')
	'<center><img <<$set_imgh>> src="images/pc/body/tits/breast_feed.jpg"></center>'
	'你把<<$kidname[ks]>>从<<$kidPosPro[ks]>>的婴儿床里抱起来，温存了一会儿，然后露出乳房。'
	if rand(1,10) = 10:
		'你抬起<<$kidname[ks]>>想喂奶，但你的<<$sdtext[ks]>>没有含住乳头。'
		gs 'mood', 'lower', 'tiny'
		minut += 5

		if milkedvolume < 800:
			'你抬起<<$kidname[ks]>>，给宝宝<<$polreb[ks]>>喂奶。你的宝宝似乎对奶量不太满意。'
		else
			'你抬起<<$kidname[ks]>>，给宝宝<<$polreb[ks]>>喂奶。你的宝宝似乎对奶量很满意。'
		end
	end
	'你轻轻地把<<$kidname[ks]>>放回<<$babyloc>>。'
	act '完成':
		if $pcs_mothername = '':
			$pcs_mothername = input("你希望孩子们怎么称呼你？（留空则为''妈妈'')")
			if $pcs_mothername = '': $pcs_mothername = 'Mom'
		end
		gt 'kid', 'start'
	end
}

$homework = {
	cla
	*clr
	if polkid[k] = 0:
			$kidPosPronn[ks] = '她'
		else
			$kidPosPronn[ks] = '他'
	end
	gs 'stat'
	$func('wrap', 'header center b', '<<$kidname[ks]>> <<$pcs_lastname>>')
	'<center><img <<$set_imgh>> src="images/characters/pavlovsk/resident/kolka/event/homework/homework1.jpg"></center>'
	'你看到<<$kidname[ks]>>正在做一大堆作业。现在的孩子作业真多。'
	'"我的小宝贝想让<<$pcs_mothername>>帮你分担一些吗？"你问道'
	'<<$kidname[ks]>>回答："<<$pcs_mothername>>？！…<<$pcs_mothername>>别再把我当小孩了…不过，是的，我想要些帮助。"'
	'<br>'
	'你坐到<<$kidname[ks]>>旁边的桌边，开始帮<<$kidPosPronn[ks]>>做作业'
	if class['school_grade_average'] > 50:
		'多亏你上学时认真听讲，作业很快就做完了。拿个A应该很容易。'
	else
		'你真希望自己上学时多用点心。作业虽然不算太复杂，但对你来说也不容易。'
	end
	homework[ks] = daystart
	minut += rand(30,80)
	act '完成':
		gt 'kid', 'start'
	end
}

if $ARGS[0] = 'start':
	gs 'stat'
	if $pcs_mothername = '':
		$pcs_mothername = input("你希望孩子们怎么称呼你？（留空则为''妈妈'')")
		if $pcs_mothername = '': $pcs_mothername = 'Mom'
	end
	act '离开':
		cla
		killvar 'ks'
		gt $loc, $loc_arg
	end
	if kid > 0: k = 0
	if lactation['active'] > 0 and pain['nipples'] > 60:
		'你现在不能喂母乳。你的乳头太疼了。'
	end
	:loop
	if k < kid:
		gs 'kid', 'kidlist'
		k += 1
		jump 'loop'
	end
	'已选择孩子：<<$kidname[ks]>>'
	if ks < (kid - 1):
		'<a href="exec:ks += 1& gt ''kid'', ''start'' ">选择下一个孩子</a>'
		act '选择下一个孩子': ks += 1 & gt 'kid', 'start'
	end
	if ks > 0:
		'<a href="exec:ks -= 1& gt ''kid'', ''start'' ">选择上一个孩子</a>'
		act '选择上一个孩子': ks -= 1 & gt 'kid', 'start'
	end
end


if $ARGS[0] = 'kidlist':
	if kidage[k] < 1:
		if yearkid[k] = year:
			if (month - monthkid[k]) < 1 and (day - daykid[k]) < 7:
				kiddaycalc[k] = day - daykid[k]
				if kiddaycalc[k] = 1:
					$kidagetext[k] = '<<kiddaycalc[k]>>天大'
				else
					$kidagetext[k] = '<<kiddaycalc[k]>>天大'
				end
			elseif (month - monthkid[k]) < 1 :
				kiddaycalc[k] = (day - daykid[k]) / 7
				if kiddaycalc[k] = 1:
					$kidagetext[k] = '<<kiddaycalc[k]>>周大'
				else
					$kidagetext[k] = '<<kiddaycalc[k]>>周大'
				end
			else
				kidmonthcalc[k] = (month - monthkid[k])
				if kidmonthcalc[k] = 1:
					$kidagetext[k] = '<<kidmonthcalc[k]>>个月大'
				else
					$kidagetext[k] = '<<kidmonthcalc[k]>>个月大'
				end
			end
		else
			if month = 1 and monthkid[k] = 12:
				if 31 - daykid[k] + day < 7:
					kiddaycalc[k] = 31 - daykid[k] + day
					if kiddaycalc[k] = 1:
						$kidagetext[k] = '<<kiddaycalc[k]>>天大'
					else
						$kidagetext[k] = '<<kiddaycalc[k]>>天大'
					end
				elseif 31 - daykid[k] + day >= 7 :
					kiddaycalc[k] = 31 - daykid[k] + day / 7
					if kiddaycalc[k] = 1:
						$kidagetext[k] = '<<kiddaycalc[k]>>周大'
					else
						$kidagetext[k] = '<<kiddaycalc[k]>>周大'
					end
				elseif daykid[k] > day:
					$kidagetext[k] = '<<kidmonthcalc[k]>>个月大'
				end
			else
				if daykid[k] - day <= 0:
					kidmonthcalc[k] = (12 - monthkid[k] + month)
				else
					kidmonthcalc[k] = (11 - monthkid[k] + month)
				end
				if kidmonthcalc[k] = 1:
					$kidagetext[k] = '<<kidmonthcalc[k]>>个月大'
				else
					$kidagetext[k] = '<<kidmonthcalc[k]>>个月大'
				end
			end
		end
	else
		if kidage[k] >= 1:
			$kidagetext[k] = '<<kidage[k]>>岁'
		else
			$kidagetext[k] = '<<kidage[k]>>岁'
		end
	end
	if $polreb[k] = 'girl':
		$sdtext[k] = '女儿'
	elseif $polreb[k] = 'boy':
		$sdtext[k] = '儿子'
	end
	if polkid[k] = 0:
		$kidPosPro[k] = '她'
	else
		$kidPosPro[k] = 'his'
	end
	if polkid[k] = 0:
		$kidPosProc[k] = '她'
	else
		$kidPosProc[k] = '他'
	end
	if polkid[k] = 0:
		$kidPosProPN[k] = '她'
	else
		$kidPosProPN[k] = 'He'
	end
	if ks = k:
		'<font color= #06992D>你的<<$sdtext[k]>><<$kidname[k]>>已经<<$kidagetext[k]>>了。<<$kidPosProc[k]>>的出生日期是<<monthkid[k]>>/<<daykid[k]>>/<<yearkid[k]>>。</font>'
	else
		'你的<<$sdtext[k]>><<$kidname[k]>>已经<<$kidagetext[k]>>了。<<$kidPosProc[k]>>的出生日期是<<monthkid[k]>>/<<daykid[k]>>/<<yearkid[k]>>。'
	end

	if kidage[k] < 1:
		if hour >= 12 and hour < 18:
			$babyloc = '毯子上'
			'<<$kidname[k]>></a>正躺在毯子上'
		else
			$babyloc = '婴儿床里'
			'<<$kidname[k]>>睡着了，躺在<<$kidPosPro[k]>>的婴儿床里。'
		end

	elseif kidage[k] >= 1 and kidage[k] < 7:
		if hour >= 9 and hour < 21 and rkidloc < 8:
			$totloc = '在地板上玩玩具'
			'<<$kidname[k]>>正在玩玩具。'
		elseif hour < 9 or hour >= 21:
			'<<$kidname[k]>>正在<<$kidPosPro[k]>>的床上睡觉。'
		else
			$totloc = '看电视'
			'<<$kidname[k]>>正在看电视。'
		end
	elseif kidage[k] >= 7 and kidage[k] < 18:
		if hour >= 21 or hour < 8:
			'<<$kidname[k]>>正在<<$kidPosPro[k]>>的床上睡觉。'
		elseif hour >= 8 and hour < 9:
			'<<$kidname[k]>>正在准备去学校。'
		elseif hour >= 9 and hour < 16:
			'<<$kidname[k]>>在学校。'
		elseif hour >= 16 and hour < 18 and homework[k] < daystart:
			$kidloc = '做作业'
			'<<$kidname[k]>>正在做作业。'
		else
			$kidloc = '看电视'
			'<<$kidname[k]>>正在看电视。'
		end
	end
	if kid > 0 and kidage[ks] < 1:

		if lactation['active'] > 0:
			act '母乳喂养<<$kidname[ks]>>': dyneval($breastfeed)
		end
		if $babyloc = '毯子上':
			act '和<<$kidname[ks]>> (0:20)': dyneval($playWithKid)
		end
	end
	if kid > 0 and kidage[ks] >= 1 and kidage[ks] < 7:
		if $totloc = '在地板上玩玩具':
			act '和<<$kidname[ks]>>(0:30)': dyneval($totplay)
		elseif $totloc = '看电视':
			act '和<<$kidname[ks]>>(1:00)': dyneval($tottv)
		end
	end
	if kid > 0 and kidage[ks] >= 7 and kidage[ks] < 18:
		if $kidloc = '看电视':
			act '和<<$kidname[ks]>> (1:00)': dyneval($kidtv)
		elseif $kidloc = '做作业':
			act '帮助<<$kidname[ks]>>做作业。': dyneval($homework)
		end
	end
	"<hr>"
end
