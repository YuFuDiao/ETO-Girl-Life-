if $ARGS[0] = '11_buy_goods':
	!! Buy goods

	!! Flag zapolanski storage, reduction
	!! this flag allows you to go shopping
	DoNotBuy = 2

	!! Amount and Quantity that we can buy(store)
	BuyQuantity = 0
	MaxQuantityHome = 0
	MaxQuantityGarage = 0
	MaxCost = 0

	!! Description size of storage
	TovarLimitHomeRepository = 30
	TovarLimitGarageRepository = 100

	'你和经理谈论购买小饰品的事…'

	!! Get current balances
	if mc_inventory['trinkets_home'] >= TovarLimitHomeRepository:
		'家里没有多余的空间了。'
	else
		DoNotBuy -= 1
		'你有 ' + mc_inventory['trinkets_home'] + ' 个小饰品存放在家里。你最多只能存放<<TovarLimitHomeRepository>>个小饰品。'
		MaxQuantityHome = TovarLimitHomeRepository - mc_inventory['trinkets_home']
	end

	if YouCanGar > 0:
		!! Get the current balances in the garage
		if mc_inventory['trinkets_garage'] >= TovarLimitGarageRepository:
			'车库里没有空间了。'
		else
			DoNotBuy -= 1
			'你有 ' + mc_inventory['trinkets_garage'] + ' 个小饰品存放在车库里。你最多只能存放<<TovarLimitGarageRepository>>个小饰品。'
			MaxQuantityGarage = TovarLimitGarageRepository - mc_inventory['trinkets_garage']
		end
	end

	if DoNotBuy = 2:
		'你想了一会儿，意识到自己根本没有地方存放货物，于是决定放弃这次购买。'
	else
		!! Produce purchase
		act '小饰品的价格为<<$func(''money'', ''string_price'', 100)>>。请输入你想购买的小饰品数量。':
			minut += 13
			BuyQuantity = input("你想以每件<<$func('money', 'string_price', 100)>>的价格购买多少个小饰品？")

			if BuyQuantity > (MaxQuantityHome + MaxQuantityGarage):
				'<b><font color=red>你无处存放多余的<<BuyQuantity-(MaxQuantityHome + MaxQuantityGarage)>>个' + iif(BuyQuantity - (MaxQuantityHome + MaxQuantityGarage) = 1, '小饰品', '小饰品') + '。</front></b>'
				cla
				act '离开': gt 'TerminalOffice', '11'
			end

			MaxCost = BuyQuantity * 100

			if func('money', 'can_afford', MaxCost, 'cash') = 0:
				!! If not enough money
				'你身上没有足够的现金。你需要先取钱才能继续购买。'
				if func('money', 'can_afford', MaxCost) = 1:
					!! The player can be plastic money
					'你的信用卡上有足够的钱，你提议用信用卡付款。'
					'"抱歉，我们没有刷卡机，"经理回答说，"我们只收现金。"'
				end
				cla
				act '离开': gt 'TerminalOffice', '11'
			else
				!! If the money is written off money to fill warehouses

				gs 'money', 'pay', MaxCost, 'cash'

				'你购买了<<BuyQuantity>>小饰品，价值<<MaxCost>>.'

				!! First fill home storage
				if BuyQuantity >= MaxQuantityHome:
					mc_inventory['trinkets_home'] = mc_inventory['trinkets_home'] + MaxQuantityHome
					BuyQuantity -= MaxQuantityHome
					if YouCanGar > 0 : *PL '你将把<<MaxQuantityHome>>个小饰品存放在家里。'
				else
					if YouCanGar > 0 : *PL '你将把<<BuyQuantity>>个小饰品存放在家里。'
					mc_inventory['trinkets_home'] = mc_inventory['trinkets_home'] + BuyQuantity
					BuyQuantity = 0
				end

				!! Now fill garage storage
				if BuyQuantity > 0:
					mc_inventory['trinkets_garage'] = mc_inventory['trinkets_garage'] + BuyQuantity
					'你将存放<<BuyQuantity>>个小饰品在车库里。'
				end

				if YouCanGar = 0 and func('money', 'can_afford', 100, 'cash') = 1:
					'你已经买了很多小玩意儿，但你一直在考虑扩大生意。但你能把它们都存放在哪里呢？也许你应该问问你的继父，能否用他车库里的空余空间…'
				end

				cla
				act '完成购买': gt 'TerminalOffice', '11'
			end
		end

		act '购买小饰品，价格为<<$func(''money'', ''string_price'', 100)>>名为 "眼球"。':
			!! Purchase spend time
			minut += 13

			BuyQuantity = MaxQuantityHome + MaxQuantityGarage
			MaxCost = BuyQuantity * 100

			!! Hammered warehouses
			if func('money', 'can_afford', MaxCost, 'cash') = 0:

				if func('money', 'can_afford', MaxCost) = 1:

				end

				cla
				act '离开': gt 'TerminalOffice', '11'
			else

				!! If the money is written off money to fill warehouses

				gs 'money', 'pay', MaxCost, 'cash'

				'你购买了<<BuyQuantity>>小饰品，价值<<MaxCost>>.'

				!! First fill home storage
				if BuyQuantity >= MaxQuantityHome:
					mc_inventory['trinkets_home'] = mc_inventory['trinkets_home'] + MaxQuantityHome
					BuyQuantity -= MaxQuantityHome
					if YouCanGar > 0 : '你将把<<MaxQuantityHome>>个小饰品存放在家里。'
				else
					if YouCanGar > 0 : '你将把<<BuyQuantity>>个小饰品存放在家里。'
					mc_inventory['trinkets_home'] = mc_inventory['trinkets_home'] + BuyQuantity
					BuyQuantity = 0
				end

				!! Now fill garage storage
				if BuyQuantity > 0:
					mc_inventory['trinkets_garage'] = mc_inventory['trinkets_garage'] + BuyQuantity
					'你将存放<<BuyQuantity>>个小饰品在车库里。'
				end

				if YouCanGar = 0 and func('money', 'can_afford', 100, 'cash') = 1:
					'你已经买了很多小玩意儿，但你一直在考虑扩大生意。但你能把它们都存放在哪里呢？也许你应该问问你的继父，能否用他车库里的空余空间…'
				end

				cla
				act '完成购买': gt 'TerminalOffice', '11'
			end
		end
	end
