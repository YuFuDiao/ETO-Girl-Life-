					end
				end
				i += 1
			if i < arrsize('class_list_institution'): jump 'exam_loop'
			killvar 'i'
		end
	end

				end
				act '继续': gt 'stol', 'start'
			end
		end
	end

		if pcs_willpwr < will_cost:
			act '完成你所有的作业' + $func('willpower', 'get_willcost_string'): $noWillpower
		else

				gs 'stat'
				'<center><img <<$set_imgh>> src="images/locations/shared/apartment/homework.jpg"></center>'
				if mc_inventory['tech_computer'] = 1:
					'这花了你<<lernHome>>小时来完成你的作业。'
					gs 'grades', 'homework', 'school', 'yes', lernHome, 0, 0
				else
					'这花了你<<lernHome>>小时来完成你的作业，但由于你没有电脑，计算机课的部分做得不够好。以后你可以去社区中心的图书馆做作业，那里有电脑可用。'
					gs 'grades', 'homework', 'school', 'no', lernHome, 0, 0
				end

				act '继续': gt 'stol', 'start'
			end
		end
	end
end

	else
		if study_mod = 0:
			'你学习了半个小时，但觉得自己没什么进步。'
		elseif study_mod = 1:
			'你学习了半个小时，觉得自己有一点进步。'
		elseif study_mod = 2:
			'你学习了半个小时，觉得自己为考试做的准备进展良好。'
		else
			'你学习了半个小时，觉得自己进步很大。'
		end
		gs 'grades', 'grade_award', '<<$ARGS[1]>>', '<<$ARGS[2]>>', study_mod
	end

	gs 'stat'

	killvar 'study_mod'
	killvar 'nod_study'

	act '从书桌前站起来': gt 'stol', '<<$ARGS[3]>>'
end
