!! This is the code for choosing the aspect of the player character. This is called from a file by $func('npc_reactions', 'general', npc_identifier/opinion) and returns a string of text that can be added to the to screen as its own line or to an other string by +
!! $ARGS[1] is either the npc_identifirer for a npc with entry in npc_startic or the name of the npc.
!! $ARGS[2] is npc''s opinion of the player character and is only used for npc's that don't have an identifier. It set to either 'love', 'like', 'indifferent' or 'dislike', if it is not set it will default to the opinion of hate. So remember to set it unless the npc hate the player character.
if $ARGS[0] = 'general':
	!! This reset the npc expectation about that player character is tanned if the player character is no longer tanned.
	if npc_reaction_first_time['<<$ARGS[1]>>_tan'] = 1 and pcs_tan = 0: npc_reaction_first_time['<<$ARGS[1]>>_tan'] = 0

	if func('pcs_has_attr', 'cum_face') and npc_reaction_hourly['<<$ARGS[1]>>_cum_on_face'] = 0:
		$result = $func('npc_reactions', 'cum_on_face', $ARGS[1], $ARGS[2])

	elseif func('pcs_has_attr', 'cum_clothes') and npc_reaction_hourly['<<$ARGS[1]>>_cum_on_clothes'] = 0:
		$result = $func('npc_reactions', 'cum_on_clothes', $ARGS[1], $ARGS[2])

	elseif func('pcs_has_attr', 'body_pregnant') and npc_reaction_pregnant['<<$ARGS[1]>>'] = 0:
		$result = $func('npc_reactions', 'pregnant', $ARGS[1], $ARGS[2])

	else
		!! Body

		if npc_reaction_weekly['<<$ARGS[1]>>_fit'] = 0:
			if func('pcs_has_attr', 'body_fit'): $npc_reaction_choice[] = 'fit'
		end

		if npc_reaction_weekly['<<$ARGS[1]>>_piercing'] = 0:
			if func('pcs_has_attr', 'cosmetics_piercing_visible'): $npc_reaction_choice[] = 'piercing'
		end

		if npc_reaction_weekly['<<$ARGS[1]>>_strong'] = 0:
			if func('pcs_has_attr', 'stats_strength_strong'): $npc_reaction_choice[] = 'strong'
		end

		if npc_reaction_first_time['<<$ARGS[1]>>_tan'] = 0:
			if func('pcs_has_attr', 'body_tan'): $npc_reaction_choice[] = 'tan'
		end

		if npc_reaction_first_time['<<$ARGS[1]>>_tattoo'] = 0:
			if func('pcs_has_attr', 'cosmetics_tattoo_visible'): $npc_reaction_choice[] = 'tattoo'
		end



		!! Clothing

		if npc_reaction_weekly['<<$ARGS[1]>>_bimbo'] = 0:
			if func('pcs_has_attr', 'clothes_style_bimbo'): $npc_reaction_choice[] = 'bimbo_look'
		end

		if npc_reaction_weekly['<<$ARGS[1]>>_goth'] = 0:
			if func('pcs_has_attr', 'clothes_style_goth'): $npc_reaction_choice[] = 'goth_look'
		end

		if npc_reaction_weekly['<<$ARGS[1]>>_punk'] = 0:
			if func('pcs_has_attr', 'clothes_style_punk'): $npc_reaction_choice[] = 'punk_look'
		end

		if npc_reaction_daily['<<$ARGS[1]>>_short_skirt'] = 0:
			if func('pcs_has_attr', 'clothes_skirt_short'): $npc_reaction_choice[] = 'short_skirt'
		end

		if npc_reaction_daily['<<$ARGS[1]>>_clothes_thinnes'] = 0:
			if func('pcs_has_attr', 'clothes_thin_high'): $npc_reaction_choice[] = 'clothes_thinnes'
		end



		!! Cosmetics

		if npc_reaction_weekly['<<$ARGS[1]>>_makeup'] = 0:
			if func('pcs_has_attr', 'cosmetics_makeup_any'): $npc_reaction_choice[] = 'makeup'
		end



		!! To be checked. Temporarily written by Anjuna

!!		if npc_reaction_weekly['<<$ARGS[1]>>_tits'] = 0:
!!			if func('pcs_has_attr', 'body_tits_???'): $npc_reaction_choice[] = 'tits'
!!		end
!!
!!		if npc_reaction_weekly['<<$ARGS[1]>>_ass'] = 0:
!!			if func('pcs_has_attr', 'body_ass_???'): $npc_reaction_choice[] = 'ass'
!!		end


		$npc_reaction_chosen = $npc_reaction_choice[rand(0, arrsize('$npc_reaction_choice') - 1)]

		if $npc_reaction_chosen = '':
			$result = ''
		else
			$result = $func('npc_reactions', $npc_reaction_chosen, $ARGS[1], $ARGS[2])
		end

		killvar 'npc_reactions_temp_preferences'
		killvar 'npc_reaction_chosen'
		killvar 'npc_reaction_choice'
	end
end

			elseif npc_rel[$ARGS[1]] > 40 or $ARGS[2] = 'indifferent':

			elseif npc_rel[$ARGS[1]] > 20 or $ARGS[2] = 'dislike':

				$result = '"你太用力了。这身打扮在你身上毫无意义。" <<$nr_temp_npc_name>>说道'
			else

				$result = '"这身衣服穿在你这样的荡妇身上真是浪费！" <<$nr_temp_npc_name>>说道'

			!! text if the npc dislikes the bimbo look
			if npc_rel[$ARGS[1]] > 80 or $ARGS[2] = 'love':

				$result = '"别误会！你依然和以前一样迷人！只是这种打扮不适合我" <<$nr_temp_npc_name>>说道'

				$result = '"你人很不错；我只是觉得换个风格更适合你" <<$nr_temp_npc_name>>说道'
			elseif npc_rel[$ARGS[1]] > 40 or $ARGS[2] = 'indifferent':
				!! if the npc dilikes the bimbo looks and is indifferent to the player character
				$result = '"我才不在乎你长什么样。这对我没有任何影响" <<$nr_temp_npc_name>>说道'
			elseif npc_rel[$ARGS[1]] > 20 or $ARGS[2] = 'dislike':
				!! if the npc dislike the bimbo looks and dislikes the player character
				$result = '"这身打扮太适合你了！也就是说，两样我都讨厌" <<$nr_temp_npc_name>>说道'
			else
				!! if the npc like the bimbo looks and hates the player character
				$result = '"至少这外表和性格很相配：廉价又肤浅" <<$nr_temp_npc_name>>说道'
			end
		else
			!! text if the npc is love to the bimbo look
			if npc_rel[$ARGS[1]] > 80 or $ARGS[2] = 'love':
				!! if the npc is indifferent to the bimbo looks and love the player character
				$result = '"你总是那么惊艳，不管穿什么！" <<$nr_temp_npc_name>>说道'
			elseif npc_rel[$ARGS[1]] > 60 or $ARGS[2] = 'like':
				!! if the npc is indifferent to the bimbo looks and likes the player character
				$result = '"你人很好。而且这风格很适合你" <<$nr_temp_npc_name>>说道'
			elseif npc_rel[$ARGS[1]] > 40 or $ARGS[2] = 'indifferent':
				!! if the npc is indifferent to the bimbo looks and is indifferent to the player character
				$result = '"我不介意。你想穿什么就穿什么。" <<$nr_temp_npc_name>>说道'
			elseif npc_rel[$ARGS[1]] > 20 or $ARGS[2] = 'dislike':
				!! if the npc is indifferent to the bimbo looks and dislikes the player character
				$result = '"你想干什么？也许你应该少关心外表，多想想怎么别当个婊子？" <<$nr_temp_npc_name>>说道'
			else
				!! if the npc is indifferent to the bimbo looks and hates the player character
				$result = '"荡妇就是荡妇。穿什么都改变不了" <<$nr_temp_npc_name>>说道'
			end
