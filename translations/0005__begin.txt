if $ARGS[0] = 'get_random':
	if $ARGS[1] = 'loc':
		temp_rand = rand(0, 2)
		if temp_rand = 0:
			$result = 'sg'
		elseif temp_rand = 1:
			$result = 'uni'
		else
			$result = 'city'
		end
	elseif $ARGS[1] = 'magic':
		temp_rand = rand(0, 2)
		if temp_rand = 0:
			$result = 'nomagic'
		elseif temp_rand = 1:
			$result = 'magic'
		else
			$result = 'tg'
		end
	end

	killvar 'temp_rand'
	exit
end
