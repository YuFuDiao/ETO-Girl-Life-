# VolkovHome

if $ARGS[0] = 'Niko Bedroom':
	cls
	$loc_arg = 'Niko Bedroom'
	$loc = 'VolkovHome'
	minut += 5
	$location_type = 'private'
	$locclass = 'bedr'
	$menu_loc = 'VolkovHome'
	$menu_arg = 'Niko Bedroom'
	menu_off = 0
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/resident/volkovHome/Rooms/Niko/bedroom.jpg"></center>'
	'尼科的房间非常凌乱，里面有两盏台灯，一张床，以及散落各处的各种电子设备。'

	act '离开卧室': gt 'VolkovHome', 'Hallway 2'
end


if $ARGS[0] = 'Hallway 2':
	cls
	$loc_arg = 'Niko Bedroom'
	$loc = 'VolkovHome'
	minut += 5
	$location_type = 'private'
	killvar 'locclass'
	$menu_loc = 'VolkovHome'
	$menu_arg = 'Niko Bedroom'
	menu_off = 0
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/resident/volkovHome/Rooms/Hallway2.jpg"></center>'
	if NikoEv = 18:
		'这是二楼走廊，浴室，尼科的房间和尤里克的房间都在这里。'
	else
		'这是二楼走廊，就是那个<a href="exec:gt ''VolkovHome'', ''Bathroom''">浴室</a>, <a href="exec:gt ''VolkovHome'', ''Niko Bedroom''">尼科的房间</a>和<a href="exec:gt ''VolkovHome'', ''Yurik Bedroom''">Yurik的房间</a>所在的地方。'
	end

	act '下楼': gt 'NikoEv2', 'Yurik BJ'
end


if $ARGS[0] = 'Bathroom':
	cls
	$loc_arg = 'Bathroom'
	$loc = 'NikoDates'
	$locM_arg = 'Bathroom'
	$locM = 'NikoDates'
	$location_type = 'bathroom'
	minut += 5
	$menu_loc = 'VolkovHome'
	$menu_arg = 'Bathroom'
	menu_off = 0
	gs 'stat'
	'<center><img <<$set_imgh>> src="images/locations/pavlovsk/resident/volkovHome/Rooms/bathroom.jpg"></center>'
	'这间浴室看起来相当干净，里面有两块毛巾，两件浴袍，一个洗手池，一个马桶，一面<a href="exec:gt ''mirror'', ''start''">镜子</a>和一个浴缸。'

	act '快速清洗 (0:10)': gt 'NikoDates', 'Quick Wash'

	act '离开浴室':
		if NikoEv = 5:
			gt 'NikoDates', 'Yurik Intro'
		elseif NikoEv = 11:
			gt 'NikoEv2', 'Reward'
		end
	end
end


if $ARGS[0] = 'Quick Wash':
	cls
	minut += 10
	cumspclnt = 1
	gs 'cum_cleanup'
	if pcs_sweat < 30:
		pcs_sweat = 15 + rand(0, 4)
	else
		pcs_sweat -= 10 + rand(0, 4)
	end
	gs 'stat'
	$func('wrap', 'accent center h4', '浴室')
	'<center><img <<$set_imgh>> src="images/characters/pavlovsk/school/boy/fedor/fedorev2/movies/Restroom/facecleanse.jpg"></center>'
	'你抽出几张纸巾，走到洗手池前，尽可能把自己清理干净。你花了几秒钟享受温水泼在脸上的感觉，清洗完毕后，你神清气爽地离开洗手池。'

	act '擦干':
		if NikoEv = 5:
			gt 'NikoDates', 'Yurik Intro'
		elseif NikoEv = 11:
			gt 'NikoEv2', 'Reward'
		end
	end
end
