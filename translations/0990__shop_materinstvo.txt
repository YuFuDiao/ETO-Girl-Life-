# shop_materinstvo

if $shoplocation = 'items': gt 'shop_materinstvo', 'items'

if $ARGS[0] = '' or $ARGS[0] = 'start':
	CLOSE ALL
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_materinstvo', 'start'
	$location_type = 'public_indoors'
	killvar 'locclass'
	killvar 'shoplocation'
	gs 'stat'

	gs 'themes', 'indoors'
	gs 'shop_materinstvo' , 'config'

	$func('wrap', 'header center b', '妈妈风格')
	'<center><img <<$set_imgh>> src="images/locations/city/citycenter/mall/mommy/shop.jpg"></center>'
	*nl
	'这是一家专门出售与怀孕相关物品的商店。在这里你可以买到孕妇装或帮助你度过日常生活的物品。'

	act '离开商店': gt 'city_mall'
	act '查看衣服': minut += 5 & gt 'shop_materinstvo', 'clothes'
!!	act 'View items': minut += 3 & gt 'shop_materinstvo', 'items'
end


if $ARGS[0] = 'items':
	menu_off = 0
	$shoplocation = 'items'
	gs 'core_library', 'setloc', 'shop_materinstvo', 'items'
	$loc_s = 'shop_materinstvo'
	$args_s = 'items'

	gs 'stat'

	$backimage = 'images/locations/city/citycenter/mall/mommy/shop1.jpg'

!! This section is commented out until we can add function
!! Items here are not doing anything
	gs 'themes', 'indoors'
	gs 'item_cart', 'shopping_aisle', 'mothershop'
	gs 'stat'

	act '返回': gt 'shop_materinstvo', 'start'
end


if $ARGS[0] = 'clothes':
	menu_off = 0
	gs 'core_library', 'setloc', 'shop_materinstvo', 'clothes'
	$locclass = 'changingroom'
	gs 'stat'

	$func('wrap', 'header center b', '查看妈妈风格服装')
	*nl

	if func('shop_utils', 'is_init') = 0:
		gs 'shop_utils', 'init', 'start', 'clothing', 'shop'
			gs 'shop_utils', 'init', 'add_type', 'materinstvo_dress'
		gs 'shop_utils', 'init', 'end'
	end

	gs 'shop_utils', 'display', 'grid_shop'

	act '返回':
		minut += 1
		gs 'shop_utils', 'cleanup'
		gt 'shop_materinstvo', 'start'
	end
end
