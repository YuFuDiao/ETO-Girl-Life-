# tabhead


tabhead_max_i = arrsize('$tabsname')

if tabhead_max_i = 0:
	killvar 'tabhead_max_i'
	exit
end

$tabhead_table = '<center><b><table width="90%" cellspacing="0" cellpadding="0" valign="top"><tr>'

tabhead_i = 0
:tabs_loop
	$tabhead_table += '<th padding=<<40 / tabhead_max_i>>>'
	if ARGS[0] = tabhead_i:
		$tabhead_table += '<<$tabsname[tabhead_i]>>'
	else
		$tabhead_table += '<a href="exec:<<$tabsaction[tabhead_i]>>"><<$tabsname[tabhead_i]>></a>'
	end
	$tabhead_table += '</th>'
	if tabhead_i > 0 and arrpos('tabsrow', tabhead_i) >= 0: $tabhead_table += '</tr><tr>'
	tabhead_i += 1
if tabhead_i < tabhead_max_i:
	jump 'tabs_loop'
end
$tabhead_table += '</tr></table></b></center>'

*pl $tabhead_table

killvar 'tabhead_i'
killvar 'tabhead_max_i'
killvar 'tabhead_table'
killvar 'tabsname'
killvar 'tabsaction'
killvar 'tabsrow'


--- tabhead ---------------------------------
