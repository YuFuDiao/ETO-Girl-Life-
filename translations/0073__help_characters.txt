# help_characters
!! Character help pages.
!!
!! Each stub sets $_hc_key, $_hc_npcid, $_hc_portrait, $_hc_bio, $_hc_rel[0..n]
!! then calls gs 'help_characters', 'build'.
!! The build helper assembles title + content, calls render, and exits.

!! ── build helper ──────────────────────────────────────────────────────────────
!! Input vars (set by each stub before calling gs 'help_characters', 'build'):
!!   $_hc_npcid      – NPC array key, e.g. 'A28'
!!   $_hc_portrait   – image path
!!   $_hc_bio        – bio HTML; if empty, npc_notes[$_hc_npcid] is used as fallback
!!   $_hc_key        – original page key (set automatically by each stub)
!!   $_hc_rel[0..n]  – relation entries: 'relation type|char_key|npcID'
!!                     (iteration stops at first empty cell)
!! Note: $_hc_locvar is derived automatically from $_hc_npcid; stubs need not set it.
!! Output: $help_page_title, $help_page_content
!! Cleans up all $_hc_* / _hc_* vars then exits.

if $ARGS[0] = 'build':

	!! ── title (banner): only the used name ──
	$_hc_fn = $npc_firstname[$_hc_npcid]
	$_hc_nn = $npc_nickname[$_hc_npcid]
	$_hc_ln = $npc_lastname[$_hc_npcid]
	$_hc_un = $npc_usedname[$_hc_npcid]
	if $_hc_un ! '':
		$help_page_title = $_hc_un
	elseif $_hc_fn ! '':
		$help_page_title = $_hc_fn
	else
		$help_page_title = '未知'
	end

	!! ── full display name for the info panel heading ──
	$_hc_full = $_hc_fn
	if $_hc_nn ! '' and $_hc_nn ! $_hc_fn:
		$_hc_full	+= ' ''' + $_hc_nn + ''''
	end
	if $_hc_ln ! '':
		$_hc_full	+= ' ' + $_hc_ln
	end

	!! ── birthday ──
	if npc_dob[$_hc_npcid] > 0:
		$_hc_birthday = func('shortgs', 'convert_dob', $_hc_npcid)
	else
		$_hc_birthday = '未知'
	end

	!! ── residence: default to Unknown if stub did not set it ──
	if $_hc_residence = '':
		$_hc_residence = '未知'
	end

	!! ── current location: derive locat key from NPC ID ──
	$_hc_curloc = ''
	if $_hc_npcid = 'A28':
		$_hc_locvar = 'Stepdad'
	elseif $_hc_npcid = 'A29':
		$_hc_locvar = 'Mother'
	elseif $_hc_npcid = 'A33':
		$_hc_locvar = 'Anya'
	elseif $_hc_npcid = 'A34':
		$_hc_locvar = 'Kolka'
	elseif $_hc_npcid = 'A4':
		$_hc_locvar = 'igor'
	elseif $_hc_npcid = 'A14':
		$_hc_locvar = 'katja'
	elseif $_hc_npcid = 'A15':
		$_hc_locvar = 'Vicky'
	elseif $_hc_npcid = 'A13':
		$_hc_locvar = 'Lariska'
	elseif $_hc_npcid = 'A18':
		$_hc_locvar = 'Christina'
	elseif $_hc_npcid = 'A23':
		$_hc_locvar = 'A23'
	elseif $_hc_npcid = 'A248':
		$_hc_locvar = 'Silvestr'
	else
		$_hc_locvar = ''
	end
	if $_hc_locvar ! '':
		gs 'help_characters', 'loc_text'
		$_hc_curloc = $result
	end

	!! ── relations ──
	$_hc_relhtml = ''
	_hc_i = 0
	:_hc_relloop
	if $_hc_rel[_hc_i] ! '':
		!! parse 'rel_type|char_key|npc_id'
		_hc_p1 = instr($_hc_rel[_hc_i], '|')
		$_hc_reltype = $mid($_hc_rel[_hc_i], 1, _hc_p1 - 1)
		$_hc_rest = $mid($_hc_rel[_hc_i], _hc_p1 + 1)
		_hc_p2 = instr($_hc_rest, '|')
		$_hc_relkey = $mid($_hc_rest, 1, _hc_p2 - 1)
		$_hc_relnpc = $mid($_hc_rest, _hc_p2 + 1)
		!! resolve display name
		if $_hc_relnpc = 'PC':
			$_hc_relname = $pcs_firstname
		elseif $npc_usedname[$_hc_relnpc] ! '':
			$_hc_relname = $npc_usedname[$_hc_relnpc]
		else
			$_hc_relname = $npc_firstname[$_hc_relnpc]
		end
		!! build href: exec: gs 'help_characters', 'char_key'
		$_hc_href = 'exec: gs ' + '''' + 'help_characters' + '''' + ', ' + '''' + $_hc_relkey + ''''
		$_hc_relhtml += '<li>' + $_hc_reltype + ': <a href="' + $_hc_href + '">' + $_hc_relname + '</a></li>'
		_hc_i	+= 1
		jump '_hc_relloop'
	end

	!! ── assemble info panel (right of portrait) ──
	$_hc_info  = '<b style="font-size:1.05em">' + $_hc_full + '</b>'
	$_hc_info += '<table style="margin:4px 0 6px 0;border-collapse:collapse">'
	$_hc_info += '<tr><td style="padding:1px 10px 1px 0;opacity:0.65"><b>生日</b></td><td>' + $_hc_birthday + '</td></tr>'
	$_hc_info += '<tr><td style="padding:1px 10px 1px 0;opacity:0.65"><b>住所</b></td><td>' + $_hc_residence + '</td></tr>'
	if $_hc_curloc ! '':
		$_hc_info += '<tr><td style="padding:1px 10px 1px 0;opacity:0.65"><b>位置</b></td><td>' + $_hc_curloc + '</td></tr>'
	end
	$_hc_info += '</table>'
	if $_hc_relhtml ! '':
		$_hc_info += '<p style="margin:6px 0 2px 0"><b>关系</b></p>'
		$_hc_info += '<ul style="margin:2px 0 4px 0">' + $_hc_relhtml + '</ul>'
	end

	!! ── bio: fall back to npc_notes if stub left it empty ──
	if $_hc_bio = '':
		if $func('npc_notes', $_hc_npcid) ! '':
			$_hc_bio = '<p>' + $func('npc_notes', $_hc_npcid) + '</p>'
		end
	end

	!! ── two-column layout: portrait left | info panel right; bio row below ──
	$help_page_content  = '<table cellspacing="0" cellpadding="0" style="margin:4px 0 8px 0;width:100%"><tr>'
	$help_page_content += '<td style="width:1%;white-space:nowrap;vertical-align:top;padding-right:14px">'
	$help_page_content += '<img src="' + $_hc_portrait + '" style="height:200px;width:auto;max-width:180px">'
	$help_page_content += '</td>'
	$help_page_content += '<td style="vertical-align:top">' + $_hc_info + '</td>'
	$help_page_content += '</tr><tr>'
	$help_page_content += '<td colspan="2" style="padding-top:10px;border-top:1px solid rgba(128,128,128,0.25)">'
	$help_page_content += $_hc_bio
	$help_page_content += '</td></tr></table>'

	!! ── cleanup ──
	killvar '_hc_npcid'
	killvar '_hc_portrait'
	killvar '_hc_bio'
	killvar '_hc_rel'
	killvar '_hc_fn'
	killvar '_hc_nn'
	killvar '_hc_ln'
	killvar '_hc_un'
	killvar '_hc_full'
	killvar '_hc_birthday'
	killvar '_hc_residence'
	killvar '_hc_locvar'
	killvar '_hc_curloc'
	killvar '_hc_info'
	killvar '_hc_relhtml'
	killvar '_hc_i'
	killvar '_hc_p1'
	killvar '_hc_p2'
	killvar '_hc_reltype'
	killvar '_hc_rest'
	killvar '_hc_relkey'
	killvar '_hc_relnpc'
	killvar '_hc_relname'
	killvar '_hc_href'
	gs 'help', 'render', $_hc_key
	killvar '_hc_key'
	exit
end
