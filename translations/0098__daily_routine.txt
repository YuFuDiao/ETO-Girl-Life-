# daily_routine


!! gt daily_routine, start, phase  (morning|evening)
!!   Initialises the active routine and jumps to the hub. Must be gt-called.
if $ARGS[0] = 'start':
	gs 'daily_routine', 'settings_defaults'
	$droutine['phase']		= $ARGS[1]
	$droutine['return_loc'] = $loc
	$droutine['return_arg'] = $loc_arg
	droutine['active']		= 1
	droutine['active_slot'] = 0
	droutine['quick_running'] = 0
	droutine['day_anchor']	= func('daily_routine', 'logical_day', $ARGS[1])
	killvar 'droutine_done'
	gt 'daily_routine', 'hub'
end


!! gs daily_routine, finish_step, fallback_loc, fallback_arg
!!   Drop-in replacement for  menu_off=0 & gt $loc,$loc_arg  at the end of
!!   any chain action. If a routine is active, advances to the hub instead.
if $ARGS[0] = 'finish_step':
	droutine['quick_running'] = 0
	if droutine['active'] = 1:
		if droutine['active_slot'] > 0: droutine_done[droutine['active_slot']] = 1
		droutine['active_slot'] = 0
		gt 'daily_routine', 'hub'
	else
		menu_off = 0
		gt $ARGS[1], $ARGS[2]
	end
end


!! gt daily_routine, hub
!!   Routine dashboard. Shows pending steps as acts; skipped steps listed at
!!   the bottom. Returns here after each step completes.
if $ARGS[0] = 'hub':
	*clr & cla
	menu_off = 0
	droutine['active_slot'] = 0
	droutine['quick_running'] = 0
	if $droutine['phase'] = 'evening':
		'<center><b>晚间日常</b></center>'
	else
		'<center><b>晨间日常</b></center>'
	end
	temp_drc = droutine['<<$droutine[''phase'']>>_count']
	if temp_drc <= 0:
		*nl
		'这个日常还没有设置任何步骤。'
		act '完成': gt 'daily_routine', 'finish'
		killvar 'temp_drc'
		exit
	end
	*nl
	temp_dri = 0
	dr_any = 0
	$dr_unavail = ''
	$dr_done = ''
	:droutine_hub_loop
	temp_dri += 1
	if temp_dri <= temp_drc:
		$droutine['current_label'] = $droutine['<<$droutine[''phase'']>>_step_<<temp_dri>>']
		$droutine['current_runner'] = ''
		$droutine['skip_reason'] = ''
		droutine['can_run'] = 1
		droutine['current_quick'] = 0
		gs 'daily_routine', 'resolve', $droutine['<<$droutine[''phase'']>>_step_<<temp_dri>>']
		if droutine_done[temp_dri] = 1:
			$dr_done += '<li>' + $droutine['current_label'] + '.</li>'
		else
			if droutine['can_run'] ! 0 and $droutine['current_runner'] ! '':
				if dr_any = 0 and droutine_settings['quick_routine'] = 1 and droutine['current_quick'] = 1:
					droutine['active_slot'] = temp_dri
					droutine['quick_running'] = 1
					dynamic $droutine['current_runner']
					exit
				end
				dr_any = 1
				dynamic "act '<<$droutine['current_label']>>': droutine['active_slot'] = <<temp_dri>> & <<$droutine['current_runner']>>"
			else
				$dr_unavail += '<li>' + $droutine['current_label']
				if $droutine['skip_reason'] ! '': $dr_unavail += ' (<<$droutine[''skip_reason'']>>)'
				$dr_unavail += '.</li>'
			end
		end
		jump 'droutine_hub_loop'
	end
	if dr_any = 0: '所有可用步骤都已完成。' & *nl
	if $dr_unavail ! '':
		'<b>不可用步骤</b>:<ul><<$dr_unavail>></ul>'
	end
	if $dr_done ! '':
		'<b>已完成步骤</b>:<ol><<$dr_done>></ol>'
	end
	act '结束日常': gt 'daily_routine', 'finish'
	killvar 'temp_drc' & killvar 'temp_dri' & killvar 'dr_any' & killvar 'dr_unavail' & killvar 'dr_done'
end


!! gt daily_routine, finish
!!   Marks the phase done for the day, clears active state, returns to caller.
if $ARGS[0] = 'finish':
	droutine_settings['<<$droutine[''phase'']>>_done_day'] = droutine['day_anchor']
	droutine['active'] = 0
	droutine['active_slot'] = 0
	killvar 'droutine_done'
	menu_off = 0
	gt $droutine['return_loc'], $droutine['return_arg']
end


!! gs daily_routine, abort
!!   Clears active state when an external scene takes over (caller handles nav).
if $ARGS[0] = 'abort':
	droutine['active'] = 0
	droutine['active_slot'] = 0
	droutine['quick_running'] = 0
	killvar 'droutine_done'
	menu_off = 0
end


!! func(daily_routine, can_use_here)
!!   1 if the player is at a home, 0 otherwise.
if $ARGS[0] = 'can_use_here':
	result = func('homes_properties', 'is_at_a_home', '')
	exit
end


!! gs daily_routine, offer_here
!!   Adds routine entry acts for any phase that is available right now.
!!   Call from a home room; routine returns to that same room when done.
if $ARGS[0] = 'offer_here':
	if func('daily_routine', 'can_use_here') = 0: exit
	gs 'daily_routine', 'settings_defaults'
	if droutine_settings['disabled'] = 1: exit
	if droutine['morning_count'] > 0 and func('daily_routine', 'phase_available', 'morning'):
		act '开始你的晨间日常': gt 'daily_routine', 'start', 'morning'
	end
	if droutine['evening_count'] > 0 and func('daily_routine', 'phase_available', 'evening'):
		act '开始你的晚间日常': gt 'daily_routine', 'start', 'evening'
	end
end


!! ============================================================================
!! Availability windows and settings
!! ============================================================================

!! gs daily_routine, settings_defaults
!!   Seeds droutine_settings on first call (no-op after that).
!!   Step lists are seeded by intro_initialization, not here.
if $ARGS[0] = 'settings_defaults':
	if droutine_settings['init'] = 1: exit
	droutine_settings['init'] = 1
	droutine_settings['disabled'] = 1
	droutine_settings['morning_use_wake'] = 1
	droutine_settings['morning_wake_min'] = 120
	droutine_settings['morning_use_abs'] = 0
	droutine_settings['morning_abs_start'] = 5
	droutine_settings['morning_abs_end'] = 11
	!! Evening uses only an absolute hour window (no after-waking window).
	droutine_settings['evening_use_abs'] = 1
	droutine_settings['evening_abs_start'] = 20
	droutine_settings['evening_abs_end'] = 1
	!! Appearance step settings (persist across sessions).
	droutine_settings['makeup_level'] = 1
	exit
end


!! func(daily_routine, in_hour_window, start_hour, end_hour)
!!   1 if the current hour is in [start, end). Wraps midnight when end < start.
if $ARGS[0] = 'in_hour_window':
	result = 0
	if ARGS[1] = ARGS[2]:
		result = 0
	elseif ARGS[1] < ARGS[2]:
		if hour >= ARGS[1] and hour < ARGS[2]: result = 1
	else
		if hour >= ARGS[1] or hour < ARGS[2]: result = 1
	end
	exit
end


!! func(daily_routine, logical_day, phase)
!!   The day this phase''s current fixed hour window belongs to. Equal to
!!   daystart, except during the tail of a window that wraps past midnight
!!   (e.g. evening 20-1), where it stays pinned to the day the window
!!   started on so the whole wrapped window counts as a single occurrence.
if $ARGS[0] = 'logical_day':
	$dr_ld = $ARGS[1]
	if $dr_ld ! 'evening': $dr_ld = 'morning'
	result = daystart
	if droutine_settings['<<$dr_ld>>_use_abs'] = 1 and droutine_settings['<<$dr_ld>>_abs_end'] < droutine_settings['<<$dr_ld>>_abs_start'] and hour < droutine_settings['<<$dr_ld>>_abs_end']:
		result = daystart - 1
	end
	killvar 'dr_ld'
	exit
end


!! func(daily_routine, phase_available, phase)
!!   1 if the phase can be started now: either within the after-waking window
!!   or inside the fixed hour range (whichever are enabled). 0 if already done.
if $ARGS[0] = 'phase_available':
	gs 'daily_routine', 'settings_defaults'
	result = 0
	$dr_pa = $ARGS[1]
	if $dr_pa ! 'evening': $dr_pa = 'morning'
	if droutine_settings['<<$dr_pa>>_done_day'] = func('daily_routine', 'logical_day', $dr_pa):
		killvar 'dr_pa'
		exit
	end
	if droutine_settings['<<$dr_pa>>_use_wake'] = 1 and droutine['woke_at_min'] > 0:
		dr_pa_elapsed = totminut - droutine['woke_at_min']
		if dr_pa_elapsed >= 0 and dr_pa_elapsed <= droutine_settings['<<$dr_pa>>_wake_min']: result = 1
	end
	if droutine_settings['<<$dr_pa>>_use_abs'] = 1:
		if func('daily_routine', 'in_hour_window', droutine_settings['<<$dr_pa>>_abs_start'], droutine_settings['<<$dr_pa>>_abs_end']): result = 1
	end
	killvar 'dr_pa' & killvar 'dr_pa_elapsed'
	exit
end


!! ============================================================================
!! Step completion (instant workers)
!! ============================================================================

!! gs daily_routine, visit_item_done, step_id
!!   Completion hook for visit-style steps (kit_din workers). If a routine is
!!   active, marks the slot done and returns to hub; otherwise redraws the room.
if $ARGS[0] = 'visit_item_done':
	if droutine['active'] = 1:
		if droutine['active_slot'] > 0: droutine_done[droutine['active_slot']] = 1
		droutine['active_slot'] = 0
		gt 'daily_routine', 'hub'
	else
		gt $loc, $loc_arg
	end
end

!! ============================================================================

!! func(daily_routine, step_pos, phase, step_id)  -> 1-based index or 0
if $ARGS[0] = 'step_pos':
	result = 0
	temp_sp_n = droutine['<<$ARGS[1]>>_count']
	temp_sp_i = 0
	:step_pos_loop
	temp_sp_i += 1
	if temp_sp_i <= temp_sp_n:
		if $droutine['<<$ARGS[1]>>_step_<<temp_sp_i>>'] = $ARGS[2]:
			result = temp_sp_i
		else
			jump 'step_pos_loop'
		end
	end
	killvar 'temp_sp_n' & killvar 'temp_sp_i'
	exit
end

!! gs daily_routine, remove_step_core, phase, index
!!   Shifts elements after index down and decrements count. No nav.
if $ARGS[0] = 'remove_step_core':
	temp_drc = droutine['<<$ARGS[1]>>_count']
	temp_dri = ARGS[2]
	:droutine_rsc_loop
	if temp_dri < temp_drc:
		$droutine['<<$ARGS[1]>>_step_<<temp_dri>>'] = $droutine['<<$ARGS[1]>>_step_<<temp_dri + 1>>']
		temp_dri += 1
		jump 'droutine_rsc_loop'
	end
	$droutine['<<$ARGS[1]>>_step_<<temp_drc>>'] = ''
	droutine['<<$ARGS[1]>>_count'] = temp_drc - 1
	killvar 'temp_drc' & killvar 'temp_dri'
end

!! gs daily_routine, remove_step_id, phase, step_id  - no-op if not present, no nav
if $ARGS[0] = 'remove_step_id':
	temp_rsid = func('daily_routine', 'step_pos', $ARGS[1], $ARGS[2])
	if temp_rsid > 0:
		gs 'daily_routine', 'remove_step_core', $ARGS[1], temp_rsid
	end
	killvar 'temp_rsid'
end

!! gt daily_routine, toggle_step, phase, step_id  - appends if absent, removes if present
if $ARGS[0] = 'toggle_step':
	temp_tp = func('daily_routine', 'step_pos', $ARGS[1], $ARGS[2])
	if temp_tp > 0:
		gs 'daily_routine', 'remove_step_core', $ARGS[1], temp_tp
	else
		temp_drc = droutine['<<$ARGS[1]>>_count'] + 1
		$droutine['<<$ARGS[1]>>_step_<<temp_drc>>'] = $ARGS[2]
		droutine['<<$ARGS[1]>>_count'] = temp_drc
		killvar 'temp_drc'

!! gt daily_routine, set_bath, phase, bath_id  (shower|bath|quickwash|none)
!!   Swaps the active bathing step in place, or appends if none was present.
if $ARGS[0] = 'set_bath':
	!! Remember where the old bathing step was (0 if none active).
	temp_sb_pos = func('daily_routine', 'step_pos', $ARGS[1], 'shower')
	if temp_sb_pos = 0: temp_sb_pos = func('daily_routine', 'step_pos', $ARGS[1], 'bath')
	if temp_sb_pos = 0: temp_sb_pos = func('daily_routine', 'step_pos', $ARGS[1], 'quickwash')
	gs 'daily_routine', 'remove_step_id', $ARGS[1], 'shower'
	gs 'daily_routine', 'remove_step_id', $ARGS[1], 'bath'
	gs 'daily_routine', 'remove_step_id', $ARGS[1], 'quickwash'
	if $ARGS[2] ! 'none':
		if temp_sb_pos > 0:
			!! Insert at the preserved position: shift everything from that slot up.
			temp_sb_n = droutine['<<$ARGS[1]>>_count'] + 1
			droutine['<<$ARGS[1]>>_count'] = temp_sb_n
			temp_sb_i = temp_sb_n
			:set_bath_ins_loop
			if temp_sb_i > temp_sb_pos:
				$droutine['<<$ARGS[1]>>_step_<<temp_sb_i>>'] = $droutine['<<$ARGS[1]>>_step_<<temp_sb_i - 1>>']
				temp_sb_i -= 1
				jump 'set_bath_ins_loop'
			end
			$droutine['<<$ARGS[1]>>_step_<<temp_sb_pos>>'] = $ARGS[2]
			killvar 'temp_sb_n' & killvar 'temp_sb_i'
		else
			!! No previous bathing step - append at the end.
			temp_drc = droutine['<<$ARGS[1]>>_count'] + 1
			$droutine['<<$ARGS[1]>>_step_<<temp_drc>>'] = $ARGS[2]
			droutine['<<$ARGS[1]>>_count'] = temp_drc
			killvar 'temp_drc'
		end
	end
	killvar 'temp_sb_pos'
	gt 'daily_routine', 'manage_phase', $ARGS[1]
end

!! gt daily_routine, manage_phase, phase
!!   Phase editor. Two sections: current steps (reorderable, removable) and
!!   catalog steps available to add, grouped by category.
if $ARGS[0] = 'manage_phase':
	*clr & cla
	menu_off = 0
	$dr_ph = $ARGS[1]
	if $dr_ph ! 'evening': $dr_ph = 'morning'
	if $dr_ph = 'evening':
		*p '<center><b>晚间日常</b></center>'
	else
		*p '<center><b>晨间日常</b></center>'
	end
	gs 'daily_routine', 'settings_defaults'
	gs 'daily_routine', 'catalog_build'

	!! ---- Section 1: the routine, in execution order (no numbers) ----
	*p '<b>你的日常</b> <i>（按顺序自上而下播放）</i>'
	if droutine['<<$dr_ph>>_count'] <= 0:
		*p '<i>还没有步骤 - 从下面的列表中添加一些。</i>'
	else
		*p '<table width=100% cellpadding=4 cellspacing=0>'
		temp_dpi = 0
		:droutine_mp_orderloop
		temp_dpi += 1
		if temp_dpi <= droutine['<<$dr_ph>>_count']:
			$dr_sid = $droutine['<<$dr_ph>>_step_<<temp_dpi>>']
			gs 'daily_routine', 'resolve', $dr_sid
			$dr_col2 = ''
			if temp_dpi > 1: $dr_col2 += '<a href="exec:gt ''daily_routine'', ''move_up'', ''<<$dr_ph>>'', <<temp_dpi>>">[↑]</a>&nbsp;'
			if temp_dpi < droutine['<<$dr_ph>>_count']: $dr_col2 += '<a href="exec:gt ''daily_routine'', ''move_down'', ''<<$dr_ph>>'', <<temp_dpi>>">[↓]</a>&nbsp;'
			$dr_col2 += '&nbsp;&nbsp;'
			if $dr_sid = 'shower' or $dr_sid = 'bath' or $dr_sid = 'quickwash':
				$dr_col1 = '<b>洗澡</b>'
				gs 'daily_routine', 'bath_picker', $dr_ph
				$dr_col2 += $dr_pick
			elseif $dr_sid = 'apply_makeup':
				$dr_col1 = '<b>化妆</b>'
				gs 'daily_routine', 'makeup_picker', $dr_ph
				$dr_col2 += $dr_pick
			elseif $dr_sid = 'apply_lashes':
				$dr_col1 = '<b>贴上假睫毛</b>'
				gs 'daily_routine', 'lashes_picker', $dr_ph
				$dr_col2 += $dr_pick
			else
				$dr_col1 = '<b><<$droutine[''current_label'']>></b>'
				$dr_col2 += '<a href="exec:gt ''daily_routine'', ''toggle_step'', ''<<$dr_ph>>'', ''<<$dr_sid>>''">[remove]</a>'
			end
			*p '<tr><td width=35%><<$dr_col1>></td><td><<$dr_col2>></td></tr>'
			killvar 'dr_col1' & killvar 'dr_col2'
			jump 'droutine_mp_orderloop'
		end
		*p '</table>'
	end
	killvar 'dr_pick'
	killvar 'temp_dpi'

	*nl
	!! ---- Section 2: steps available to add, grouped by category ----
	*p '<b>添加步骤</b>'
	*p '<table width=100% cellpadding=4 cellspacing=0>'
	$dr_prevcat = ''
	dr_bath_shown = 0
	temp_dci = 0
	:droutine_mp_addloop
	temp_dci += 1
	if temp_dci <= droutine_catalog['count']:
		$dr_sid = $droutine_catalog['id_<<temp_dci>>']
		!! Steps already in the routine are shown in section 1 above.
		if func('daily_routine', 'step_pos', $dr_ph, $dr_sid) > 0: jump 'droutine_mp_addloop'
		gs 'daily_routine', 'resolve', $dr_sid
		dr_show = 1
		$dr_col1 = ''
		$dr_col2 = ''
		if $dr_sid = 'shower' or $dr_sid = 'bath' or $dr_sid = 'quickwash':
			!! One bathing chooser only, and only when no bathing step is active.
			if func('daily_routine', 'step_pos', $dr_ph, 'shower') > 0 or func('daily_routine', 'step_pos', $dr_ph, 'bath') > 0 or func('daily_routine', 'step_pos', $dr_ph, 'quickwash') > 0:
				dr_show = 0
			elseif dr_bath_shown = 1:
				dr_show = 0
			else
				dr_bath_shown = 1
				$dr_col1 = '洗澡'
				$dr_col2 = '<a href="exec:gt ''daily_routine'', ''set_bath'', ''<<$dr_ph>>'', ''shower''">淋浴</a>&nbsp;&nbsp;<a href="exec:gt ''daily_routine'', ''set_bath'', ''<<$dr_ph>>'', ''bath''">泡澡</a>&nbsp;&nbsp;<a href="exec:gt ''daily_routine'', ''set_bath'', ''<<$dr_ph>>'', ''quickwash''">快速清洗</a>'
			end
		elseif $dr_sid = 'apply_makeup':
			$dr_col1 = '化妆'
			$dr_col2 = '<a href="exec:gt ''daily_routine'', ''add_makeup'', ''<<$dr_ph>>''">[add]</a>'
		elseif $dr_sid = 'apply_lashes':
			$dr_col1 = '贴上假睫毛'
			$dr_col2 = '<a href="exec:gt ''daily_routine'', ''add_lashes'', ''<<$dr_ph>>''">[add]</a>'
		else
			$dr_col1 = '<<$droutine[''current_label'']>>'
			$dr_col2 = '<a href="exec:gt ''daily_routine'', ''toggle_step'', ''<<$dr_ph>>'', ''<<$dr_sid>>''">[add]</a>'
		end
		if dr_show = 1:
			if $droutine['current_category'] ! $dr_prevcat:
				$dr_prevcat = $droutine['current_category']
				*p '<tr><td colspan=2 style="padding-top:10px;border-bottom:1px solid #555"><b><<$dr_prevcat>></b></td></tr>'
			end
			*p '<tr><td width=35%><font color="#888"><<$dr_col1>></font></td><td><<$dr_col2>></td></tr>'
		end
		killvar 'dr_col1' & killvar 'dr_col2'
		jump 'droutine_mp_addloop'
	end
	*p '</table>'
	killvar 'temp_dci' & killvar 'dr_sid' & killvar 'dr_prevcat' & killvar 'dr_bath_shown' & killvar 'dr_show'
	*nl
	act '返回': gt 'daily_routine', 'manage'
end

!! gt daily_routine, remove_step, phase, index  - removes by index, shifts down
if $ARGS[0] = 'remove_step':
	temp_drc = droutine['<<$ARGS[1]>>_count']
	temp_dri = ARGS[2]
	:droutine_remove_loop
	if temp_dri < temp_drc:
		$droutine['<<$ARGS[1]>>_step_<<temp_dri>>'] = $droutine['<<$ARGS[1]>>_step_<<temp_dri + 1>>']
		temp_dri += 1
		jump 'droutine_remove_loop'
	end
	!! Blank the now-duplicate last slot. The count drop below keeps it unread.
	$droutine['<<$ARGS[1]>>_step_<<temp_drc>>'] = ''
	droutine['<<$ARGS[1]>>_count'] = temp_drc - 1
	killvar 'temp_drc' & killvar 'temp_dri'
	gt 'daily_routine', 'manage_phase', $ARGS[1]
end

		$dr_line += $droutine['current_label']
		jump 'droutine_rli_loop'
	end
	if $dr_line = '': $dr_line = '（无）'
	'  <<$dr_line>>'
	killvar 'temp_drc' & killvar 'temp_dri' & killvar 'dr_line'
end
